+
+We use a single AsyncEngine per process and a sessionmaker bound to it.
+The `get_session` dependency yields a session and guarantees cleanup even
+on exceptions (the `try/finally` in the FastAPI dependency pattern).
+"""

+
+from collections.abc import AsyncIterator
+
+from sqlalchemy.ext.asyncio import (
+    AsyncEngine,
+    AsyncSession,
+    async_sessionmaker,
+    create_async_engine,
+)
+from sqlalchemy.orm import DeclarativeBase
+
+from .config import settings
+
+
+class Base(DeclarativeBase):
+    """Declarative base for all ORM models."""
+
+
+# Module-level engine. Tests reset this via ``create_test_engine()`` so each
+# test session can use a fresh in-memory SQLite without polluting the next.
+engine: AsyncEngine = create_async_engine(
+    settings.database_url,
+    echo=False,
+    future=True,
+)
+
+SessionLocal: async_sessionmaker[AsyncSession] = async_sessionmaker(
+    engine,
+    expire_on_commit=False,
+    class_=AsyncSession,
+)
+
+
+async def get_session() -> AsyncIterator[AsyncSession]:
+    """FastAPI dependency. Yields a session, commits on success, rolls back
+    on exception, always closes the connection."""
+    async with SessionLocal() as session:
+        try:
+            yield session
+            await session.commit()
+        except Exception:
+            await session.rollback()
+            raise
+
+
+async def init_db() -> None:
+    """Create all tables. Called once at app startup."""
+    # Importing models here registers them on Base.metadata.
+    from . import models  # noqa: F401
+
+    async with engine.begin() as conn:
+        await conn.run_sync(Base.metadata.create_all)

+"""Rate limiter setup (slowapi).
+
+Two pieces to wire correctly:
+  1. ``Limiter`` is the singleton. Per-route decorators (``@limiter.limit``)
+     pull the request from the route's first arg, so every protected route
+     MUST accept ``request: Request`` as its first parameter.
+  2. ``SlowAPIMiddleware`` must be added so the ``Retry-After`` header is
+     set on 429 responses. The exception handler is registered separately
+     because slowapi raises its own ``RateLimitExceeded`` exception.
+
+Storage: defaults to ``memory://`` (single-process). For multi-worker prod,
+swap to ``redis://...`` via ``TODO_API_RATE_LIMIT_STORAGE``.
+"""
+from __future__ import annotations
+
+from slowapi import Limiter
+from slowapi.errors import RateLimitExceeded
+from slowapi.middleware import SlowAPIMiddleware
+from slowapi.util import get_remote_address
+from starlette.requests import Request
+from starlette.responses import JSONResponse
+
+from .config import settings
+
+
+def _key_func(request: Request) -> str:
+    """Bucket by client IP. Falls back to a constant if no IP (e.g. test)."""
+    return get_remote_address(request) or "anonymous"
+
+
+limiter = Limiter(
+    key_func=_key_func,
+    storage_uri=settings.rate_limit_storage_uri,
+    default_limits=[settings.rate_limit],
+)
+
+
+async def rate_limit_exceeded_handler(
+    request: Request, exc: RateLimitExceeded
+) -> JSONResponse:
+    """Convert slowapi's RateLimitExceeded into a JSON 429 with Retry-After."""
+    return JSONResponse(
+        status_code=429,
+        content={"detail": f"rate limit exceeded: {exc.detail}"},
+        headers={"Retry-After": str(exc.detail)},
+    )
+
+
+__all__ = ["limiter", "rate_limit_exceeded_handler", "SlowAPIMiddleware"]

+## 3. 設計決策（有疑問的取捨）
+
+### 3.1 為什麼 slowapi + SQLAlchemy 2.0 async（不是手刻限流 / 純 sqlite3）
+
+任務留白沒指定。我用 `clarify` 詢問（已 timeout），採下列預設：
+
+| 選擇 | 理由 |
+|------|------|
+| **slowapi** 限流 | FastAPI 生態標準、文件齊、鍵函數可換；自刻限流會需要 token bucket + storage + 重置語意，純重造輪子 |
+| **SQLAlchemy 2.0 async** | 與 FastAPI 異步模型一致；aiosqlite 是 oficial async driver；ORM 易擴展（如未來加 user_id） |
+| **純 sqlite3（stdlib）** | 程式碼短但 FastAPI sync def 在 DB 阻塞時會跨界事件循環，**反模式** |
+
+### 3.2 為什麼寫在 todo_api/todo_api/ 雙層嵌套
+
+`pyproject.toml` 用 `setuptools.packages.find` 找 `todo_api*` 套件，**必須**有可匯入的命名空間。
+第一次安裝時 setuptools 預設吃 inner `app/` 為 top-level（因為 `where = ["."]` 沒指對），
+導致 `ModuleNotFoundError`。**修正**：建立 `todo_api/todo_api/` 雙層目錄。
+
+### 3.3 為什麼用 StaticPool + 同 connection 處理 in-memory SQLite
+
+最常見的 SQLite in-memory 測試坑：每條 connection 看到**獨立**的 `:memory:` DB，
+第一個 `SELECT` 就 `no such table: todos`。
+`StaticPool` 強制所有 session 共用同一 connection → 共用同一 schema。
+這是 `learning_1785165605_3` 沒處理好的地方（用文件 SQLite，跨 process 不穩）。
+
+### 3.4 為什麼每個限流路由都接收 `request: Request`
