+load_env_file() {
+  # This parser intentionally supports only KEY=VALUE lines.  It does not
+  # source shell code, so a compromised .env cannot execute commands.
+  local env_file="$1"
+  [[ -f "$env_file" ]] || return 0
+  while IFS= read -r line || [[ -n "$line" ]]; do
+    [[ -z "$line" || "$line" == \#* ]] && continue
+    if [[ "$line" != *=* ]]; then
+      printf '錯誤：.env 含無效行（必須是 KEY=VALUE）\n' >&2
+      return 2
+    fi
+    local key="${line%%=*}"
+    local value="${line#*=}"
+    if [[ ! "$key" =~ ^[A-Za-z_][A-Za-z0-9_]*$ ]]; then
+      printf '錯誤：.env 含無效變數名稱\n' >&2
+      return 2
+    fi
+    value="${value%$'\r'}"
+    if [[ "$value" == \"*\" && "$value" == *\" ]]; then
+      value="${value:1:${#value}-2}"
+    elif [[ "$value" == \'*\' && "$value" == *\' ]]; then
+      value="${value:1:${#value}-2}"
+    fi
+    printf -v "$key" '%s' "$value"
+    export "$key"
+  done < "$env_file"

+# Telegram 天氣機器人
+
+純 Python 3 標準函式庫實作：接收 Telegram 文字，透過 Open-Meteo 查詢目前天氣，再回傳給使用者。Open-Meteo 公開端點不需 API key。
+
+## 架構
+
+```text
+Telegram getUpdates → WeatherBot 路由/allowlist/限流
+                    → OpenMeteoClient（geocoding + current weather）
+                    → Telegram sendMessage
+```
+
+- `weather_api.py`：Open-Meteo client，可獨立 CLI 使用。
+- `telegram_weather_bot.py`：Telegram long polling、訊息路由、限流與訊號處理。
+- `test_bot.py`：離線 mock 單元與整合測試。
+- `run.sh`：安全載入 `.env`、測試、檢查與啟動入口。
+- `IMPLEMENTATION_NOTES.md`：完整實作決策與驗收證據。
+
+## 快速開始
+
+1. 在 Telegram 向 `@BotFather` 建立 bot，取得 token。

+cp .env.example .env

+

+from __future__ import annotations
+
+import json
+import os
+import signal
+import subprocess
+import sys
+import threading
+import time
+from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
+from pathlib import Path
+
+PROJECT = Path('/home/hoonsoropenclaw/.hermes/projects/learning_1785203404_3')
+TEST_TOKEN = '123' + '456:' + 'signal_smoke_secret_value_123'
+
+
+class Handler(BaseHTTPRequestHandler):
+    def do_POST(self):  # noqa: N802
+        length = int(self.headers.get('Content-Length', '0'))
+        self.rfile.read(length)
+        raw = json.dumps({'ok': True, 'result': []}).encode()
+        self.send_response(200)
+        self.send_header('Content-Type', 'application/json')
+        self.send_header('Content-Length', str(len(raw)))
+        self.end_headers()
+        self.wfile.write(raw)
+
+    def log_message(self, _format, *_args):
+        return
+
+
+def main() -> int:
+    server = ThreadingHTTPServer(('127.0.0.1', 0), Handler)
+    thread = threading.Thread(target=server.serve_forever, daemon=True)
+    thread.start()
+    env = os.environ.copy()
+    env.update({
+        'TELEGRAM_BOT_TOKEN': TEST_TOKEN,
+        'TELEGRAM_ALLOWED_USERS': '42',
+        'TELEGRAM_API_BASE': f'http://127.0.0.1:{server.server_port}',
+        'RATE_LIMIT_SECONDS': '0',
+        'LOG_LEVEL': 'INFO',
+    })
+    process = subprocess.Popen(
+        [sys.executable, str(PROJECT / 'telegram_weather_bot.py')],
+        cwd=PROJECT,
+        env=env,
+        stdout=subprocess.PIPE,
+        stderr=subprocess.PIPE,
+        text=True,
+    )
+    try:
+        time.sleep(0.3)
+        if process.poll() is not None:
+            stdout, stderr = process.communicate()
+            raise AssertionError(f'bot exited before SIGTERM: {process.returncode} {stdout} {stderr}')
+        process.send_signal(signal.SIGTERM)
+        stdout, stderr = process.communicate(timeout=10)
+    finally:
+        if process.poll() is None:
+            process.kill()
+            process.wait(timeout=5)
+        server.shutdown()
+        server.server_close()
+        thread.join(timeout=2)
