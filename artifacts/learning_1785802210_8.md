  axialToPixelPointy: (q, r, size) => ({ x: size * Math.sqrt(3) * (q + r/2), y: size * 1.5 * r }),
  axialToPixelFlat:    (q, r, size) => ({ x: size * 1.5 * q, y: size * Math.sqrt(3) * (r + q/2) }),
  pixelToAxialPointy:  (x, y, size) => ({ q: (Math.sqrt(3)/3 * x - y/3) / size, r: (2/3 * y) / size }),
  pixelToAxialFlat:    (x, y, size) => ({ q: (2/3 * x) / size, r: (-x/3 + Math.sqrt(3)/3 * y) / size }),
  hexRound: (qf, rf) => {
    const xf=qf, zf=rf, yf=-xf-zf;

+  raw = raw.replace(/(^|\n)([ \t]*#[^\n]*)/g, (m, lead, body) => lead + stash('c', body));

+      'break','continue','match','enum','class_name'
+    ].join('|') + ')\\b', 'g'
+  );
+  raw = raw.replace(KEYWORDS, '<span class="k">$1</span>');

+  raw = raw.replace(
+    /\b(Vector2|Vector2i|Color|Array|Dictionary|PackedVector2Array|String|int|float|bool|void|Node2D|Node|Resource|RefCounted)\b/g,

+  raw = raw.replace(/\b(\d+\.?\d*)\b/g, '<span class="n">$1</span>');
+
+  // Restore sentinels
