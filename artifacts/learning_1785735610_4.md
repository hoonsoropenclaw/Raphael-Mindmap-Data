+#!/usr/bin/env python3
+"""Deterministic Godot visual regression runner.
+
+Captures the same frame twice, hashes PNG bytes, and optionally compares against
+an approved baseline. A hash-only gate is deliberate here: Godot's renderer is
+fixed to GL Compatibility and the scene contains no external assets or fonts,
+so byte equality is a useful, strict signal for this fixture.
+"""
+from __future__ import annotations
+import argparse, hashlib, json, shutil, subprocess, sys, time
+from pathlib import Path
+
+ROOT = Path(__file__).resolve().parent
+ARTIFACTS = ROOT / "tests" / "artifacts"
+BASELINE = ARTIFACTS / "baseline.png"
+CURRENT = ARTIFACTS / "current.png"
+DIFF = ARTIFACTS / "diff.png"
+REPORT = ARTIFACTS / "visual_report.json"
+
+def sha(path: Path) -> str:
+    return hashlib.sha256(path.read_bytes()).hexdigest()
+
+def capture(path: Path, seconds: float) -> tuple[int, str]:
+    path.parent.mkdir(parents=True, exist_ok=True)
+    cmd = ["godot", "--path", str(ROOT), "--editor-pid", "-1", "--headless", "--write-movie", str(path), "--quit-after", str(seconds)]
+    # Godot's movie writer is a deterministic frame capture path. We keep its
+    # stderr/stdout in the report so failures are diagnosable.
+    proc = subprocess.run(cmd, capture_output=True, text=True, timeout=45)
+    return proc.returncode, (proc.stdout + proc.stderr)[-4000:]
+
+def main() -> int:
+    ap = argparse.ArgumentParser()
+    ap.add_argument("--update", action="store_true", help="approve current capture as baseline")
+    ap.add_argument("--seconds", type=float, default=1.0)
+    args = ap.parse_args()
+    ARTIFACTS.mkdir(parents=True, exist_ok=True)
+    code, log = capture(CURRENT, args.seconds)
+    result = {"capture_exit_code": code, "baseline": str(BASELINE), "current": str(CURRENT), "timestamp": time.time()}
+    if code != 0 or not CURRENT.exists():
+        result.update(status="error", log=log)
+        REPORT.write_text(json.dumps(result, indent=2), encoding="utf-8")
+        print(json.dumps(result, indent=2)); return 2
+    if args.update or not BASELINE.exists():
+        shutil.copy2(CURRENT, BASELINE)
+        result.update(status="baseline-created" if not args.update else "baseline-updated", current_sha256=sha(CURRENT), baseline_sha256=sha(BASELINE))
+        REPORT.write_text(json.dumps(result, indent=2), encoding="utf-8")
+        print(json.dumps(result, indent=2)); return 0
+    same = sha(BASELINE) == sha(CURRENT)
+    result.update(status="passed" if same else "regression", same_bytes=same, current_sha256=sha(CURRENT), baseline_sha256=sha(BASELINE), log=log)
+    if not same:
+        try:
+            from PIL import Image, ImageChops
+            a, b = Image.open(BASELINE).convert("RGBA"), Image.open(CURRENT).convert("RGBA")
+            if a.size == b.size: ImageChops.difference(a, b).save(DIFF)
+            result["image_size"] = {"baseline": a.size, "current": b.size}
+            result["diff"] = str(DIFF) if DIFF.exists() else None
+        except Exception as exc:
+            result["diff_error"] = str(exc)
+    REPORT.write_text(json.dumps(result, indent=2), encoding="utf-8")
+    print(json.dumps(result, indent=2))
+    return 0 if same else 1
+
+if __name__ == "__main__": raise SystemExit(main())
