+# ─── 純函式（單元測試） ───
+
+def add(a: float, b: float) -> float:
+    return a + b
+
+def divide(a: float, b: float) -> float:
+    if b == 0:
+        raise ZeroDivisionError("b must not be zero")
+    return a / b
+
+def normalize_scores(scores: Iterable[float]) -> list[float]:
+    """Min-max 標準化到 [0, 1]。"""
+    arr = list(scores)
+    if not arr:
+        raise ValueError("empty input")
+    lo, hi = min(arr), max(arr)
+    if hi == lo:
+        return [0.0] * len(arr)
+    return [(x - lo) / (hi - lo) for x in arr]
+
+
+# ─── 重 CPU（基準測試） ───
+
+def heavy_primes(n: int) -> list[int]:
+    """找出前 n 個質數。基準測試用。"""
+    if n <= 0:
+        return []
+    primes: list[int] = []
+    candidate = 2
+    while len(primes) < n:
+        is_prime = True
+        sqrt_c = math.isqrt(candidate)
+        for p in primes:
+            if p > sqrt_c:
+                break
+            if candidate % p == 0:
+                is_prime = False
+                break
+        if is_prime:
+            primes.append(candidate)
+        candidate += 1
+    return primes
+
+
