+/**
+ * Zero-dependency static file server.
+ *
+ * Why:  Playwright's built-in `webServer` command runner expects a stable
+ *       URL we can poll. A 60-line http.createServer keeps the test run
+ *       hermetic — no global Python/Node process to manage, no port
+ *       collisions, no extra npm packages.
+ *
+ * Usage: `node utils/serve.js [port] [rootDir]`
+ *        PORT env var also honoured.
+ */
+const http = require('http');
+const fs = require('fs');
+const path = require('path');
+
+const PORT = Number(process.argv[2] || process.env.PORT || 4173);
+const ROOT = path.resolve(process.argv[3] || path.join(__dirname, '..', '..'));
+
+const MIME = {
+  '.html': 'text/html; charset=utf-8',
+  '.js':   'application/javascript; charset=utf-8',
+  '.mjs':  'application/javascript; charset=utf-8',
+  '.css':  'text/css; charset=utf-8',
+  '.json': 'application/json; charset=utf-8',
+  '.svg':  'image/svg+xml',
+  '.png':  'image/png',
+  '.jpg':  'image/jpeg',
+  '.jpeg': 'image/jpeg',
+  '.gif':  'image/gif',
+  '.ico':  'image/x-icon',
+  '.txt':  'text/plain; charset=utf-8',
+  '.map':  'application/json; charset=utf-8',
+};
+
+function safeJoin(root, urlPath) {
+  // strip query/hash, decode, prevent traversal
+  const clean = decodeURIComponent(urlPath.split('?')[0].split('#')[0]);
+  const target = path.normalize(path.join(root, clean));
+  if (!target.startsWith(root)) return null;
+  return target;
+}
+
+const server = http.createServer((req, res) => {
+  let urlPath = req.url || '/';
+  if (urlPath === '/') urlPath = '/web_output.html';
+  const filePath = safeJoin(ROOT, urlPath);
+  if (!filePath) { res.writeHead(403); res.end('forbidden'); return; }
+
+  fs.stat(filePath, (err, stat) => {
+    if (err || !stat.isFile()) {
+      res.writeHead(404, { 'content-type': 'text/plain' });
+      res.end('not found: ' + urlPath);
+      return;
+    }
+    const ext = path.extname(filePath).toLowerCase();
+    res.writeHead(200, {
+      'content-type': MIME[ext] || 'application/octet-stream',
+      'cache-control': 'no-store',
+      'x-content-type-options': 'nosniff',
+    });
+    fs.createReadStream(filePath).pipe(res);
+  });
+});
+
+server.listen(PORT, '127.0.0.1', () => {
+  // Plain stdout is the easiest signal for the runner to scrape
+  console.log(`[serve] listening on http://127.0.0.1:${PORT}  (root=${ROOT})`);
+});
+
+function shutdown() { server.close(() => process.exit(0)); }
+process.on('SIGINT', shutdown);
+process.on('SIGTERM', shutdown);

+/**
+ * Cross-Browser Test Runner
+ * -------------------------
+ * Lightweight wrapper around the `playwright` library that runs the same
+ * test suite against Chromium / Firefox / WebKit, in parallel, with:
+ *   - per-viewport capture (desktop / tablet / mobile)
+ *   - automatic retry with exponential backoff
+ *   - screenshot-on-failure (full page + viewport)
+ *   - console + pageerror + network-failure capture
+ *   - structured JSON report at reports/runner-report.json
+ *   - optional suite filter (`--suite smoke|functional|visual|a11y|responsive`)
+ *   - optional browser filter (`--browsers chromium,firefox`)
+ *   - optional viewport filter (`--viewports desktop,mobile`)
+ *
+ * Why not just `npx playwright test`?  Because (a) the report format this
+ * runner emits is the same shape the user-facing portal consumes, (b) it
+ * avoids the extra `@playwright/test` dependency when the user already
+ * has plain `playwright`, and (c) it makes the cross-browser matrix a
+ * first-class object instead of being hidden in `playwright.config.js`.
+ */

+const fs = require('fs');
+const { chromium, firefox, webkit, devices } = require('playwright');

+# Playwright Cross-Browser Test Framework
+
+A self-contained cross-browser test framework built on **Playwright 1.62** that
+stabilises front-end applications by running the same test catalogue against
+**Chromium / Firefox / WebKit** across **desktop / tablet / mobile** viewports —
+in parallel, with auto-retry, network-error capture, visual regression, and an

+## Why this framework
+
+- **No third-party services.** Runs offline. The static server is 60 lines of
+  Node, the runner is 300 lines, the report generator is 90 lines. Everything
+  you ship is auditable.
+- **Same matrix, two runners.** The catalogue lives in `utils/runner.js` for
+  zero-dep execution; the same cases are also exposed as `*.spec.js` for
+  `npx playwright test` if you install `@playwright/test`.
+- **Failures ship evidence.** Every failed run saves a full-page screenshot
+  under `screenshots/` and embeds it into `reports/runner-report.html`.
+- **Visual regression with sensible defaults.** First run seeds the baseline;
+  subsequent runs fail when the SHA-256 of the screenshot drifts, with a
+  pixel-independent tolerance for small layout shifts.
