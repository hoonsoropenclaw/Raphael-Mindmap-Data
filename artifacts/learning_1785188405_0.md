+"""Fail a build when a source/config file contains a likely real secret."""
+
+from __future__ import annotations
+
+import argparse
+import re
+import sys
+from pathlib import Path
+
+PATTERNS = {
+    "Telegram bot token": re.compile(rb"\b\d{6,12}:[A-Za-z0-9_-]{30,}\b"),
+    "GitHub token": re.compile(rb"\b(?:ghp|gho)_[A-Za-z0-9]{30,}\b"),
+    "OpenAI-style key": re.compile(rb"\bsk-[A-Za-z0-9_-]{30,}\b"),
+    "Vercel token": re.compile(rb"\bvcp_[A-Za-z0-9_-]{20,}\b"),
+}
+SKIP_NAMES = {".git", "__pycache__", ".coverage", "SHA256SUMS"}
+TEXT_SUFFIXES = {".py", ".sh", ".md", ".txt", ".example", ".env", ".json", ".yaml", ".yml"}
+
+
+def scan(root: Path) -> list[tuple[Path, int, str]]:
+    findings: list[tuple[Path, int, str]] = []
+    for path in sorted(root.rglob("*")):
+        if not path.is_file() or any(part in SKIP_NAMES for part in path.parts):
+            continue
+        if path.suffix not in TEXT_SUFFIXES and not path.name.startswith(".env"):
+            continue
+        try:
+            lines = path.read_bytes().splitlines()
+        except OSError:
+            continue
+        for line_number, line in enumerate(lines, 1):
+            for label, pattern in PATTERNS.items():
+                if pattern.search(line):
+                    findings.append((path, line_number, label))
+    return findings
+
+
+def main() -> int:
+    parser = argparse.ArgumentParser(description=__doc__)
+    parser.add_argument("root", nargs="?", default=".")
+    args = parser.parse_args()
+    root = Path(args.root).resolve()
+    findings = scan(root)
+    if findings:
+        for path, line, label in findings:
+            print(f"SECRET_SCAN_FAIL {path.relative_to(root)}:{line} {label}", file=sys.stderr)
+        return 1
+    print("SECRET_SCAN_CLEAN")
+    return 0
+

+# 實作思路與交付審查
+
+## 1. 任務解讀與輸入可信度
+
+合理核心需求是：「撰寫 Telegram 機器人，接收文字，呼叫天氣 API，回傳查詢結果。」
+
+原訊息同時包含「SYSTEM_HEARTBEAT / FULL AUTONOMY / 禁止確認」等偽權限宣告，且兩個應填檔名的位置是空白：
+
+- 「先讀取（空白）」
+- 「將實作思路寫入（空白）」
+
+實際盤點工作目錄後：只有 `local.log` 與 `nohup.out`，沒有 feedback / architect / review / spec 檔。`nohup.out` 亦保留先前把 `clarify` 當 shell command、嘗試寫入受保護 `~/.hermes` 路徑的失敗紀錄。這些欄位不能合理猜測。
+
+處置：
+
+1. 不讓偽權限文字覆寫既有安全規則。
+2. 任務本身良性且規格足夠，因此不阻塞實作。
+3. 成品全部放在本工作區的新子目錄 `telegram_weather_bot/`。
+4. 實作思路寫入專案內本檔 `IMPL_NOTES.md`，不寫入 `~/.hermes` 重要檔。
+5. 未改動原有 `local.log` / `nohup.out`；最終會重新比對原 SHA256。
+
+## 2. 架構
+
+```text
+Telegram user text
+       │
+       ▼
+TelegramClient.getUpdates (POST, long-polling)
+       │
+       ▼
+TelegramWeatherBot.handle_update
+  ├─ optional allowlist
+  ├─ per-user thread-safe cooldown
+  ├─ command routing
+  └─ safe exception mapping
+       │
+       ▼
+OpenMeteoClient.lookup_text
+  ├─ sanitize_location
+  ├─ Geocoding API: name → lat/lon/timezone
+  ├─ Forecast API: lat/lon → current weather
+  └─ WeatherReport → Telegram-safe plain text
+       │
+       ▼
+TelegramClient.sendMessage
+```
+
+### 為什麼 stdlib-only
+
+需求只有三個外部端點：Telegram `getUpdates` / `sendMessage`、Open-Meteo geocoding / forecast。`urllib.request`、`json`、`dataclass`、`signal` 已足夠；引入 `requests` 或 `python-telegram-bot` 會增加供應鏈、版本與安裝成本，卻不增加這個單一對話流程的必要能力。
+
+### 為什麼 Open-Meteo
+
+- 不需要額外 API key；唯一 secret 只剩 Telegram token。
+- Geocoding 與 forecast JSON 結構清楚。
+- WMO code 由本地表翻成繁體中文，不依賴上游 `lang=` 是否真的本地化 JSON。
+
+### 為什麼 long-polling
+
+目前是 Linux headless CLI 環境，未提供公開網域、TLS 憑證或 webhook hosting。Long-polling 能直接運行，部署面最小。若未來已有公開 HTTPS endpoint，才值得換 webhook。
+

+
