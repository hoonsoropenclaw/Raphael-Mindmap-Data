+class RingBuffer:
+    """固定容量 FIFO,滿了丟最舊的。"""
+
+    def __init__(self, capacity: int = 10000):
+        if capacity <= 0:
+            raise ValueError("capacity must be > 0")
+        self._cap = capacity
+        self._buf: deque[LogEntry] = deque(maxlen=capacity)
+        self._lock = threading.Lock()
+        self._dropped = 0  # 累計因容量滿而被淘汰的數量
+
+    def append(self, entry: LogEntry) -> None:
+        # deque(maxlen=N) 在 append 時若已滿會自動丟左邊,
+        # 但我們想精確計數「被淘汰」幾次,所以走 with-lock 手動檢查。
+        with self._lock:
+            if len(self._buf) >= self._cap:
+                self._dropped += 1
+            self._buf.append(entry)
+
+    def extend(self, entries: Iterable[LogEntry]) -> None:
+        with self._lock:
+            for e in entries:
+                if len(self._buf) >= self._cap:

+| `POST /ingest` | 寫入單筆 log（自動給 trace_id / span_id） |

+| `GET /cluster/gossip` | 對外暴露 gossip heartbeat（buffer_size / capacity / dropped / unique_traces） |
+| `POST /cluster/sync` | 強制觸發一次 gossip sync（拉 peer logs → dedupe → merge） |

+| `GET /stats` | 各 node buffer 狀態 |
