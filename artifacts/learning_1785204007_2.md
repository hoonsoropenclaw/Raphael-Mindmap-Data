+"""Rate limiting — slowapi-based。
+
+為什麼用 slowapi:
+- 是 FastAPI 官方推薦的 rate limit 套件,社群維護活躍
+- 內建 IP-based limiting（用 request.client.host）與自訂 key_func
+- 預設 in-memory storage（單機夠用,production 可換 Redis）
+
+設計:
+- 預設限制:每個 IP 每分鐘 10 次 request（測試時 fixture 會放寬到 100/min 避免測試觸發限流）
+- 路由層級限制:寫入操作（POST/PUT/DELETE）每分鐘 5 次
+"""
+from fastapi import Request
+from slowapi import Limiter
+from slowapi.util import get_remote_address
+
+
+def _key_func(request: Request) -> str:
+    """用 client IP 當 key。
+
+    為什麼不考慮 X-Forwarded-For:N100 在 proxy 後方,但本機測試不需要處理
+    proxy header。如果未來部署到 reverse proxy 後面,改用 slowapi 的
+    get_remote_address_with_proxy 或自訂解析 X-Forwarded-For。
+    """
+    return get_remote_address(request)
+
+
+limiter = Limiter(
+    key_func=_key_func,
+    default_limits=["10/minute"],  # 全域預設:每 IP 每分鐘 10 次
+    headers_enabled=True,  # 自動加 X-RateLimit-* response headers
