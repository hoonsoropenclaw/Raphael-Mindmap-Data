+def parse_due_and_priority(raw: str, now: Optional[datetime] = None) -> ParsedAdd:
+    now = now or datetime.now()
+    text = raw.strip()
+    priority = "medium"
+    match_priority = re.search(r"#(高|中|低)(?:優先)?", text)
+    if match_priority:
+        priority = PRIORITY_MAP[match_priority.group(1)]
+        text = (text[:match_priority.start()] + text[match_priority.end():]).strip()
+
+    due_at: Optional[datetime] = None
+    match = DATE_RE.search(text)
+    if match:
+        time_text = match.group("time") or "23:59"
+        try:
+            due_at = datetime.strptime(

+            )

+            raise ValueError("期限格式錯誤，請用 YYYY-MM-DD [HH:MM]") from exc
+        text = (text[:match.start()] + text[match.end():]).strip()

+        match = RELATIVE_RE.search(text)
+        if match:
+            delta = {"今天": 0, "明天": 1, "後天": 2}[match.group("day")]
+            target = now + timedelta(days=delta)
+            time_text = match.group("time") or "23:59"

+class TelegramClient:
+    def __init__(self, token: str, api_base: str = "https://api.telegram.org",
+                 request_timeout: int = 40) -> None:
+        if not token or ":" not in token:
+            raise ValueError("Telegram token 格式錯誤")
+        parsed = urllib.parse.urlparse(api_base)
+        loopback = parsed.hostname in {"127.0.0.1", "localhost", "::1"}
+        if parsed.scheme != "https" and not loopback:
+            raise ValueError("Telegram API 必須使用 HTTPS；HTTP 僅允許 loopback 測試")
+        self.token = token
+        self.api_base = api_base.rstrip("/")
+        self.request_timeout = request_timeout
+
+    def call(self, method: str, params: Optional[dict[str, object]] = None) -> object:
+        url = f"{self.api_base}/bot{self.token}/{method}"
+        data = urllib.parse.urlencode({k: v for k, v in (params or {}).items()
+                                      if v is not None}).encode("utf-8")
+        request = urllib.request.Request(
+            url, data=data, method="POST",
+            headers={"Content-Type": "application/x-www-form-urlencoded",
+                     "User-Agent": "telegram-task-bot/1.0"},
+        )
+        try:
+            with urllib.request.urlopen(request, timeout=self.request_timeout) as response:
+                payload = json.loads(response.read().decode("utf-8"))
+        except urllib.error.HTTPError as exc:
+            raise TelegramTransportError(f"Telegram HTTP {exc.code}") from None
+        except (urllib.error.URLError, TimeoutError, OSError, json.JSONDecodeError) as exc:
+            raise TelegramTransportError(f"Telegram transport {type(exc).__name__}") from None
+        if not payload.get("ok"):
+            raise TelegramTransportError(
