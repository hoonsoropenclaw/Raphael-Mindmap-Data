+// --- CLI flag parser ---------------------------------------------------------
+// Recognised flags (any suite can use them; missing = default):
+//   --browsers=chromium,firefox          subset of {chromium,firefox,webkit}
+//   --devices=desktop-1280,iphone-13     subset of {desktop-1280,ipad,iphone-13}
+//   --pages=home.html,form.html          subset of fixture files in pages/
+//   --threshold=0.18                     pixelmatch YIQ distance tolerance
+//   --max-diff-ratio=0.02                maximum allowed diff ratio (0..1)
+//   --update                             write fresh baselines instead of comparing
+//   --strict                             missing-baseline counts as a failure
+function parseFlags(argv) {
+  const flags = {
+    browsers: null, devices: null, pages: null,
+    threshold: 0.18, maxDiffRatio: 0.02,
+    update: false, strict: false,
+  };
+  for (const a of argv) {
+    if (a === '--update') flags.update = true;
+    else if (a === '--strict') flags.strict = true;
+    else if (a.startsWith('--browsers=')) flags.browsers = a.split('=')[1].split(',').map(s => s.trim()).filter(Boolean);
+    else if (a.startsWith('--devices='))  flags.devices  = a.split('=')[1].split(',').map(s => s.trim()).filter(Boolean);
+    else if (a.startsWith('--pages='))    flags.pages    = a.split('=')[1].split(',').map(s => s.trim()).filter(Boolean);
+    else if (a.startsWith('--threshold='))         flags.threshold     = parseFloat(a.split('=')[1]);
+    else if (a.startsWith('--max-diff-ratio='))    flags.maxDiffRatio  = parseFloat(a.split('=')[1]);
+  }
+  return flags;

+
+A self-contained Playwright + pixelmatch + axe-core lab that exercises a small
+set of HTML fixtures across **Chromium / Firefox / WebKit** desktop engines and
+**desktop / iPad / iPhone 13** device emulation, and produces a single
+self-contained dashboard with diff-overlays and a run-history trend chart.

+## Why these design choices
+
+* **Responsive suite runs chromium only** — firefox/webkit mobile rendering diverges too much from real iOS Safari/Chrome on Android to give a clean RWD signal. RWD is CSS media-query behaviour; engine variation there is thin.
+* **Network mock uses window.fetch monkey-patch** — Playwright `page.route()` cannot intercept `fetch()` over `file://` URLs (Chromium blocks the request before the route handler fires). `addInitScript` patches `window.fetch` before page scripts run.
+* **Accessibility fails on critical only** — most demo pages trip a few contrast violations. Failing the suite on every warning would obscure the high-signal critical issues. The dashboard lists all of them by severity.
+* **Trend data in `reports/trends.jsonl`** — append-only JSONL means the dashboard can plot history without any database. The first 30 entries are shown in the SVG trend chart.
