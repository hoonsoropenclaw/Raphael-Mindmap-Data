+"""DefParser — defensive BeautifulSoup parser.
+
+Borrowed from data_synchronization_and_analysis micro-skill:
+    DefParser is the third layer, normalizes raw HTML into structured data.
+
+Key defensive patterns:
+    1. Multiple selector strategies with fallback
+    2. Tolerate missing/empty fields (return "" or None, never crash)
+    3. Auto-trim whitespace and normalize unicode
+    4. Validate types before returning records
+    5. Skip records missing required fields
+"""
+
+from __future__ import annotations
+
+import html
+import re
+import unicodedata
+from dataclasses import dataclass, field, asdict
+from typing import Any, Callable, Dict, List, Optional, Sequence
+from bs4 import BeautifulSoup, Tag
+
+
+# --------------------------------------------------------------------- model
+@dataclass
+class ParsedRecord:
+    """A single scraped record. Fields vary per schema."""
+    title: str = ""
+    url: str = ""
+    summary: str = ""
+    tags: List[str] = field(default_factory=list)
+    extras: Dict[str, Any] = field(default_factory=dict)
+
+    def to_row(self, columns: Sequence[str]) -> List[str]:
+        """Project to a row in the order of `columns`.
+
+        Unknown columns fall through to self.extras; missing fields become "".
+        """
+        base = asdict(self)
+        out: List[str] = []
+        for col in columns:
+            if col in base:
+                v = base[col]
+                if isinstance(v, list):
+                    out.append(", ".join(str(x) for x in v))
+                elif v is None:
+                    out.append("")
+                else:
+                    out.append(str(v))
+            else:
+                out.append(str(self.extras.get(col, "")))
+        return out
+

+## Architecture
+Four-layer pipeline, borrowed from data_synchronization_and_analysis:

+                          (mock store)  (BeautifulSoup)  (gspread / CSV)

+## 1. PolitenessGate (Layer 1)

+class PolitenessGate:
+    def __init__(self, user_agent="bot/1.0", rate_per_sec=1.0):
+        self._robots_cache = {}
+        self._last_request_at = {}
+        self._lock = threading.Lock()
+        ...
+
+    def check(self, url: str) -> bool:
+        rp = self._fetch_robots(host)
+        if rp and not rp.can_fetch(self.user_agent, url):
+            raise PermissionError(f"robots.txt disallows {url}")
+        return True
+
+    def wait(self, url: str):
+        host = urllib.parse.urlparse(url).netloc
+        with self._lock:
+            last = self._last_request_at.get(host, 0.0)
+            wait_s = (1.0 / self.rate_per_sec) - (time.monotonic() - last)
+            if wait_s > 0:
+                time.sleep(wait_s)
+            self._last_request_at[host] = time.monotonic()
+```
