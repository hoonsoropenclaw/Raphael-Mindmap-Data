+"""
+Notification delivery worker.
+Runs in a background thread; drains the outbox into STORE.notified.
+This is the seam where most chaos experiments focus: it is asynchronous,
+stateful, and depends on the application correctly handling message loss,
+worker crashes, and backpressure.
+"""
+from __future__ import annotations
+import time
+import threading
+import random
+
+from common import STORE, maybe_fail, maybe_delay
+
+
+def _loop():
+    while True:
+        time.sleep(0.1)
+        try:
+            # Pop one event off the outbox (FIFO).
+            with STORE.with_lock():
+                if not STORE.outbox:
+                    continue
+                event = STORE.outbox.pop(0)
+            # Failures here are visible downstream.
+            maybe_delay("notification-worker")
+            maybe_fail("notification-worker")
+            # Chaos knob: drop messages at random.
+            if STORE.failure_rate.get("notification-drop", 0) > 0 and \
+                    random.random() < STORE.failure_rate["notification-drop"]:
+                with STORE.with_lock():
+                    STORE.counters["outbox_dropped"] += 1
+                continue
+            with STORE.with_lock():
+                STORE.notified.append(event)
+                STORE.counters["outbox_delivered"] += 1
+        except Exception:
+            # In a real system we'd retry with backoff and a dead-letter queue.
+            # Here we just keep the loop alive so the chaos report can see it.
+            time.sleep(0.2)
+
+
+def start_worker():
+    t = threading.Thread(target=_loop, daemon=True, name="notification-worker")
+    t.start()
+    return t

+# 混沌工程實踐 SOP — Cloud-Native Chaos Engineering

+> 一份給 SRE / 平台工程師的實戰手冊,以本倉庫的可執行程式碼為例,

+## 1. 為什麼需要混沌工程
+
+雲原生系統的失敗模式是**統計性**的:個體元件偶爾失敗,但只要系統設計正確,
+使用者體驗應該不受影響。傳統測試(單元、整合、E2E)在 happy path 給我們信心,
+但對**故障傳播路徑**的可見度不足。
+
+混沌工程的核心主張:
+
+> 在受控條件下,主動把故障注入生產系統,觀察系統反應,把「未知風險」
+> 變成「已知、量化、有 playbook 對應的風險」。
+
+Netflix 在 2010 年率先建立 Chaos Monkey,後續演進出完整的「Chaos Engineering
+成熟度模型」(Principles of Chaos, 2017):
+
+1. **建立穩態假設**(Steady-State Hypothesis):用指標量化「系統正常」。
+2. **多樣化真實事件**:延遲、失敗、訊息丟失、資源耗盡。
+3. **在生產環境跑**:至少在 staging 環境跑。
+4. **自動化實驗**:跑成 cron,不靠手動。
+5. **最小爆炸半徑**:一次只壞一個元件。
+6. **持續演進**:擴大到更多 failure mode。
+
+本倉庫的 `chaos_engine.py` 把這 6 條原則對應到具體程式碼。

+# 雲原生系統概覽
+
+> 雲原生(Cloud-Native)是一種「為雲端而設計」的方法論,核心是把系統拆成
+> 小而獨立的服務,每個服務都能獨立部署、擴展、失敗恢復。混沌工程是驗證
+> 雲原生系統**真的**具備這些特性的唯一可靠手段。
+
+## 1. 雲原生的四大支柱
+
+| 支柱 | 技術實作 | 失敗時的特徵 |
+|------|---------|------------|
+| **容器化** | Docker / OCI / containerd | OOM、CPU throttling、image pull failure |
+| **編排** | Kubernetes / Nomad | pod eviction、node drain、rolling update 卡住 |
+| **微服務** | gRPC / REST / message queue | 跨服務 timeout、訊息丟失、partition |
+| **可觀測性** | Prometheus / OpenTelemetry / Loki | 指標 / log / trace 缺失或延遲 |
+
+雲原生 ≠ 上雲。**把 monolithic 部署到 EC2 不算雲原生**——除非它具備上述四個特性。
+
+## 2. 雲原生失敗模式
+
+雲原生系統有四種典型的失敗模式,每一種都應該在 chaos 實驗中模擬:
+
+### 2.1 Crash Failure
+- **症狀**:進程/Pod 直接消失,連 graceful shutdown 都沒有。
+- **真實場景**:OOM kill、kubelet 砍掉過載節點、container runtime bug。
+- **混沌對應**:`FailureAttack(c, rate=1.0)` 立即殺掉所有 call。
+
+### 2.2 Slow Failure (Brown-out)
+- **症狀**:服務還活著,但回應時間從 ms 級拉到秒級。
+- **真實場景**:資料庫慢查、CPU 飽和、GC stop-the-world。
+- **混沌對應**:`LatencyAttack(c, ms=2000)` 注入 2 秒延遲。
+
+### 2.3 Partial Failure
+- **症狀**:部分 request 失敗,部分成功,且難以預測哪個會失敗。
+- **真實場景**:負載平衡器健康檢查滯後、circuit breaker 沒覆蓋所有路徑。
+- **混沌對應**:`FailureAttack(c, rate=0.3)` 30% 機率失敗。
+
+### 2.4 Byzantine Failure
+- **症狀**:服務回應了,但回應內容是錯的(資料不一致、時序錯亂)。
+- **真實場景**:Split-brain、cache stampede、訊息亂序。
+- **混沌對應**:`MessageDropAttack` 模擬訊息丟失,`PartitionAttack` 模擬 split-brain。
+
+## 3. 雲原生 + 混沌工程的關係

+```

+Level 5: Continuous Chaos,      "Chaos 是 deploy pipeline 的一環"
