+def make_engine(database_url: str) -> Engine:
+    """Create an engine.
+
+    SQLite needs `check_same_thread=False` for use across the
+    threadpool that uvicorn/pytest-asyncio drives.
+    """
+    connect_args = {}
+    if database_url.startswith("sqlite"):
+        connect_args["check_same_thread"] = False
+    return create_engine(database_url, future=True, connect_args=connect_args)
+

+    """Configure the global session factory. Call from app startup / test fixture."""

+"""In-process sliding-window rate limiter.
+
+Why sliding-window rather than fixed-window?
+- Fixed-window allows up to 2N requests across the boundary (N at end of window N,
+  N at start of window N+1). Sliding-window is closer to a true N req / window.
+- We use `time.monotonic()` so the window is immune to wall-clock jumps.
+
+Limitations (documented so we don't over-claim):
+- State lives in-process: per worker. For multi-worker / multi-pod deployments
+  you need Redis or similar. We pick the simple option and note the upgrade path.
+- Client identity comes from the `X-Client-ID` header; falls back to peer IP.
+"""
+from __future__ import annotations
+
+import threading
+import time
+from collections import defaultdict, deque
+from typing import Deque, Dict
+
+from fastapi import HTTPException, Request, status
+
+
+class SlidingWindowRateLimiter:
+    """Tracks (client_id, request timestamps in a deque), prunes old entries per call."""
+
+    def __init__(self, max_requests: int, window_seconds: float) -> None:
+        if max_requests <= 0:
+            raise ValueError("max_requests must be > 0")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be > 0")
+        self.max_requests = max_requests
+        self.window = float(window_seconds)
+        self._buckets: Dict[str, Deque[float]] = defaultdict(deque)
+        self._lock = threading.Lock()
+
+    def _client_key(self, request: Request) -> str:
+        client_id = request.headers.get("x-client-id")
+        if client_id:
+            return f"cid:{client_id}"
+        if request.client is not None:
+            return f"ip:{request.client.host}"
+        return "ip:unknown"
+
+    def check(self, request: Request) -> None:
+        """Raise 429 if this client has exceeded the limit in the current window."""
+        key = self._client_key(request)
+        now = time.monotonic()
+        cutoff = now - self.window
+        with self._lock:
+            bucket = self._buckets[key]
+            # prune entries older than the window
+            while bucket and bucket[0] <= cutoff:
+                bucket.popleft()
+            if len(bucket) >= self.max_requests:
+                retry_after = max(0.0, bucket[0] + self.window - now)
+                raise HTTPException(
+                    status_code=status.HTTP_429_TOO_MANY_REQUESTS,
+                    detail={
+                        "error": "rate_limit_exceeded",
+                        "max_requests": self.max_requests,
+                        "window_seconds": self.window,
+                        "retry_after_seconds": round(retry_after, 3),
+                    },
+                    headers={"Retry-After": str(int(retry_after) + 1)},
+                )
+            bucket.append(now)
+
+    def reset(self) -> None:
+        """Test helper — clear all state."""
+        with self._lock:
+            self._buckets.clear()

+def _int_env(name: str, default: int) -> int:
+    raw = os.environ.get(name)

+        return default
+    try:
+        return int(raw)
+    except ValueError:
+        return default
+
+

+    raw = os.environ.get(name)

+## What was built
+
+A FastAPI service exposing CRUD for `Todo` items, persisted to SQLite,
+with an in-process sliding-window rate limiter applied to every

+| ORM | SQLAlchemy 2.0 (typed Mapped[]) | Modern typed mapping; swappable to Postgres by changing `TODO_DB_URL`. |
+| Validation | Pydantic v2 | Same runtime as FastAPI, single source of truth for request shapes. |
+| Tests | pytest + httpx TestClient | No need to spin up a real server. |

+## Design decisions worth flagging
+
+### 1. Rate limiter — sliding window, in-process
+
+`SlidingWindowRateLimiter` keeps a `deque[float]` of monotonic-clock
+timestamps per client. On each call it:
+
+1. Prunes entries older than `now - window`.
+2. Rejects if `len(bucket) >= max_requests` (HTTP 429 with `Retry-After`).
+3. Otherwise appends `now` and allows.
+
+**Tradeoffs**:
+- ✅ Truly N requests / window — no boundary doubling that fixed-window has.
+- ✅ Uses `time.monotonic()` so wall-clock jumps don't break the limiter.
+- ✅ Lock-protected so it's safe under uvicorn workers.
+- ❌ State lives in-process → each uvicorn worker has its own bucket.
+  For multi-worker / multi-pod: replace with Redis (`ZADD` + `ZREMRANGEBYSCORE`).
+  This is noted in the module docstring so future readers see it.
+
+Client identity is `X-Client-ID` header, falling back to peer IP. Easy
+to upgrade to a JWT subject later without rewriting the limiter.
+
+### 2. SQLite — file or `:memory:` via env var
+
+`TODO_DB_URL` defaults to `sqlite:///./todos.db`. Tests use
+`sqlite:///:memory:`. Two SQLite-specific concerns I had to fix during
+development:

+- **Cross-thread access**: `connect_args={"check_same_thread": False}` so
+  the pool's threads (uvicorn workers / TestClient) can share the engine.

+  test run.
