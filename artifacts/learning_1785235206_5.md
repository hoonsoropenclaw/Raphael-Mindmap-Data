+"""Token-bucket rate limiter.
+
+Design notes — derived from Architect feedback:
+
+#1  No ``defaultdict`` lazy initialization: buckets are created EXPLICITLY with
+    the correct rate/capacity for their endpoint class (read vs write). The
+    first request for an IP MUST use the real bucket parameters, never a
+    generic default.
+
+#5  X-Forwarded-For parsing: when configured to trust proxies, take the
+    FIRST IP in the chain (``X-Forwarded-For: client, proxy1, proxy2``).
+    Spoofing-resistant for upstream-proxy deployments; spoofable if you set
+    ``trust_forwarded_for`` without a reverse proxy in front — we expose the
+    flag so deployments are explicit about the trust assumption.

+#7  TTL/sweep: each bucket stores ``last_seen``; a periodic sweep removes
+    idle buckets so the dict doesn't grow unbounded. A background asyncio
+    task is started in lifespan and cancelled on shutdown.

+@dataclass
+class TokenBucket:
+    """Classic token bucket with refill-rate tokens/sec and burst capacity.
+
+    Uses a single asyncio.Lock per limiter instance (held at the limiter level)
+    to keep the critical section short.
+    """
+
+    rate: float
+    capacity: float
+    tokens: float = 0.0
+    last_refill: float = field(default_factory=time.monotonic)
+    last_seen: float = field(default_factory=time.monotonic)
+
+    def __post_init__(self) -> None:
+        # Pre-fill to capacity so a brand-new bucket can absorb its first burst
+        if self.tokens == 0.0:
+            self.tokens = self.capacity
+
+    def try_consume(self, cost: float = 1.0, now: float | None = None) -> bool:
+        """Attempt to consume ``cost`` tokens. Returns True if allowed."""
+        now = now if now is not None else time.monotonic()
+        elapsed = now - self.last_refill
+        if elapsed > 0:
+            self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
+            self.last_refill = now
+        self.last_seen = now
+        if self.tokens >= cost:
+            self.tokens -= cost
+            return True
+        return False
+
+    def retry_after(self, cost: float = 1.0, now: float | None = None) -> float:
+        """Seconds until the next ``cost`` tokens are available."""
+        now = now if now is not None else time.monotonic()
+        if self.try_consume(cost=0.0, now=now):
+            return 0.0
+        if self.rate <= 0:
+            return math.inf
+        missing = cost - self.tokens
+        return missing / self.rate
