+def _get_json(url: str, timeout: float = 15.0) -> dict:
+    request = urllib.request.Request(url, headers={"User-Agent": "telegram-weather-bot/1.0"})
+    try:
+        with urllib.request.urlopen(request, timeout=timeout) as response:
+            if response.status != 200:
+                raise WeatherError(f"API HTTP {response.status}")
+            return json.loads(response.read().decode("utf-8"))
+    except (OSError, json.JSONDecodeError) as exc:
+        raise WeatherError(f"天氣服務暫時無法使用：{exc}") from exc
+
+
+def lookup_weather(place: str) -> Weather:
+    place = " ".join(place.strip().split())
+    if not place or len(place) > 100:
+        raise WeatherError("請輸入有效的地點名稱（最多 100 個字元）。")
+    query = urllib.parse.urlencode({"name": place, "count": 1, "language": "zh", "format": "json"})
+    geo = _get_json(f"https://geocoding-api.open-meteo.com/v1/search?{query}")
+    results = geo.get("results") or []
+    if not results:
+        raise WeatherError(f"找不到地點：{place}")
+    location = results[0]
+    params = urllib.parse.urlencode({

+        "current": "temperature_2m,apparent_temperature,weather_code,wind_speed_10m",
+        "timezone": "auto",
+    })
+    data = _get_json(f"https://api.open-meteo.com/v1/forecast?{params}")
+    current = data.get("current") or {}
+    try:
+        return Weather(

+            temperature=float(current["temperature_2m"]),
+            apparent_temperature=float(current["apparent_temperature"]),

+    return data["result"]

+# Telegram 天氣機器人
+
+使用 Python 標準函式庫、Telegram Bot API 長輪詢，以及免費 Open-Meteo 地理編碼/天氣 API。
+
+## 啟動
+
+```bash
+chmod +x run.sh
+export TELEGRAM_BOT_TOKEN='從 BotFather 取得的 token'
+# 可選：限制可使用的 Telegram user ID（逗號分隔）
+export TELEGRAM_ALLOWED_USERS='123456789,987654321'
+./run.sh
+```
+
+使用者傳送 `/start`、`/help` 或地點文字（例如 `台北`、`Tokyo`），機器人會查詢目前氣溫、體感溫度、天氣狀況與風速。
+
+## 檔案
+
+- `telegram_weather_bot.py`: Telegram 長輪詢、訊號處理、權限 allowlist、冷卻限制與訊息路由。
+- `weather_api.py`: Open-Meteo 地點搜尋、目前天氣查詢、中文狀況格式化；可獨立匯入測試。
+- `run.sh`: 安全啟動器。
+
+## 架構與安全決策
+
+1. 使用 stdlib `urllib`，避免不必要的第三方依賴。
+2. 長輪詢適合無公開 HTTPS endpoint 的 headless 主機。
+3. `offset = update_id + 1` 避免 Telegram 重複投遞。
+4. `TELEGRAM_ALLOWED_USERS` 是可選的第一層存取控制。
+5. 每位使用者有請求冷卻，避免 API 洪水。
+6. token 只從環境變數讀取，不寫入檔案。
+7. API 回傳錯誤對使用者顯示安全訊息，細節只記錄在標準輸出。
+8. 直接使用 emoji literal，避免 surrogate pair 的 UnicodeEncodeError。
+9. 依照 Senior Architect 既有建議，HTTP 請求加入明確 User-Agent，並以模組分離保持可測試性。
+
+## 驗證

+python3 -m py_compile telegram_weather_bot.py weather_api.py
+python3 - <<'PY'

+PY

+
