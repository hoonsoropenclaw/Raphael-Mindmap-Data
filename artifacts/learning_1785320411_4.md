+var _peak_in_use: int = 0
+var _instantiations: int = 0  # count of actual new() calls (lower = better)

+func _ready() -> void:
+	if scene == null:
+		push_error("ObjectPool: scene not assigned")
+		return
+	if parent_for_instances == null:
+		parent_for_instances = get_parent()
+	_grow(initial_size)
+
+func _grow(n: int) -> void:
+	for i in n:
+		var inst := scene.instantiate()
+		parent_for_instances.add_child(inst)
+		inst.set_process(false)
+		inst.set_physics_process(false)
+		inst.visible = false
+		# Park far off-screen so it doesn't interact
+		if inst is Node2D:
+			(inst as Node2D).position = Vector2(-10000, -10000)
+		_free.append(inst)
+		_instantiations += 1
+
+func acquire() -> Node:
+	if _free.is_empty():
+		_grow(grow_step)
+	var n: Node = _free.pop_back()
+	_alive.append(n)

+	n.set_physics_process(true)

+## Performance monitor — samples physics tick time, frame time, jitter, and
+## physics step count. Designed to be usable from a headless script (no GUI).
+
+var _start_msec: int = 0
+var _frame_samples: Array[float] = []
+var _physics_samples: Array[float] = []
+var _spawn_count: int = 0
+var _query_count: int = 0
+var _signal_count: int = 0
+var _frame_jitter_threshold_ms: float = 4.0
+
+func begin() -> void:
+	_start_msec = Time.get_ticks_msec()
+	_frame_samples.clear()
+	_physics_samples.clear()
+	_spawn_count = 0
+	_query_count = 0
+	_signal_count = 0
+
+func record_frame(delta_ms: float) -> void:
+	_frame_samples.append(delta_ms)
+
+func record_physics(delta_ms: float) -> void:
+	_physics_samples.append(delta_ms)
+
+func count_spawn() -> void:
+	_spawn_count += 1
+
+func count_query() -> void:
+	_query_count += 1
+
+func count_signal() -> void:
+	_signal_count += 1
+
+func _stat(arr: Array, fn: String) -> float:
+	if arr.is_empty():
+		return 0.0
+	match fn:
+		"mean":
+			var s := 0.0
+			for v in arr:
+				s += v
+			return s / arr.size()
+		"max":
+			var m := -INF
+			for v in arr:
+				if v > m:
+					m = v
+			return m
+		"stddev":
+			var mean := _stat(arr, "mean")
+			var s := 0.0
+			for v in arr:
+				s += pow(v - mean, 2)
+			return sqrt(s / arr.size())
+	return 0.0
+
+func summary(label: String) -> Dictionary:
+	var wall_ms := Time.get_ticks_msec() - _start_msec
+	var mean_frame := _stat(_frame_samples, "mean")
+	var jitter := _stat(_frame_samples, "stddev")
+	var max_frame := _stat(_frame_samples, "max")
+	var mean_phys := _stat(_physics_samples, "mean")
+	return {

+optimization techniques** by running an identical physics workload under a

+the deltas in frame time, jitter, and query count.
+
+## Run
+
+```bash
+godot --headless res://scenes/main.tscn
+# results printed to stdout + persisted to res://results/bench_latest.txt
+```
+
+The runner runs 600 physics frames (~10s @ 60Hz) for each phase, then
+prints a side-by-side comparison and writes the report to
+`results/bench_latest.txt`.
+
+## What gets benchmarked
+
+| # | Strategy | BEFORE (anti-pattern) | AFTER (optimized) |
+|---|----------|----------------------|-------------------|
+| 1 | **Object Pool** | `instantiate() + queue_free()` per bullet | `ObjectPool.acquire/release()` reuses 64 pre-built bodies |
+| 2 | **Tight Collision Mask** | `mask = 0xFFFFFFFF` (all 32 layers) | `mask = (1<<0) | (1<<5)` (2 layers) |
+| 3 | **Disable Contact Monitor** | `contact_monitor = true` (4 contacts) | `contact_monitor = false` (0 contacts) |
+| 4 | **Single-Phase `_physics_process`** | physics work unconditional | off when phase inactive |
+| 5 | **Signal-driven area query** | `intersect_shape(q, 512)` × 2/frame | `Area2D.body_entered/exited` signals |
+| 6 | **Compact 2D physics layer layout** | unused 26 layers in mask | named layers, only 5 in use |
