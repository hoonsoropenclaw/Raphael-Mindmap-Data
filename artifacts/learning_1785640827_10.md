+"""
+Async Event Bus — pub/sub for inter-module communication.
+Reuses pattern from skills/event_bus_async_pub_sub.
+"""
+import asyncio
+from collections import deque
+from dataclasses import dataclass, field
+from typing import Any, AsyncGenerator, Deque, List
+
+
+@dataclass
+class BusEvent:
+    topic: str
+    payload: Any
+    ts: float = field(default_factory=lambda: asyncio.get_event_loop().time())
+
+
+class EventBus:
+    """Asyncio fan-out event bus with bounded history snapshot."""
+
+    def __init__(self, history_size: int = 500):
+        self._subscribers: List[asyncio.Queue] = []
+        self._history: Deque[BusEvent] = deque(maxlen=history_size)
+        self._lock = asyncio.Lock()
+
+    async def subscribe(self, topics: List[str] = None) -> AsyncGenerator[BusEvent, None]:
+        """Yield events; optionally filtered by topic list (None = all)."""
+        queue: asyncio.Queue = asyncio.Queue()
+        self._subscribers.append(queue)
+        try:
+            while True:
+                event = await queue.get()
+                if topics is None or event.topic in topics:
+                    yield event
+        finally:
+            self._subscribers.remove(queue)
+
+    async def publish(self, topic: str, payload: Any) -> None:
+        """Non-blocking fan-out to all subscribers + history."""
+        event = BusEvent(topic=topic, payload=payload)
+        self._history.append(event)
+        # Snapshot subscribers under lock to avoid mutation during iteration
+        async with self._lock:
+            subs = list(self._subscribers)
+        for q in subs:
+            try:
+                q.put_nowait(event)
+            except asyncio.QueueFull:
+                # Drop on overflow; subscriber will be cleaned by caller.
+                pass
+
+    def history(self, topic: str = None, n: int = 50) -> List[BusEvent]:
+        items = [e for e in self._history if topic is None or e.topic == topic]
+        return items[-n:]
+
+    def stats(self) -> dict:
+        topics: dict = {}
+        for e in self._history:
+            topics[e.topic] = topics.get(e.topic, 0) + 1
+        return {"total_events": len(self._history), "topics": topics,
