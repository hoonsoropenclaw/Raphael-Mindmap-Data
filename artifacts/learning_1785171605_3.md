+def _load_config(path: Path) -> dict:
+    with path.open() as f:

+def run_once(cfg: dict, base_dir: Path) -> int:
+    """單次採集 + 評估 + 報告 + 告警。回傳有發送的告警數。"""
+    collector = Collector(cfg)
+    engine = AlertEngine(cfg)
+    engine.reset_streaks()
+
+    snap = collector.collect()
+    raw_alerts = engine.evaluate(snap)
+    emitted = engine.emit(raw_alerts, base_dir)
+    write_reports(base_dir, cfg, snap, emitted)
+
+    return len(emitted)
+
+
+def run_daemon(cfg: dict, base_dir: Path) -> None:
+    interval = int(cfg.get("collection_interval_seconds", 30))
+    print(f"[INFO] sysmon-agent daemon 啟動 (interval={interval}s, base={base_dir})", flush=True)
+
+    def _stop(*_):
+        global _RUNNING
+        print("\n[INFO] 收到停止訊號, 準備離場…", flush=True)
+        _RUNNING = False
+    signal.signal(signal.SIGINT, _stop)
+    signal.signal(signal.SIGTERM, _stop)
+
+    while _RUNNING:
+        start = time.time()
+        try:
+            n = run_once(cfg, base_dir)
+            print(f"[INFO] {time.strftime('%Y-%m-%d %H:%M:%S')} 採集完成, 告警 {n} 條",

    6. Journal 預設關閉：預設值會被常見子字淹沒，改成精準關鍵字，使用者自行開啟
