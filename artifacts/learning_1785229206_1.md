+// ThemeContext — 第二個獨立 state slice，示範「多 context 拆分」設計。
+// 主題與 todos 無相依，所以分兩個 Provider 比單一 mega-context 乾淨。
+// 同樣用 lazy initializer 從 localStorage 回灌，避免 hydration flash。
+
+import { createContext, useContext, useEffect, useState } from 'react';
+
+const STORAGE_KEY = 'react-todo:v1:theme';
+
+const THEMES = {
+  LIGHT: 'light',
+  DARK: 'dark'
+};
+
+function loadTheme() {
+  if (typeof window === 'undefined') return THEMES.LIGHT;
+  try {
+    const raw = window.localStorage.getItem(STORAGE_KEY);
+    if (raw === THEMES.DARK || raw === THEMES.LIGHT) return raw;
+    // 自動偵測系統偏好（如果使用者沒設定過）
+    if (window.matchMedia?.('(prefers-color-scheme: dark)').matches) {
+      return THEMES.DARK;
+    }
+    return THEMES.LIGHT;
+  } catch {
+    return THEMES.LIGHT;
+  }
+}
+
+const ThemeContext = createContext(null);
+
+export function ThemeProvider({ children }) {
+  const [theme, setTheme] = useState(loadTheme);
+
+  // 寫 localStorage + 同步到 <html data-theme> 讓 CSS 變數能切換
+  useEffect(() => {
+    if (typeof window === 'undefined') return;
+    document.documentElement.setAttribute('data-theme', theme);
+    try {
+      window.localStorage.setItem(STORAGE_KEY, theme);
+    } catch {
+      // 忽略 quota / private mode 錯誤
+    }
+  }, [theme]);
+
+  const toggle = () =>
+    setTheme((t) => (t === THEMES.DARK ? THEMES.LIGHT : THEMES.DARK));
+
+  const value = { theme, toggle, isDark: theme === THEMES.DARK };
+
+  return <ThemeContext.Provider value={value}>{children}</ThemeContext.Provider>;
+}
+
+export function useTheme() {
+  const ctx = useContext(ThemeContext);
+  if (ctx === null) {
+    throw new Error('useTheme must be used inside <ThemeProvider>');
+  }
+  return ctx;

+## 三、複雜狀態管理的體現在哪
+
+**訊息要求「複雜的狀態管理」**，不是「能動就好」。具體複雜度：
+
+### 1. 兩個互相關聯的 state slice
+
+`todos[]` 和 `filter` 雖然是兩個獨立的 slice，但 `selectVisibleTodos` 衍生計算把它們綁在一起。同樣 `selectStats` 從 todos 衍生出 active/done 計數。**衍生邏輯放在 selector 函數而非 component 內**，避免每次重複計算邏輯、也方便測試。
+
+### 2. 完整的 CRUD + 衍生操作 reducer
+
+7 種 action type（ADD / TOGGLE / EDIT / DELETE / CLEAR_COMPLETED / SET_FILTER / HYDRATE），每個都有：
+- 輸入驗證（空字串 / 純空白 / 非法 filter value 守住）
+- Immutability（永遠 return new state，不 mutate）
+- 雙重防呆（元件先守、reducer 再守一次）
+
+### 3. lazy initializer + useEffect 持久化
+
+`useReducer(reducer, undefined, loadFromStorage)`：
+- 第三參數 lazy initializer 只在 mount 跑一次（避免每次 render 都讀 localStorage）

+
+### 4. localStorage 壞資料防禦
+

+

+
