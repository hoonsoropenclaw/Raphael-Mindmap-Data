+// Returns a debounced copy of `value` — only updates after `delay` ms of stability.

+// Calls `handler` on every keydown matching `key`, regardless of case.
+// Pass an `enabled` flag for conditional listeners (e.g. only inside modal).
+export function useKey(key: string, handler: (e: KeyboardEvent) => void, enabled = true): void {
+  const handlerRef = useRef(handler);
+  useEffect(() => {
+    handlerRef.current = handler;
+  }, [handler]);
+
+  useEffect(() => {
+    if (!enabled) return;
+    const listener = (e: KeyboardEvent) => {
+      if (e.key.toLowerCase() === key.toLowerCase()) {
+        handlerRef.current(e);
+      }
+    };
+    window.addEventListener('keydown', listener);
+    return () => window.removeEventListener('keydown', listener);

+## 2. Architecture overview

+### 2.1 State shape (`src/types.ts`)
+
+```
+AppState
+├── tasks: Task[]                // domain data
+├── filter: FilterState          // query, priorities[], tags[], sortBy, sortDir
+├── ui: UIState                  // theme, sidebarOpen, modal, draggedTaskId, dragOverColumn

+history slice — this matches the **slice pattern** common in Redux Toolkit,
+but rolled by hand to keep the dependency surface minimal (zero runtime deps
+beyond React itself).

+### 2.2 Action surface (`src/types.ts` + `src/state/reducer.ts`)
+
+A discriminated union — every action has a `type` literal tag, so the reducer
+gets exhaustiveness checking for free (`const _: never = action`).

+* `bulk/loadTasks`, `bulk/seedDemo`

+
+Implemented as a **ring-buffer snapshot stack** (capacity = 20). Each mutating
+action (`task/*` except no-op `task/move` onto same column) calls
