import asyncio
from playwright.async_api import async_playwright

async def main():
    async with async_playwright() as p:
        browser = await p.chromium.launch()
        page = await browser.new_page(viewport={"width": 1440, "height": 900})
        errors = []
        page.on("pageerror", lambda e: errors.append(f"PAGEERROR: {e}"))
        await page.goto("http://127.0.0.1:5180/", wait_until="networkidle", timeout=15000)
        await page.wait_for_selector(".kanban-card", timeout=10000)

        # 1. 點擊「新增卡片」按鈕 (第一個 column)
        await page.locator(".kanban-column").first.locator(".kanban-column__add-card-btn").click()
        await page.wait_for_selector(".modal", timeout=5000)
        modal_visible = await page.locator(".modal").is_visible()
        title_focused = await page.locator(".form-field__input").first.evaluate("el => el === document.activeElement")
        print(f"STEP1_OPEN_MODAL:{modal_visible} autoFocused:{title_focused}")

        # 2. 填表單
        await page.locator(".form-field__input").first.fill("測試新卡片")
        await page.locator(".form-field__input--textarea").fill("這是測試描述")
        # 改優先級
        await page.locator(".form-field__input").nth(2).select_option("high")
        # 填 tags
        await page.locator(".form-field__input").nth(4).fill("test, automation")
        await page.screenshot(path="/tmp/kanban-shot-modal.png", full_page=False)
        # 3. 提交
        await page.locator(".modal__footer button[type=submit]").click()
        await page.wait_for_selector(".modal", state="detached", timeout=5000)
        new_card_count = await page.locator(".kanban-card").count()
        print(f"STEP2_AFTER_CREATE:{new_card_count}")

        # 4. 開啟統計面板
        await page.locator("text=📊 統計").click()
        await page.wait_for_selector(".stats-sidebar", timeout=3000)
        await page.screenshot(path="/tmp/kanban-shot-stats.png", full_page=False)
        stats_visible = await page.locator(".stats-sidebar").is_visible()
        print(f"STEP3_STATS_OPEN:{stats_visible}")

        # 5. 搜尋測試
        await page.locator(".kanban-search").fill("test")
        await asyncio.sleep(0.3)
        filtered_count = await page.locator(".kanban-card").count()
        print(f"STEP4_FILTERED:{filtered_count}")
        await page.locator(".kanban-search").fill("")

        # 6. 切換優先級過濾
        await page.locator(".kanban-priority-filter").select_option("urgent")
        await asyncio.sleep(0.3)
        urgent_count = await page.locator(".kanban-card").count()
        print(f"STEP5_URGENT_ONLY:{urgent_count}")
        await page.locator(".kanban-priority-filter").select_option("all")

        # 7. 編輯現有卡片 (點擊卡片本體)
        first_card = page.locator(".kanban-card").first
        first_card_title_before = await first_card.locator(".kanban-card__title").text_content()
        await first_card.click()
        await page.wait_for_selector(".modal", timeout=3000)
        edit_title = await page.locator(".modal__title").text_content()
        print(f"STEP6_EDIT_OPEN:{edit_title}")
        await page.locator(".modal__close-btn").click()

        # 8. 鍵盤操作:用 Delete 鍵刪第一張卡
        before_del = await page.locator(".kanban-card").count()
        # 自動處理 confirm 對話框
        page.on("dialog", lambda d: asyncio.ensure_future(d.accept()))
        await first_card.focus()
        await page.keyboard.press("Delete")
        await asyncio.sleep(0.3)
        after_del = await page.locator(".kanban-card").count()

        # 9. 截最終畫面

        print(f"ERRORS:{len(errors)}")
        for e in errors[:10]:
            print(f"  {e}")
        await browser.close()

asyncio.run(main())
