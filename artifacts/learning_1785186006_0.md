+使用 sqlite3 內建模組，搭配 FastAPI 的 Depends 注入，
+讓每次 request 拿到獨立 connection，request 結束自動 close。
+"""
+import sqlite3
+from contextlib import contextmanager
+from typing import Iterator
+
+
+# SQLite 連線 factory。check_same_thread=False 才能在 FastAPI 多 thread 下使用。
+def _make_connection(db_path: str) -> sqlite3.Connection:
+    conn = sqlite3.connect(db_path, check_same_thread=False)
+    conn.row_factory = sqlite3.Row  # 讓查詢結果可以用 dict / 屬性存取
+    return conn
+
+
+def init_db(db_path: str = "todos.db") -> None:
+    """建立 todos table（若不存在）。"""
+    with _make_connection(db_path) as conn:
+        conn.execute(
+            """
+            CREATE TABLE IF NOT EXISTS todos (
+                id INTEGER PRIMARY KEY AUTOINCREMENT,
+                title TEXT NOT NULL,
+                description TEXT DEFAULT '',
+                completed INTEGER NOT NULL DEFAULT 0,
+                created_at TEXT NOT NULL DEFAULT (datetime('now'))
+            )
+            """
+        )
+        conn.commit()
+
+
+@contextmanager
+def get_db_connection(db_path: str = "todos.db") -> Iterator[sqlite3.Connection]:
+    """Context manager：yield 一條 connection，離開時自動 close。"""
+    conn = _make_connection(db_path)
+    try:
+        yield conn
+    finally:
+        conn.close()
+
