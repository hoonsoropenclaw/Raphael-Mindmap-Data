+class TelegramError(RuntimeError):
+    """Raised when the Telegram Bot API returns an unsuccessful response."""
+
+
+class TelegramAPI:
+    def __init__(
+        self,
+        token: str,
+        transport: Callable[[str, dict[str, Any], float], dict[str, Any]] | None = None,
+        timeout: float = 40.0,
+    ) -> None:
+        token = token.strip()
+        if not token:
+            raise ValueError("Telegram bot token is required")
+        self.base_url = f"https://api.telegram.org/bot{token}"
+        self.transport = transport or self._request
+        self.timeout = timeout
+
+    def call(self, method: str, payload: dict[str, Any] | None = None) -> dict[str, Any]:
+        result = self.transport(method, payload or {}, self.timeout)
+        if not result.get("ok"):
+            description = result.get("description", "unknown Telegram API error")
+            raise TelegramError(f"{method}: {description}")
+        return result
+
+    def get_updates(self, offset: int, timeout: int = 30) -> list[dict[str, Any]]:
+        payload = {"offset": offset, "timeout": timeout, "allowed_updates": ["message"]}
+        return self.call("getUpdates", payload).get("result", [])
+
+    def send_message(self, chat_id: int, text: str) -> dict[str, Any]:
+        return self.call("sendMessage", {"chat_id": chat_id, "text": text})
+
+    def _request(self, method: str, payload: dict[str, Any], timeout: float) -> dict[str, Any]:
+        body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
+        request = Request(
+            f"{self.base_url}/{method}",
+            data=body,
+            headers={"Content-Type": "application/json"},
+            method="POST",
+        )
+        with urlopen(request, timeout=timeout) as response:
+            raw = response.read().decode("utf-8")

+- Fake 測試 helper **必須反映系統動態行為**（從 URL 解析 city），不能回傳固定 payload；不然會「fake 通過、production 失敗」。
+- CLI 入口若用 `argparse.error(WeatherError(...))`，會把例外訊息當 usage 印，造成混淆；用 `parser.error` 應只用於**真正**的 CLI 參數錯誤。
