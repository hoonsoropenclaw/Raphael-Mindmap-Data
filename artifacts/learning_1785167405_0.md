+"""
+weather_client.py — 天氣 API 查詢層
+
+設計原則:
+- 抽象出 WeatherProvider protocol,讓 wttr.in / OpenWeatherMap / mock 都能 plug-in
+- 輸入:城市名(中文或英文)
+- 輸出:結構化的 WeatherReport(溫度、體感、濕度、風速、簡短描述)
+- 失敗時 raise WeatherQueryError,不要回 None(避免 caller 忘了處理)
+"""
+
+from __future__ import annotations
+
+import json
+from dataclasses import dataclass
+from typing import Protocol
+from urllib.parse import quote
+
+import requests
+
+
+class WeatherQueryError(Exception):
+    """對外暴露的查詢失敗例外。bot 層會攔截並回使用者友善訊息。"""
+
+
+@dataclass(frozen=True)
+class WeatherReport:
+    """結構化的天氣查詢結果。"""
+
+    location: str
+    temperature_c: float
+    feels_like_c: float
+    humidity_pct: int
+    wind_speed_kmh: float
+    description: str  # 例:「多雲」、「Sunny」
+
+    def to_message(self) -> str:
+        """轉成 Telegram 友善的純文字訊息。"""
+        return (
+            f"🌤 *{self.location}* 現在天氣\n"
+            f"溫度:{self.temperature_c:.1f}°C(體感 {self.feels_like_c:.1f}°C)\n"
+            f"濕度:{self.humidity_pct}%\n"
+            f"風速:{self.wind_speed_kmh:.1f} km/h\n"
+            f"概況:{self.description}"
+        )
+
+
+class WeatherProvider(Protocol):
+    """所有天氣 provider 都要實作 query(location) -> WeatherReport。"""
+
+    def query(self, location: str) -> WeatherReport: ...
+
+
+class WttrInProvider:
+    """wttr.in — 免 API key、單 endpoint 的輕量服務。
+
+    Endpoint: https://wttr.in/<location>?format=j1
+    回傳 JSON。可直接 curl 驗證:
+        curl 'https://wttr.in/Taipei?format=j1' | jq .current_condition[0]
+    """
+
+    ENDPOINT = "https://wttr.in/{location}?format=j1"
+    TIMEOUT = 8.0
+
+    def query(self, location: str) -> WeatherReport:
+        location = (location or "").strip()
+        if not location:
+            raise WeatherQueryError("城市名不能是空的。")
+
+        url = self.ENDPOINT.format(location=quote(location, safe=""))
