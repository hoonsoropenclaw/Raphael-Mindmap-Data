+"""Multi-language routing table and prompt enrichment.
+
+Each entry is small, curated and safe-by-default: prompts steer Whisper towards
+the right vocabulary while staying inside OpenAI's recommended guidelines.
+"""
+from __future__ import annotations
+
+from collections.abc import Iterable, Mapping
+from typing import Any
+
+from .models import LanguageProfile, LanguageRouteError
+
+
+class LanguageRegistry:
+    """Immutable lookup of curated language profiles."""
+
+    def __init__(self, profiles: Iterable[LanguageProfile]) -> None:
+        deduped: dict[str, LanguageProfile] = {}
+        for profile in profiles:
+            code = profile.code.strip().lower()
+            if not code or len(code) > 8:
+                raise LanguageRouteError(f"無效的語言代碼: {profile.code!r}")
+            deduped[code] = profile
+        if not deduped:
+            raise LanguageRouteError("語言註冊表不可為空")
+        self._profiles: Mapping[str, LanguageProfile] = deduped
+
+    # ---- read-only accessors --------------------------------------------------
+
+    def __contains__(self, code: object) -> bool:
+        return isinstance(code, str) and code.strip().lower() in self._profiles
+
+    def __len__(self) -> int:
+        return len(self._profiles)
+
+    def __iter__(self):
+        return iter(self._profiles)
+
+    @property
+    def codes(self) -> tuple[str, ...]:
+        return tuple(sorted(self._profiles))
+
+    def get(self, code: str) -> LanguageProfile | None:
+        if not code:
+            return None
+        return self._profiles.get(code.strip().lower())
+
+    def require(self, code: str) -> LanguageProfile:
+        profile = self.get(code)
+        if profile is None:
+            raise LanguageRouteError(
+                f"不支援的語言代碼: {code!r}；可用: {', '.join(self.codes)}"
+            )
+        return profile
+
+    def resolve_bcp47(self, code: str) -> str | None:
+        profile = self.get(code)
+        if not profile:
+            return None
+        return profile.bcp47[0] if profile.bcp47 else None
+
+    # ---- prompt construction --------------------------------------------------
+
+    def build_prompt(self, code: str, *, extra: str | None = None) -> str:
+        """Combine curated vocabulary with caller-supplied context."""
+        profile = self.require(code)
+        pieces: list[str] = [profile.whisper_prompt]
+        if profile.common_phrases:
+            pieces.append("常見詞: " + "、".join(profile.common_phrases))
+        if extra:
+            pieces.append(extra.strip())
+        return "\n".join(piece for piece in pieces if piece)
+
+    # ---- bulk export ----------------------------------------------------------
+
+    def to_dict(self) -> dict[str, dict[str, Any]]:
+        return {code: profile.to_dict() for code, profile in self._profiles.items()}

+# Whisper Multi-Language Desktop Automation
+
+把 OpenAI Whisper API 從「轉出一段文字」升級成「**多語言語音 → 結構化輸出 → 桌面自動化任務**」的完整管線。
+
+## 與 `whisper_desktop` 的差異
+
+| 面向 | 上一輪 `whisper_desktop` | 本輪 `whisper_multilang` |
+|---|---|---|
+| 語言處理 | `language=` 直傳 Whisper | `LanguageRegistry` 內建 16 種語言 prompt/BCP-47/常見詞；自動產生多語 prompt |
+| 桌面副作用 | copy / type / notify 三件 | copy / type / notify + **shell allowlist + webhook + log sink** |
+| 文字內指令 | 無 | 偵測內嵌的「提醒 / log this / コピー」等觸發詞 |
+| 任務組合 | 全部或全不 | `--tasks copy,notify,!shell` 細粒度 |
+| Web 前端 | 無 | `web_output.html` 內建瀏覽器端 STT 控制台 |
+| 測試覆蓋 | 略 | 12+ 個 unittest（HTTP retry / multipart / registry / workflow） |
+
