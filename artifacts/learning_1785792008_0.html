⑤ Playwright 互動劇本（pseudo code） runner.js

// test_framework/runner.py: 跨瀏覽器 + 跨 viewport 跑測試
const INTERACTION_SCRIPT = `
  // 1. 等 DOM ready
  await new Promise(r => setTimeout(r, 50));
  // 2. 點中文按鈕
  document.querySelector('.lang-btn[data-lang="zh"]')?.click();
  // 3. 點食譜範例
  document.querySelector('.sample-btn[data-sample="recipe"]')?.click();
  // 4. 確認 output 有內容
  window.__testMarker = document.getElementById('output').textContent.length;
`;

// 對 9 個 case 各跑一次：
for (const browser of ['chromium','firefox','webkit']) {
  for (const vp of ['desktop','tablet','mobile']) {
    const page = await playwright[browser].newPage({ viewport: vp });
    await page.goto('file://web_output.html');
    await page.evaluate(INTERACTION_SCRIPT);
    await page.screenshot({ path: 'screenshots/' + browser + '_' + vp + '.png' });
    // 比對 baseline + 記錄 traversal_path
  }
}
