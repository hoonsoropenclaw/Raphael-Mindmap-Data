+"""
+config.py - 從環境變數讀 Telegram bot token 等敏感設定
+
+設計：
+- 不寫死任何 token（避免推到版控時洩漏）
+- 提供 load_config() 統一讀取
+- 缺必要設定時直接 raise ConfigError，避免「bot 起來卻不能運作」

+If 未來要加更多設定 → 加到 Config dataclass + 對應 env var 即可
+"""
+
+from __future__ import annotations
+
+import os
+from dataclasses import dataclass
+
+
+class ConfigError(Exception):
+    """設定缺失或無效。"""
+
+
+@dataclass(frozen=True)
+class Config:
+    telegram_token: str
+    weather_provider: str          # "wttr" | "openweathermap"
+    openweathermap_api_key: str | None
+    log_level: str
+    request_timeout_sec: int
+
+
+def load_config() -> Config:
+    """從環境變數讀取設定。
+
+    必要環境變數：
+      - TELEGRAM_BOT_TOKEN: Telegram bot token（@BotFather 拿）
+
+    選用環境變數（都有預設值）：
+      - WEATHER_PROVIDER: "wttr"（預設）或 "openweathermap"
+      - OPENWEATHERMAP_API_KEY: 選填，只在用 OpenWeatherMap 時需要
+      - LOG_LEVEL: "INFO"（預設）/ "DEBUG" / "WARNING"
+      - REQUEST_TIMEOUT_SEC: 10（預設）
+    """
+    token = os.environ.get("TELEGRAM_BOT_TOKEN", "").strip()
+    if not token:
+        raise ConfigError(
+            "缺少環境變數 TELEGRAM_BOT_TOKEN\n"
+            "請到 @BotFather 申請 bot token，然後 export TELEGRAM_BOT_TOKEN=..."
+        )
+
+    provider = os.environ.get("WEATHER_PROVIDER", "wttr").strip().lower()
+    if provider not in ("wttr", "openweathermap"):
+        raise ConfigError(

+        )
+
+    api_key = os.environ.get("OPENWEATHERMAP_API_KEY", "").strip() or None

+
+    try:
+        timeout = int(os.environ.get("REQUEST_TIMEOUT_SEC", "10"))
+    except ValueError:
+        timeout = 10
+    timeout = max(5, min(timeout, 60))  # clamp 5~60 秒
+
+    return Config(
+        telegram_token=token,
+        weather_provider=provider,
+        openweathermap_api_key=api_key,
+        log_level=log_level,
+        request_timeout_sec=timeout,
+    )

+# Weather Bot (Telegram)
+
+一個輕量 Telegram 機器人，接收文字訊息後查詢該地點的天氣並回傳。
+
+## 特色
+
+- **零設定啟動**：預設使用 [wttr.in](https://wttr.in)，無需任何 API key
+- **多 provider**：可切換到 OpenWeatherMap（需 API key）
+- **純 Python 標準工具**：只依賴 `requests`，無第三方 bot framework
+- **錯誤處理完整**：網路錯誤、Telegram rate limit、無效地點都有對應訊息
+- **Long polling**：不需要 webhook 或公開網域，本機或 N100 都能跑
+
+## 快速啟動
+
+```bash
+# 1. 安裝依賴
+pip install -r requirements.txt
+
+# 2. 設定 token（用 @BotFather 申請）
+export TELEGRAM_BOT_TOKEN=*** 3. 啟動
+python3 weather_bot.py
+# 或
+./run.sh
+```
+
+跟 bot 對話：
+
+```
+你: /start
+bot: 👋 你好！我是天氣查詢機器人...
+
+你: 台北
+bot: 📍 台北, 台灣
+     🌡️ 溫度: 28.0°C (體感 31.0°C)
+     ☁️ 天氣: Partly cloudy
+     💧 濕度: 78%
+     🌬️ 風: 12.0 km/h NE
+     🔧 來源: wttr
+
+你: /weather Tokyo
+bot: 📍 Tokyo, Japan
+     ...
+```
+
+## 指令列表
+
+| 指令 | 說明 |
+|------|------|
+| `/start` | 歡迎訊息 |
+| `/help` | 完整說明 |

+# Implementation Thoughts — Telegram 天氣查詢 Bot
+
+## 1. 核心目標
+
+實作一個 Telegram bot：
+- 接收使用者文字訊息（城市名）
+- 呼叫天氣 API 查詢該地點目前天氣
+- 把結果格式化回傳給使用者
+
+要求：可立即執行的成品，不需要 webhook / 公開網域。
+
+## 2. 架構決策
+
+### 2.1 選 wttr.in 作為預設 provider
+
+- **無需 API key**：降低使用門檻，第一次啟動就能跑
+- **HTTP GET 簡單**：requests 兩行就搞定
+- **支援中文城市名**：測過「台北」「高雄」都能解析
+
+備用 OpenWeatherMap 是給「需要 SLA / 更穩定資料源」的場景，但要多一個 API key。
+
+### 2.2 Long polling 而非 webhook
+
+- 不需要 HTTPS endpoint、不需要公開網域
+- 本機 / N100 / 任何能出網的環境都能跑
+- 程式碼量少很多（不用處理 webhook certificate、port binding）
+
+### 2.3 不用 python-telegram-bot 框架
+
+- 只用 `requests` 就能 cover 本 bot 需要的兩個 API：`getUpdates` / `sendMessage`
+- 少一個依賴 = 少一個更新災難
+- 程式碼反而更透明，使用者一看就懂流程
+- 缺點：要自己處理 polling offset、429 retry 等眉角（已實作）
+
+### 2.4 模組分離
+
+```
+weather.py     ← 純天氣邏輯（純函式 + dataclass）
+config.py      ← 環境變數讀取
+weather_bot.py ← Telegram 主程式
+```
+
+理由：
+- `weather.py` 可獨立測試（已驗證過真實打 wttr.in）
+- `weather_bot.py` 專注 Telegram 協定，不混天氣 API 細節
+- `config.py` 集中管理環境變數，未來加設定不會散落
+
+## 3. 錯誤處理設計
+
+**原則：所有錯誤都變成使用者訊息，不讓 bot crash。**

+- **Telegram 429 rate limit**：按 `Retry-After` header 等
+- **Telegram 4xx**（chat not found / bot blocked）：catch 後只 log，不重試
+- **天氣 API 4xx/5xx**：丟 WeatherError，handle 包成「⚠️ 查詢失敗」訊息
+- **未知指令**：回 /help 而不是錯誤
