+"""Database engine + session factory.
+
+Design choices:
+- SQLAlchemy 2.0 ``DeclarativeBase`` style (modern, type-safe)
+- ``check_same_thread=False`` for SQLite — required when sharing the
+  connection across threads (FastAPI runs sync endpoints in a threadpool)
+- ``StaticPool`` keeps an in-memory SQLite alive across connections
+  (used by tests)
+- ``get_db`` is a generator dependency so the session is always closed
+  even if the endpoint raises
+"""
+from __future__ import annotations
+
+from typing import Generator
+
+from sqlalchemy import create_engine
+from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker
+
+from app.config import get_settings
+
+
+class Base(DeclarativeBase):
+    """Common declarative base for all ORM models."""
+
+
+def _make_engine(url: str):
+    """Build a SQLAlchemy engine with sensible SQLite defaults."""
+    if url.startswith("sqlite"):
+        # SQLite needs special args for FastAPI threadpool + in-memory tests
+        connect_args = {"check_same_thread": False}
+        if url.endswith(":memory:"):
+            # In-memory DB must use StaticPool so the same DB is shared
+            # across all connections in the connection pool
+            from sqlalchemy.pool import StaticPool
+            return create_engine(
+                url,
+                connect_args=connect_args,
+                poolclass=StaticPool,
+            )
+        return create_engine(url, connect_args=connect_args)
+    return create_engine(url)
+
+
+settings = get_settings()
+engine = _make_engine(settings.database_url)
+SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
+
+
+def init_db() -> None:
+    """Create all tables. Safe to call repeatedly (idempotent)."""
+    # Import models so they register with Base.metadata before create_all
+    from app import models  # noqa: F401
+
+    Base.metadata.create_all(bind=engine)
+
+
+def get_db() -> Generator[Session, None, None]:
+    """FastAPI dependency that yields a DB session and closes it after."""
+    db = SessionLocal()
+    try:
+        yield db
+    finally:
+        db.close()

+Design:

+  configure trusted hosts, otherwise falls back to ``request.client.host``)
+- Custom 429 handler so the test suite can assert on JSON shape, not raw text

+  (which is the whole point — we need memory of past requests)
+"""
+from __future__ import annotations
+
+from fastapi import Request
+from fastapi.responses import JSONResponse

+from slowapi.errors import RateLimitExceeded
+from slowapi.util import get_remote_address

+

+limiter = Limiter(

+    default_limits=[settings.rate_limit_default],
+    headers_enabled=True,  # sets X-RateLimit-* headers automatically

+    )

+## 架構決策（為何這樣做）
+
+### 1. 分層：Config / Database / Model / Schema / Route
+- **Config** (`config.py`): Pydantic Settings 單一設定源；環境變數驅動，無散落 `os.getenv()`
+- **Database** (`database.py`): SQLAlchemy 2.0 Declarative Base + `get_db` 依賴產生器（確保一定 close）
+- **Models** (`models.py`): ORM 對應 DB table schema
+- **Schemas** (`schemas.py`): Pydantic 對應 HTTP API contract；兩個分開避免 persistence 細節洩漏到 API
+- **Rate limit** (`rate_limit.py`): 獨立模組，全域 `Limiter` 實例 + 自訂 429 JSON handler
+- **Main** (`main.py`): FastAPI app + 路由 + lifespan
+
+### 2. SQLite 連線細節
+- `check_same_thread=False`：FastAPI 同步端點在 threadpool 跑，必須設
+- `:memory:` 模式用 `StaticPool`：確保同一 in-memory DB 在 connection pool 內共享
+- `init_db()` 寫在 `lifespan` 內：啟動時建表，無需手動執行 SQL
+
+### 3. Rate Limit 選 `slowapi` + `SlowAPIASGIMiddleware`
+- slowapi 是 FastAPI 官方推薦的 Flask-Limiter port
+- **為什麼用 ASGI middleware 而不是 BaseHTTPMiddleware**：BaseHTTPMiddleware 在 streaming response 會卡住；ASGI 版支援完整 ASGI 介面
+- `key_func=get_remote_address`：以 client IP 為 key
+- `default_limits=["60/minute"]`：所有未帶裝飾器路由的預設上限
+- `@limiter.exempt` on `/health`：健康檢查不算限流
+- Custom 429 handler：回 JSON `{"detail":..., "limit":..., "path":...}`，方便客戶端解析
