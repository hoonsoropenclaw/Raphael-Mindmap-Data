+// tests-visual/visual.spec.js
+// Visual regression: full-page + element-level screenshots compared to baselines
+// using pixelmatch. Reference: skills/comprehensive_visual_regression_testing.md
+import { test, expect } from '@playwright/test';
+import fs from 'node:fs';
+import path from 'node:path';
+import { PNG } from 'pngjs';
+import pixelmatch from 'pixelmatch';
+
+const BASELINE_DIR = path.join(process.cwd(), 'baselines');
+const ACTUAL_DIR   = path.join(process.cwd(), 'test-results', 'actual');
+const DIFF_DIR     = path.join(process.cwd(), 'test-results', 'diff');
+for (const d of [BASELINE_DIR, ACTUAL_DIR, DIFF_DIR]) fs.mkdirSync(d, { recursive: true });
+
+// Tunable thresholds
+const PAGE_THRESHOLD    = 0.01;  // 1% pixel mismatch tolerance for full pages
+const ELEMENT_THRESHOLD = 0.005; // 0.5% for individual elements (stricter)
+
+function diffPng(actualPath, baselinePath, diffPath, threshold) {
+  const actual   = PNG.sync.read(fs.readFileSync(actualPath));
+  const baseline = PNG.sync.read(fs.readFileSync(baselinePath));
+  if (actual.width !== baseline.width || actual.height !== baseline.height) {
+    throw new Error(
+      `Dimension mismatch: actual=${actual.width}x${actual.height} ` +
+      `baseline=${baseline.width}x${baseline.height}`
+    );
+  }
+  const diff  = new PNG({ width: actual.width, height: actual.height });
+  const ratio = pixelmatch(
+    baseline.data, actual.data, diff.data,
+    actual.width, actual.height,
+    { threshold }
+  );
+  fs.writeFileSync(diffPath, PNG.sync.write(diff));
+  return ratio; // 0..1
+}
+
+function ensureBaseline(project, name, generator) {
+  const file = path.join(BASELINE_DIR, project, `${name}.png`);
+  if (!fs.existsSync(file)) {
+    fs.mkdirSync(path.dirname(file), { recursive: true });
+    generator().toFile(file);
+  }
+  return file;
+}

+
+## 概述
+本目錄實作 Playwright 跨瀏覽器測試框架，整合三大類測試：
+1. **跨瀏覽器 smoke test** — 互動流程、DOM 更新、零 JS error
+2. **視覺回歸測試** — 整頁 + 元素級 PNG 比對（pixelmatch）
+3. **Request interception** — `addInitScript` 攔截 fetch/XHR + `page.route()` mock API
+
+## 瀏覽器矩陣
+| 專案 | 視口 | 模擬裝置 |
+|------|------|----------|
+| `chromium-desktop` | 1440x900 | Desktop Chrome |
+| `firefox-tablet`   | 768x1024 | Desktop Firefox (平板) |
+| `webkit-mobile`    | 390x844 (DPR=3 → 1170x2532 截圖) | iPhone 13 |
+
+## 安裝
+```bash
+npm install          # 裝 @playwright/test + pixelmatch + pngjs
+# 瀏覽器已預裝於 /home/hoonsoropenclaw/.cache/ms-playwright/
+```
+
+## 執行
+```bash
+# 全部 30 個測試 (3 瀏覽器 × 10 測試)
+npx playwright test
+
+# 各子集
+npx playwright test tests/             # 12 smoke tests
+npx playwright test tests-visual/      # 9 visual regression
+npx playwright test tests-intercept/   # 9 request interception
+
+# 重新產生 baselines（HTML 有意改動後）
+UPDATE_BASELINE=1 npx playwright test tests-visual/
+
+# 開 HTML 報告
+npx playwright show-report
+```

+## 設計要點
+
+### 1. 動態內容穩定化
+- `data-testid="user-agent"` 在視覺測試時用 `addStyleTag` 隱藏 — 不同瀏覽器的 UA 文字不污染 baseline
+- 每張截圖前 `waitForTimeout(300~500ms)` 等字型載入 + JS 跑完
+- `data-testid` 屬性統一給測試抓，避開 i18n / class name 變動
+
+### 2. Baseline 管理
+- 預設 `baselines/<project>/<name>.png`（每個瀏覽器獨立 baseline — 因為字型/反鋸齒本來就會不同）
+- `UPDATE_BASELINE=1` 觸發再生；沒設環境變數時 baseline 缺失會 `test.skip` 並提示
+- `pixelmatch threshold = 0.01` 整頁 / `0.005` 元素級（更嚴格）
+
+### 3. Request interception 三層
+- **L1 (Network)**：`page.route('**/api/items.json', ...)` 攔 HTTP
+- **L2 (Page addInitScript)**：in-page fetch wrapper 紀錄所有 fetch 呼叫
+- **L3 (Page addInitScript)**：XHR open hook 紀錄 XHR 呼叫
+- 三層各司其職：L1 提供 mock 資料、L2/L3 提供觀察力
+
+### 4. 跨瀏覽器一致性
+- 不用 CSS 變數以外的瀏覽器特異語法
+- `@supports not (backdrop-filter)` 提供 fallback
+- `data-state` 屬性作狀態機 (`ready` / `passed` / `failed`) 而非 class，方便測試斷言
+
+## 檔案
+- `web_output.html` — 被測目標（SUT）
+- `playwright.config.js` — 測試配置 + webServer + projects
+- `tests/smoke.spec.js` — 12 跨瀏覽器 smoke tests
+- `tests-visual/visual.spec.js` — 9 視覺回歸 tests
+- `tests-intercept/intercept.spec.js` — 9 request interception tests
+- `baselines/<project>/` — 每瀏覽器的 baseline 截圖
+- `test-results/actual/` — 實際截圖
+- `test-results/diff/` — pixelmatch 差異圖
+- `playwright-report/` — HTML 報告
