+ITERATIONS = 310_000
+

+def hash_password(password: str) -> str:
+    salt = os.urandom(16)
+    digest = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, ITERATIONS)
+    return f"pbkdf2_sha256${ITERATIONS}${base64.urlsafe_b64encode(salt).decode()}${base64.urlsafe_b64encode(digest).decode()}"
+
+
+def verify_password(password: str, encoded: str) -> bool:
+    try:
+        algorithm, iterations, salt_b64, digest_b64 = encoded.split("$", 3)
+        if algorithm != "pbkdf2_sha256":
+            return False
+        salt = base64.urlsafe_b64decode(salt_b64)
+        expected = base64.urlsafe_b64decode(digest_b64)
+        actual = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, int(iterations))
+        return hmac.compare_digest(actual, expected)
+    except (ValueError, TypeError):
+        return False
+
+
+def _b64(data: bytes) -> str:
+    return base64.urlsafe_b64encode(data).rstrip(b"=").decode()
+
+
+def _unb64(data: str) -> bytes:
+    return base64.urlsafe_b64decode(data + "=" * (-len(data) % 4))
+
+
+def create_access_token(subject: str) -> tuple[str, int]:
+    settings = get_settings()
+    now = int(time.time())
+    ttl = settings.jwt_ttl_minutes * 60
+    header = {"alg": "HS256", "typ": "JWT"}
+    payload = {"sub": subject, "iat": now, "exp": now + ttl, "jti": str(uuid.uuid4())}
+    signing_input = f"{_b64(json.dumps(header, separators=(',', ':')).encode())}.{_b64(json.dumps(payload, separators=(',', ':')).encode())}"
+    signature = hmac.new(settings.jwt_secret.encode(), signing_input.encode(), hashlib.sha256).digest()
+    return f"{signing_input}.{_b64(signature)}", ttl
+

+# FastAPI 動態 RBAC
+
+可執行的角色型動態權限管理後端，支援：
+
+- 使用者 ↔ 角色多對多
+- 角色 ↔ 權限多對多
+- 角色傳遞繼承（child inherits parent）
+- cycle 偵測與 409 拒絕
+- PBKDF2-SHA256 密碼雜湊
+- HS256 Bearer JWT 驗證
+- request-time DB 權限解析：授權/撤權與角色指派對既有 JWT 立即生效
+- 權限管理稽核紀錄
+- OpenAPI：`/docs`
+
+## 啟動
+
+```bash
+python3 -m venv .venv
+.venv/bin/pip install -r requirements.txt
+export RBAC_JWT_SECRET="$(openssl rand -hex 32)"
+.venv/bin/uvicorn app.main:app --host 127.0.0.1 --port 8000
+```
+
+首次啟動自動建立 SQLite schema 與 demo 資料：
+
+| 帳號 | 密碼 | 角色 |
+|---|---|---|
+| admin | ChangeMe123! | admin |
+| alice | ChangeMe123! | viewer |
+| bob | ChangeMe123! | editor |
+
+這些帳密只供本機 demo，正式環境必替換 seed 策略。
+
+## 快速操作
+
+```bash
+TOKEN=$(curl -sS -X POST http://127.0.0.1:8000/auth/login \
+  -H 'Content-Type: application/json' \
+  -d '{"username":"admin","password":"ChangeMe123!"}' | python3 -c 'import json,sys; print(json.load(sys.stdin)["access_token"])')
+
+curl -sS http://127.0.0.1:8000/me/permissions -H "Authorization: Bearer $TOKEN"
+
+curl -i -X POST http://127.0.0.1:8000/admin/permissions \
+  -H "Authorization: Bearer $TOKEN" -H 'Content-Type: application/json' \
+  -d '{"name":"reports:export","description":"Export reports"}'
+```
+
+## API 摘要
+
+- `POST /auth/login`
+- `GET /me/permissions`
+- `GET/POST /documents`
+- `GET/POST /admin/roles`
+- `POST /admin/permissions`
+- `PUT/DELETE /admin/roles/{role}/permissions/{permission}`
+- `PUT/DELETE /admin/roles/{child}/parents/{parent}`
+- `PUT/DELETE /admin/users/{user}/roles/{role}`
+- `GET /admin/audit-logs`
+
+## 安全設計
+
+1. JWT 不保存角色或權限，避免 stale authorization。
+2. 生產模式 `RBAC_ENV=production` 時，JWT secret 不足 32 字元即 fail closed。
+3. 所有 SQL 經 SQLAlchemy parameter binding。
+4. 後端 permission dependency 是唯一安全邊界，UI 隱藏按鈕不能取代 API 授權。
+5. 角色繼承在寫入前做 graph cycle 檢查。
+6. 權限變更與 audit log 在同一 transaction commit。
