+    paths: tuple[str, ...] = ("/",)
+    log_files: tuple[str, ...] = ("/var/log/syslog", "/var/log/auth.log", "/var/log/kern.log")
+    journal: bool = True
+
+@dataclass
+class Alert:
+    rule: str
+    severity: str
+    message: str
+    value: float | int | None = None
+
+class State:
+    def __init__(self, path: Path):
+        self.path = path
+        self.data = self._load()
+    def _load(self):
+        try:
+            return json.loads(self.path.read_text())
+        except (FileNotFoundError, json.JSONDecodeError, OSError):
+            return {"cursors": {}, "alerts": {}}
+    def save(self):
+        self.path.parent.mkdir(parents=True, exist_ok=True)
+        tmp = self.path.with_suffix(self.path.suffix + ".tmp")
+        tmp.write_text(json.dumps(self.data, indent=2, sort_keys=True))
+        os.replace(tmp, self.path)
+
+class AlertManager:
+    def __init__(self, state: State, cooldown: int, webhook: str | None = None):
+        self.state, self.cooldown, self.webhook = state, cooldown, webhook
+    def emit(self, alert: Alert) -> bool:
+        key = f"{alert.rule}:{alert.severity}"
+        now = time.time()
+        if now - float(self.state.data["alerts"].get(key, 0)) < self.cooldown:
+            return False
+        self.state.data["alerts"][key] = now
+        logging.warning("ALERT [%s] %s", alert.severity.upper(), alert.message)
+        if self.webhook:
+            try:
+                import urllib.request
+                payload = json.dumps(asdict(alert)).encode()
+                req = urllib.request.Request(self.webhook, payload, {"Content-Type": "application/json"})

+# Linux 系統日誌與資源監控器
+
+## 功能
+- 讀取 `/proc/stat`、`/proc/meminfo`、`shutil.disk_usage()`，監測 CPU、記憶體、磁碟與 load average。
+- 增量掃描 syslog/auth/kern 等檔案，支援 inode 變更與 truncate 後自動重設 cursor。
+- 使用 `journalctl` 補抓 systemd journal 的錯誤事件。
+- JSON state 原子寫入，保存 cursor、CPU 基準與告警冷卻時間。
+- stderr logging、SIGTERM/SIGINT graceful shutdown、每條規則獨立 cooldown。
+- 可選 webhook JSON 警告；核心功能僅依賴 Python 3.12 標準函式庫。
+
+## 使用
+```bash
+chmod +x linux_monitor.py
+python3 linux_monitor.py --once
+python3 linux_monitor.py -c monitor.conf --interval 60
+python3 linux_monitor.py --once --webhook https://example.invalid/hook
+```
+
+設定檔採用簡化 YAML（key: value），例如：
+```text
+cpu_warn: 80
+mem_warn: 80
+disk_warn: 85
+load_warn: 2
+log_error_threshold: 5
+log_window: 60
+cooldown: 300
+paths: [ / ]
+log_files: [ /var/log/syslog, /var/log/auth.log, /var/log/kern.log ]
+journal: true
+```
+
+## 設計與限制
+- 使用增量 cursor 避免每輪重掃整份日誌；cursor key 同時保存 inode 與 offset，因此 logrotate 和原地 truncate 都能恢復。
+- 告警以 `rule:severity` 做冷卻鍵，避免重複告警但不抑制更嚴重事件。
+- `journalctl` 不存在、權限不足或 timeout 時會安全降級，不讓資源監控整體失敗。
+- 這是 read-only 監控器，不進行自動修復；webhook 失敗只記錄錯誤。
+- 設定 parser 不是完整 YAML 1.2 parser；若設定需求變複雜，應改用已安裝的 YAML library 或明確 JSON 設定。
+

+# 實作審查筆記：Linux 系統日誌與資源監控器
+
+## 架構
+1. `linux_monitor.py` 採單檔、stdlib-first，避免在最小化 Linux 環境引入第三方依賴。
+2. 監控來源分成三類：`/proc` 資源、檔案日誌、systemd `journalctl`；來源失敗時局部降級，不中斷整輪檢查。
+3. state 使用 JSON temporary file + `os.replace()` 原子替換，避免程序中斷留下半個 state。
+4. 日誌 cursor 同時記錄 inode/offset：inode 改變代表 rotation，offset 變小代表 truncate，兩者都從新檔開始。
+5. 告警 cooldown 以 `rule:severity` 分組，抑制相同告警風暴，但不讓 warn cooldown 阻擋 crit。
+6. `SIGTERM`/`SIGINT` 設為 graceful stop；每輪結束都保存 state，停止後再保存一次。
+7. webhook 是可選通知 sink；網路或 webhook 錯誤只記錄，不影響本地告警。
+
+## 依 Senior Architect 回饋採用的原則
+- stdlib-first、清楚的責任分界、外部邊界錯誤隔離、rate limiting、避免重複讀取，以及 production logging。
+- 參照上一輪指出的風險：本實作未假設 journal 一定存在，也未把 webhook 失敗視為監控器致命錯誤。
+

+- `monitor.conf` parser 只支援本工具所需的簡化 YAML 子集，不是完整 YAML 1.2。
