+@dataclass(frozen=True)

+    @classmethod
+    def from_environment(cls) -> "BotConfig":
+        token = os.environ.get("TELEGRAM_BOT_TOKEN", "").strip()

+# Telegram 天氣機器人
+
+這是一個零第三方 Python 依賴的 Telegram 長輪詢機器人。使用者傳送城市名稱後，程式會呼叫 Open-Meteo Geocoding + Forecast API，並以繁體中文回傳目前天氣。
+
+## 功能
+
+- 接收純文字城市：`Taipei`
+- 接收命令：`/weather Taipei`
+- `/start`、`/help` 使用說明
+- 目前溫度、體感溫度、濕度、風速、天氣代碼中文化
+- Telegram user ID 白名單（可選）
+- 每個使用者獨立 cooldown 限流
+- 所有外部 HTTP 請求有 timeout、暫時性錯誤有 bounded retry
+- SIGINT/SIGTERM graceful shutdown
+- Token 不寫進原始碼或日誌
+- Python 3.12 stdlib only，無需 `pip install`
+

+

+def sanitize_text(value: object, max_length: int = 500) -> str:
+    """Sanitise a string for safe inclusion in a Telegram message.
+
+    - Removes NUL bytes and ASCII control characters.
+    - Converts ``\\r`` into ``\\n`` so multi-line weather output keeps
+      its structure inside the chat, while stripping lone surrogates
