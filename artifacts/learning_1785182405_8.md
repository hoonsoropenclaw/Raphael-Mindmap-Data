+"""
+Weather lookup module — talks to OpenWeatherMap.
+
+Design: returns a plain dict with `ok` flag so the Telegram layer can
+present a friendly error without dealing with HTTP exceptions.
+
+Two-step lookup because OWM v2.5 needs lat/lon for accurate results:
+  1. geocode city name -> (lat, lon, country)
+  2. weather at (lat, lon) -> current conditions
+
+Both endpoints are independent and unit-testable without Telegram.
+"""
+from __future__ import annotations
+
+import logging
+from typing import Any
+
+import requests
+from requests import RequestException
+
+import config
+
+log = logging.getLogger(__name__)
+
+
+class WeatherError(Exception):
+    """Raised when the weather lookup fails for any reason."""
+
+
+def geocode(city: str, limit: int = 1) -> list[dict[str, Any]]:
+    """Resolve a city name to lat/lon via OWM Geocoding API."""
+    try:
+        resp = requests.get(
+            config.OWM_GEO_URL,
+            params={
+                "q": city,
+                "limit": limit,
+                "appid": config.load_config().openweather_api_key,
+            },
+            timeout=10,
+        )
+        resp.raise_for_status()
+    except RequestException as e:
+        log.warning("Geocode request failed for %r: %s", city, e)
+        raise WeatherError(f"地理編碼請求失敗: {e}") from e
+    return resp.json()
+
+
+def fetch_current(lat: float, lon: float, lang: str, units: str) -> dict[str, Any]:

+                "lang": lang,
+                "units": units,
+            },
+            timeout=10,
+        )
+        resp.raise_for_status()

+        log.warning("Weather request failed for (%.2f, %.2f): %s", lat, lon, e)

+# Telegram Weather Bot
+
+A minimal, well-organized Telegram bot that answers with current weather
+when you send it a city name.
+
+## Architecture
+
+```
+bot.py        ← Telegram handlers (thin layer)
+weather.py    ← OpenWeatherMap client (testable in isolation)
+config.py     ← Env loader + constants
+run.sh        ← Launcher with safe defaults
+test_weather.py ← Offline + live smoke tests for weather.py
+```
+
+The bot layer never touches HTTP directly — it calls `get_weather(city)`
+which either returns a user-ready string or raises `WeatherError`.
+
+## Setup
+
+### 1. Get your keys
+
+| Key | Where to get it |
+|-----|-----------------|
+| `TELEGRAM_BOT_TOKEN` | [@BotFather](https://t.me/BotFather) → `/newbot` |
+| `OPENWEATHER_API_KEY` | [openweathermap.org/api](https://openweathermap.org/api) → free tier |
+
+### 2. Install dependencies
+
+```bash
+python3 -m venv .venv
+source .venv/bin/activate
+pip install -r requirements.txt
+```
+
+### 3. Configure
+
+```bash
+cp .env.example .env
+# edit .env and paste your two tokens
+```
+
+### 4. Run
+
+```bash
+./run.sh
+# or:  python3 bot.py
+```
+
+You'll see `[INFO] Bot is up. Press Ctrl+C to stop.` when it's live.
+Open Telegram, talk to your bot, send a city name.
+
+## Usage examples
+
+```
+/start

+San Francisco, US
+```
+
+## Tests
+
