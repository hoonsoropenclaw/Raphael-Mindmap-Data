+#!/usr/bin/env bash
+# sysmon.sh — Linux 系統日誌 + 資源監控腳本
+# 使用方式:
+#   ./sysmon.sh                  # 跑一次（cron 友好）
+#   ./sysmon.sh --check-deps      # 檢查相依工具
+#   ./sysmon.sh --self-test       # 跑內建的 smoke test
+#   CRITICAL=1 ./sysmon.sh        # 強制走 critical 分支（測試用）
+#
+# 設計: 每個監控項獨立函式 → 警告統一收斂到 emit_alert()
+#       冷卻機制防止同一個警告被狂炸（避免 cron 高頻時刷屏）
+#       任何失敗走 "降級" 不是 "爆炸"（disk 沒 df 還能跑 mem, etc.）
+set -u
+# 故意不用 set -e: 我要每個檢查獨立 try/fail、即使一個區塊壞了其他照跑
+IFS=$'\n\t'
+
+SCRIPT_DIR="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" &>/dev/null && pwd)"
+# shellcheck disable=SC1091
+source "${SCRIPT_DIR}/sysmon_config.sh"
+
+# ─────────── 工具檢查 / 降級 ───────────
+have() { command -v "$1" >/dev/null 2>&1; }
+
+HAVE_JOURNALCTL=0; have journalctl && HAVE_JOURNALCTL=1
+HAVE_FREE=0;       have free       && HAVE_FREE=1
+HAVE_DF=0;         have df         && HAVE_DF=1
+HAVE_UPTIME=0;     have uptime     && HAVE_UPTIME=1
+HAVE_AWK=0;        have awk        && HAVE_AWK=1
+HAVE_BC=0;         have bc         && HAVE_BC=1
+HAVE_LOGGER=0;     have logger     && HAVE_LOGGER=1
+HAVE_SYSTEMCTL=0;  have systemctl  && HAVE_SYSTEMCTL=1
+HAVE_SS=0;         have ss         && HAVE_SS=1
+HAVE_TAIL=0;       have tail       && HAVE_TAIL=1
+
+# ─────────── 告警收斂器（含冷卻）───────────
+COOLDOWN_DIR="$(mktemp -d -t sysmon.XXXXXX)"
+cleanup() { rm -rf "$COOLDOWN_DIR"; }
+trap cleanup EXIT
+
+emit_alert() {
+  local level="$1"; shift    # WARN | CRIT
+  local src="$1"; shift      # DISK | MEM | CPU | OOM | AUTH | SVC
+  local msg="$1"             # 人讀的訊息
+  local key="${src}:$(printf '%s' "$msg" | awk '{print $1"_"$2"_"$3}')"
+  local now; now="$(date +%s)"
+  local stamp; stamp="$(date -Iseconds)"
+
+  # 冷卻：同一個 key N 秒內只發一次
+  local cooldown_file="${COOLDOWN_DIR}/${key}.lock"
+  if [[ -f "$cooldown_file" ]]; then
+    local last; last="$(cat "$cooldown_file")"
+    if (( now - last < SYSMON_COOLDOWN_SEC )); then
+      return 0
+    fi
+  fi
+  echo "$now" > "$cooldown_file"
+
+  local line="[$stamp] [$level] [$src] $msg"
+  # 雙軌：寫 syslog（用 -t 設 tag） + 寫本地 alert log
+  if (( HAVE_LOGGER )); then
+    logger -t "$SYSMON_TAG" -p "user.${level,,}" "$src: $msg"
+  fi
+  printf '%s\n' "$line" >> "$SYSMON_ALERT_LOG"
+  printf '⚠  %s\n' "$line" >&2
+}
