+"""TokenBucket rate limiter — single-process, asyncio.Lock per bucket.
+
+Design (per architect_feedback.md, carried from learning_1785198004_3):
+
+- `capacity` tokens refill at `refill_per_second` tokens/sec, capped at
+  capacity. A burst of `capacity` requests is allowed; steady-state
+  throughput is `refill_per_second` req/sec.
+- One `asyncio.Lock` per bucket — *not* per request — so concurrent
+  callers serialize on the bucket, not on each other.
+- `consume(key, n=1)` is the public API. Returns `(granted: bool,
+  retry_after_seconds: float, tokens_left: float)`.
+- `Retry-After` header in the HTTP layer reads the
+  `retry_after_seconds` field of a denied token.
+
+The limiter is intentionally injected into `app.state` rather than
+imported as a module global. This lets tests instantiate a fresh
+limiter with a small capacity (3 tokens) to trigger 429 in unit tests
+without sleeping or wall-clock dependency.
+"""
+from __future__ import annotations
+
+import asyncio
+import time
+from dataclasses import dataclass, field
+from typing import Dict
+
+
+@dataclass
+class _Bucket:
+    capacity: float
+    refill_per_second: float
+    tokens: float
+    last_refill: float
+    lock: asyncio.Lock = field(default_factory=asyncio.Lock)
+
+    def refill_to(self, now: float) -> None:
+        elapsed = max(0.0, now - self.last_refill)
+        self.tokens = min(self.capacity, self.tokens + elapsed * self.refill_per_second)
+        self.last_refill = now
+
+
+@dataclass
+class TokenBucket:
+    """Token-bucket rate limiter, async-safe."""
+    capacity: float = 60.0
