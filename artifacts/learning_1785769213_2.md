+"""
+pdf_table_extractor.py — 跨平台桌面自動化核心
+PDF → OCR (Tesseract) → 表格偵測 → CSV/JSON 輸出
+
+設計原則:
+- 雙路徑:有 selectable text 直接抽(text layer),沒有就走 OCR
+- OCR 用 Tesseract psm=6 (uniform block) 拿座標 + image_to_data 拿 token 級信心
+- 表格偵測用「水平投影 + 連續空白行」找表格上下界,再用「垂直距離谷底」切欄
+- 純 stdlib + pytesseract/pdf2image/Pillow,跨 macOS/Linux/Windows
+
+引用:advanced_ocr_integration.md (Section 1 Tesseract OCR Integration)
+"""
+from __future__ import annotations
+
+import argparse
+import csv
+import json
+import logging
+import sys
+from collections import defaultdict
+from dataclasses import dataclass, field, asdict
+from pathlib import Path
+from typing import Iterable
+
+from PIL import Image
+from pdf2image import convert_from_path
+import pytesseract
+
+log = logging.getLogger("pdf_table_extractor")
+
+
+# ─── 資料結構 ───────────────────────────────────────────────────────────────
+@dataclass
+class Token:
+    text: str
+    conf: float          # 0-100
+    left: int
+    top: int
+    right: int
+    bottom: int
+    page: int
+
+
+@dataclass
+class TableCell:
+    text: str
+    row: int
+    col: int
+    conf: float          # 平均信心
+
+
+@dataclass
+class ExtractedTable:
+    page: int
+    bbox: tuple[int, int, int, int]   # left, top, right, bottom
+    n_rows: int
+    n_cols: int
+    cells: list[list[TableCell]] = field(default_factory=list)
+
+    def to_rows(self) -> list[list[str]]:
+        return [[c.text for c in row] for row in self.cells]
+
+
+# ─── PDF → image → OCR tokens ────────────────────────────────────────────────
+def pdf_to_images(pdf_path: Path, dpi: int = 300) -> list[Image.Image]:
+    """PDF 全部頁轉 PIL Image(單一 dpi 統一,讓座標可跨頁比對)
+    注意:300dpi 是 OCR 甜蜜點;200 以下中文會掉字,400 以上加時間無明顯提升。"""
+    log.info("[pdf] %s → images @%ddpi", pdf_path.name, dpi)
+    images = convert_from_path(
+        str(pdf_path), dpi=dpi, fmt="png",
+        thread_count=2,  # 跨平台安全,Windows/Mac 也能跑
+    )
+    log.info("[pdf] %d page(s) loaded", len(images))
+    return images
+
+
