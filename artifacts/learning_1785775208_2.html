How it works

/* 1. 定義 token（每個主題一段，獨立、易讀） */
:root[data-theme="aurora"] {
  --bg-base: #0a0d18;
  --accent: #b89cff;
  --glass-blur: 24px;
  --gradient-bg: radial-gradient(...), linear-gradient(...);
}

/* 2. 元件只引用變數，不寫顏色 */
.card { background: var(--bg-elev-1); border: 1px solid var(--border-subtle); }

/* 3. 切換：JS 改 :root 的 data-* 即可 */
document.documentElement.dataset.theme = 'aurora';

/* 4. 過場：View Transition API（支援時） */
document.startViewTransition(() => { document.documentElement.dataset.theme = next; });
