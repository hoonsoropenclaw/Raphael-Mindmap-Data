+from __future__ import annotations
+
+import asyncio
+import math
+import time
+from collections.abc import Callable
+from dataclasses import dataclass
+
+
+@dataclass(frozen=True, slots=True)
+class RateLimitResult:
+    allowed: bool
+    remaining: int
+    retry_after: int
+
+
+class TokenBucket:
+    """Concurrency-safe token bucket for one rate-limit identity."""
+
+    def __init__(
+        self,
+        capacity: int,
+        refill_rate: float,
+        *,
+        clock: Callable[[], float] = time.monotonic,
+    ) -> None:
+        if capacity <= 0:
+            raise ValueError("capacity must be positive")
+        if refill_rate < 0:
+            raise ValueError("refill_rate cannot be negative")
+        self.capacity = capacity
+        self.refill_rate = refill_rate
+        self._tokens = float(capacity)
+        self._clock = clock
+        self._updated_at = clock()
+        self._lock = asyncio.Lock()
+
+    async def consume(self) -> RateLimitResult:
+        async with self._lock:
+            now = self._clock()
+            elapsed = max(0.0, now - self._updated_at)
+            self._tokens = min(
+                float(self.capacity), self._tokens + elapsed * self.refill_rate
+            )
+            self._updated_at = now
+
+            if self._tokens >= 1.0:
+                self._tokens -= 1.0
+                return RateLimitResult(True, int(self._tokens), 0)
+
+            retry_after = (
+                max(1, math.ceil((1.0 - self._tokens) / self.refill_rate))
+                if self.refill_rate > 0
+                else 1
+            )
+            return RateLimitResult(False, 0, retry_after)
+
+
+class RateLimiter:
+    """Process-local limiter keyed by client identity."""
+
+    def __init__(self, capacity: int, refill_rate: float) -> None:
+        self.capacity = capacity
+        self.refill_rate = refill_rate
+        self._buckets: dict[str, TokenBucket] = {}
+        self._buckets_lock = asyncio.Lock()
+
+    async def consume(self, key: str = "global") -> RateLimitResult:
+        async with self._buckets_lock:
+            bucket = self._buckets.get(key)
+            if bucket is None:
+                bucket = TokenBucket(self.capacity, self.refill_rate)
+                self._buckets[key] = bucket
+        return await bucket.consume()

+# Todo Rate Limit API
+
+A small FastAPI CRUD service backed by SQLite.
+
+## Features
+
+- `GET /health` health endpoint
+- `POST /todos` create a todo
+- `GET /todos` list todos with `completed`, `limit`, and `offset` filters
+- `GET /todos/{id}` read one todo
+- `PATCH /todos/{id}` partially update a todo
+- `DELETE /todos/{id}` delete a todo
+- Per-client process-local token-bucket rate limit on todo endpoints
+- `Retry-After`, `X-RateLimit-Limit`, and `X-RateLimit-Remaining` headers
+- Pydantic input validation and parameterized SQLAlchemy queries
+- SQLite directory/database permissions tightened to `0700`/`0600`
+
+## Run
+
+```bash
+python3 -m venv .venv
+.venv/bin/pip install -e '.[dev]'
+.venv/bin/uvicorn todo_api.main:app --reload
+```
+
+Configuration is available through environment variables:
+
+- `DATABASE_URL` (default `sqlite+pysqlite:///./data/todos.db`)
+- `DATA_DIR` (default `./data`)
+- `RATE_LIMIT_CAPACITY` (default `60`)
+- `RATE_LIMIT_REFILL_RATE` (default `1` token/second)
+
+## Test
+
+```bash
+.venv/bin/python -m pytest --cov=todo_api --cov-report=term-missing
+```
+
+The test suite covers CRUD, validation, pagination/filtering, 404/429 errors, retry headers, concurrent limiter capacity, refill behavior, identity isolation, packaging-safe settings, and SQLite file permissions.
+
+## Known limitation
+
+The limiter is process-local. For multiple application instances, replace `RateLimiter` with a shared Redis-backed implementation behind the same `consume()` interface. SQLite is appropriate for this simple single-writer service; PostgreSQL is the next scaling step.

+# Implementation Notes
+
+## Scope
+
+Built a self-contained Python FastAPI Todo API in this working directory. The requested implementation path was unspecified, so the project was placed in the current workspace and this file records the concrete decisions.
+
+## Architecture decisions
+
+1. **Synchronous SQLAlchemy**: endpoints use `def`, `sessionmaker`, and SQLAlchemy 2.x `select()` calls. This avoids mixing an async endpoint with a synchronous session and keeps the CRUD path simple for SQLite.
+2. **SQLite**: the default database is `./data/todos.db`; tests use an in-memory database with `StaticPool` so all test connections see the same schema.
+3. **Token bucket**: each client IP gets an independent `TokenBucket`. Each bucket has an `asyncio.Lock`, so concurrent requests cannot overspend the bucket. The limiter is process-local by design.
+4. **Validation**: Pydantic v2 schemas enforce non-blank titles, title length, description length, and safe pagination bounds. SQLAlchemy handles parameterization rather than string-built SQL.
+5. **Database lifecycle**: tables initialize during FastAPI lifespan startup. Sessions are request-scoped and always close in `finally`; writes call `commit()` then `refresh()` before returning.
+6. **File security**: lifespan startup applies a private umask and reasserts `0700` for the data directory and `0600` for SQLite database/sidecar files.
+7. **Test isolation**: application settings and limiter can be injected into `create_app()`, avoiding global state leakage across tests.
+
+## TDD and verification record
+
+- Wrote API and limiter tests before implementation; initial collection failed with `ModuleNotFoundError` because `todo_api` did not yet exist.
+- Implemented the package until the core suite passed.
+- Strengthening pass found a packaging discovery issue: a generated `data/` directory was detected as a second top-level package. Added explicit setuptools discovery for `todo_api*`.
+- Final automated verification: `20 passed`, `96.07%` total coverage, exceeding the configured `80%` threshold.
+- The suite includes an `asyncio.gather()` concurrency test proving exactly 10 grants from 100 simultaneous requests against a capacity-10 bucket.
+- A live Uvicorn HTTP smoke test is run as a separate delivery gate.
+
+## Deliberate limitations
+
+- No authentication was requested; add an auth dependency and use a per-user limiter key before exposing sensitive multi-user data.
+- The limiter is not distributed. Use Redis or another shared store when horizontally scaling.
+- SQLite is single-writer and intended here for a simple todo workload.
