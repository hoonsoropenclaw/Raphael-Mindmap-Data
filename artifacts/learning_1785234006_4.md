+benefit at this scope.

+"""Simple in-memory fixed-window rate limiter.
+
+Why fixed-window + in-memory:
+- Spec only says "rate limit", doesn't require distributed/Redis.
+- In-memory avoids a new dependency and works for single-process deployment.
+- Fixed window is the simplest algorithm that's correct under the constraints
+  (N requests per W seconds). Sliding window would be more accurate but adds
+  bookkeeping for no functional gain at this scope.
+- Per-client key = `client. host` from request. For tests we override via
+  the X-Test-Client header so concurrent tests don't share buckets.
+
+Atomicity: the counter is incremented inside the same handler that returns
+429, so two concurrent requests may both pass when the last token is consumed
+within microseconds of each other (a tiny race), but never will a client
+exceed `limit + 1` requests in a window — acceptable.
+"""
+from __future__ import annotations
+
+import threading
+import time
+from collections import deque
+from dataclasses import dataclass
+
+from fastapi import HTTPException, Request, status
+
+
+@dataclass
+class _Bucket:
+    timestamps: deque  # float seconds
+
+
+class RateLimiter:
+    def __init__(self, limit: int, window_seconds: float) -> None:
+        self.limit = limit
+        self.window_seconds = float(window_seconds)
+        self._buckets: dict[str, _Bucket] = {}
+        self._lock = threading.Lock()
+
+    def _client_key(self, request: Request) -> str:
+        # Test override: lets tests run in parallel without sharing buckets.
+        test = request.headers.get("x-test-client")
+        if test:
+            return f"test:{test}"
+        # In production: client.host is the only reliable client identifier
+        # without proxy headers. For real deployment behind a proxy, parse
+        # X-Forwarded-For — out of scope for this minimal spec.
+        if request.client and request.client.host:
+            return request.client.host
+        return "anonymous"
+
+    def check(self, request: Request) -> None:
+        now = time.monotonic()
+        key = self._client_key(request)
+        with self._lock:
+            bucket = self._buckets.get(key)
+            if bucket is None:
+                bucket = _Bucket(timestamps=deque())
+                self._buckets[key] = bucket
+            # Drop timestamps outside the window.
+            cutoff = now - self.window_seconds
+            while bucket.timestamps and bucket.timestamps[0] <= cutoff:
+                bucket.timestamps.popleft()
+            if len(bucket.timestamps) >= self.limit:
+                retry_after = max(
+                    1, int(self.window_seconds - (now - bucket.timestamps[0])) + 1
+                )
+                raise HTTPException(
+                    status_code=status.HTTP_429_TOO_MANY_REQUESTS,
+                    detail="Rate limit exceeded",
+                    headers={"Retry-After": str(retry_after)},
+                )
+            bucket.timestamps.append(now)
+
+    def reset(self) -> None:
+        """For tests: clear all buckets."""
+        with self._lock:
+            self._buckets.clear()

+## 架構決策（每條都有 SOP 依據）
+
+### 1. Sync SQLAlchemy，非 async
+
+**Why**: `trial-and-error/references/by-category/fastapi-async-session-pitfalls.md`（Cycle 539, 教訓 37）歸納 async session 三類坑：
+- **stale read**: `yield Depends(get_db)` 跨越整個請求生命週期 + identity map 快取 → 外部服務改 DB 後查詢返回舊資料
+- **deadlock**: `async def` endpoint 內用同步 `sessionmaker` + `.query()` → 100 concurrent requests 直接凍結 event loop
+- **DetachedInstanceError**: commit 後懶加載 relationships 拋例外（須設 `expire_on_commit=False`）
+
+且 `fastapi-sync-vs-async-pattern-20260727.md` 證明 Quote API 72 個 sync tests 達到 production quality，async 在本 scope 沒好處。
+
+**Pick**: `sessionmaker(bind=engine, autoflush=False, expire_on_commit=False)` + sync `def` endpoints。
+
+### 2. Rate limit 用 in-memory fixed-window + module singleton
+
+**Why**:
+- 規格沒指定 distributed/Redis，in-memory 對單機部署夠用、零外部依賴
+- Fixed window 比 sliding window 簡單且對本場景正確（沒要求絕對精準限流）
+- Module singleton `default_limiter` 讓 middleware 直接呼叫，免 dependency_overrides 設定（簡化測試）
+
+**取捨**：
+- 多 process 部署（gunicorn workers > 1）會繞過限流 → 真要 distributed rate limit 應改 Redis token bucket
+- 客戶端識別用 `client.host` + `X-Test-Client` 測試 header；生產環境若在 proxy 後須解析 `X-Forwarded-For`（不在本 spec 範圍）

+### 3. Rate limit middleware 是 `async def` + `try_consume()`，不 raise HTTPException
+
