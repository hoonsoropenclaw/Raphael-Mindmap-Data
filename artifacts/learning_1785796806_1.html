import asyncio
from async_crawler import AsyncCrawler, CrawlerConfig, RetryPolicy

async def main():
    config = CrawlerConfig(
        max_concurrency=8,
        rate_per_sec=10.0,
        max_pages=50,
        timeout_sec=15.0,
        cache_max_size=1024,
        cache_disk_path="./cache.jsonl",
        retry_policy=RetryPolicy(max_attempts=5, base_delay=0.5, max_delay=30.0, jitter=0.5),
    )

    async with AsyncCrawler(config) as crawler:
        crawler.on_page = lambda p: print(p.title, p.url)
        await crawler.crawl(["https://example.com", "https://example.org"])

asyncio.run(main())
