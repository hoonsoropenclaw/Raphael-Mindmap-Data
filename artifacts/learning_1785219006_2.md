+"""Clock abstraction.
+
+The rate limiter needs to know "how much time has passed" between requests.
+Using ``time.monotonic()`` directly is fine in production, but it makes tests
+flaky (timing assertions get racy under load).
+
+By injecting a clock, tests can return whatever timestamps they want and
+exercise edge cases like "request at exactly bucket refill boundary".
+"""
+
+from __future__ import annotations
+
+import time
+from typing import Callable, Protocol
+
+
+class Clock(Protocol):
+    """A monotonic clock returning seconds as float."""
+
+    def now(self) -> float: ...
+
+
+class SystemClock:
+    """Production clock — wraps :func:`time.monotonic`."""
+
+    def now(self) -> float:
+        return time.monotonic()
+
+
+class FrozenClock:
+    """Test clock — returns a value that only advances when :meth:`advance` is called.
+
+    >>> c = FrozenClock(start=10.0)
+    >>> c.now()
+    10.0
+    >>> c.advance(2.5)
+    12.5
+    >>> c.now()
+    12.5
+    """
+
+    def __init__(self, start: float = 0.0) -> None:
+        self._now = float(start)
+
+    def now(self) -> float:
+        return self._now
+
+    def advance(self, seconds: float) -> None:
+        self._now += float(seconds)
+
+    def set(self, value: float) -> None:
+        self._now = float(value)
+
+
+def make_clock(kind: str = "system") -> Clock:
+    """Factory used by :class:`Settings`.
+
+    ``kind == "system"`` returns :class:`SystemClock`; anything else returns
+    :class:`FrozenClock` (useful for ad-hoc REPL experiments).
+    """
+    if kind == "system":
+        return SystemClock()
+    return FrozenClock()
+
+
+ClockFactory = Callable[[], Clock]

+"""Environment-driven configuration.
+
+Why env-driven? Because:
+
+1. The repo is a learning exercise; we don't want a config file committed.
+2. It keeps ``create_app()`` callable with zero args in tests, while still
+   letting a real operator tune behaviour via env vars.
+"""
+
+from __future__ import annotations
+
+import os
+from dataclasses import dataclass
+
+
+def _env_int(name: str, default: int) -> int:
+    raw = os.environ.get(name)
+    if raw is None or raw == "":
+        return default
+    try:
+        return int(raw)
+    except ValueError as exc:
+        raise ValueError(f"env {name}={raw!r} is not an int") from exc
+
+
+def _env_str(name: str, default: str) -> str:
+    raw = os.environ.get(name)
+    return default if raw is None or raw == "" else raw
+
+
+@dataclass(frozen=True)
+class Settings:
+    """All knobs the service exposes, with safe defaults for tests.

+"""SQLite connection lifecycle and schema bootstrap.
+
+We deliberately use stdlib ``sqlite3`` rather than SQLAlchemy: the data shape
+is trivial (one table, five columns), and avoiding an ORM dependency keeps
+the surface area auditable in a learning exercise. Every connection opens
+with ``check_same_thread=False`` because FastAPI's threadpool can hand a
+request to any worker, then enables ``row_factory=sqlite3.Row`` so callers
+can use dict-like access without an extra mapping layer.
+"""
+
+from __future__ import annotations
+
+import sqlite3
+from contextlib import contextmanager
+from typing import Iterator
+
+
+SCHEMA_SQL = """
+CREATE TABLE IF NOT EXISTS todos (
+    id          INTEGER PRIMARY KEY AUTOINCREMENT,
+    title       TEXT    NOT NULL,
+    completed   INTEGER NOT NULL DEFAULT 0,
+    created_at  TEXT    NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ', 'now')),
+    updated_at  TEXT    NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))
+);
+
+CREATE INDEX IF NOT EXISTS idx_todos_completed ON todos(completed);
+CREATE INDEX IF NOT EXISTS idx_todos_created_at ON todos(created_at);
+"""
+
+
+def open_connection(database_path: str) -> sqlite3.Connection:
+    """Open a SQLite connection with sane defaults.
+
+    - ``row_factory=sqlite3.Row`` so callers can do ``row["title"]``.
+    - ``PRAGMA foreign_keys=ON`` so future FK constraints are honoured.
+    - ``PRAGMA journal_mode=WAL`` is *not* enabled because the test suite uses
+      ``:memory:`` which doesn't support WAL; production code reading a file
+      DB can layer WAL on later if needed.
+
+    Caller owns the connection and must close it (or use :func:`connection`).
+    """
+    conn = sqlite3.connect(
+        database_path,
+        check_same_thread=False,
+        isolation_level=None,  # autocommit; we use explicit BEGIN/COMMIT
+    )
+    conn.row_factory = sqlite3.Row
+    conn.execute("PRAGMA foreign_keys = ON")
+    return conn
+
+
+def initialise_schema(conn: sqlite3.Connection) -> None:
+    """Create tables/indexes if they don't exist. Idempotent."""
+    conn.executescript(SCHEMA_SQL)
+
+
+@contextmanager
+def connection(database_path: str) -> Iterator[sqlite3.Connection]:
+    """Context-managed connection that closes on exit, even on error.
+
+    Use as::
+
+        with connection(":memory:") as conn:
+            ...
+    """
+    conn = open_connection(database_path)
+    try:
+        initialise_schema(conn)
+        yield conn
+    finally:
+        conn.close()

+"""Repository — pure data-access functions over a sqlite connection.
+
+Why a hand-rolled layer instead of SQLAlchemy Core?
+
+- The schema is one table. SQLAlchemy's machinery costs more lines than the
+  SQL itself.
+- Keeping SQL visible (not hidden behind a DSL) makes the learning exercise
+  honest about what's actually happening on the wire.

+  in-memory connection without monkey-patching globals.
+
+All functions use ``sqlite3.Row`` for dict-like access and return either
+a single :class:`TodoOut` or a list of them. They never raise HTTP exceptions;
+HTTP-layer errors come from the routes module.
+"""
+

+
+import sqlite3

+from typing import List, Optional
+

+
+
+def _row_to_todo(row: sqlite3.Row) -> TodoOut:
+    """Map a DB row to the API schema.
+
+    ``completed`` is stored as INTEGER (0/1) but exposed as a Python bool.
+    Timestamps come back as ISO-8601 strings; pydantic parses them.
+    """
+    return TodoOut(
+        id=row["id"],
+        title=row["title"],
+        completed=bool(row["completed"]),
+        created_at=datetime.fromisoformat(row["created_at"].replace("Z", "+00:00")),
+        updated_at=datetime.fromisoformat(row["updated_at"].replace("Z", "+00:00")),
+    )
+
+
+def list_todos(conn: sqlite3.Connection) -> List[TodoOut]:
+    """Return all todos, newest first."""
+    cur = conn.execute(
+        "SELECT id, title, completed, created_at, updated_at "
+        "FROM todos ORDER BY created_at DESC, id DESC"
+    )
+    return [_row_to_todo(row) for row in cur.fetchall()]

+# todo_api
+
+A small FastAPI Todo service backed by SQLite with a token-bucket rate limiter.
+
+## Quick start
+
+```bash
+# Run the API server (defaults to :memory: DB — set TODO_API_DB for persistence)
+python3 -m uvicorn todo_api.main:create_app --factory --host 127.0.0.1 --port 8000
+
+# In another terminal:
+curl -X POST http://127.0.0.1:8000/todos \
+  -H "Content-Type: application/json" \
+  -d '{"title": "buy milk"}'
+```
+
+## Endpoints
+
+| Method | Path             | Status | Description                          |
+|--------|------------------|--------|--------------------------------------|
+| GET    | `/health`        | 200    | Liveness probe (no DB hit)           |
+| GET    | `/todos`         | 200    | List all todos, newest first         |
+| POST   | `/todos`         | 201    | Create a todo                        |
+| GET    | `/todos/{id}`    | 200    | Fetch a todo by id                   |
+| PATCH  | `/todos/{id}`    | 200    | Partial update (title and/or done)   |
+| DELETE | `/todos/{id}`    | 204    | Delete a todo                        |
+
+## Configuration
+
+All knobs read from environment variables, with safe defaults:
+
+| Variable                        | Default              | Meaning                                |
+|---------------------------------|----------------------|----------------------------------------|
+| `TODO_API_DB`                   | `:memory:`           | SQLite path (use a file for persistence)|
+| `TODO_API_RATE_CAPACITY`        | `10`                 | Max tokens per bucket                  |
+| `TODO_API_RATE_REFILL_PER_SEC`  | `2`                  | Tokens refilled per second             |
+| `TODO_API_RATE_KEY_HEADER`      | `x-forwarded-for`    | Header used to scope the bucket        |
+
+## Running tests
+
+```bash
+cd todo_api
+python3 -m pytest tests/ -v
+```
+
+43 tests cover the rate limiter (10), repository (11), middleware (6), endpoints (14),
+and a full-lifecycle smoke test (2).
+
+## Architecture
+
+```
+src/todo_api/
+  __init__.py     # public re-exports
+  clock.py        # clock abstraction (System + Frozen) for testability
+  settings.py     # env-driven config dataclass
+  db.py           # sqlite connection lifecycle + schema bootstrap
+  schemas.py      # pydantic request/response models
+  repository.py   # pure data-access functions over sqlite
+  rate_limit.py   # token-bucket algorithm
+  middleware.py   # pure ASGI middleware wrapping the limiter
+  routes/         # HTTP endpoints (thin layer)
+    __init__.py
+  main.py         # create_app factory — single entry point
+```
+
+Key design choices:
+
+- **stdlib `sqlite3`** instead of SQLAlchemy — the data shape is trivial.
+- **Token-bucket rate limiter** instead of a sliding window — same burst tolerance,
+  smoother behaviour under steady load (see `rate_limit.py` docstring).
+- **Pure ASGI middleware** instead of `@app.middleware("http")` — gives us precise
+  control over `Retry-After` and `X-RateLimit-*` headers without Starlette's
+  streaming-response wrapping interfering.
