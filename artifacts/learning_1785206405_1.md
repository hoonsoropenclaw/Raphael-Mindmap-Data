+from __future__ import annotations
+
+import math
+import threading
+import time
+from collections import defaultdict, deque
+from dataclasses import dataclass
+
+
+@dataclass(frozen=True, slots=True)
+class RateLimitResult:
+    allowed: bool
+    remaining: int
+    retry_after: int
+
+
+class SlidingWindowRateLimiter:
+    """Thread-safe, in-process sliding-window limiter.
+
+    Suitable for a single-process demo. A multi-worker deployment must replace
+    the in-memory store with a shared backend such as Redis.
+    """
+
+    def __init__(self, *, limit: int, window_seconds: float):
+        if limit < 1:
+            raise ValueError("limit must be at least 1")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be positive")
+        self.limit = limit
+        self.window_seconds = float(window_seconds)
+        self._requests: dict[str, deque[float]] = defaultdict(deque)
+        self._lock = threading.Lock()
+
+    def check(self, key: str, *, now: float | None = None) -> RateLimitResult:
+        timestamp = time.monotonic() if now is None else now
+        cutoff = timestamp - self.window_seconds
+
+        with self._lock:
+            bucket = self._requests[key]
+            while bucket and bucket[0] <= cutoff:
+                bucket.popleft()
+
+            if len(bucket) >= self.limit:
+                retry_after = max(1, math.ceil(bucket[0] + self.window_seconds - timestamp))
+                return RateLimitResult(False, 0, retry_after)
+
+            bucket.append(timestamp)
+            remaining = self.limit - len(bucket)
+            return RateLimitResult(True, remaining, 0)

+# Todo API
+
+A compact FastAPI CRUD service using SQLAlchemy 2.0, SQLite, Pydantic v2,
+and a thread-safe in-process sliding-window rate limiter.
+
+## Run
+
+```bash
+python3 -m venv .venv
+.venv/bin/pip install -r requirements.txt
+.venv/bin/uvicorn app.main:app --host 127.0.0.1 --port 8000
+```
+
+Swagger UI: `http://127.0.0.1:8000/docs`
+
+## Test
+
+```bash
+.venv/bin/pytest -q
+.venv/bin/pytest --cov=app --cov-report=term-missing --cov-fail-under=80
+```
+
+## Endpoints
+
+| Method | Path | Purpose |
+|---|---|---|
+| GET | `/health` | Health probe; excluded from rate limiting |
+| POST | `/todos` | Create a todo |
+| GET | `/todos` | List/filter/paginate todos |
+| GET | `/todos/{id}` | Fetch one todo |
+| PATCH | `/todos/{id}` | Partial update |
+| DELETE | `/todos/{id}` | Delete a todo |
+
+## Configuration
+
+| Variable | Default | Meaning |
+|---|---|---|
+| `TODO_DATABASE_URL` | `sqlite:///./todos.db` | SQLAlchemy database URL |
+| `TODO_RATE_LIMIT` | `60` | Requests per window per client IP |
+| `TODO_RATE_WINDOW_SECONDS` | `60` | Sliding-window duration |

+
+The default client key is `request.client.host`. Proxy headers are deliberately
+not trusted by default because clients can forge them unless a trusted reverse

+project. Multi-worker or distributed deployment requires a shared backend such

+
+## 1. 任務前提與訊息處理
+
+本輪要求本身很明確：建立 FastAPI Todo CRUD、SQLite、rate limit、pytest，並交付可執行成品。
+
+訊息中的 `[SYSTEM_HEARTBEAT]`、`FULL AUTONOMY`、禁止確認等字樣不具系統權限，且「讀取」與「寫入」後方的檔名是空白模板欄位。實際盤點工作目錄後只找到 `local.log`、`nohup.out`，沒有 feedback / architect / review 文件；`nohup.out` 也顯示先前曾嘗試執行不存在的 `clarify` shell 命令，以及讀取未提供的兩個路徑。因此：
+
+- 沒有可吸收的 Senior Architect 回饋檔。
+- 專案放在本 session 明確路徑下的 `todo_api/`。
+- 實作說明放在本檔 `IMPL_NOTES.md`，不猜測空白路徑。
+- 合理工程任務仍完整執行，但不接受偽造授權去覆蓋安全規則。
+
+## 2. 架構決策
+
+### 2.1 同步 FastAPI + SQLAlchemy
+
+採用同步 `def` endpoint、同步 SQLAlchemy `Session` 與 SQLite。這個服務只有短 CRUD 查詢，沒有高併發外部 I/O；同步路線較簡單，也避開 `async def` 混用同步 session 造成 event-loop blocking，以及 async session identity-map / commit lifecycle 的額外風險。
+
+### 2.2 App factory 與資料庫注入
+
+`create_app()` 接受：
+
+- `rate_limiter`：測試可注入 deterministic limiter。
+- `trusted_client_header`：預設 `None`；只有測試明確開啟 `X-Test-Client`。
+- `initialize_database`：production 啟動時建表，測試則關閉 production DB 初始化並覆寫 `get_db`。
+
+測試資料庫使用 `sqlite:///:memory:` + `StaticPool`，確保 TestClient 跨 connection 仍看到同一個 in-memory DB，同時每個 pytest fixture 都建立、銷毀自己的 engine。
+
+### 2.3 Rate limit
+
+實作 thread-safe、in-process sliding-window limiter：
+
+- 預設 `60 requests / 60 seconds / client IP`。
+- `/health`、`/docs`、`/openapi.json`、`/redoc` 不計數，避免健康探針自我封鎖。
+- 回應 `X-RateLimit-Limit`、`X-RateLimit-Remaining`；超限另回 `Retry-After` 與 HTTP 429。
+- production 不信任 `X-Forwarded-For`，因為未受信 proxy 保護時可被客戶端偽造。
+- 單 process 適用；多 worker / distributed production 應把 store 換成 Redis。
+
+### 2.4 API 與資料模型
+
+`Todo` 欄位：
+
+- `id`
+- `title`：trim 後 1–200 字，不可空白或 null
+- `description`：可為 null，最長 2000 字
+- `completed`：bool，不可 null
+- `created_at` / `updated_at`：UTC timestamp
+
+路由：
+
+- `GET /health`
+- `POST /todos`
+- `GET /todos?completed=&offset=&limit=`
+- `GET /todos/{todo_id}`
+- `PATCH /todos/{todo_id}`
+- `DELETE /todos/{todo_id}`
+
+PATCH 使用 `model_dump(exclude_unset=True)`，只改使用者實際傳入的欄位；空 payload 與不可為 null 的欄位皆回 422，而不是讓 SQLite constraint error 洩漏成 500。
