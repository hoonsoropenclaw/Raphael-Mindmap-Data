+"""Per-client token-bucket rate limiter, in-memory.
+
+Design choices:
+- In-memory dict keyed by client identifier (default = IP address). Suitable
+  for single-process deployments. For multi-worker / multi-host, swap the
+  `buckets` dict for a Redis-backed implementation.
+- Token-bucket via `slowapi` is the lazy choice, but pulling in `slowapi`
+  adds another dep. A 30-line implementation gives full control and is
+  trivially testable.
+- We expose a FastAPI dependency `RateLimiter(...)` so tests can construct
+  per-test limits and per-test bucket stores.
+"""
+from __future__ import annotations
+
+import time
+from dataclasses import dataclass, field
+from typing import Callable, Dict
+
+from fastapi import Depends, HTTPException, Request, status
+
+
+@dataclass
+class Bucket:
+    """A single client's token bucket.
+
+    `capacity` = max tokens (burst size). `refill_rate` = tokens per second.
+    Each request consumes 1 token. When tokens run out, request is rejected.
+    """
+    capacity: float
+    refill_rate: float  # tokens / second
+    tokens: float = field(init=False)
+    last_refill: float = field(init=False)
+
+    def __post_init__(self) -> None:
+        self.tokens = self.capacity
+        self.last_refill = time.monotonic()
+
+    def try_consume(self, cost: float = 1.0) -> bool:
+        """Return True if `cost` tokens were available and consumed."""
+        now = time.monotonic()
+        elapsed = now - self.last_refill
+        # Refill: capped at capacity.
+        self.tokens = min(self.capacity, self.tokens + elapsed * self.refill_rate)
+        self.last_refill = now
+        if self.tokens >= cost:
+            self.tokens -= cost
+            return True
+        return False
+
+    def retry_after(self) -> float:
+        """Seconds until the next single token is available."""
+        if self.tokens >= 1.0:
+            return 0.0
+        deficit = 1.0 - self.tokens
+        return deficit / self.refill_rate if self.refill_rate > 0 else float("inf")
+
