+const initial = { missions: seed, selected: 1, filter: 'all', search: '', events: ['系統同步完成 · 09:42'], undo: [] };
+function reducer(state, action) {
+  const checkpoint = () => [...state.undo, state.missions].slice(-10);
+  switch (action.type) {
+    case 'select': return { ...state, selected: action.id };
+    case 'filter': return { ...state, filter: action.value };
+    case 'search': return { ...state, search: action.value };
+    case 'advance': return { ...state, undo: checkpoint(), missions: state.missions.map(m => m.id === action.id ? { ...m, progress: Math.min(100, m.progress + 10), status: m.progress >= 90 ? 'done' : 'active' } : m), events: [`任務 #${action.id} 推進 10% · 現在`, ...state.events].slice(0, 5) };
+    case 'add': {
+      const id = Math.max(0, ...state.missions.map(m => m.id)) + 1;
+      const mission = { id, title: action.title, owner: 'YOU', priority: 'normal', status: 'queue', progress: 0 };
+      return { ...state, undo: checkpoint(), missions: [...state.missions, mission], selected: id, events: [`新增任務 #${id} · 現在`, ...state.events].slice(0, 5) };
+    }
+    case 'remove': return { ...state, undo: checkpoint(), missions: state.missions.filter(m => m.id !== action.id), selected: null };
+    case 'undo': {
+      if (!state.undo.length) return state;
+      return { ...state, missions: state.undo.at(-1), undo: state.undo.slice(0, -1), events: ['已復原前次異動 · 現在', ...state.events].slice(0, 5) };
