+"""Weather API client.
+
+Supports OpenWeatherMap geocoding + current weather.
+Falls back to a deterministic mock when MOCK_WEATHER=1 or no API key.
+"""
+
+from __future__ import annotations
+
+import os
+from dataclasses import dataclass
+from typing import Optional
+
+import httpx
+
+
+OWM_BASE = "https://api.openweathermap.org"
+TIMEOUT = 10.0
+
+
+@dataclass(frozen=True)
+class Weather:
+    city: str
+    country: str
+    temp_c: float
+    feels_like_c: float
+    humidity: int
+    description: str
+    wind_mps: float
+
+
+class WeatherError(RuntimeError):
+    """Raised when the weather API cannot produce a result."""
+
+
+def _mock_weather(city: str) -> Weather:
+    """Deterministic mock keyed off the city name so tests are stable."""
+    h = abs(hash(city)) % 30
+    return Weather(
+        city=city,
+        country="MOCK",
+        temp_c=15.0 + h * 0.4,
+        feels_like_c=14.0 + h * 0.4,
+        humidity=40 + h,
+        description="partly cloudy (mock)",
+        wind_mps=2.5,
+    )
+
+
+def _normalize_lang(lang: str) -> str:
+    lang = (lang or "zh_tw").strip().lower()
+    return "zh_tw" if lang in ("zh", "zh_tw", "zh-tw", "tw") else lang
+
+
+def _http() -> httpx.Client:
+    return httpx.Client(base_url=OWM_BASE, timeout=TIMEOUT)
+
+
+def geocode(city: str, api_key: str, lang: str = "zh_tw") -> tuple[float, float, str, str]:
+    """Resolve city -> (lat, lon, resolved_name, country).
+
+    Returns coordinates that OpenWeatherMap recognizes.
+    Raises WeatherError if the city cannot be resolved.
+    """
+    if not city.strip():
+        raise WeatherError("city name is empty")
+    lang = _normalize_lang(lang)
+    params = {"q": city, "limit": 1, "appid": api_key, "lang": lang}
+    with _http() as c:
+        r = c.get("/geo/1.0/direct", params=params)
+    if r.status_code != 200:
+        raise WeatherError(f"geocoding HTTP {r.status_code}")
+    data = r.json()
+    if not data:
+        raise WeatherError(f"city not found: {city!r}")
+    first = data[0]
+    return float(first["lat"]), float(first["lon"]), str(first.get("local_names", {}).get(lang, first["name"])), str(first.get("country", ""))
