+export class EventStore {
+  constructor({ maxEvents = 1000 } = {}) {
+    this.maxEvents = maxEvents;
+    this.events = [];
+    this.listeners = new Set();
+    this.sequence = 0;
+  }
+
+  publish(input) {
+    const metric = String(input.metric ?? '').trim();
+    const source = String(input.source ?? '').trim();
+    const value = Number(input.value);
+    if (!metric || metric.length > 80) throw new Error('metric must be 1-80 characters');
+    if (!source || source.length > 80) throw new Error('source must be 1-80 characters');
+    if (!Number.isFinite(value)) throw new Error('value must be a finite number');
+    const event = Object.freeze({
+      id: String(++this.sequence), metric, source, value,
+      timestamp: new Date().toISOString()
+    });
+    this.events.push(event);
+    if (this.events.length > this.maxEvents) this.events.shift();
+    for (const listener of this.listeners) listener(event);
+    return event;
+  }
+
+  query({ metric, source, limit = 50 } = {}) {
+    const safeLimit = Math.min(Math.max(Number(limit) || 50, 1), 200);
+    return this.events
+      .filter((e) => (!metric || e.metric === metric) && (!source || e.source === source))
+      .slice(-safeLimit).reverse();
+  }
+
+  asyncIterator(filter = {}) {
+    const queue = [];
+    let pending;
+    let active = true;
+    const accepts = (e) => (!filter.metric || e.metric === filter.metric) && (!filter.source || e.source === filter.source);
+    const listener = (event) => {
+      if (!active || !accepts(event)) return;
+      if (pending) { const resolve = pending; pending = undefined; resolve({ value: { eventAdded: event }, done: false }); }
+      else queue.push(event);
+    };
+    this.listeners.add(listener);
+    return {
+      next: () => {
+        if (!active) return Promise.resolve({ done: true });
+        if (queue.length) return Promise.resolve({ value: { eventAdded: queue.shift() }, done: false });
+        return new Promise((resolve) => { pending = resolve; });
+      },
+      return: () => {
+        active = false; this.listeners.delete(listener);
+        if (pending) pending({ done: true });
+        return Promise.resolve({ done: true });
+      },
+      throw: (error) => { active = false; this.listeners.delete(listener); return Promise.reject(error); },
+      [Symbol.asyncIterator]() { return this; }
+    };
+  }

+# GraphQL 即時資料查詢系統：實作與除錯紀錄
+
+## 成品架構
+
+- HTTP GraphQL endpoint：`POST /graphql`
+- WebSocket GraphQL Subscription：`ws://127.0.0.1:4000/graphql`
+- 健康檢查：`GET /health`
+- 零編譯展示頁：`GET /`，原始檔為 `demo.html`
+- 記憶體事件儲存：最多 1000 筆，查詢最多回傳 200 筆
+- GraphQL 操作：
+  - Query：`health`、`events(metric, source, limit)`
+  - Mutation：`publishEvent(input)`
+  - Subscription：`eventAdded(metric, source)`
+
+## 技術選擇
+
+採 Node.js + 官方 `graphql` 執行器 + `graphql-ws` + `ws`。HTTP 層使用 Node 內建 `http`，避免引入 Express/Apollo Server 的額外依賴。Subscription 使用標準 `graphql-transport-ws` 子協定，瀏覽器可以直接訂閱。
+
+## 安全與韌性
+
+- HTTP / WebSocket payload 上限 64 KiB
+- GraphQL query 長度上限 10,000 字元
+- schema validation 後才執行
+- metric/source 長度與空值驗證
+- value 必須為有限數字
+- 查詢 limit 強制限定 1-200
+- 不回傳 stack trace
+- HTML 加入 X-Frame-Options、Referrer-Policy
+- 無密鑰、無 `.env`、npm audit 為 0 vulnerability
+
+## 除錯過程
+
+首次 `npm test`：2/3 通過。失敗原因不是業務輸出錯誤，而是 GraphQL 回傳物件使用 null prototype；Node `assert.deepStrictEqual` 將它與一般 object 視為不同 prototype，即使欄位和值相同仍失敗。
+
+修正：改為逐欄位斷言，測試真正的 API contract，而非 GraphQL 執行器內部的 object prototype。修正後 3/3 測試通過。
+
