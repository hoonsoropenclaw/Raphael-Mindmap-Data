+def _load_dotenv(dotenv_path: Path) -> None:
+    """Minimal .env loader — avoids extra dependency on python-dotenv.
+
+    Format: KEY=value per line, # for comments, blank lines ignored, no quoting magic.
+    """
+    if not dotenv_path.is_file():
+        return
+    for raw_line in dotenv_path.read_text(encoding="utf-8").splitlines():
+        line = raw_line.strip()
+        if not line or line.startswith("#"):
+            continue
+        if "=" not in line:
+            continue
+        key, _, value = line.partition("=")
+        key = key.strip()
+        value = value.strip().strip('"').strip("'")
+        # Only set if not already in process env (env vars win over .env file)
+        os.environ.setdefault(key, value)
+
+
+@dataclass(frozen=True)
+class Settings:
+    telegram_token: str
+    weather_timeout_seconds: float
+    default_city: str
+    log_level: str
+
+

+    """Build Settings from env. Raises a clear error if the bot token is missing."""
