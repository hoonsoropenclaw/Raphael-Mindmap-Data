+
+We use SQLAlchemy 2.x's async API together with aiosqlite. The engine and
+session factory are module-level singletons that read their target URL
+from `Settings`. The `get_db` dependency yields a session per request and
+is responsible for committing on success / rolling back on error.
+"""
+from __future__ import annotations
+
+from collections.abc import AsyncIterator
+
+from sqlalchemy.ext.asyncio import (
+    AsyncSession,
+    async_sessionmaker,
+    create_async_engine,
+)
+from sqlalchemy.orm import DeclarativeBase
+
+from app.config import get_settings
+
+
+class Base(DeclarativeBase):
+    """Declarative base for all ORM models."""
+
+

+# intentionally don't use a connection pool with SQLite because aiosqlite
+# serialises writes; the default NullPool behaviour is fine for tests.
+_settings = get_settings()
+engine = create_async_engine(
+    _settings.database_url,
+    echo=False,
+    future=True,
+)

+    engine,
+    expire_on_commit=False,
+    class_=AsyncSession,
+)
+
+
+async def get_db() -> AsyncIterator[AsyncSession]:
+    """FastAPI dependency that yields an `AsyncSession` per request.
+
+    The session is committed if the route handler succeeded, otherwise
+    the in-flight transaction is rolled back. The session itself is
+    always closed so connections are returned to the pool.
+    """
+    async with AsyncSessionLocal() as session:
+        try:
+            yield session
+            await session.commit()
+        except Exception:
+            await session.rollback()
+            raise
+        finally:
+            await session.close()

+"""In-process sliding-window rate limiter.
+
+Design
+------
+We keep a per-key deque of monotonic timestamps. On each request we drop
+timestamps that fell out of the window, then check the remaining count.
+This is exact (no "leaky bucket" approximation) and cheap for the small
+request volumes we expect in this demo. State is process-local; if the
+process is restarted the window resets. For a multi-replica deployment
+you would back this with Redis, but that is out of scope for SQLite.
+
+The limiter is a single module-level instance so it can be reset between
+tests without re-wiring the FastAPI app.
+"""
+from __future__ import annotations
+
+import asyncio
+import time
+from collections import defaultdict, deque
+from collections.abc import Iterable
+from typing import Deque
+
+from fastapi import HTTPException, Request, status
+
+from app.config import get_settings
+
+
+class SlidingWindowRateLimiter:
+    """Simple per-key sliding window counter."""
+
+    def __init__(self, *, max_requests: int, window_seconds: float) -> None:
+        if max_requests < 1:
+            raise ValueError("max_requests must be >= 1")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be > 0")
+        self.max_requests = max_requests
+        self.window_seconds = float(window_seconds)
+        self._buckets: dict[str, Deque[float]] = defaultdict(deque)
+        self._lock = asyncio.Lock()
+
+    async def check(self, key: str) -> tuple[bool, int]:
+        """Record a hit for `key` and return `(allowed, retry_after_seconds)`.
+
+        When the request is allowed, `retry_after_seconds` is 0. When it
+        is denied, the value is how long the caller should wait before
+        the oldest hit in their window expires.
+        """
+        now = time.monotonic()
+        cutoff = now - self.window_seconds
+        async with self._lock:
+            bucket = self._buckets[key]
+            # Drop expired entries from the left.
+            while bucket and bucket[0] <= cutoff:
+                bucket.popleft()
+            if len(bucket) >= self.max_requests:
+                retry_after = max(1, int(bucket[0] + self.window_seconds - now) + 1)
+                return False, retry_after
+            bucket.append(now)
+            return True, 0
+
+    def reset(self) -> None:
+        """Clear all state. Intended for tests."""
+        self._buckets.clear()
+
+    def keys(self) -> Iterable[str]:
+        """Return all keys currently being tracked (test introspection)."""
+        return list(self._buckets.keys())
+
+

+# `_limiter` directly.

+    window_seconds=_settings.rate_limit_window_seconds,
+)
