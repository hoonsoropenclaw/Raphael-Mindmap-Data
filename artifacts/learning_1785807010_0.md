+# ---------- 視覺回歸核心 ----------
+def pixel_diff_pct(baseline_path: Path, current_path: Path) -> tuple[float, Image.Image | None]:
+    """回傳 (diff_pct, diff_image_or_None)。baseline 不存在時回傳 (0.0, None)。"""

+        curr = curr.resize(base.size)

+    return (pct, enhanced)

+## 觀察到的差異
