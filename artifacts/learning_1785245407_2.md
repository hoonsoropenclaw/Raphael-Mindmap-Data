+/**
+ * usePersistentReducer — wraps useReducer with localStorage persistence.
+ *
+ * Design choices:
+ *  - HYDRATE only runs once on mount (via the empty dep array), not on
+ *    every state change; this avoids fighting the storage layer.
+ *  - We persist AFTER every change but inside useEffect, never inside
+ *    the dispatch path — keeping the reducer pure.
+ *  - We swallow JSON parse errors silently: corrupted storage should
+ *    not crash the app, only fall back to initialState.
+ *  - The storage key is namespaced so multiple apps on the same origin
+ *    do not collide.
+ */
+
+const STORAGE_KEY = 'react-todo::state::v1'
+
+const safeRead = (): TodoState | null => {
+  try {
+    const raw = localStorage.getItem(STORAGE_KEY)
+    if (!raw) return null
+    const parsed = JSON.parse(raw) as TodoState
+    return parsed
+  } catch {
+    return null
+  }
+}
+
+const safeWrite = (state: TodoState): void => {
+  try {
+    localStorage.setItem(STORAGE_KEY, JSON.stringify(state))
+  } catch {
+    // quota exceeded, private mode, or storage disabled — silently ignore
+  }
+}
+
+export function usePersistentReducer(): readonly [TodoState, React.Dispatch<TodoAction>] {
+  const [state, dispatch] = useReducer(todoReducer, initialState)
+
+  // 1) Hydrate once on mount.
+  useEffect(() => {
+    const persisted = safeRead()
+    if (persisted) {
+      dispatch({ type: 'HYDRATE', state: persisted })
+    }
+  }, [])
+
+  // 2) Persist on every change AFTER hydration. We skip the very first
+  //    run (when state still equals initialState and we haven't hydrated
+  //    yet) by relying on the fact that hydrated state ≠ initialState
+  //    unless the persisted state was itself empty.
+  useEffect(() => {
+    safeWrite(state)
+  }, [state])
+
+  return [state, dispatch] as const

+import { useEffect } from 'react'

+/**
+ * useGlobalKey — tiny keyboard-shortcut helper.
+ *
+ * Patterns supported:
+ *   '/'         → focus the new-todo input
+ *   'ctrl+shift+k' (or 'mod+shift+k') → clear all completed
+ *
+ * The hook is intentionally minimal — it adds one document-level listener
+ * on mount and removes it on unmount, and it routes matched combos back to
+ * the supplied callbacks.
+ */
+
+type Combo = string
+
+export interface KeyHandlers {
+  onFocusInput?: () => void
+  onClearCompleted?: () => void
+}
+
+const parse = (e: KeyboardEvent): Combo => {
+  const parts: string[] = []
+  if (e.ctrlKey || e.metaKey) parts.push('mod')
+  if (e.shiftKey) parts.push('shift')
+  if (e.altKey) parts.push('alt')
+  const k = e.key.toLowerCase()
+  if (k !== 'control' && k !== 'shift' && k !== 'meta' && k !== 'alt') {
+    parts.push(k)
+  }
+  return parts.join('+')
+}
+
+export function useGlobalKey(handlers: KeyHandlers): void {
+  const { onFocusInput, onClearCompleted } = handlers
+
+  useEffect(() => {
+    const onKey = (e: KeyboardEvent) => {
+      // Don't hijack typing in an editable element.
+      const tgt = e.target as HTMLElement | null
+      const editable =
+        tgt?.tagName === 'INPUT' ||
+        tgt?.tagName === 'TEXTAREA' ||
+        tgt?.isContentEditable
+      const combo = parse(e)
+
+      if (combo === '/' && !editable) {
+        e.preventDefault()
+        onFocusInput?.()
+        return
+      }
+      if (combo === 'mod+shift+k' && !editable) {
+        e.preventDefault()
+        onClearCompleted?.()
+        return
+      }
+      if (combo === 'escape' && editable) {
+        ;(tgt as HTMLInputElement).blur()
+      }
+    }
+    document.addEventListener('keydown', onKey)
+    return () => document.removeEventListener('keydown', onKey)
+  }, [onFocusInput, onClearCompleted])

+## 3. 架構決策（為什麼這樣切）
+
+### 3.1 三層 Context 切分
+
+`TodoContext.tsx` 把 state、derived、actions、ref 分成 4 個 context，不是塞在一個 context object：
+
+```
+StateContext        → 訂閱者只在 state 變才 re-render
+DerivedContext      → 訂閱者只在 derived（visible[] / stats）變才 re-render
+ActionsContext      → 訂閱者永遠不 re-render（callback 都是 useCallback stable）
+InputRefContext     → inputRef.current 不變就永遠不 re-render
+```
+
+**為什麼要切**：如果把所有東西塞一個 context，任何 state 改變都會 re-render 所有 consumer。常見反模式「為什麼我的 React App 慢」的 root cause 就是這個。
+
+### 3.2 Reducer 是純函數
+
+`todoReducer.ts` 不 import React，不讀 storage，**純 (state, action) → state**：
+- 12 個 unit test 沒有用 React 就能跑（happy-dom 也不需要）
+- 將來要升級 Redux/Zustand 也是單檔改 import 就好
+- 用 `never` exhaustiveness check，新 action 不處理會在 TS 編譯期就爆
+
+### 3.3 Persistence 的兩個 useEffect 分離
+
+`usePersistentReducer.ts` 兩個 effect 分開：
+1. mount 時 hydrate 一次（empty deps）
+2. 之後每次 state 變就 persist
+
+**為什麼不寫在 reducer 裡**：reducer 必須是純函數。Storage I/O 是 side effect，會破壞可測試性。Effect-after-reducer 是 React 推薦的副作用分層。
+
+**為什麼 input 不持久化**：HYDRATE 時主動把 input 清空——input 是 session-local 的「正在輸入中」狀態，重新整理不該還原（會很煩人）。
+
+### 3.4 useGlobalKey 故意精簡
+
+只用 12 行實作 `/`、`Ctrl+Shift+K`、`Esc` 三個組合：
+- 不引進 react-hotkeys / mousetrap（增加 50KB+）
+- 不擋輸入框（`editable` 檢查），避免搶使用者正在打字
