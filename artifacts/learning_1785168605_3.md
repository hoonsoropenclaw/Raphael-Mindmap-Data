+"""SQLite database layer for the todo API.
+
+Design notes:
+- Uses stdlib `sqlite3` (no ORM) to keep dependencies minimal and the
+  schema easy to inspect. For a real product I'd reach for SQLAlchemy,
+  but for a "simple todo API" this is the right level of abstraction.
+- A single `Database` class owns the connection and exposes typed
+  methods (`list_todos`, `create_todo`, `get_todo`, `update_todo`,
+  `delete_todo`). Each method opens its own transaction.
+- `check_same_thread=False` is required because FastAPI may dispatch
+  requests across worker threads.
+- `row_factory = sqlite3.Row` makes results behave like dicts.
+"""
+
+from __future__ import annotations
+
+import sqlite3
+from contextlib import contextmanager
+from typing import Iterator, Optional
+
+SCHEMA = """
+CREATE TABLE IF NOT EXISTS todos (
+    id          INTEGER PRIMARY KEY AUTOINCREMENT,
+    title       TEXT    NOT NULL,
+    done        INTEGER NOT NULL DEFAULT 0,  -- 0 = false, 1 = true (SQLite has no native bool)
+    created_at  TEXT    NOT NULL DEFAULT CURRENT_TIMESTAMP
+);
+"""
+
+
+class Database:
+    def __init__(self, path: str) -> None:
+        self.path = path
+        # One connection per Database instance. FastAPI dependency
+        # injection gives a fresh Database() per request when wired
+        # via `get_db`, so we don't need a connection pool here.
+        self._conn = sqlite3.connect(self.path, check_same_thread=False)
+        self._conn.row_factory = sqlite3.Row
+        self._conn.execute("PRAGMA foreign_keys = ON;")
+        self._conn.execute(SCHEMA)
+        self._conn.commit()
+
+    def close(self) -> None:
+        try:
+            self._conn.close()
+        except sqlite3.ProgrammingError:
+            pass
+
+    @contextmanager
+    def _tx(self) -> Iterator[sqlite3.Connection]:
+        try:
+            yield self._conn
+            self._conn.commit()
+        except Exception:
+            self._conn.rollback()
+            raise
+
+    # ---------- CRUD ----------
+
+    def list_todos(self, limit: int = 100, offset: int = 0) -> list[sqlite3.Row]:
+        cur = self._conn.execute(
+            "SELECT id, title, done, created_at FROM todos "
+            "ORDER BY id DESC LIMIT ? OFFSET ?",

+        )

+"""FastAPI application — todo API with rate limiting.
+
+Rate limiting: uses `slowapi` with an in-memory storage backend

+The default limit is 30 requests/minute per IP. The limit is
+configurable via the `RATE_LIMIT` env var.
+"""
+
+from __future__ import annotations
+
+import os
+from typing import Iterator
+
+from fastapi import Depends, FastAPI, HTTPException, Request, status
+from slowapi import Limiter
+from slowapi.errors import RateLimitExceeded
+from slowapi.middleware import SlowAPIMiddleware
+from slowapi.util import get_remote_address
+from starlette.responses import JSONResponse

+from .db import Database
+from .schemas import TodoCreate, TodoOut, TodoUpdate

+def _rate_limit() -> str:
+    return os.environ.get("RATE_LIMIT", "30/minute")
+
+
+# `get_remote_address` uses the client's IP. Behind a proxy you'd
+# want a wrapper that reads X-Forwarded-For — out of scope here.
+limiter = Limiter(key_func=get_remote_address, default_limits=[_rate_limit()])
+
+
+def create_app(db_path: str | None = None) -> FastAPI:
+    """App factory — `db_path` lets tests pass a temp file."""
+    app = FastAPI(

+        version="0.1.0",
+        description="Simple todo API with SQLite storage and per-IP rate limiting.",
+    )
+    app.state.limiter = limiter
+    app.add_middleware(SlowAPIMiddleware)
+
+    # Default DB path: ./todos.db in CWD, overridable for tests.
+    if db_path is None:

+    def get_db() -> Iterator[Database]:

+# IMPL_NOTES.md — Implementation Notes
+
+## 1. Goal
+Build a small but production-shaped FastAPI service that demonstrates:
+- A clean SQL CRUD layer
+- Request/response validation
+- Per-IP rate limiting
+- A real test suite that exercises both happy paths and edge cases
+
+The task brief said "simple" — so I deliberately did not pull in SQLAlchemy,
+Alembic, or a heavier web stack. The point is to show the same shape you'd
+ship, just smaller.
+
+## 2. Architecture decisions
+
+### 2.1 Why a custom `Database` class instead of an ORM
+- The schema is one table with four columns. An ORM adds setup cost and
+  indirection without buying us anything.
+- Stdlib `sqlite3` is already thread-safe enough for FastAPI's threadpool
+  when paired with `check_same_thread=False` and a fresh `Database()` per
+  request via `Depends(get_db)`.
+- For a real product I would reach for SQLAlchemy + Alembic, but I want
+  this codebase to be readable end-to-end in one sitting.
+
+### 2.2 Why `slowapi` for rate limiting
+- It's the de-facto FastAPI integration for `limits` (the same engine
+  used by Flask-Limiter).
+- The default in-memory backend is enough for a single-process dev
+  server. Swapping to Redis is a one-line change to `storage_uri`.
+- The limit is configurable via `RATE_LIMIT` env var so the test suite
+  can lower it (3/min) to exercise 429 behavior quickly.
+
+### 2.3 Why PATCH (not PUT) for updates
+- Updates are partial — clients might change just `done` or just `title`.
+- PUT implies "replace the whole resource"; PATCH is the correct verb.
+- `TodoUpdate` has all fields optional, and the service only mutates
+  fields the client actually sent.
+
+### 2.4 Why an app factory (`create_app`)
+- Lets the test suite pass a temp DB path and a custom `RATE_LIMIT`
+  per test.
+- The module-level `app = create_app()` is a convenience for
+  `uvicorn app.main:app`; tests don't use it.
+
+## 3. Test isolation gotcha (and the fix)
+
+First run: 12/13 pass, but `test_list_pagination` failed with 429

+applied to *all* tests in that single client fixture.

+counter doesn't reset when a new `TestClient` is created — it only
+resets on `app.state.limiter.reset()`.

+- `client` — `RATE_LIMIT=100/minute` (CRUD tests stay clear)
+- `rate_limited_client` — `RATE_LIMIT=3/minute` (rate-limit test fires fast)

+I also added `app.state.limiter.reset()` per fixture so the counter
+starts at zero for each test, not at whatever the last test left it.
