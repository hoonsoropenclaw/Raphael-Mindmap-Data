+// framework/test-runner.js
+// Parallel Playwright test runner.
+// Key design choices:
+//  - Worker pool sized by --workers (default = CPU/2, min 1, max 8)
+//  - Each worker owns a single Chromium browser + isolated BrowserContext
+//    → tests share zero state, but the browser process is reused (much faster
+//      than launching per-test, much more isolated than sharing a context).
+//  - Test cases are plain async functions (page) => Promise<void> registered
+//    via describe()/it() like Mocha — no DSL lock-in, no codegen.
+//  - Per-test timeout, retry policy (flaky-test tolerance), and per-test
+//    custom viewport override (handy for responsive-design runs).
+//  - Failures do not crash the pool; the worker picks up the next case.
+//  - Reporter streams JSONL events so external tools (CI, dashboards) can
+//    consume results without scraping console output.
+//
+// Usage:
+//   const { defineConfig, run } = require('./test-runner');
+//   run({ file: __filename, config: defineConfig({ baseURL, workers: 4 }) });
+
+'use strict';
+
+const path = require('path');
+const fs = require('fs');
+const os = require('os');
+const { chromium } = require('playwright');
+
+// ---------------------------------------------------------------------------
+// Public DSL
+// ---------------------------------------------------------------------------
+
+const REGISTRY = new Map(); // file -> { suites, config }
+
+function describe(name, fn) {
+  const file = module._currentFile || '__root__';
+  const reg = REGISTRY.get(file) || { suites: [], config: {} };
+  reg.suites.push({ name, file, tests: [] });
+  module._currentSuite = reg.suites[reg.suites.length - 1];
+  try {
+    fn();
+  } finally {
+    module._currentSuite = null;
+  }
+  REGISTRY.set(file, reg);
+}
+
+function it(name, fn, opts = {}) {
+  if (!module._currentSuite) {
+    throw new Error(`it("${name}") called outside describe()`);
+  }
+  module._currentSuite.tests.push({
+    name,
+    fn,
+    timeout: opts.timeout || 30000,
+    retries: opts.retries != null ? opts.retries : 1,
+    viewport: opts.viewport || null,
+    tags: opts.tags || [],
+  });
+}
+

+  return {

+# Playwright Parallel Test Framework
+
+A self-contained parallel test runner for web applications, built on top of
+Playwright. Designed to be small enough to read end-to-end (≈400 LOC) yet
+production-grade: worker pool, retry policy, JSONL reporting, automatic
+failure screenshots, Playwright traces, and CI-friendly exit codes.
+
+## Quick start
+
+```bash
+# 1. Install (Playwright + Chromium headless shell)
+npm install
+npx playwright install chromium
+
+# 2. Start the demo app
+PORT=3011 node demo-app/server.js &
+
+# 3. Run the test suite (4 workers, headless)
+node run-tests.js
+# or
+WORKERS=8 node run-tests.js
+```
+
+Expected output:

+## Architecture
+
+```
+┌─────────────────────────────────────────────────────────────────┐
+│  DSL: describe / it  (test cases registered per file)           │
+└────────────────────────────┬────────────────────────────────────┘
+                             │
+                             ▼
+┌─────────────────────────────────────────────────────────────────┐
+│  WorkerPool.run()                                               │
+│  - spawns N workers, each pulling cases from a shared cursor    │
+│  - one Browser per worker, one BrowserContext per worker        │
+│  - one Page per test → full cookie/DOM isolation                │
+└────────────────────────────┬────────────────────────────────────┘
+                             │
+                             ▼
+┌─────────────────────────────────────────────────────────────────┐
+│  _runOne(): execute fn(page, { baseURL, browser, context })     │
+│  - per-test timeout                                             │
+│  - retry policy on failure                                      │
+│  - screenshot + trace on retain-on-failure                      │
+└────────────────────────────┬────────────────────────────────────┘
+                             │
+                             ▼
+┌─────────────────────────────────────────────────────────────────┐
+│  Reporter: console + JSONL stream + summary JSON                │
+│  Process exit code = 1 if any test failed                       │
+└─────────────────────────────────────────────────────────────────┘
+```
+
+### Key design choices
+
+| Concern | Decision | Why |
+|---|---|---|
+| Process model | 1 Chromium process per worker | True parallelism, not green-thread mimicry |
+| Context sharing | 1 BrowserContext per worker | Cookies stay scoped to a worker, not globally |
+| Page lifecycle | new Page per test | DOM/storage reset is automatic — no manual cleanup |
+| Failure isolation | try/finally per test | A crash in one test doesn't tear down the pool |
+| Auth hermeticity | Tests that need cookies call `newContext()` | Workers don't leak sessions across tests |
+| Flaky tolerance | Per-test `retries` (default 1) | A single 500 doesn't fail the whole run |
+| Reporting | Console + JSONL + summary JSON | Local devs see results; CI consumes JSONL |
+| Artifacts | Timestamped dirs under `reports/` | Each run is preserved, never overwritten |
