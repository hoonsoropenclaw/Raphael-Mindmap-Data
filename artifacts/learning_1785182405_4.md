+/**
+ * useLocalStorage — typed wrapper around localStorage with cross-tab sync.
+ *
+ * Pitfalls handled:
+ *   1. JSON parse errors (corrupted storage) — falls back to initialValue
+ *   2. SSR / no-window environments — guards with typeof check
+ *   3. Storage events fire in OTHER tabs only — we listen to sync across tabs
+ *   4. JSON.stringify can throw on circular refs — try/catch the setter too
+ */
+
+import { useCallback, useEffect, useRef, useState } from 'react';
+
+export function useLocalStorage<T>(
+  key: string,
+  initialValue: T,
+): [T, (value: T | ((prev: T) => T)) => void] {
+  const readValue = useCallback((): T => {
+    if (typeof window === 'undefined') return initialValue;
+    try {
+      const raw = window.localStorage.getItem(key);
+      return raw ? (JSON.parse(raw) as T) : initialValue;
+    } catch (err) {
+      console.warn(`useLocalStorage: failed to read "${key}"`, err);
+      return initialValue;
+    }
+  }, [key, initialValue]);
+
+  const [stored, setStored] = useState<T>(readValue);
+  const storedRef = useRef(stored);
+  storedRef.current = stored;
+
+  const setValue = useCallback(
+    (value: T | ((prev: T) => T)) => {
+      const next = value instanceof Function ? value(storedRef.current) : value;
+      try {
+        if (typeof window !== 'undefined') {
+          window.localStorage.setItem(key, JSON.stringify(next));
+        }
+        setStored(next);
+      } catch (err) {
+        console.warn(`useLocalStorage: failed to write "${key}"`, err);
+      }
+    },
+    [key],
+  );
+
+  // Cross-tab sync: when another tab writes the same key, update here.
+  useEffect(() => {
+    if (typeof window === 'undefined') return;
+    const onStorage = (e: StorageEvent) => {
+      if (e.key !== key || e.newValue === null) return;
+      try {
+        setStored(JSON.parse(e.newValue) as T);
+      } catch {
+        /* ignore corrupt event */
+      }
+    };
+    window.addEventListener('storage', onStorage);
+    return () => window.removeEventListener('storage', onStorage);
+  }, [key]);
+
+  return [stored, setValue];

+/**
+ * TodoContext — single source of truth for todo state + dispatch.
+ *
+ * Pattern:
+ *   - Provider owns the reducer (useReducer)
+ *   - Persists todos (not filter/sort prefs) to localStorage on every change
+ *   - On mount, hydrates from localStorage (handles the "first render uses empty,
+ *     then async hydration flashes a full list" anti-pattern with lazy init)
+ *
+ * Why split state into two useEffects (hydrate-once + persist-on-change)?
+ *   - hydrate-once runs on mount; if we re-hydrate on every render we'd clobber
+ *     in-memory edits with stale localStorage.
+ *   - persist-on-change syncs localStorage without coupling filter/sort UI state
+ *     to disk (those stay ephemeral session prefs).
+ */
+
+import {
+  createContext,
+  useContext,
+  useEffect,
+  useMemo,
+  useReducer,
+  type Dispatch,
+  type ReactNode,
+} from 'react';
+import { todoReducer, initialState } from './todoReducer';
+import { useLocalStorage } from '../hooks/useLocalStorage';
+import type { Todo, TodoAction, TodoState } from '../types/todo';
+
+interface TodoContextValue {
+  state: TodoState;
+  dispatch: Dispatch<TodoAction>;
+}
+
+const TodoContext = createContext<TodoContextValue | null>(null);
+
+const STORAGE_KEY = 'react-todo-app.todos.v1';
+
+export function TodoProvider({ children }: { children: ReactNode }) {
+  // Lazy init: read localStorage SYNCHRONOUSLY on first render to avoid
+  // the "flash of empty state then full list" hydration jank.
+  const [persisted, setPersisted] = useLocalStorage<Todo[]>(STORAGE_KEY, []);
+  const [state, dispatch] = useReducer(todoReducer, {
+    ...initialState,
+    todos: persisted,
+  });
+
+  // Persist todos (only) on change. filter/sort are session-only.
+  useEffect(() => {
+    if (state.todos !== persisted) {
+      setPersisted(state.todos);
+    }
+    // We intentionally omit `persisted` from deps to avoid an infinite loop:
+    // setPersisted updates `persisted`, which would re-trigger this effect.
+    // eslint-disable-next-line react-hooks/exhaustive-deps
+  }, [state.todos]);
+
+  const value = useMemo(() => ({ state, dispatch }), [state]);
+
+  return <TodoContext.Provider value={value}>{children}</TodoContext.Provider>;
+}
+
+export function useTodoContext(): TodoContextValue {
+  const ctx = useContext(TodoContext);
+  if (!ctx) {
+    throw new Error('useTodoContext must be used inside <TodoProvider>');
+  }
+  return ctx;

+### 2.1 `useReducer` 而不是多個 `useState`
+
+```tsx
