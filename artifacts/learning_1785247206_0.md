+def _ensure_dir(path: Path) -> None:
+    path.parent.mkdir(parents=True, exist_ok=True)

+
+def atomic_write_json(path: Path, data: dict[str, Any]) -> None:
+    """Write dict as JSON atomically. mode 0o600."""
+    _ensure_dir(path)
+    dir_ = path.parent
+    fd, tmp = tempfile.mkstemp(prefix=path.name + ".", dir=str(dir_))
+    try:
+        with os.fdopen(fd, "w") as f:
+            json.dump(data, f, indent=2, default=str)
+            f.flush()
+            os.fsync(f.fileno())
+        os.chmod(tmp, 0o600)
+        os.replace(tmp, path)
+        os.chmod(path, 0o600)  # defense-in-depth
+    except Exception:
+        try:
+            os.unlink(tmp)
+        except OSError:
+            pass
+        raise
+
+
+def read_state(path: Path) -> dict[str, Any]:
+    """Read state JSON; tolerate missing / empty / corrupt."""
+    if not path.exists():
+        return {}
+    try:
+        text = path.read_text().strip()
+        if not text:
+            return {}
+        data = json.loads(text)
+        if not isinstance(data, dict):
+            return {}
+        return data
+    except (json.JSONDecodeError, OSError):
+        # corrupt → start fresh but don't blow up
+        return {}
+

+# syswatch — Architecture
+
+## Purpose
+
+A pure-stdlib Linux system-health monitor for environments where `pip install` is
+not an option (N100 mini-PCs, homelab boxes, air-gapped servers, minimal containers).
+It samples CPU, memory, load, disk and inspects `/var/log` for security signals
+(authentication failures, kernel OOM, panic), then emits NDJSON alerts when
+thresholds are breached.
+
+## Design Constraints (from `stdlib-production-tool-pattern` skill)
+
+1. **No third-party deps.** CPU%, memory, load, disk come from `/proc/*`, `/sys/*`
+   and `os.statvfs()`. HTTP webhook uses `urllib.request`. Dataclasses and
+   regex are stdlib.
+2. **Pure functions for collectors and evaluators.** Each takes its
+   `proc_root` or path as a parameter → trivially testable with mock
+   `/proc` directories.
+3. **Thin orchestrator.** `bin/main.py` is the only file with I/O side effects.
+   It composes `core` (collectors + evaluators) + `config` (validated) +
+   `sink` (NDJSON + webhook).
+4. **Append-only NDJSON alerts.** Truncation-safe, pipe-friendly, schema
+   evolves per line. State file is JSON for inspectability.
+5. **Strict config validation.** Unknown keys raise at startup. Numeric
+   bounds enforced. `warn < crit` sanity-checked.
+6. **Atomic writes (no torn state).** State and alerts use temp file +
+   `os.replace()` + `fsync()`. Mode 0o600.
+7. **SIGTERM-graceful daemon.** Daemon mode sleeps in 0.5s chunks so signals
+   are responsive within ≤0.5s.
+8. **Rotation-safe log scanning.** File-shrink detection → reset offset.
+

+# syswatch — Usage Guide

+
+```bash
+# Single check, exit code reflects worst alert severity (0=OK, 1=warn, 2=crit)
+./bin/syswatch.sh --once

+
+```bash
