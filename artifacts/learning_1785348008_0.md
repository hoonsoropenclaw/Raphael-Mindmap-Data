+// permission-engine.js
+// Pure functions for a dynamic RBAC system built on React Flow.
+//
+// Permission model
+// ----------------
+// tri-state per (role, resource): 'allow' | 'deny' | 'inherit'
+// inheritance forms a DAG of roles; edge {source, target} means "source inherits from target"
+// resolution priority: deny > explicit allow > inherited allow > implicit deny
+//
+// Public API
+//   detectCycle, wouldCreateCycle,
+//   resolveRolePermissions (returns Map<resourceId, 'allow' | 'deny'>),
+//   resolveAllRoles (Map<roleId, Map<resourceId, 'allow' | 'deny'>>),
+//   explainPermission (returns the granting/denying role along the inheritance chain),
+//   findDenyPath (returns the chain that gives a specific deny verdict),
+//   validateState, makeSeed, schemaVersion
+
+export const SCHEMA_VERSION = 2;
+
+export function detectCycle(edges) {
+  const adj = new Map();
+  for (const e of edges) {
+    if (!adj.has(e.source)) adj.set(e.source, []);
+    adj.get(e.source).push(e.target);
+  }
+  const WHITE = 0, GRAY = 1, BLACK = 2;
+  const color = new Map();
+  for (const e of edges) {
+    if (!color.has(e.source)) color.set(e.source, WHITE);
+    if (!color.has(e.target)) color.set(e.target, WHITE);
+  }
+  function dfs(u) {
+    color.set(u, GRAY);
+    const next = adj.get(u) || [];
+    for (const v of next) {
+      const c = color.get(v) ?? WHITE;
+      if (c === GRAY) return true;
+      if (c === WHITE && dfs(v)) return true;
+    }
+    color.set(u, BLACK);
+    return false;
+  }
+  for (const u of color.keys()) {
+    if (color.get(u) === WHITE && dfs(u)) return true;
+  }
+  return false;
+}
+
+export function wouldCreateCycle(edges, src, dst) {
+  if (src === dst) return true;
+  return detectCycle([...edges, { source: src, target: dst }]);
+}

+// BFS up the inheritance tree (closest first), used by every resolver.
+// Edge semantics in this app: edge {source, target} means "source inherits from target".
+// So a role's parents are the targets of its outgoing edges.
+export function inheritanceChain(roleId, edges) {
+  const parentsOf = new Map();
+  for (const e of edges) {
+    if (!parentsOf.has(e.source)) parentsOf.set(e.source, []);
+    parentsOf.get(e.source).push(e.target);
+  }
+  const order = [];
+  const visited = new Set();
+  const queue = [roleId];
+  while (queue.length) {
+    const cur = queue.shift();
+    if (visited.has(cur)) continue;
+    visited.add(cur);
+    order.push(cur);
+    for (const p of (parentsOf.get(cur) || [])) queue.push(p);
+  }
+  return order;
+}
