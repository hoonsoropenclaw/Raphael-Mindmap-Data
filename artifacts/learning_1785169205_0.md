+    title = Column(String(200), nullable=False)

+Base.metadata.create_all(bind=engine)
+
+class TodoCreate(BaseModel):

+
+class TodoUpdate(BaseModel):
+    title: str | None = Field(default=None, min_length=1, max_length=200)
+    completed: bool | None = None
+
+class TodoRead(BaseModel):
+    id: int
+    title: str
+    completed: bool
+
+    model_config = {"from_attributes": True}
+
+limiter = Limiter(key_func=get_remote_address)
+app = FastAPI(title="Todo API", version="1.0.0")
+app.state.limiter = limiter
+app.add_exception_handler(RateLimitExceeded, _rate_limit_exceeded_handler)
+
+@app.get("/health")
+def health() -> dict[str, str]:
+    return {"status": "ok"}
+
+@app.post("/todos", response_model=TodoRead, status_code=status.HTTP_201_CREATED)
+@limiter.limit("10/minute")
+def create_todo(request: Request, payload: TodoCreate) -> Todo:
+    with SessionLocal() as db:
+        todo = Todo(title=payload.title)
+        db.add(todo)
+        db.commit()
+        db.refresh(todo)
+        return todo

+## Review
+
+- Chose synchronous FastAPI endpoints with synchronous SQLAlchemy sessions because this is a small CRUD API with no external I/O in request handlers.
+- Used SQLite with `check_same_thread=False` so FastAPI's threaded sync execution can safely share the engine.
+- Kept database access inside short-lived `SessionLocal()` context managers; each endpoint commits or rolls back through session cleanup and refreshes created/updated rows before serialization.
+- Added Pydantic constraints for title length and non-empty values, and explicit 404 responses for missing resources.
+- Added `slowapi` middleware integration with separate limits for creation and listing, keyed by client IP.
+- Tests cover health, create/list, validation, read/update/delete, missing resources, and the rate-limit boundary.
