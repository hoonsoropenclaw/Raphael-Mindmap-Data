+## This resource adds:

+##   * friction_curve / bounce_curve — optional Curve resources for
+##     velocity-dependent or impact-angle-dependent response

+##
+## A small helper static, `apply_to_body()`, wires it onto a PhysicsBody2D.

+@export var friction: float = 0.3 :
+	set(value):
+		friction = clamp(value, 0.0, 1.0)
+
+@export var bounce: float = 0.0 :
+	set(value):
+		bounce = clamp(value, 0.0, 1.0)
+
+@export_range(0.0, 50.0, 0.01) var density: float = 1.0
+@export var surface_type: SurfaceType = SurfaceType.GENERIC
+@export var tangential_damping: float = 1.0
+
+@export var friction_curve: Curve
+@export var bounce_curve: Curve
+
+## Calculate friction adjusted by relative velocity magnitude (m/s).
+## friction_curve maps |v| (m/s) -> multiplier (typically 0..1.5).
+func effective_friction(relative_velocity: float) -> float:
+	if friction_curve == null:
+		return friction
+	return friction * friction_curve.sample_baked(clamp(relative_velocity, 0.0, 50.0))
+
+## Calculate bounce adjusted by impact angle in radians (0 = head-on, π/2 = graze).
+## bounce_curve maps angle (rad) -> multiplier.
+func effective_bounce(impact_angle: float) -> float:
+	if bounce_curve == null:
+		return bounce
+	return bounce * bounce_curve.sample_baked(clamp(impact_angle, 0.0, PI))
+
+## Estimate mass from collision shape area (m^2) and density.
+func compute_mass(area_m2: float) -> float:
+	return max(0.0001, area_m2 * density)

+## Used by WindForceField2D, BlackHole2D, etc.

+## for fixed-timestep simulations.
