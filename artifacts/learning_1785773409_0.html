src/cross_browser.py
"""
Cross-browser Playwright screenshot harness.

Reads HTTP target from environment or CLI arg, runs the page through
chromium / firefox / webkit at a fixed viewport, dumps full-page PNGs
plus a tiny JSON sidecar with browser version + viewport + load metrics.

Usage:
    python3 cross_browser.py --url http://127.0.0.1:8765/ \
        --out /tmp/shots --browsers chromium,firefox,webkit

Design notes (micro-skills applied):
  - playwright_testing_and_scraping (SKILL_CATALOG #18)
  - playwright_relative_url_fix (relative URLs must be absolute for goto)
  - visual_regression_dimension_mismatch_handling (fixed viewport per browser)
"""

from __future__ import annotations

import argparse
import json
import os
import sys
import time
from pathlib import Path
from typing import Iterable

from playwright.sync_api import sync_playwright

BROWSERS_DEFAULT = ("chromium", "firefox", "webkit")
VIEWPORT = {"width": 1280, "height": 800}


def _browser_launcher(playwright, name: str):
    if name == "chromium":
        return playwright.chromium.launch(headless=True)
    if name == "firefox":
        return playwright.firefox.launch(headless=True)
    if name == "webkit":
        return playwright.webkit.launch(headless=True)
    raise ValueError(f"unsupported browser: {name}")


def _browser_version(playwright, name: str, browser) -> str:
    try:
        if name == "chromium":
            return browser.version
        if name == "firefox":
            return browser.version
        if name == "webkit":
            return browser.version
    except Exception:
        pass
    return "unknown"


def run(url: str, out_dir: Path, browsers: Iterable[str]) -> list[dict]:
    out_dir.mkdir(parents=True, exist_ok=True)
    results: list[dict] = []

    with sync_playwright() as p:
        for name in browsers:
            t0 = time.perf_counter()
            browser = _browser_launcher(p, name)
            try:
                context = browser.new_context(
                    viewport=VIEWPORT,
                    device_scale_factor=1,
                    color_scheme="dark",
                    ignore_https_errors=True,
                )
                page = context.new_page()
                page.set_default_timeout(15000)
                # playwright_relative_url_fix: must be absolute
                response = page.goto(url, wait_until="networkidle")
                status = response.status if response else None
                title = page.title()
                # let CSS settle (fonts, layout)
                page.wait_for_timeout(250)
                shot_path = out_dir / f"{name}.png"
                page.screenshot(path=str(shot_path), full_page=True)
                elapsed_ms = int((time.perf_counter() - t0) * 1000)
                results.append(
                    {
                        "browser": name,
                        "version": _browser_version(p, name, browser),
                        "status": status,
                        "title": title,
                        "viewport": VIEWPORT,
                        "screenshot": str(shot_path),
                        "elapsed_ms": elapsed_ms,
                        "url": url,
                    }
                )
                print(
                    f"[ok] {name:<8} status={status} title={title!r} "
                    f"elapsed={elapsed_ms}ms -> {shot_path}"
                )
            except Exception as exc:
                results.append(
                    {
                        "browser": name,
                        "error": f"{type(exc).__name__}: {exc}",
                        "url": url,
                        "viewport": VIEWPORT,
                    }
                )
                print(f"[FAIL] {name}: {exc}", file=sys.stderr)
            finally:
                try:
                    context.close()
                except Exception:
                    pass
                browser.close()

    sidecar = out_dir / "shots.json"
    sidecar.write_text(json.dumps(results, indent=2))
    return results


def _parse_args(argv: list[str]) -> argparse.Namespace:
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--url", default=os.environ.get("TARGET_URL"))
    p.add_argument("--out", required=True, help="output directory for screenshots")
    p.add_argument(
        "--browsers",
        default=",".join(BROWSERS_DEFAULT),
        help="comma-separated list of browsers",
    )
    return p.parse_args(argv)


def main(argv: list[str] | None = None) -> int:
    args = _parse_args(sys.argv[1:] if argv is None else argv)
    if not args.url:
        print("ERROR: --url or TARGET_URL is required", file=sys.stderr)
        return 2
    browsers = tuple(b.strip() for b in args.browsers.split(",") if b.strip())
    results = run(args.url, Path(args.out), browsers)
    failed = [r for r in results if "error" in r]
    return 1 if failed else 0


if __name__ == "__main__":
    raise SystemExit(main())

"""
Visual regression diff using pixelmatch.

Compares two PNGs and produces:
  - a binary diff PNG (colored where pixels differ)
  - a JSON report with diff_px / total_px / diff_ratio + threshold verdict

Usage:
    python3 visual_diff.py --baseline a.png --current b.png \
        --out-diff /tmp/diff.png --threshold 0.001

Micro-skills applied:
  - comprehensive_visual_regression_testing_with_pixelmatch
  - visual_regression_threshold_calibration (ratio, not raw px)
  - visual_regression_dimension_mismatch_handling (resize-to-match policy)
"""

from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

from PIL import Image
import pixelmatch
from pixelmatch.contrib.PIL import pixelmatch as pil_pixelmatch  # type: ignore


def load_image(path: Path) -> Image.Image:
    img = Image.open(path)
    return img.convert("RGBA") if img.mode != "RGBA" else img


def diff(
    baseline: Path,
    current: Path,
    diff_out: Path,
    threshold: float,
) -> dict:
    a = load_image(baseline)
    b = load_image(current)

    if a.size != b.size:
        # Resize current to baseline (top-left) — deterministic for demos
        b_resized = b.resize(a.size, Image.LANCZOS)
        resized = True
    else:
        b_resized = b
        resized = False

    diff_img = Image.new("RGBA", a.size, (245, 245, 250, 255))
    diff_px = pil_pixelmatch(
        a,
        b_resized,
        diff_img,
        threshold=threshold,
        includeAA=False,
    )
    diff_out.parent.mkdir(parents=True, exist_ok=True)
    diff_img.save(diff_out)

    total_px = a.size[0] * a.size[1]
    diff_ratio = (diff_px / total_px) if total_px else 0.0
    return {
        "baseline": str(baseline),
        "current": str(current),
        "diff_image": str(diff_out),
        "width": a.size[0],
        "height": a.size[1],
        "total_px": total_px,
        "diff_px": diff_px,
        "diff_ratio": diff_ratio,
        "threshold": threshold,
        "resized_current_to_baseline": resized,
        "verdict": "pass" if diff_ratio <= threshold else "fail",
    }


def _parse_args(argv: list[str]) -> argparse.Namespace:
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--baseline", required=True)
    p.add_argument("--current", required=True)
    p.add_argument("--out-diff", required=True)
    p.add_argument("--threshold", type=float, default=0.001)
    return p.parse_args(argv)


def main(argv: list[str] | None = None) -> int:
    args = _parse_args(sys.argv[1:] if argv is None else argv)
    report = diff(
        Path(args.baseline),
        Path(args.current),
        Path(args.out_diff),
        args.threshold,
    )
    print(json.dumps(report, indent=2))
    return 0 if report["verdict"] == "pass" else 1


if __name__ == "__main__":
    raise SystemExit(main())

#!/usr/bin/env bash
# End-to-end runner for learning_1785773409_0.
# Spins up a local HTTP server, runs cross-browser screenshot harness,
# captures a baseline set, then runs pixelmatch diff vs baseline.

set -euo pipefail

ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
cd "$ROOT"

PORT="${PORT:-8765}"
HOST="127.0.0.1"
URL="http://${HOST}:${PORT}/"
SHOTS_DIR="$ROOT/artifacts/shots"
BASELINES_DIR="$ROOT/baselines"
REPORT="$ROOT/artifacts/report.json"
LOG="$ROOT/artifacts/runner.log"

mkdir -p "$SHOTS_DIR" "$BASELINES_DIR" "$(dirname "$REPORT")"

# Start a tiny stdlib HTTP server pointing at src/ (so demo_target.html is the index)
# We rename demo_target.html -> index.html in a temp dir so the bare URL serves it.
SERVE_DIR="$(mktemp -d)"
cp "$ROOT/src/demo_target.html" "$SERVE_DIR/index.html"
python3 -m http.server -b "$HOST" -d "$SERVE_DIR" "$PORT" >"$LOG" 2>&1 &
SERVER_PID=$!
trap 'kill "$SERVER_PID" 2>/dev/null || true; rm -rf "$SERVE_DIR"' EXIT

# Wait for the server to be reachable (max 5s)
for _ in $(seq 1 25); do
  if curl -fsS "$URL" >/dev/null 2>&1; then
    break
  fi
  sleep 0.2
done
curl -fsS "$URL" >/dev/null

echo "[runner] server up at $URL (pid=$SERVER_PID)"

# 1) Cross-browser screenshots
python3 "$ROOT/src/cross_browser.py" \
  --url "$URL" \
  --out "$SHOTS_DIR" \
  --browsers chromium,firefox,webkit

# 2) Establish baselines on first run, then diff on subsequent runs.
mkdir -p "$BASELINES_DIR"
first_run=0
for b in chromium firefox webkit; do
  if [[ ! -f "$BASELINES_DIR/$b.png" ]]; then
    cp "$SHOTS_DIR/$b.png" "$BASELINES_DIR/$b.png"
    first_run=1
  fi
done
if [[ "$first_run" -eq 1 ]]; then
  echo "[runner] first run: baselines captured at $BASELINES_DIR"
fi

# 3) Visual regression diff
python3 - "$SHOTS_DIR" "$BASELINES_DIR" "$REPORT" <<'PY'
import json
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parent / "src"))
from visual_diff import diff  # type: ignore

shots = Path(sys.argv[1])
baselines = Path(sys.argv[2])
report_path = Path(sys.argv[3])

THRESHOLD = 0.001  # 0.1% of pixels
report = {"threshold": THRESHOLD, "browsers": {}}
for browser in ("chromium", "firefox", "webkit"):
    cur = shots / f"{browser}.png"
    base = baselines / f"{browser}.png"
    diff_out = shots / f"{browser}.diff.png"
    if not cur.exists() or not base.exists():
        report["browsers"][browser] = {"error": "missing shot or baseline"}
        continue
    report["browsers"][browser] = diff(base, cur, diff_out, THRESHOLD)

report_path.write_text(json.dumps(report, indent=2))
print(f"[runner] diff report -> {report_path}")

failed = [b for b, r in report["browsers"].items()
          if isinstance(r, dict) and r.get("verdict") == "fail"]
if failed:
    print(f"[runner] VISUAL REGRESSION in: {', '.join(failed)}", file=sys.stderr)
    sys.exit(1)
PY

echo "[runner] OK"
