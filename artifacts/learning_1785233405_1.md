+// utils/persistence.js — versioned localStorage adapter.
+// Schema-versioning lets us evolve the state shape without breaking
+// existing saved boards. (Architect #6: plan for migration.)
+
+const STORAGE_KEY = 'kanban-state-v1'
+const CURRENT_SCHEMA_VERSION = 1
+
+export function loadState() {
+  if (typeof localStorage === 'undefined') return null
+  try {
+    const raw = localStorage.getItem(STORAGE_KEY)
+    if (!raw) return null
+    const parsed = JSON.parse(raw)
+    if (!parsed || typeof parsed !== 'object') return null
+
+    // Future-proofing: if we ever bump CURRENT_SCHEMA_VERSION, run a
+    // migration here instead of refusing the data.
+    if (parsed.schemaVersion !== CURRENT_SCHEMA_VERSION) {
+      // For v1 -> v1 this branch is never taken. When it is, call
+      // migrate(parsed) and reassign. No migrations yet, so bail.
+      console.warn(
+        `[kanban] saved schema ${parsed.schemaVersion} differs from current ${CURRENT_SCHEMA_VERSION}; ignoring`,
+      )
+      return null
+    }
+
+    // Validate minimal shape; missing fields mean corrupted save.
+    if (!parsed.cards || !parsed.columnOrder) return null
+    return parsed
+  } catch (err) {
+    // Corrupted JSON — log and start fresh. Don't crash the app.
+    console.warn('[kanban] failed to load saved state:', err)
+    return null
+  }
+}
+
+export function saveState(state) {
+  if (typeof localStorage === 'undefined') return

+## 3. State architecture — why this shape
+
+### Normalized, not nested
+
+`state.cards` is `{ [id]: card }`, NOT `state.columns[].cards[]`. `state.columnOrder` is `{ [columnId]: cardId[] }`. Two reasons:
+
+1. **O(1) lookup by id** when a reducer action like `UPDATE_CARD` or `DELETE_CARD` arrives — no column scan needed.
+2. **Move operations are array splices**, not deep clones of nested objects. Cheap.
+
+### Single reducer, explicit actions
+
+All mutations live in `boardReducer.js`. No scattered `setState` calls. This gives us:
+
+- **One place to audit nullability** — `UPDATE_CARD` rejects unknown `columnId` values rather than writing `null` to a required field. This is directly the lesson from **Architect feedback §3** (don't write `None`/null to a required field; be explicit).
+- **One place to enforce schema** — `makeCard()` validates `priority` against the `PRIORITIES` allowlist and rejects bad values silently rather than writing garbage.
+- **One place to gate history** — UI actions (`SET_FILTER_TEXT`, `SELECT_CARD`) deliberately do not push to the undo stack; data actions do. Mixing them would let users "undo" their search box.
+
+### History stack — snapshot, not inverse-operation
+
+Each entry in `state.past` is a complete past snapshot of `{ cards, columnOrder }`. We considered storing inverse ops (e.g. `{ type: 'UNDO_MOVE', from, to }`) but snapshots win because:
+
+- **Trivially correct** — no chance of an inverse op getting out of sync with the reducer's actual mutation logic.
