+import type { BoardState, PersistedShape } from '../types';
+import { STATE_SCHEMA_VERSION } from '../types';
+import { seedBoard } from './seed';
+
+const STORAGE_KEY = 'kanban.state.v1';
+
+/**
+ * Load state from localStorage with a hard schema-version gate.
+ *
+ * If the persisted version doesn't match STATE_SCHEMA_VERSION we treat
+ * the data as stale and return the seed. This is intentionally brutal:
+ * localStorage is a personal, single-user cache — losing it is a UX
+ * nuisance, not a data-loss event, so we prefer a clean slate over a
+ * half-broken migration on every release.
+ */
+export function loadState(): BoardState {
+  if (typeof window === 'undefined') return seedBoard();
+  try {
+    const raw = window.localStorage.getItem(STORAGE_KEY);
+    if (!raw) return seedBoard();
+    const parsed = JSON.parse(raw) as Partial<PersistedShape>;
+    if (parsed.version !== STATE_SCHEMA_VERSION) return seedBoard();
+    if (!parsed.state || typeof parsed.state !== 'object') return seedBoard();
+    if (!isValidBoardState(parsed.state)) return seedBoard();
+    return parsed.state;
+  } catch {
+    return seedBoard();
+  }
+}
+
+export function saveState(state: BoardState): void {
+  if (typeof window === 'undefined') return;
+  const payload: PersistedShape = { version: STATE_SCHEMA_VERSION, state };
+  try {
+    window.localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
+  } catch {
+    // Quota errors are non-fatal — the in-memory state still works.
+  }
+}
+
+/**
+ * Defensive shape check. Storage can be tampered with via DevTools, so
+ * we never trust parsed JSON blindly before handing it to the reducer.
+ */
+function isValidBoardState(value: unknown): value is BoardState {
+  if (!value || typeof value !== 'object') return false;
+  const v = value as Partial<BoardState>;
+  if (!v.columns || typeof v.columns !== 'object') return false;
+  if (!v.cards || typeof v.cards !== 'object') return false;
+  if (!Array.isArray(v.columnOrder)) return false;
+  return true;

+/**
+ * Generic undo/redo history for any reducer-driven state.
+ *
+ * Design choices:
+ *   - History lives in a ref to avoid re-renders when we push / pop.
+ *     Re-renders happen on the *index* change, which is the only thing
+ *     components actually depend on.
+ *   - We cap history at MAX_HISTORY so memory stays bounded.
+ *   - Each commit (push) snapshots the *previous* state. That makes
+ *     `undo` a no-op when index === 0 (we don't have anything older)
+ *     and `redo` a no-op when index === length - 1.
+ *
+ * Usage:
+ *   const { state, dispatch, undo, redo, canUndo, canRedo, reset } =
+ *     useUndoableState(initial, reducer);
+ */

+  redo: () => void;
+  canUndo: boolean;
+  canRedo: boolean;
+  /** Replace state and clear history (used by hydrate / reset). */
+  reset: (next: S) => void;

+import { useEffect, useRef } from 'react';
+import type { BoardState } from '../types';
+import { saveState } from '../state/persistence';
+
+/**
+ * Debounced persistence. We don't want to JSON.stringify the whole
+ * state on every keystroke (which can fire 10x/sec during a title
+ * edit). 250ms is short enough that the user never notices a delay
+ * between the visual change and the persistence, but long enough to
+ * coalesce typical input bursts into a single write.
+ */
+const DEBOUNCE_MS = 250;
+
+export function useDebouncedPersist(state: BoardState): void {
+  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
+  const latestRef = useRef(state);
+
+  latestRef.current = state;
+
+  useEffect(() => {
+    if (timerRef.current) clearTimeout(timerRef.current);
+    timerRef.current = setTimeout(() => {
+      saveState(latestRef.current);
+      timerRef.current = null;
+    }, DEBOUNCE_MS);
+    return () => {
+      if (timerRef.current) clearTimeout(timerRef.current);
+    };
+  }, [state]);
+}

+## 1. 為什麼挑 Kanban（不是 todo list 或 counter）

+要對得起「複雜狀態管理」四個字，需要這些特徵：

+- **跨元件共享狀態**：Context 有意義 — Toolbar / Board / Column / Card 都要讀 dispatch
+- **副作用持久化**：每次 state 變更要寫 localStorage，要 debounce 防止 keystroke 風暴
+- **可撤銷 / 重做**：history snapshot pattern
+- **跨分頁同步**：storage event 監聽

+### 為什麼 useReducer 而不是 useState + 多 setter
+
+- **單一入口**：所有變更走 `dispatch(action)`，未來加 Redux DevTools / 中介層（logging, optimistic update）零成本
+- **不可變更新強制**：reducer 必須回傳新物件，這對 undo/redo 是必要條件（history snapshot）
+- **可序列化**：整個 state tree 一次 JSON.stringify 寫入 localStorage，不需要 patch diff
+
+### 為什麼 useReducer + Context 而不是 Redux / Zustand
+
+- **規模**：這個元件樹規模在 ~30 個節點以內，Context 重渲染問題不存在（reducer 回傳 immutable new reference，未改變的 column / card 還是有同樣 reference，memo 的 Card 不會 re-render）
+- **依賴面**：useReducer + Context 是 React 內建、零依賴、build size 友善（最終 bundle 253KB / 80KB gzip 已含 dnd-kit）
+- **可教學**：越用越聰明 — 自己手刻比用套件學到更多
