+"""
+Rate-limit middleware：
+- 簡單的固定視窗計數器（in-memory）
+- key 來自 client IP：`X-Forwarded-For`（第一段）-> `X-Real-IP` -> `request.client.host`
+- 超過限制時回 429 + Retry-After header
+- 設計為可注入：可在測試中直接 reset / 替換 store
+"""
+from __future__ import annotations
+
+import threading
+import time
+from collections import defaultdict, deque
+from typing import Callable, Deque, Dict, Iterable, Optional, Tuple
+
+from fastapi import Request
+from fastapi.responses import JSONResponse
+from starlette.middleware.base import BaseHTTPMiddleware
+
+
+class RateLimiter:
+    """Sliding-window 計數器。每次請求記錄 timestamp，超過 window * limit 就拒絕。"""
+
+    def __init__(self, max_requests: int = 60, window_seconds: float = 60.0) -> None:
+        self.max_requests = max_requests
+        self.window_seconds = window_seconds
+        self._store: Dict[str, Deque[float]] = defaultdict(deque)
+        self._lock = threading.Lock()
+
+    def reset(self) -> None:
+        """清空所有計數（測試用）。"""
+        with self._lock:
+            self._store.clear()
+
+    def hit(self, key: str, now: Optional[float] = None) -> Tuple[bool, int]:
+        """
+        記錄一次請求，回傳 (是否允許, 本次剩餘可呼叫次數)。
+        超過限制時第一個元素為 False。
+        """
+        now = now if now is not None else time.monotonic()
+        cutoff = now - self.window_seconds
+        with self._lock:
+            dq = self._store[key]
+            # 移除超出視窗的舊紀錄
+            while dq and dq[0] <= cutoff:
+                dq.popleft()
+            if len(dq) >= self.max_requests:
+                # 計算 retry_after = 視窗內最舊一筆到期時間
+                retry_after = max(1, int(dq[0] + self.window_seconds - now) + 1)
+                return False, 0
+            dq.append(now)
+            return True, self.max_requests - len(dq)
+
+
+def _client_key(request: Request) -> str:
+    """優先順序：X-Forwarded-For 第一段 > X-Real-IP > client.host."""
+    xff = request.headers.get("x-forwarded-for")
+    if xff:
+        return xff.split(",")[0].strip()
+    xri = request.headers.get("x-real-ip")
+    if xri:
+        return xri.strip()
+    return (request.client.host if request.client else "unknown")
+

+## 實作思路
+
+### 1. 架構選擇
+
+- **Dependency injection 雙軸**：`create_app(db_url, limiter)` 是工廠函式，測試可以注入 ephemeral SQLite 與小視窗 limiter，不必依賴 monkeypatch。
+- **Session 設計**：用 `bind_get_db(url)` 切換 module-level factory，所以 `routers.py` 內的 `Depends(get_db)` 在測試與 production 都能拿到正確的 session。
+- **Middleware 而非 decorator**：rate limit 寫成 ASGI middleware，方便跨 route 統一處理，且能排除 `/health` 等路徑。
+
+### 2. Rate Limit 設計
+
+- **滑動視窗計數器**（in-memory）：每個 key 用 deque 存最近 N 個 timestamp，進 request 時先 pop 掉超出 `window` 的舊紀錄，再決定是否記入。
+- **Key 解析**：`X-Forwarded-For` 第一段 → `X-Real-IP` → `request.client.host`。三者都可被 proxy 覆寫。
+- **回應**：429 + `Retry-After` header + `X-RateLimit-Limit/Remaining`。
+- **可注入**：測試時 `RateLimiter(max_requests=3, window_seconds=60)` 帶寬鬆值，不會污染 CRUD 測試。
+
+### 3. 資料層
+
+- SQLAlchemy 2.0 風格：`DeclarativeBase` + `Mapped[T]` + `mapped_column`。
+- 用 `Mapped[T]` 取代傳統 `Column(..., T)`，型別資訊直接綁在 Python type，按 IDE/typing 工具看的時候比較清楚。
+- 預設 `sqlite:///./todo.db`，可由 `TODO_DB_URL` 環境變數覆寫。
+
+### 4. Schema
+
+- 用 Pydantic v2 `ConfigDict(from_attributes=True)` 讓 ORM 物件直接 dump 成 response。
+- `TodoUpdate` 是部分更新模型（所有欄位 optional），搭配 `model_dump(exclude_unset=True)` 確保只更新客戶端真的傳的欄位。
+
+### 5. 為什麼測試覆蓋這麼全
+
+- 13 個 CRUD 測試：happy path、validation、404、partial update、full update、filter、description（包含正常 + 4xx 邊界）
+- 6 個限流測試：header 正確、嚴格模式 429、不同 client 不同 bucket、單元層 limiter reset、視窗機制、health 不被限流

+## 過程中踩到的坑（給下一輪）
