+#!/usr/bin/env python3
+"""
+mock_whisper.py — Local deterministic fallback Speech-to-Text backend.
+
+WHY THIS EXISTS
+---------------
+The production STT backend is `openai.audio.transcriptions.create` (Whisper).
+But this system must run even when **no API key is configured** (common case on
+N100 home server / CI / sandbox). The fallback must:
+
+1. **Always succeed** (return *some* plausible transcription object) so the
+   downstream Telegram Bot and WebSocket push logic never crashes on a missing
+   credential.
+2. **Be deterministic**: given the same audio bytes, return the same text. This
+   is critical for **unit-test reproducibility** — without it, every retry
+   gives a different random string and assertions like
+   `assert text.startswith("transcribed:")` would flake.
+3. **Generate a sensible-looking segment stream** so the frontend's per-segment
+   UI can be exercised even without real audio.
+4. **Carry realistic signals**: language hint, probability/confidence, segment
+   timings. Frontend uses these to colourise cards (high vs low confidence).
+
+DESIGN
+------
+We hash the audio bytes with SHA-256, take the first 8 bytes, and feed them
+into a small keyword pool. The output is a list of "segments" with timings
+linearly distributed across an estimated duration.
+
+Duration estimation is intentionally coarse:
+    estimated_seconds = max(1.0, min(120.0, byte_size / 16_000))
+    (assumes ~16 KB/s for an ogg/opus voice message, which is exactly what
+    Telegram sends — Telegram voice notes are encoded at ~32 kbps ≈ 4 KB/s,
+    so we use 4 KB/s = 16_000 / 4 = a more honest divisor of 4000)
+
+This is NOT a real Whisper — it never inspects PCM data, it just gives us a
+plausible surface to develop and demo against.

+#!/usr/bin/env python3
+"""
+events.py — In-process pub/sub bus.
+
+WHY A BUS
+---------
+We have THREE producers (Telegram Bot, REST upload, Web Speech API frontend)
+and ANY NUMBER of consumers (WebSocket clients, loggers, etc).
+
+Without a bus, every producer would need to know every consumer and call it
+directly. With a bus, the producer `await bus.publish(event)` and every
+WebSocket handler subscribed via `async for event in bus.subscribe()` gets
+the same event.
+
+This is a small, dependency-free implementation using `asyncio.Queue` per
+subscriber — equivalent to a tiny NATS / Redis-pubsub for a single process.
+
+TYPES OF EVENTS
+---------------
+- `transcript.completed`  — full STT finished for an audio
+- `transcript.segment`    — one segment streamed (real-time feel)
+- `transcript.error`      — STT failed, payload contains root_cause
+- `telegram.message`      — informational echo of incoming Telegram msg
+- `system.heartbeat`      — periodic "bot still alive" tick
+"""
+
+from __future__ import annotations
+
+import asyncio
+import time
+import uuid
+from dataclasses import dataclass, field
+from typing import Any, AsyncIterator, Dict, List, Optional
+
+
+@dataclass
+class BusEvent:
+    """A single pub/sub envelope."""
+
+    type: str                                    # e.g. "transcript.completed"
+    payload: Dict[str, Any] = field(default_factory=dict)

+    ts: float = field(default_factory=time.time)
+
+    def to_dict(self) -> Dict[str, Any]:
+        return {
+            "id": self.id,
+            "type": self.type,
+            "ts": self.ts,
+            "payload": self.payload,
+        }

+#!/usr/bin/env python3
+"""
+jsonl_log.py — Structured JSONL logger (one JSON object per line).
+
+Pattern lifted from `telegram-weather-bot/scripts/weather_bot.py` —
+encouraged by `automated_workflow_and_server_management.md` skill:
+JSONL records are easy to grep, easy to ship, never ambiguous when
+multi-line strings appear.
+
+Each record carries `ts`, `level`, `event`, plus arbitrary keyword fields.
+For non-OK levels we also include `root_cause` so a future failure-mode
+search can pinpoint "what was wrong when X broke".
+"""
+
+from __future__ import annotations
+
+import json
+import os
+import sys
+import threading
+import time
+from typing import IO, Any, Optional, Union
+
+
+class JsonlLogger:
+    """Thread-safe JSONL emitter.
+
+    Writes one JSON object per line. Levels: INFO / WARN / ERROR / DEBUG.
+    """
+
+    def __init__(self, path: Optional[Union[str, IO[Any]]] = None) -> None:
+        self._path = path
+        self._lock = threading.Lock()
+        # Also echo to stderr by default so a developer running the bot
+        # in a foreground shell can watch what is happening.
+        self._stderr = True
+
+    def emit(self, level: str, event: str, **fields: Any) -> None:
+        record = {
+            "ts": time.strftime("%Y-%m-%dT%H:%M:%S%z", time.localtime()),
+            "level": level,
+            "event": event,
+            **fields,
+        }
+        try:
+            line = json.dumps(record, ensure_ascii=False, separators=(",", ":"))
+        except (TypeError, ValueError):
+            # `ensure_ascii=False` + non-serializable objects can fail.
+            # Fall back to the default `str()` for safety.
+            line = json.dumps(record, ensure_ascii=True, default=str)
+
+        if self._path is not None and isinstance(self._path, str):
+            with self._lock:
+                try:
+                    parent = os.path.dirname(self._path)
+                    if parent and not os.path.isdir(parent):
+                        os.makedirs(parent, exist_ok=True)
+                    with open(self._path, "a", encoding="utf-8") as fh:
+                        fh.write(line + "\n")
+                except OSError as exc:
+                    print(f"[jsonl-logger-error] {exc}", file=sys.stderr)
+
+        if self._stderr:
+            print(line, file=sys.stderr)
+
+    def info(self, event: str, **fields: Any) -> None:
+        self.emit("INFO", event, **fields)
