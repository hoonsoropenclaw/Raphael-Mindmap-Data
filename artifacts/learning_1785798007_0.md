+/**
+ * Compare two PNG buffers and return diff metrics.
+ * @param {Buffer} baselineBuf
+ * @param {Buffer} actualBuf
+ * @param {object} opts
+ * @param {number} [opts.threshold=0.1]   pixelmatch threshold (0..1, lower = stricter)
+ * @param {number} [opts.maxRatio=0.01]  acceptable mismatch ratio before failing
+ * @returns {{
+ *   ok: boolean,
+ *   mismatch: number,
+ *   ratio: number,
+ *   reason?: string,
+ *   width: number,
+ *   height: number,
+ *   diffBuf: Buffer
+ * }}
+ */
+export function diffPngBuffers(baselineBuf, actualBuf, opts = {}) {
+  const threshold = opts.threshold ?? 0.1;
+  const maxRatio = opts.maxRatio ?? 0.01;
+
+  const baseline = PNG.sync.read(baselineBuf);
+  const actual = PNG.sync.read(actualBuf);
+
+  if (baseline.width !== actual.width || baseline.height !== actual.height) {
+    return {
+      ok: false,
+      mismatch: Infinity,
+      ratio: 1,
+      reason: `dimension mismatch: baseline=${baseline.width}x${baseline.height} actual=${actual.width}x${actual.height}`,
+      width: actual.width,
+      height: actual.height,
+      diffBuf: Buffer.alloc(0),
+    };
+  }
+
+  const diff = new PNG({ width: baseline.width, height: baseline.height });
+  const mismatch = pixelmatch(
+    baseline.data,
+    actual.data,
+    diff.data,
+    baseline.width,
+    baseline.height,
+    { threshold }
+  );
+  const ratio = mismatch / (baseline.width * baseline.height);
+  const ok = ratio <= maxRatio;
+  return {
+    ok,
+    mismatch,
+    ratio,
+    width: baseline.width,
+    height: baseline.height,
+    diffBuf: PNG.sync.write(diff),
+  };

+const PORT = Number(process.env.PORT || 4173);
+
+const MIME = {
+  '.html': 'text/html; charset=utf-8',
+  '.js':   'application/javascript; charset=utf-8',
+  '.css':  'text/css; charset=utf-8',
+  '.svg':  'image/svg+xml',
+  '.png':  'image/png',
+  '.json': 'application/json; charset=utf-8',
+};
+
+const server = http.createServer((req, res) => {
+  const url = new URL(req.url, `http://${req.headers.host}`);
+  let filePath = path.join(ROOT, url.pathname === '/' ? 'index.html' : url.pathname);
+  if (!filePath.startsWith(ROOT)) {
+    res.writeHead(403); res.end('forbidden'); return;
+  }
+  fs.readFile(filePath, (err, data) => {
+    if (err) { res.writeHead(404); res.end('not found'); return; }
+    const ext = path.extname(filePath).toLowerCase();
+    res.writeHead(200, { 'Content-Type': MIME[ext] || 'application/octet-stream' });
+    res.end(data);
+  });
+});
+
+server.listen(PORT, '127.0.0.1', () => {
+  console.log(`[serve-demo] http://127.0.0.1:${PORT}/`);
