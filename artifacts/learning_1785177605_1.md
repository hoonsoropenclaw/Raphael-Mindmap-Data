+"""Rate limiter — 記憶體版 sliding-window。
+
+設計：
+- 每個 key（預設為 client IP）保留一個 deque，記錄最近 N 個 request 的時間戳
+- 每次 request 進來，先清掉超出 window 的舊紀錄，再看 deque 長度
+- 超過上限 → 回 (False, retry_after) 給 middleware 決定回 429
+- 故意設計成 class、未來要換 Redis/TokenBucket 都能無痛替換
+"""
+from __future__ import annotations
+
+import time
+from collections import defaultdict, deque
+from dataclasses import dataclass
+from typing import Deque, Dict
+
+
+@dataclass
+class RateLimitResult:
+    allowed: bool
+    remaining: int
+    retry_after: float  # seconds; 0 when allowed
+
+
+class RateLimiter:
+    """Sliding-window rate limiter.
+
+    參數:
+        max_requests: 在 window 內允許的最大 request 數
+        window_seconds: 滑動視窗大小（秒）
+    """
+
+    def __init__(self, max_requests: int = 10, window_seconds: float = 1.0) -> None:
+        if max_requests < 1:
+            raise ValueError("max_requests must be >= 1")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be > 0")
+        self.max_requests = max_requests
+        self.window_seconds = window_seconds
+        self._hits: Dict[str, Deque[float]] = defaultdict(deque)
+
+    def check(self, key: str, now: float | None = None) -> RateLimitResult:
+        """檢查一個 key 是否允許再發一個 request。
+
+        Args:
+            key: 限制對象（通常是 IP），空字串或 None 會用 '__anonymous__'
+            now: 注入用於測試的時間戳;None 時用 time.monotonic()
+        """
+        if now is None:
+            now = time.monotonic()
+        effective_key = key or "__anonymous__"
+        cutoff = now - self.window_seconds
+        bucket = self._hits[effective_key]
+
+        # 清掉超出 window 的舊紀錄
+        while bucket and bucket[0] <= cutoff:
+            bucket.popleft()
+
+        if len(bucket) >= self.max_requests:
+            # 至少要等 bucket[0] (最早的) 退出 window 才能再進
+            retry_after = max(bucket[0] + self.window_seconds - now, 0.0)
+            return RateLimitResult(
+                allowed=False,
+                remaining=0,
+                retry_after=retry_after,
+            )
+
+        bucket.append(now)
+        return RateLimitResult(
+            allowed=True,
+            remaining=self.max_requests - len(bucket),
+            retry_after=0.0,
+        )
+
+    def reset(self) -> None:
+        """清空所有計數 — 給測試 fixture 用。"""
+        self._hits.clear()

+## 2. 架構決策
+
+### 2.1 為什麼選 FastAPI + SQLAlchemy 2.0 async + aiosqlite?
+- **FastAPI**:任務指定;async-first、Pydantic 內建、OpenAPI 自動產生
+- **SQLAlchemy 2.0 async + aiosqlite**:純 async stack、SQLite 不需要額外 daemon
+- **httpx `ASGITransport`**:不開 uvicorn server 就跑 integration 測試,CI 友善
+
+### 2.2 Rate limit 為什麼選「記憶體 sliding window」而不是 `slowapi` / Redis?
+- 任務只有「簡單待辦事項 API」,1 個 process 的記憶體版本就夠
+- 不引入 `slowapi`(要再裝相依、跟 FastAPI exception handler 耦合)
+- 不引入 Redis(要再起一個 process,違反「簡單」精神)
+- **介面故意設計成可抽換**:`RateLimiter.check(key)` 簽名小,
+  之後要換 Redis 只要替換 `app/rate_limit.py`,`main.py` 不動
+
+### 2.3 Rate limit 為什麼掛在 middleware 而不是在 dependency?
+- 全域所有 `/todos*` 路由都要計數(GET 列表、GET 單筆、PATCH、DELETE 都算)
+- middleware 一次套用,不會忘了在某個新路由加 `Depends(rate_limit)`
+- middleware 內部 **skip `/health`、`/docs`、`/openapi.json`、`/redoc`**,
+  否則健康檢查會自我雷擊
+- **204 DELETE** 不能帶 response body,所以 `delete_todo` 用 `response_class=Response` 回空 body
+
