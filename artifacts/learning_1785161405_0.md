+"""SQLite 資料庫層 — 待辦事項 CRUD。

+設計重點：
+1. 每個 request 用獨立連線（sqlite3 不是 thread-safe，需 per-thread connection）
+2. context manager 自動 commit/rollback
+3. Row 轉 dict 透過 sqlite3.Row factory，方便序列化
+4. schema 在 get_conn() 內做 CREATE TABLE IF NOT EXISTS，啟動時不需額外 migration
+"""

+
+import os
+import sqlite3
+from contextlib import contextmanager
+from pathlib import Path
+from typing import Iterator
+
+
+# 預設 DB 路徑，可被環境變數覆蓋（測試時用 tmp file）
+DEFAULT_DB_PATH = os.environ.get("TODO_DB_PATH", "todo.db")
+
+
+def _connect(db_path: str) -> sqlite3.Connection:
+    """建立 sqlite 連線，啟用 Row factory + foreign keys。"""
+    conn = sqlite3.connect(db_path, check_same_thread=False, isolation_level=None)
+    conn.row_factory = sqlite3.Row
+    conn.execute("PRAGMA foreign_keys = ON")
+    conn.execute("PRAGMA journal_mode = WAL")  # 並行讀寫更友善
+    return conn
+
+
+# Thread-local 連線池 — 每個 thread 維持一個連線，避免反覆 open/close
+import threading
+_local = threading.local()
+
+
+def get_conn(db_path: str | None = None) -> sqlite3.Connection:
+    """取得目前 thread 的連線（lazy init）。
+
+    第一次呼叫會建連線並執行 schema，後續重用同 thread 的連線。
+    """
+    path = db_path or DEFAULT_DB_PATH
+    # 確保父目錄存在
+    parent = Path(path).expanduser().resolve().parent
+    parent.mkdir(parents=True, exist_ok=True)
+
+    key = f"conn::{path}"
+    conn = getattr(_local, key, None)
+    if conn is None:
+        conn = _connect(path)
+        _init_schema(conn)
+        setattr(_local, key, conn)
+    return conn
