+function extractFields(text) {
+  const extracted = {};
+  const firstLine = text.split('\n').map(l => l.trim()).find(l => l.length > 0);
+  if (firstLine) extracted.title = firstLine.replace(/會議紀錄|記錄|Minutes?$/i, '').trim();
+
+  const dateMatch = text.match(/(\d{4})[\/\-年.](\d{1,2})[\/\-月.](\d{1,2})/);
+  if (dateMatch) {
+    const [, y, m, d] = dateMatch;
+    extracted.date = `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`;
+    extracted.rawDate = `${y}/${m}/${d}`;
+  }
+
+  const timeMatch = text.match(/(\d{1,2}):(\d{2})/);
+  if (timeMatch) {
+    extracted.start = `${timeMatch[1].padStart(2, '0')}:${timeMatch[2]}`;
+  }
+
+  const rangeMatch = text.match(/(\d{1,2}:\d{2})\s*[-–~至到]\s*(\d{1,2}:\d{2})/);
+  if (rangeMatch) {
+    const start = timeToMinutes(rangeMatch[1]);
+    const end   = timeToMinutes(rangeMatch[2]);
+    const diff  = end - start;
+    if (diff > 0) {
+      const hours = Math.floor(diff / 60);
+      const mins  = diff % 60;
