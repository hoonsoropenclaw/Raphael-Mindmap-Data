+import type { Priority, Todo, TodoAction } from './types'
+
+/**
+ * Pure reducer — 不允許副作用,localStorage 由 Provider 的 useEffect 處理。
+ *
+ * Why pure?
+ *   - 易測試 (丟進 state + action 就得到 next state,沒有 race condition)
+ *   - 易序列化 (state 形狀穩定 → localStorage round-trip 安全)
+ *   - 易 replay (之後想做 optimistic update 撤銷/重做直接基於此)
+ */
+export function todoReducer(state: Todo[], action: TodoAction): Todo[] {
+  switch (action.type) {
+    case 'add': {
+      const now = Date.now()
+      const newTodo: Todo = {
+        id: crypto.randomUUID?.() ?? `t-${now}-${Math.random().toString(36).slice(2, 8)}`,
+        title: action.title.trim(),
+        done: false,
+        priority: action.priority,
+        tags: dedupeTags(action.tags),
+        createdAt: now,
+        updatedAt: now,
+      }
+      return [newTodo, ...state]
+    }
+
+    case 'toggle':
+      return state.map((t) =>
+        t.id === action.id ? { ...t, done: !t.done, updatedAt: Date.now() } : t
+      )
+
+    case 'edit':
+      return state.map((t) =>
+        t.id === action.id ? { ...t, title: action.title.trim(), updatedAt: Date.now() } : t
+      )
+
+    case 'remove':
+      return state.filter((t) => t.id !== action.id)
+
+    case 'addTag':
+      return state.map((t) => {
+        if (t.id !== action.id) return t
+        if (t.tags.includes(action.tag)) return t
+        return { ...t, tags: [...t.tags, action.tag], updatedAt: Date.now() }
+      })
+
+    case 'removeTag':
+      return state.map((t) =>
+        t.id === action.id
+          ? { ...t, tags: t.tags.filter((tag) => tag !== action.tag), updatedAt: Date.now() }
+          : t
+      )
+
+    case 'clearCompleted':
+      return state.filter((t) => !t.done)
+
+    case 'reorder': {
+      const idx = state.findIndex((t) => t.id === action.id)
+      if (idx === -1) return state
+      const swapWith = action.direction === 'up' ? idx - 1 : idx + 1
+      if (swapWith < 0 || swapWith >= state.length) return state
+      const next = state.slice()
+      ;[next[idx], next[swapWith]] = [next[swapWith], next[idx]]
+      return next
+    }
+
+    case 'hydrate':
+      // 用 hydrate action 而非直接 setState — 之後可以加 schema 驗證 / migration
+      return action.todos
+
+    default:
+      // exhaustive check — TypeScript 編譯時就會報錯
+      const _exhaustive: never = action
+      return _exhaustive
+  }
