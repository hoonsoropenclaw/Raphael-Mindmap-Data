+    p = Path(path)
+    if not p.exists():

+# 計算 summary statistics

+def _summary(results: list) -> dict:
+    total = len(results)
+    passed = sum(1 for r in results if not r.get("error") and r.get("diff_passed", False))
+    failed = total - passed
+    errors = sum(1 for r in results if r.get("error"))
+    avg_diff_ratio = (
+        sum(r.get("diff_ratio", 0) for r in results) / total if total > 0 else 0.0
+    )
+    avg_duration = (
+        sum(r.get("duration_ms", 0) for r in results) / total if total > 0 else 0.0
+    )

+        "passed": passed,
