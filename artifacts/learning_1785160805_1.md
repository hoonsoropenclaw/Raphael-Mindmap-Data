+from typing import Any

+class WeatherError(RuntimeError):
+    """找不到城市或 API 回應異常。"""
+
+
+@dataclass(frozen=True)
+class Location:
+    name: str
+    country: str | None
+    latitude: float
+    longitude: float
+
+    @property
+    def label(self) -> str:
+        if self.country:
+            return f"{self.name}, {self.country}"
+        return self.name
+
+
+def _http_get(url: str, params: dict[str, Any]) -> dict[str, Any]:
+    try:
+        resp = requests.get(
+            url,
+            params=params,
+            timeout=config.REQUEST_TIMEOUT_SEC,
+            headers={"User-Agent": config.USER_AGENT},
+        )
+    except requests.RequestException as exc:
+        raise WeatherError(f"網路錯誤：{exc}") from exc
+
+    if resp.status_code != 200:
+        raise WeatherError(
+            f"API 回應 {resp.status_code}：{resp.text[:160] or '(empty body)'}"
+        )
+
+    try:
+        return resp.json()
+    except ValueError as exc:
+        raise WeatherError(f"API 回傳非 JSON：{exc}") from exc
+
+
+def geocode(city: str) -> Location:
+    """把城市名轉成經緯度。找不到就拋 WeatherError。"""
+    city = city.strip()
+    if not city:
+        raise WeatherError("城市名不可為空")
+
+    data = _http_get(
+        config.OPEN_METEO_GEOCODE_URL,
+        {"name": city, "count": 1, "language": "zh", "format": "json"},
+    )
+
+    results = data.get("results") or []
+    if not results:
+        raise WeatherError(f"找不到『{city}』這個城市（請用英文或加國家名）")
+
+    top = results[0]
+    try:
+        return Location(
+            name=str(top["name"]),
+            country=top.get("country"),
+            latitude=float(top["latitude"]),
+            longitude=float(top["longitude"]),
+        )

+## 架構決策
+
+### 為什麼選 Open-Meteo 而非 OpenWeatherMap？
+- **免 API key**：不用註冊、不用管 secret rotation、部署零摩擦
+- **無 quota**：個人用途不被擋
+- **支援中文 geocoding**（`language=zh`）→ 回傳「台北市, 台湾」更友善
+- **單一 vendor**：geocoding + current weather 同站提供，少一次失敗點
+
+### 模組化（單一職責）
+- `config.py` 只負責讀 env，給出明確的 `ConfigError`
+- `weather.py` 只負責「城市 → 天氣」，定義 `WeatherError`
+- `bot.py` 只負責 Telegram 路由與回覆樣式
+- 每層不知道彼此的內部實作 → 任何一層可獨立換掉（例如改用 weatherapi.com）
+
+### Error handling 原則
+- `requests` 任何網路/JSON 失敗 → `WeatherError` 帶人話訊息
+- Geocoding 找不到城市 → 明確訊息 + 提示用英文/加國家名
+- 找不到 token → `SystemExit(1)` + log，**不要 polling 失敗了才發現**
+- handler 內 finally 兜底，未預期錯誤不會讓整個 polling 掛掉
+
+### 不用 polling 以外的機制
+任務規格沒要 webhook（要 public HTTPS URL + reverse proxy），polling 對單人 bot 已足夠。
