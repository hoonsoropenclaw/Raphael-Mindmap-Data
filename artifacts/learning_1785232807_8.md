+// Cross-tab sync via the 'storage' event. Handles JSON parse failures gracefully
+// (returns fallback, never throws) so a corrupted localStorage can't crash the app.

+import { useEffect, useRef, useState, useCallback } from 'react';

+export function useLocalStorage(key, initialValue) {
+  const keyRef = useRef(key);
+  keyRef.current = key;
+
+  const read = useCallback(() => {
+    if (typeof window === 'undefined') {
+      return typeof initialValue === 'function' ? initialValue() : initialValue;
+    }
+    try {
+      const raw = window.localStorage.getItem(keyRef.current);
+      if (raw == null) {
+        return typeof initialValue === 'function' ? initialValue() : initialValue;
+      }
+      return JSON.parse(raw);
+    } catch (_err) {
+      // Corrupted JSON or storage disabled (private mode etc.) — silently fall back.
+      return typeof initialValue === 'function' ? initialValue() : initialValue;
+    }
+  }, [initialValue]);
+
+  const [value, setValue] = useState(read);
+
+  const set = useCallback((next) => {
+    setValue((prev) => {
+      const resolved = typeof next === 'function' ? next(prev) : next;
+      try {
+        window.localStorage.setItem(keyRef.current, JSON.stringify(resolved));
+      } catch (_err) {
+        // quota exceeded or storage disabled — keep in-memory value, don't crash.
+      }
+      return resolved;
+    });
+  }, []);
+
+  // Cross-tab sync: when another tab writes to the same key, update ours.
+  useEffect(() => {
+    const onStorage = (event) => {
+      if (event.key !== keyRef.current) return;
+      try {
+        setValue(event.newValue == null

+## 3. State 設計（single source of truth）
+
+```js
+{
+  columns: ['Backlog', 'In Progress', 'Review', 'Done'],
+  tasks: { [id]: { id, title, description, priority, assignee, column, createdAt, completedAt? } },
+  order: { [columnId]: [taskId, ...] },   // 分開儲存排序，避免任務物件裡藏冗餘 column 屬性
+  filter: { priority: null, assignee: null },
+  search: '',
+  theme: 'light' | 'dark',
+  history: { past: [...], future: [...] },  // 50 步上限
+}
+```
+
+**關鍵決策**：
+- **tasks 是 flat lookup，order 是 per-column array** — 移動任務時只要 splice 兩個 array，不用 mutate task 物件（除了更新 column 屬性以方便 selector）
+- **filter / search 不進 history** — UI-only state 進 undo 會變得不可預測
+- **HYDRATE / RESET 不進 history** — 初始化操作不污染 undo stack
