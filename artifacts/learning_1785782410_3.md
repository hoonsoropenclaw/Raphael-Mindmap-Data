+"""Centralised configuration for the multilingual Telegram bot.
+
+Everything tunable lives here so the bot body stays focused on routing and
+i18n. Secrets come from environment variables — never hard-coded.
+"""
+from __future__ import annotations
+
+import os
+from dataclasses import dataclass, field
+from typing import FrozenSet
+
+
+# ---------------------------------------------------------------------------
+# Environment-driven secrets / endpoints
+# ---------------------------------------------------------------------------
+
+def _env(name: str, default: str | None = None, *, required: bool = False) -> str:
+    val = os.environ.get(name, default)
+    if required and not val:
+        raise RuntimeError(
+            f"Missing required env var: {name}. "
+            f"Set it before running the bot (see README)."
+        )
+    return val or ""
+
+
+@dataclass(frozen=True)
+class Secrets:
+    telegram_bot_token: str = field(default_factory=lambda: _env("TELEGRAM_BOT_TOKEN", required=True))
+    openai_api_key: str = field(default_factory=lambda: _env("OPENAI_API_KEY", required=False))
+    openai_base_url: str = field(default_factory=lambda: _env("OPENAI_BASE_URL", default="https://api.openai.com/v1"))
+    openai_model: str = field(default_factory=lambda: _env("OPENAI_WHISPER_MODEL", default="whisper-1"))
+    openai_chat_model: str = field(default_factory=lambda: _env("OPENAI_CHAT_MODEL", default="gpt-4o-mini"))
+    # If empty, the bot still runs but voice messages will reply with a polite error.
+
+
+# ---------------------------------------------------------------------------
+# Behaviour toggles
+# ---------------------------------------------------------------------------
+
+@dataclass(frozen=True)
+class Settings:
+    # Default UI language for users who never set one (Telegram often gives
+    # us a hint via language_code, which we honour first when "auto").
+    default_language: str = field(default_factory=lambda: _env("BOT_DEFAULT_LANG", default="zh-Hant"))
+
+    # When True, after a successful transcription the bot will also translate
+    # the recognised text into the user's preferred UI language. Users can
+    # toggle this with /translate on|off.

+# Polyglot — Multilingual Telegram Bot with Whisper
+
+A production-ready Telegram bot that transcribes voice / video notes to text
+using OpenAI **Whisper**, then (optionally) translates the recognised text
+into the user's preferred language via GPT. Built on the async-first
+`python-telegram-bot` v22 stack.
+
+## Features
+
+- 🌐 **8 UI / transcription languages** out of the box: 繁體中文, 简体中文,
+  English, 日本語, 한국어, Español, Français, Deutsch. Add more by appending
+  one dict to `i18n.py`.
+- 🎙️ **Voice + video note + audio** routing — the bot picks the largest
+  available attachment and forwards it to Whisper.
+- ⚡ **Async-first** — `Application.concurrent_updates(True)` so one slow
+  Whisper call never blocks another user's message.
+- 🛡️ **Retry with exponential backoff + jitter** for transient OpenAI
+  errors (429, 5xx, network). Auth and bad-request errors fail fast.
+- 💾 **Per-user preferences persisted** in a tiny SQLite (WAL mode, atomic
+  upserts). Restart-safe.
+- 🌐 **Telegram language hint** auto-maps `zh-TW` → `zh-Hant`, `zh-CN` →
+  `zh-Hans`, etc., so first-time users land in the right UI immediately.
+- 🧪 **10 offline smoke tests** — no network, no real Telegram, no real
+  OpenAI needed. Run `python test_bot.py`.
+- 📜 **Graceful shutdown** — `SIGINT` / `SIGTERM` cleanly stops the
+  updater, then closes the Whisper HTTP session.
+- 🔌 **No external DB / Redis / broker** — single-file state for hobby
+  deployments; swap in Postgres later if you scale.
+
