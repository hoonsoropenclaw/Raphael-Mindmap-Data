+"""SQLite 資料庫連線與 schema 管理。"""
+from __future__ import annotations
+
+import sqlite3
+from contextlib import contextmanager
+from pathlib import Path
+from typing import Iterator
+
+DEFAULT_DB_PATH = Path(__file__).resolve().parent.parent / "todos.db"
+
+SCHEMA = """
+CREATE TABLE IF NOT EXISTS todos (
+    id          INTEGER PRIMARY KEY AUTOINCREMENT,
+    title       TEXT    NOT NULL,
+    description TEXT    NOT NULL DEFAULT '',
+    completed   INTEGER NOT NULL DEFAULT 0,
+    created_at  TEXT    NOT NULL DEFAULT (datetime('now'))
+);
+CREATE INDEX IF NOT EXISTS idx_todos_completed ON todos(completed);
+"""
+
+
+def _connect(db_path: str | Path) -> sqlite3.Connection:
+    conn = sqlite3.connect(str(db_path), check_same_thread=False)
+    conn.row_factory = sqlite3.Row
+    conn.execute("PRAGMA foreign_keys = ON")
+    return conn
+
+
+def init_db(db_path: str | Path = DEFAULT_DB_PATH) -> None:
+    """初始化 schema,並對每條 CREATE 語句逐條執行。"""
+    conn = _connect(db_path)
+    try:
+        for stmt in SCHEMA.strip().split(";"):
+            sql = stmt.strip()
+            if sql:
+                conn.execute(sql)
+        conn.commit()
+    finally:
+        conn.close()
+
+
+@contextmanager
+def get_conn(db_path: str | Path = DEFAULT_DB_PATH) -> Iterator[sqlite3.Connection]:
+    """Context manager:自動 commit 或 rollback,確保連線關閉。"""
+    conn = _connect(db_path)
+    try:
+        yield conn
+        conn.commit()
+    except Exception:
+        conn.rollback()
+        raise
+    finally:
+        conn.close()

+## 1. 整體設計哲學
+
+**目標**:寫一個 production-ready、可獨立運作的 Todo API,主軸是「正確性、測試覆蓋、結構清晰」。

+**分層架構** (符合 FastAPI 慣例):
+```
+routes (main.py) → CRUD (crud.py) → SQLite (database.py)
+                ↓
+            Pydantic schemas (schemas.py)
+                ↓
+        Rate limiting (rate_limit.py)
+```
+
+每一層可以單獨測試,介面透過 type hints 嚴格定義。
+
+## 2. 關鍵決策
+
+### 2.1 `create_app(db_path, rate_limit)` Factory 模式
+
+```python
+def create_app(db_path: str | None = None, rate_limit: str | None = None) -> FastAPI:
+```
+
+**為什麼不用 module-level app?** 因為測試需要:
+- 注入 db_path (每次 test 用 tempfile,避免 state leak)
+- 注入 rate_limit (rate-limit 測試需要嚴格上限)
+- 兩個 create_app 應有獨立 limiter storage (避免跨測試污染)
+
+這個設計在 `test_rate_limit_independent_per_ip_storage` 被驗證。
+
+### 2.2 SQLite + stdlib (不選 SQLAlchemy)
+
+- 題目要求簡單 Todo API,SQLAlchemy 是 overkill
+- stdlib `sqlite3` 已足夠,沒有額外 dependency
+- `check_same_thread=False` + `get_conn` context manager 處理 thread-safety (FastAPI 雖然是 async 但 sync route 會在 threadpool 跑)
+
+### 2.3 Rate Limiter 設計:用 SlowAPI + per-app limiter
+
+每個 `create_app` 透過 `make_limiter(rate_limit)` 拿到獨立 `Limiter` 實例。
+- 預設 60/minute per IP (SlowAPI `get_remote_address`)
+- 超過回 429 + 自訂 JSON `{"detail": "...", "limit": "..."}`
+- 環境變數 `RATE_LIMIT` 可覆寫 (`60/minute` / `10/minute` 等都行)
+
+**踩坑**:slowapi 的 `@limiter.limit` decorator 必須**先 application decorator 內,後 slowapi 限流外** — 順序換了會讓 FastAPI 看不到正確的 signature。已驗證正確順序。
+
+### 2.4 PATCH 端點用 `TodoUpdate` 部分更新
+
+```python
+class TodoUpdate(BaseModel):

+    description: str | None = None
+    completed: bool | None = None
+```
+
+允許 client 只送要改的欄位,沒送的欄位保持不動 — CRUD 層用 `if x is not None: ...` 動態組 UPDATE SQL。
