+##   - Polls every `poll_interval_sec` to stay portable across all platforms.
+##   - Captures a directory snapshot (path -> mtime) and diffs it on each tick.
+##   - Supports glob filters (e.g. "*.txt" or "*.{log,txt}") and recursive mode.
+##   - All signals are emitted on the main thread (poll runs in `_process`).
+
+signal event_detected(event_type: String, path: String, is_directory: bool, dest_path: String)
+signal watcher_started(path: String)
+signal watcher_stopped()
+signal watcher_error(message: String)
+
+enum EventType { CREATED, MODIFIED, DELETED, MOVED }
+
+const DEFAULT_POLL_INTERVAL := 0.6

+var is_running: bool = false
+
+# Snapshot: path -> mtime (Unix seconds, float).
+var _snapshot: Dictionary = {}
+# Pending rename tracking (path -> old path); helps us emit MOVED with dest.
+var _pending_renames: Dictionary = {}
+var _accumulator: float = 0.0
+
+
