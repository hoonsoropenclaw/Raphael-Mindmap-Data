+"""SQLite persistence layer for todos.
+
+Design notes
+------------
+* Single connection per request via FastAPI dependency injection.
+  SQLite handles concurrency through its own locking; for a single-process
+  FastAPI app this is enough. For multi-writer production we would switch
+  to a pool or Postgres, but that is out of scope.
+* Timestamps are stored as ISO-8601 strings — sortable, human-readable,
+  no timezone arithmetic needed at the storage layer.
+* The `done` flag is stored as INTEGER (0/1) because SQLite has no native
+  boolean type.

+from __future__ import annotations
+
+import sqlite3
+from contextlib import contextmanager
+from datetime import datetime, timezone
+from typing import Iterator
+
+SCHEMA = """
+CREATE TABLE IF NOT EXISTS todos (
+    id          INTEGER PRIMARY KEY AUTOINCREMENT,
+    title       TEXT    NOT NULL,
+    done        INTEGER NOT NULL DEFAULT 0,
+    created_at  TEXT    NOT NULL
+);
+CREATE INDEX IF NOT EXISTS idx_todos_done ON todos(done);
+"""
+
+
+def _now_iso() -> str:
+    return datetime.now(timezone.utc).isoformat()
+
+
+def init_db(conn: sqlite3.Connection) -> None:
+    """Create tables. Safe to call repeatedly (IF NOT EXISTS)."""
+    conn.executescript(SCHEMA)
+    conn.commit()
+
+
+@contextmanager
+def transaction(conn: sqlite3.Connection) -> Iterator[sqlite3.Connection]:
+    """Wrap a unit of work in a transaction with commit/rollback."""
+    try:
+        yield conn
+        conn.commit()
+    except Exception:
+        conn.rollback()
+        raise

+        "created_at": created,

+"""In-process sliding-window rate limiter.
+
+Why sliding window
+------------------
+A fixed-window counter (e.g. "reset every minute on the wall clock") allows
+clients to send 2*N requests in the second spanning a window boundary.
+A sliding window over the most recent N seconds is fairer and only costs
+keeping a per-key deque of timestamps.
+
+Why in-process
+--------------
+This is sufficient for a single-process FastAPI app. For multi-worker /
+multi-host deployments swap for Redis (`INCR` + `EXPIRE`) or a token-bucket
+proxy. Keeping the implementation here in pure stdlib makes tests fast
+and the contract clear.
+
+Thread-safety
+-------------
+A `threading.Lock` guards the per-key deques. The lock is held only for
+the small critical section that prunes old timestamps and appends the new
+one, so contention is low.
+"""
+from __future__ import annotations
+
+import threading
+import time
+from collections import defaultdict, deque
+from dataclasses import dataclass
+
+
+class RateLimitExceeded(Exception):
+    """Raised when a caller exceeds the configured quota."""
+
+    def __init__(self, retry_after: float):
+        super().__init__(f"rate limit exceeded; retry after {retry_after:.3f}s")
+        self.retry_after = retry_after
+
+
+@dataclass
+class RateLimitResult:
+    allowed: bool
+    remaining: int
+    retry_after: float
+
+
+class SlidingWindowRateLimiter:
+    """Per-key sliding window over a configurable interval."""
+
+    def __init__(self, max_requests: int, window_seconds: float):
+        if max_requests <= 0:
+            raise ValueError("max_requests must be > 0")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be > 0")
+        self._max = max_requests
+        self._window = float(window_seconds)
+        self._buckets: dict[str, deque[float]] = defaultdict(deque)
+        self._lock = threading.Lock()
+
+    def check(self, key: str, *, now: float | None = None) -> RateLimitResult:
+        """Atomically check and record a hit for ``key``.
+
+        Returns whether the request is allowed, how many quota remain in
+        the current window, and — if denied — how long until the oldest
+        hit falls outside the window.
+        """
+        ts = time.monotonic() if now is None else now
+        cutoff = ts - self._window
+        with self._lock:
+            bucket = self._buckets[key]
+            # Prune timestamps that fell out of the window.
+            while bucket and bucket[0] <= cutoff:
+                bucket.popleft()
+            if len(bucket) >= self._max:
+                retry_after = max(0.0, bucket[0] + self._window - ts)
+                return RateLimitResult(False, 0, retry_after)
+            bucket.append(ts)
+            return RateLimitResult(True, self._max - len(bucket), 0.0)

+# Todo API
+
+A small FastAPI service that exposes CRUD over todos, persists them in SQLite, and rate-limits every request by client IP using a sliding-window counter.
+
+## Layout
+
+```
+app/

+  config.py        # Settings loaded from env vars

+  main.py          # FastAPI app factory + routes
+tests/
+  conftest.py      # Shared fixtures (in-memory DB, TestClient)
+  test_db.py       # CRUD unit tests
+  test_rate_limit.py  # Limiter unit tests
+  test_api.py      # HTTP-level tests via TestClient
+pyproject.toml
+```
+
+## Setup
+
+```bash
+python3 -m venv .venv
+source .venv/bin/activate
+pip install -e ".[dev]"

+uvicorn app.main:app --reload
+```
+

+
+## Endpoints
+
+| Method | Path              | Body                                | Notes                              |
+|--------|-------------------|-------------------------------------|------------------------------------|
+| GET    | `/healthz`        | —                                   | Bypasses rate limit                |

+| POST   | `/todos`          | `{"title": "..."}`                  | 201 on success, 422/400 on bad input |
+| GET    | `/todos/{id}`     | —                                   | 404 if missing                     |
+| PATCH  | `/todos/{id}`     | `{"title"?: "...", "done"?: bool}`  | Partial update                    |
+| DELETE | `/todos/{id}`     | —                                   | 204 on success, 404 if missing    |
+

+
+> Working notes captured during the build. Project lives at
+> `/home/hoonsoropenclaw/.hermes/projects/learning_1785216005_0/`.

+## 1. Scope & Constraints
+
+The task was intentionally small but had three explicit requirements that
+each shape the design:
+
+1. **FastAPI** for the HTTP layer (routing, validation, OpenAPI for free).
+2. **Rate limit** — must reject excess traffic, must surface it via headers,
+   must be testable.
+3. **SQLite** — single-file persistence, no external service, no migrations
+   framework needed at this size.
+4. **Unit tests** — not "tests that happen to pass", but tests that pin
+   behaviour so refactors stay safe.
+
+## 2. Module Decomposition
+
+```
+app/
+  config.py        — env-driven Settings dataclass
+  db.py            — SQLite schema + CRUD helpers (no FastAPI imports)
+  rate_limit.py    — pure-stdlib sliding-window limiter
+  schemas.py       — Pydantic models for request/response
+  main.py          — FastAPI factory + routes
+tests/
+  test_db.py             — CRUD unit tests (no HTTP)
+  test_rate_limit.py     — Limiter unit tests (no HTTP)
+  test_api.py            — TestClient HTTP tests
+```
+
+The split is deliberate:
+
+* `db.py` and `rate_limit.py` know **nothing** about FastAPI. They are
+  pure logic modules that can be exercised in microseconds without an
+  ASGI stack. That is what makes their tests fast and the contracts
+  obvious.
+* `main.py` is the only file that wires everything together. It is also
+  the only file that needs to change when adding endpoints.
+* `config.py` exists because I wanted runtime knobs (DB path, limit,
+  window) but did not want to sprinkle `os.environ.get(...)` calls
+  across the codebase.
+
+## 3. Key Design Decisions
+
+### 3.1 Sliding window over fixed window
+
+A fixed-window counter (reset every N seconds on the wall clock) is the
+naïve rate-limit implementation and it has a well-known flaw: a client
+can send `2*limit` requests in the span of two seconds around a window
+boundary. The sliding window I implemented keeps a deque of timestamps
+per key and prunes anything older than `now - window`. This is still
+O(1) amortised (each timestamp is appended once and popped once) and
+gives exact behaviour at the boundary.
+
+### 3.2 In-process state with a clear swap-out path
+
+The limiter is in-process because that's the simplest thing that
+satisfies the task. A docstring in `rate_limit.py` calls out the
+production swap (Redis `INCR` + `EXPIRE` or a token-bucket proxy). The
+reasoning is: do not invent abstractions before they are needed, but
+document the upgrade path so the next person does not have to.
+
+### 3.3 `check_same_thread=False` for SQLite
+
+FastAPI workers can dispatch requests on different threads when sync
+defs are used. SQLite connections are thread-unsafe by default, so
+`check_same_thread=False` is required for the app to work. The DB layer
+never shares mutable state across connections — every CRUD call goes
+through a single shared `Connection` whose access is serialised by
+SQLite's own locking.
