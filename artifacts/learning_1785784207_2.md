+class _JsonFormatter(logging.Formatter):
+    """JSON formatter（單行 JSON，方便 Loki/CloudWatch 解析）"""
+
+    def format(self, record: logging.LogRecord) -> str:
+        payload: dict[str, Any] = {
+            "ts": datetime.fromtimestamp(record.created, tz=timezone.utc).isoformat(),
+            "level": record.levelname,

+                payload[k] = _scrub(k, v)
+        # 任何自定欄位（不屬於 LogRecord 內建）
+        for key, value in record.__dict__.items():
+            if key in {
+                "name", "msg", "args", "levelname", "levelno", "pathname",
+                "filename", "module", "exc_info", "exc_text", "stack_info",
+                "lineno", "funcName", "created", "msecs", "relativeCreated",
+                "thread", "threadName", "processName", "process", "taskName",
+            }:
+                continue

+## ✨ 核心特性
+
+| 特性 | 說明 |
+|------|------|
+| **批次廣播** | 一個 API 呼叫 → 同一訊息送到 N 個 chat_id（並行派送） |
+| **訊息格式** | text / MarkdownV2 / HTML / 靜音，多種模式自動對映 Telegram parse_mode |
+| **Provider 抽象** | 同一套介面可在「Mock provider（無需 token）」與「真實 Telegram API」之間切換 |
+| **自動 retry** | tenacity 指數回退 + 區分 transient / permanent / rate-limit 錯誤 |

+| **Webhook 入口** | 接收 Telegram push 的 update，含 secret token 驗證 |
