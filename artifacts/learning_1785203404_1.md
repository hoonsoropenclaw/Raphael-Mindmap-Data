+import { boardReducer, buildInitialState } from './boardReducer.js'

+const STORAGE_KEY = 'kanboard:state:v1'
+const BoardContext = createContext(null)
+
+function serialize(state) {
+  // Set → Array 才能 JSON 化
+  return {
+    ...state,
+    filters: { ...state.filters, tagSet: Array.from(state.filters.tagSet) },
+  }
+}
+
+function deserialize(raw) {
+  return {
+    ...raw,
+    filters: { ...raw.filters, tagSet: new Set(raw.filters?.tagSet ?? []) },
+  }
+}
+
+export function BoardProvider({ children }) {
+  const [state, dispatch] = useReducer(boardReducer, undefined, buildInitialState)
+  const hydratedRef = useRef(false)
+
+  // 從 localStorage hydrate (只在 mount 時跑一次)
+  useEffect(() => {
+    try {
+      const raw = localStorage.getItem(STORAGE_KEY)
+      if (raw) {
+        const parsed = JSON.parse(raw)
+        dispatch({ type: 'HYDRATE', payload: deserialize(parsed) })
+      }
+    } catch (err) {
+      console.warn('[kanboard] hydrate failed:', err)
+    } finally {
+      hydratedRef.current = true
+    }
+  }, [])
+
+  // 持久化:state 變更後 debounce 寫回
+  useEffect(() => {
+    if (!hydratedRef.current) return
+    const id = setTimeout(() => {
+      try {
+        localStorage.setItem(STORAGE_KEY, JSON.stringify(serialize(state)))
+      } catch (err) {
+        console.warn('[kanboard] persist failed:', err)
+      }
+    }, 200)
+    return () => clearTimeout(id)
+  }, [state])
+
+  return <BoardContext.Provider value={{ state, dispatch }}>{children}</BoardContext.Provider>

+### 2.2 狀態管理:useReducer + Context,刻意不引入 Redux/Zustand
+
+**動機**:這個應用的 state graph 在一個檔記憶體內裝得下,但跨多個元件共享。用 useState 提到頂層會 prop drilling,用 Redux 會 over-engineering。

+{

+  columnOrder: [id, ...],                              // 欄位順序
+  filters: { query, tagSet, priority },                // 獨立維度
+  ui:      { selectedCardId, ... },                    // UI state

+```
+
+採用 **normalize pattern** (Redux 官方推薦):以 ID 為 key 存實體、另存 ID 陣列維護順序。理由:
+- O(1) lookup card by ID
+- MOVE_CARD 只動 cardIds、其餘不動

+- **Column CRUD**:`ADD_COLUMN` / `RENAME_COLUMN` / `DELETE_COLUMN`
