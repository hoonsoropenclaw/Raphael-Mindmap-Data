+// ──────────────────────────────────────────────────────────
+// Debounce utility. Architect feedback Round 2: "When
+// persisting to localStorage via effects, consider adding
+// a debounce or throttle". We use this in persist layer.
+// ──────────────────────────────────────────────────────────
+
+export function debounce(fn, wait) {
+  let timer = null;
+  const debounced = (...args) => {
+    if (timer) clearTimeout(timer);
+    timer = setTimeout(() => {
+      timer = null;
+      fn.apply(null, args);
+    }, wait);
+  };
+  debounced.cancel = () => {
+    if (timer) {
+      clearTimeout(timer);
+      timer = null;
+    }
+  };
+  debounced.flush = (...args) => {
+    if (timer) {
+      clearTimeout(timer);
+      timer = null;
+    }
+    fn.apply(null, args);
+  };
+  return debounced;

+// ──────────────────────────────────────────────────────────
+// useAsyncAction — simulates a network round-trip with
+// configurable latency and failure rate. Architect
+// feedback: "The simulated network latency + failure
+// rate is a brilliant touch to prove real-world UI
+// resilience."
+//
+// Use case: optimistic UI — dispatch start, call this
+// promise, dispatch success or error. Reducer tracks
+// pending / error state per task id.
+// ──────────────────────────────────────────────────────────
+
+import { useCallback } from "react";
+
+/**
+ * Returns a function: ({ latencyMs, failureRate, shouldFail })
+ * => Promise that resolves after latencyMs (or rejects with
+ * a synthetic "network" error if shouldFail / random).
+ */
+export function useAsyncAction() {
+  return useCallback(function asyncAction({
+    latencyMs = [120, 480],
+    failureRate = 0,
+    shouldFail = false,
+  } = {}) {
+    return new Promise((resolve, reject) => {
+      const delay =
+        Array.isArray(latencyMs)
+          ? latencyMs[0] + Math.random() * (latencyMs[1] - latencyMs[0])
+          : latencyMs;
+      setTimeout(() => {
+        if (shouldFail || Math.random() < failureRate) {
+          reject(new Error("Network request failed (simulated)"));
+        } else {
+          resolve({ ok: true });
+        }
+      }, delay);
+    });
+  }, []);

+/**

+export function isStorageAvailable() {
+  try {
+    if (typeof localStorage === "undefined") return false;
+    const probe = "__tm_probe__";
+    localStorage.setItem(probe, probe);

+    }
+    return parsed;
