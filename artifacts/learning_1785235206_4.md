+/**
+ * 副作用層: localStorage 同步 + prefers-reduced-motion 偵測
+ *
+ * 為什麼拆 useEffect 到自訂 hook:
+ * - TaskProvider 本身保持純粹(只負責 state + dispatch)
+ * - 不同環境(N100 / 瀏覽器 / SSR)各自有不同副作用需求,自訂 hook 更容易換
+ */
+
+import { useEffect, useState } from 'react';
+import type { Task } from './types';
+
+const STORAGE_KEY = 'react-task-manager:v1';
+
+export const useTaskPersistence = (tasks: Task[]) => {
+  // hydrate from localStorage on mount
+  useEffect(() => {
+    try {
+      const raw = localStorage.getItem(STORAGE_KEY);
+      if (raw) {
+        const parsed = JSON.parse(raw) as Task[];
+        if (Array.isArray(parsed) && parsed.every((t) => t.id && t.title)) {
+          window.dispatchEvent(
+            new CustomEvent('__task_hydrate__', { detail: parsed }),
+          );
+        }
+      }
+    } catch {
+      // localStorage 不可用(隱私模式 / SSR)就 silently skip
+    }
+  }, []);
+
+  // persist on change
+  useEffect(() => {
+    try {
+      localStorage.setItem(STORAGE_KEY, JSON.stringify(tasks));
+    } catch {
+      // quota 超過也是 silently skip
+    }
+  }, [tasks]);
+};
+
+export const usePrefersReducedMotion = (): boolean => {
+  const [prefers, setPrefers] = useState<boolean>(() => {
+    if (typeof window === 'undefined') return false;
+    return window.matchMedia('(prefers-reduced-motion: reduce)').matches;
+  });
+
+  useEffect(() => {
+    const mq = window.matchMedia('(prefers-reduced-motion: reduce)');
+    const handler = (e: MediaQueryListEvent) => setPrefers(e.matches);
+    mq.addEventListener('change', handler);
+    return () => mq.removeEventListener('change', handler);
+  }, []);
+
+  return prefers;
+};

+## 2. 架構決策
+
+### 2.1 狀態管理: 為什麼選 useReducer + Context,不上 Redux/Zustand
+
+**選 useReducer + Context 的理由**:
+- 規模 < 100 tasks,Redux 三大特色（time-travel devtools、middleware ecosystem、cross-slice subscribe）都用不到
+- useReducer 提供「所有 mutation 走 dispatched action」的單一入口,已經能滿足可預測性需求
+- Context 已經是 React 內建,零依賴、零 bundle cost
+
+**不上 Redux/Zustand 的理由**:
+- Redux: 對這個規模 overkill,3 個 file 的 boilerplate 換不到對應價值
+- Zustand: 雖然輕量,但「加一個依賴」要付「未來升級要付遷移成本」,現在 useReducer + Context 還撐得住
+
+**升級觸發點**(寫在程式碼註解裡,給未來的我):
+- tasks > 500,或需要 undo/redo → 加 redux-toolkit
+- 需要跨頁簽同步、需要 sync 到 server → 加 Zustand + persist middleware
+- 需要 optimistic update + 衝突解決 → 考慮 TanStack Query
+
+### 2.2 為什麼把 state 跟 dispatch 拆成兩個 Context
+
+寫在 `TaskContext.tsx` 的註解:
+```
+只讀 state 的元件不會因為 dispatch 變化 re-render
+只 dispatch 的元件不會因為 state 變化 re-render
+```
+
+對 100 個 task 的規模,re-render 影響其實有限,但這是**先做對的習慣**——元件一多就會感謝現在的自己。這條原則的「現在付的成本」= 5 個額外行;「未來省的成本」= 不用重構整個 Provider 結構。
+
+### 2.3 為什麼 filter state 不放 Reducer,放在 TaskManager 本地
+
+**這是刻意的非對稱**:
+- 任務資料 = 「業務資料」,放 Reducer 才有「所有 mutation 可追溯」的價值
+- 篩選條件 = 「UI 暫時狀態」,跟業務資料無關
+- 篩選放 Reducer 會污染任務資料的單一真實來源（reducer 不知道「這個 action 是 mutation 還是 UI state」）
+- 篩選不持久化也合理——重新整理就重置篩選（不是 bug,是 feature）
+
+**升級觸發點**: 若篩選要被深層元件（modal、popover）共享 → 升級到 Context;若要持久化 → 改用 URL query string。
+
+### 2.4 為什麼把副作用（localStorage）拆到自訂 hook
+
+`useTaskPersistence` 在 `hooks.ts` 獨立:
+- TaskProvider 本體保持純粹(只負責 state + dispatch)
+- 不同環境(N100 / 瀏覽器 / SSR)各自有不同副作用需求,自訂 hook 更容易換實作
+- **測試更容易**:要測 reducer 邏輯不必 mock localStorage,要測持久化不必 render Provider
+
