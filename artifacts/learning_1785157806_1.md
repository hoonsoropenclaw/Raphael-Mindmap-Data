+"""Configuration loader for the weather Telegram bot.
+
+Reads TELEGRAM_BOT_TOKEN from the environment, falling back to a local
+``.env`` file. The bot will not start without a valid token; the absence
+of a token is treated as a configuration error rather than silently
+running in offline mode.
+
+Design notes
+------------
+- We use ``pathlib.Path.read_text`` (raw bytes) instead of streaming
+  ``open()`` to make the loader deterministic for tests.
+- We deliberately keep the loader dependency-free so it works under any
+  Python 3.10+ interpreter with the standard library only.
+"""
+
+from __future__ import annotations
+
+import os
+from dataclasses import dataclass
+from pathlib import Path
+
+
+PROJECT_ROOT = Path(__file__).resolve().parent.parent
+DEFAULT_ENV_FILE = PROJECT_ROOT / ".env"
+
+
+@dataclass(frozen=True)
+class BotConfig:
+    """Resolved configuration for the bot."""
+
+    telegram_token: str
+    weather_timeout_seconds: float = 10.0
+    weather_base_url: str = "https://wttr.in"
+
+
+class ConfigError(RuntimeError):
+    """Raised when the bot cannot be configured (e.g. missing token)."""
+
+
+def _load_env_file(env_file: Path) -> dict[str, str]:
+    """Parse a minimal ``KEY=VALUE`` env file. Lines starting with ``#``
+    or blank lines are ignored. Values may be optionally quoted with
+    single or double quotes.
+    """
+
+    result: dict[str, str] = {}
+    if not env_file.exists():
+        return result
+
+    for raw_line in env_file.read_text(encoding="utf-8").splitlines():
+        line = raw_line.strip()
+        if not line or line.startswith("#"):
+            continue
+        if "=" not in line:
+            continue
+        key, _, value = line.partition("=")
+        key = key.strip()
+        value = value.strip()
+        if len(value) >= 2 and value[0] == value[-1] and value[0] in {"'", '"'}:
+            value = value[1:-1]
+        result[key] = value
+    return result
+
+

+# Weather Telegram Bot
+
+A small Telegram bot that answers weather questions by hitting
+[wttr.in](https://github.com/chubin/wttr.in). Built on
+`python-telegram-bot 22.x` and `aiohttp`.
+
+```
+you     -> /weather Taipei        -> bot
+you     -> Taipei                 -> bot (implicit query)
+you     -> /help                  -> bot (markdown help)
+```
+
+## Layout
+
+```
+weather-telegram-bot/
+├── src/
+│   ├── config.py     # BotConfig + .env loader (no third-party deps)

+│   ├── bot.py        # python-telegram-bot application & handlers
+│   └── __init__.py
+├── tests/
+│   ├── test_config.py   # env parsing, token validation, precedence
+│   ├── test_weather.py  # payload parsing, formatting, optional live
+│   └── test_bot.py      # handler smoke tests (no network)
+├── verify_weather.py    # standalone end-to-end script (no Telegram)
+├── requirements.txt
+├── .env.example
+└── run.sh            # local venv wrapper
+```
