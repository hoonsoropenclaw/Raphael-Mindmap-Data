+def normalize_url(url: str) -> str:
+    """URL 正規化:
+    - 小寫 scheme + host
+    - 移除 fragment
+    - 排序 query 參數
+    - 移除 default port (80/443)
+    - 移除結尾多餘斜線
+    """
+    parsed = urlparse(url.strip())
+    scheme = parsed.scheme.lower()
+    netloc = parsed.netloc.lower()
+    # 去掉預設 port
+    if (scheme == "http" and netloc.endswith(":80")) or (
+        scheme == "https" and netloc.endswith(":443")
+    ):
+        netloc = netloc.rsplit(":", 1)[0]
+
+    # 排序 query
+    query_pairs = parse_qsl(parsed.query, keep_blank_values=True)
+    query_pairs.sort()
+    query = urlencode(query_pairs)
+
+    # path 移除結尾多餘斜線(根目錄保留 /)
+    path = parsed.path or "/"
+    if len(path) > 1:
+        path = path.rstrip("/")
+
+    return urlunparse((scheme, netloc, path, parsed.params, query, ""))

+"""
+parser.py — HTML link extraction + 內容摘要
+- 用 BeautifulSoup 抽 <a href> 連結
+- 回傳 (title, text_preview, links)
+"""
+from __future__ import annotations
+
+import logging
+import re
+from typing import List, Tuple
+from urllib.parse import urljoin
+
+from bs4 import BeautifulSoup
+
+logger = logging.getLogger(__name__)
+
+_LINK_RE = re.compile(r"^https?://", re.I)
+
+
+def extract_links(html: str, base_url: str) -> List[str]:
+    """從 HTML 抽出所有 <a href>,join 絕對 URL,過濾空字串/錨點/javascript。"""
+    soup = BeautifulSoup(html, "html.parser")
+    out: List[str] = []
+    for a in soup.find_all("a", href=True):
+        href = a["href"].strip()
+        if not href or href.startswith(("javascript:", "mailto:", "tel:", "#")):
+            continue
+        absolute = urljoin(base_url, href)
+        if _LINK_RE.match(absolute):
+            out.append(absolute)
+    # 同頁去重(保持順序)
+    return list(dict.fromkeys(out))
+
+
+def extract_metadata(html: str) -> Tuple[str, str]:
+    """回傳 (title, 純文字前 240 字預覽)。"""
+    soup = BeautifulSoup(html, "html.parser")
+    title_tag = soup.find("title")
+    title = title_tag.get_text(strip=True) if title_tag else ""
+
+    for tag in soup(["script", "style", "noscript"]):
+        tag.decompose()
+    text = soup.get_text(separator=" ", strip=True)
+    text = re.sub(r"\s+", " ", text)
+    return title, text[:240]

+# Async Crawler Framework
+
+基於 `asyncio` + `aiohttp` 的高效網頁爬蟲框架,搭配完整的**重試機制**(指數回退 + 429 Retry-After + jitter 抗 thundering herd)。

+## 核心特色
+
+| 能力 | 實作 |
+|------|------|
+| 高併發非同步 I/O | `aiohttp.ClientSession` 共用 + `TCPConnector` 連線池 |
+| Worker-Pump BFS | 多 pump worker 並行消費 `asyncio.Queue` frontier |
+| URL 去重 | 正規化(小寫 host / 排 query / 去 fragment / 去預設 port)+ `asyncio.Lock` |
+| **指數回退重試** | `multiplier=2.0`、`jitter=±20%` 抗 thundering herd |
+| 429 友善 | 解析 `Retry-After` header,優先用 server 建議的等待時間 |
+| 錯誤分類 | 4xx → `PermanentError`(不重試);5xx/timeout/連線錯誤 → `TransientError`(重試) |
+| Per-host 速率限制 | Token-bucket 風格,用 lock + monotonic time 實作 |
+| 觀察性 | 即時 hook + 完整 `stats.py` 聚合 per-host metrics |
