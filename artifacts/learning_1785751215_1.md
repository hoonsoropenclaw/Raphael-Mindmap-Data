// Smoke test: 模擬 RBAC + dynamic form 渲染核心邏輯
const ROLES = {
  admin:  { permissions: { create:true, read:true, update:true, delete:true, publish:true, export:true } },
  editor: { permissions: { create:true, read:true, update:true, delete:false, publish:true, export:true } },
  viewer: { permissions: { create:false, read:true, update:false, delete:false, publish:false, export:true } },
  guest:  { permissions: { create:false, read:true, update:false, delete:false, publish:false, export:false } },
};
const FIELD_ACCESS = {
  admin:  { read: true,  write: true },
  editor: { read: true,  write: true },
  viewer: { read: true,  write: false },
  guest:  { read: false, write: false },
};
const can = role => action => ROLES[role].permissions[action];
const fieldVisible = (role, f) => f.visibleTo.includes(role);
const fieldWritable = (role, f) => fieldVisible(role, f) && FIELD_ACCESS[role].write;

// 請假表單 schema
const leaveForm = {
  fields: [
    { name:'employeeId', visibleTo:['admin','editor','viewer'] },
    { name:'reason',     visibleTo:['admin','editor'] },        // viewer/guest 看不到
    { name:'managerApproval', visibleTo:['admin'] },            // 只有 admin 看得到
  ]
};

console.log('=== 角色權限矩陣 ===');
['admin','editor','viewer','guest'].forEach(r => {
  const visible = leaveForm.fields.filter(f => fieldVisible(r, f)).map(f=>f.name);
  const writable = leaveForm.fields.filter(f => fieldWritable(r, f)).map(f=>f.name);
  console.log(`${r.padEnd(7)} | visible: [${visible.join(', ')}] | writable: [${writable.join(', ')}]`);
});

console.log('\n=== 操作權限 ===');
['create','update','delete','publish','export'].forEach(action => {
  console.log(`${action.padEnd(8)}: ${['admin','editor','viewer','guest'].map(r => can(r)(action)?'●':'○').join(' ')}`);
});

// 驗證關鍵斷言
const tests = [
  ['admin 看得到 reason', fieldVisible('admin', leaveForm.fields[1]) === true],
  ['viewer 看不到 reason', fieldVisible('viewer', leaveForm.fields[1]) === false],
  ['guest 看不到 managerApproval', fieldVisible('guest', leaveForm.fields[2]) === false],
  ['admin 可以 delete', can('admin')('delete') === true],
  ['viewer 不能 update', can('viewer')('update') === false],
  ['viewer 不能 submit (write 假裝通過)', fieldWritable('viewer', leaveForm.fields[0]) === false],
];
const passed = tests.filter(t => t[1]).length;
console.log(`\n=== 測試結果：${passed}/${tests.length} 通過 ===`);
tests.forEach(([name, ok]) => console.log(`  ${ok?'✓':'✗'} ${name}`));
process.exit(passed === tests.length ? 0 : 1);
EOF
