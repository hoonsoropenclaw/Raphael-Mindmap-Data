+
+Uses SQLite for zero-setup local development. The DATABASE_URL env var can
+override the default file location; tests point it at a temp file via the
+fixture in tests/conftest.py.
+"""
+from __future__ import annotations
+
+import os
+from collections.abc import Generator
+
+from sqlalchemy import create_engine
+from sqlalchemy.engine import Engine
+from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker
+
+# Default to a file in the project root; tests override via env var.
+DEFAULT_DATABASE_URL = "sqlite:///./todos.db"
+
+
+def _build_engine(url: str) -> Engine:
+    """Build a SQLAlchemy engine.
+
+    SQLite needs ``check_same_thread=False`` so the same connection can be
+    used from the thread that runs the FastAPI request handler. The pool is
+    sized for a tiny dev workload; production would use a real DB.
+    """
+    connect_args = {"check_same_thread": False} if url.startswith("sqlite") else {}
+    return create_engine(url, connect_args=connect_args, future=True)
+
+
+def get_engine() -> Engine:
+    """Return an engine bound to the current DATABASE_URL."""
+    url = os.getenv("DATABASE_URL", DEFAULT_DATABASE_URL)
+    return _build_engine(url)
+
+
+engine: Engine = get_engine()
+SessionLocal = sessionmaker(bind=engine, autoflush=False, autocommit=False, future=True)
+
+
+class Base(DeclarativeBase):
+    """Declarative base shared by all ORM models."""
+
+
+def get_db() -> Generator[Session, None, None]:
+    """FastAPI dependency that yields a DB session and guarantees cleanup."""
+    db = SessionLocal()
+    try:
+        yield db
+    finally:
+        db.close()
+

+    from app import models  # noqa: F401
