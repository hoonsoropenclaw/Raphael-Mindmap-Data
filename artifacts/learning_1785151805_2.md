+type Priority = 'low' | 'medium' | 'high';
+type Task = { id: number; title: string; owner: string; priority: Priority; done: boolean };
+type State = { tasks: Task[]; filter: 'all' | 'active' | 'done'; query: string; focusMode: boolean };
+type Action =
+  | { type: 'add'; title: string }
+  | { type: 'toggle'; id: number }
+  | { type: 'remove'; id: number }
+  | { type: 'filter'; value: State['filter'] }
+  | { type: 'query'; value: string }
+  | { type: 'focus' };
+
+const initialState: State = {
+  filter: 'all', query: '', focusMode: false,
+  tasks: [
+    { id: 1, title: 'Review launch telemetry', owner: 'Mika', priority: 'high', done: false },
+    { id: 2, title: 'Tune empty-state copy', owner: 'You', priority: 'medium', done: false },
+    { id: 3, title: 'Archive sprint notes', owner: 'Noah', priority: 'low', done: true },
+    { id: 4, title: 'Verify mobile breakpoints', owner: 'You', priority: 'high', done: false },
+  ],
+};
+
+function reducer(state: State, action: Action): State {
+  switch (action.type) {
+    case 'add':
+      return action.title.trim() ? { ...state, tasks: [{ id: Date.now(), title: action.title.trim(), owner: 'You', priority: 'medium', done: false }, ...state.tasks] } : state;
+    case 'toggle': return { ...state, tasks: state.tasks.map(t => t.id === action.id ? { ...t, done: !t.done } : t) };
+    case 'remove': return { ...state, tasks: state.tasks.filter(t => t.id !== action.id) };
+    case 'filter': return { ...state, filter: action.value };
+    case 'query': return { ...state, query: action.value };
+    case 'focus': return { ...state, focusMode: !state.focusMode };
+  }

+# Command Center
+
+A React task-management UI built around a single reducer-driven state machine.
+
+## Architecture notes
+
+- `useReducer` owns task mutations, filtering, search, and focus mode so transitions are explicit and easy to test.
+- `useMemo` derives visible tasks instead of duplicating filtered state.
+- The UI keeps the data model intentionally small: task identity, ownership, priority, and completion are enough to demonstrate non-trivial interactions.
+- CSS uses a restrained editorial/operations direction: lime completion signal, coral active accent, mono metadata, and asymmetric hero composition.
+- Interaction safeguards include semantic buttons, ARIA labels, keyboard-friendly form controls, responsive layout, and an empty search state.
+
+## Integration
+
+Import `CommandCenter` and `CommandCenter.css` from a React application. The component expects React 18+.
+
+## Verification checklist
+
+- Add a priority and confirm it appears at the top.
+- Toggle completion and confirm progress updates.
+- Switch All / Active / Done filters.
+- Search by title, remove a task, and toggle focus mode.
+- Resize below 700px to verify the responsive stack.
