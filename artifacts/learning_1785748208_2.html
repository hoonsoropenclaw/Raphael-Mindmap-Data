▣ 使用範例
# 1. 最簡單用法（單 URL 抓取） from hermes_spider import FetcherFactory fetcher = FetcherFactory.create("http") resp = fetcher.get("https://example.com") print(resp.text("h1")) # 2. 寫一個爬蟲 from hermes_spider import Spider class MySpider(Spider): name = "myspider" start_urls = ["https://example.com"] def parse(self, response): for item in response.css(".item"): yield {"title": item.text} # 3. API 逆向工程 from hermes_spider import PlaywrightEngine, APIReverser with PlaywrightEngine() as engine: page, har = engine.new_page() page.goto("https://target.com") har.to_json("trace.har") reverser = APIReverser("trace.har") print(reverser.summary()) # 4. 啟動 Dashboard # $ hermes-spider serve --port 8000
