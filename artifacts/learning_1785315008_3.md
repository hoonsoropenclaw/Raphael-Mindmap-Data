+#!/usr/bin/env python3
+"""Tiny local HTTP server that mimics the OpenAI Whisper /v1/audio/transcriptions
+endpoint. Used for integration tests so we don't burn API quota.
+
+Usage:
+    python3 tools/mock_whisper_server.py [--port 7878] [--transcript "go north"]
+
+It accepts multipart/form-data uploads (matching real OpenAI Whisper) and
+returns:
+    {"text": "<echoed transcript>", "elapsed_sec": 0.0}
+
+The response text is taken from --transcript (CLI arg), an env var
+WHISPER_MOCK_TRANSCRIPT (default: "go north attack slime take potion"),
+or the prompt field if present, falling back to a per-file echo of the
+filename.
+
+Run in background, then point the game at WHISPER_ENDPOINT=http://127.0.0.1:7878/v1/audio/transcriptions
+"""
+from __future__ import annotations
+import argparse
+import json
+import os
+import sys
+from http.server import BaseHTTPRequestHandler, HTTPServer
+from typing import Tuple
+
+
+class WhisperHandler(BaseHTTPRequestHandler):
+    transcript_override: str = ""
+
+    def log_message(self, fmt: str, *args) -> None:
+        sys.stderr.write("[mock-whisper] " + (fmt % args) + "\n")
+
+    def do_POST(self) -> None:  # noqa: N802 - stdlib naming
+        ctype = self.headers.get("Content-Type", "")
+        length = int(self.headers.get("Content-Length", "0"))
+        body = self.rfile.read(length) if length else b""
+        if "multipart/form-data" not in ctype:
+            self.send_response(400)
+            self.end_headers()
+            self.wfile.write(b"expected multipart/form-data")
+            return
+        # Pull the "file" filename (very lightweight).
+        filename = "audio.wav"
+        # Pick transcript.
+        if WhisperHandler.transcript_override:
+            text = WhisperHandler.transcript_override
+        else:
+            text = os.environ.get("WHISPER_MOCK_TRANSCRIPT", "go north attack slime take potion")
+        payload = json.dumps({"text": text, "elapsed_sec": 0.05}).encode("utf-8")
+        self.send_response(200)
+        self.send_header("Content-Type", "application/json")
+        self.send_header("Content-Length", str(len(payload)))
+        self.end_headers()
+        self.wfile.write(payload)
+
+
+def main() -> int:
+    ap = argparse.ArgumentParser(description=__doc__)
+    ap.add_argument("--port", type=int, default=7878)
+    ap.add_argument("--host", default="127.0.0.1")
+    ap.add_argument("--transcript", default="")
+    args = ap.parse_args()
+    WhisperHandler.transcript_override = args.transcript
+    httpd = HTTPServer((args.host, args.port), WhisperHandler)
+    sys.stderr.write("[mock-whisper] listening on %s:%d\n" % (args.host, args.port))
+    try:
+        httpd.serve_forever()
+    except KeyboardInterrupt:
+        pass
+    return 0
+
+
+if __name__ == "__main__":
+    sys.exit(main())

+"""Generate a short PCM WAV for testing the voice pipeline.
+
+Writes a 1-second 16kHz mono sine wave to the path given on the CLI.
+Useful as a smoke test fixture for VoiceInput.process_file().
+"""
+from __future__ import annotations
+import math
+import struct
+import sys
+import wave
+
+
+def make_wav(path: str, seconds: float = 1.0, freq: float = 440.0, rate: int = 16000) -> None:
+    nframes = int(seconds * rate)
+    with wave.open(path, "wb") as wf:
+        wf.setnchannels(1)
+        wf.setsampwidth(2)
+        wf.setframerate(rate)
+        for i in range(nframes):
+            t = i / rate
+            sample = int(0.4 * 32767 * math.sin(2.0 * math.pi * freq * t))
+            wf.writeframes(struct.pack("<h", sample))
+
+

+    dur = float(sys.argv[2]) if len(sys.argv) > 2 else 1.0
+    make_wav(out, dur)
