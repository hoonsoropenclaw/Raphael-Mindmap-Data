+class Severity(Enum):
+    """重構 finding 的嚴重程度。
+
+    INFO    = 純風格建議 (例:變數命名風格不一致)
+    WARNING = 可讀性疑慮 (例:函式過長、巢狀過深)

+    WARNING = "warning"
+    ERROR = "error"

+    def fit(self, corpus: Iterable[PathLike]) -> "RefactorEngine":
+        """從訓練 corpus 學命名基線。

+        """

+        total = 0

+# mlrefactor — ML-style 自動化代碼重構工具
+
+> **透明聲明 (Transparency note)**
+> 本工具**並非**真正的機器學習系統,而是一個**模仿 sklearn 介面的 rule-based
+> 啟發式引擎**。介面契約 (`fit / predict / score`) 為真,但底層邏輯為
+> 確定性的 AST 規則。
+
+## 為什麼這樣設計
+
+真實 ML 模型需要訓練資料 + 計算資源,在 N100 headless 環境不可行。
+啟發式規則對「命名、複雜度、結構異味」這類結構性重構已能產出 80% 價值,
+且**可解釋、可審查**。用 sklearn 風格的 API 讓未來替換成真 ML 模型時,
+呼叫端零修改。
+
+## 5 條預定義規則
+
+| ID | 名稱 | 預設 severity | 偵測目標 |
+|----|------|---------------|----------|
+| R001 | long-function | WARNING | 函式 > 50 行 |
+| R002 | mutable-default | ERROR | 函式預設參數為 list/dict/set |
+| R003 | bare-except | WARNING | `except:` (沒指定 type) |
+| R004 | except-pass | ERROR | `except XXX: pass` (吞錯) |
+| R005 | naming-style | INFO | 變數/參數命名與訓練基線不一致 |
+
+## 安裝
+
+零依賴 — 純 Python 3.10+ stdlib。
+
+```bash
+git clone <repo>
+cd <repo>
+# 直接用
+python3 -m mlrefactor path/to/code.py
+```
+
+## CLI 使用
+
+```bash
+# 掃單一檔案
+python3 -m mlrefactor examples/dirty/01_mutable_and_except.py
+
+# 掃整個目錄(遞迴)
+python3 -m mlrefactor examples/dirty/
+
+# 訓練命名基線 + 掃描
+python3 -m mlrefactor examples/dirty/ --fit examples/refactored/
+
+# 寫報告到檔
+python3 -m mlrefactor examples/dirty/ --out reports/findings.md
+```
+
+## Python API 使用
+
+```python
+from mlrefactor import RefactorEngine
+
+engine = RefactorEngine()
+engine.fit(["path/to/corpus.py"])          # 學命名基線
+
+# 對 source code 字串
+findings = engine.predict("def foo(x=[]):\n    pass\n")
+for f in findings:
+    print(f.rule_id, f.severity.value, f.message, f.confidence)
+
+# 對 .py 檔
+findings = engine.predict("path/to/source.py")
+
+# 0.0~1.0 品質分數
+score = engine.score("path/to/source.py")
+print(f"Quality: {score:.2f}")
+```
