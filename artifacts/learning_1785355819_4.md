+@dataclass
+class Event:
+    id: str                       # uuid4
+    summary: str                  # 標題
+    description: str = ""
+    location: str = ""
+    start: datetime               # tz-aware Asia/Taipei
+    end: datetime
+    attendees: list[str] = field(default_factory=list)
+    reminders: list[Reminder]     # 至少一個
+    recurrence: Recurrence | None = None
+    calendar_id: str = "primary"
+    status: str = "confirmed"
+
+    def to_api_payload(self) -> dict: ...   # 序列化給 Google API
+    @classmethod
+    def from_api_payload(cls, d: dict) -> "Event": ...
+    def is_due(self, now: datetime) -> bool: ...   # 是否到期
+    def should_remind(self, now: datetime, lead_minutes: int) -> bool: ...
+
+@dataclass
+class Reminder:
+    method: Literal["email","popup","telegram","local","log"]
+    minutes_before: int = 0        # 0 = 準時
+    delivered: bool = False
+    delivered_at: datetime | None = None
+
+@dataclass
+class Recurrence:
+    freq: Literal["DAILY","WEEKLY","MONTHLY","YEARLY"]
+    interval: int = 1
+    count: int | None = None      # None = forever (cap at 365)
+    until: datetime | None = None
+```

+"""Event, Reminder, Recurrence 資料模型。
+
+設計原則 (Zero-Defect Program §3):
+- 每個欄位都有一個明確語義,不可用 Optional 模糊化

+"""
+from __future__ import annotations
+
+import uuid
+from dataclasses import dataclass, field, asdict
+from datetime import datetime, timedelta, timezone
+from typing import Literal
