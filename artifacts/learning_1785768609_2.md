+# Pre-implementation assessment
+
+## PRD modules
+1. Async transport: shared aiohttp session, bounded global concurrency, per-host throttling, retry/backoff.
+2. URL safety: HTTP(S) only, private-network/loopback blocking by default, redirect-hop validation.
+3. Crawl core: URL normalization/dedup, partial-failure isolation, lifecycle events.
+4. Passive API reverse engineering: inspect HTML forms, links, inline/external JavaScript references, fetch/XHR/Axios/GraphQL/WebSocket strings, and OpenAPI-like documents supplied by the operator.
+5. Data extraction: JSON/JSONL output, dotted-path selectors, record discovery, pagination URL templates.
+6. CLI: `crawl`, `discover`, and `extract` subcommands with explicit opt-in for private/local targets.
+7. Verification: unit tests, local aiohttp integration test, coverage, syntax compile, executable smoke test.
+

+"""JSON record extraction helpers."""
+from __future__ import annotations
+
+from dataclasses import dataclass, field
+from typing import Any
+
+_MISSING = object()
+
+
+def dotted_get(value: Any, path: str, default: Any = _MISSING) -> Any:
+    """Resolve `a.b.0.c` through dict/list containers."""
+    current = value
+    if not path:
+        return current
+    try:
+        for part in path.split("."):
+            current = current[int(part)] if isinstance(current, list) else current[part]
+        return current
+    except (KeyError, IndexError, TypeError, ValueError):
+        if default is _MISSING:
+            raise KeyError(path) from None
+        return default
+
+
+def _object_lists(value: Any, prefix: str = "") -> list[tuple[str, list[dict[str, Any]]]]:
+    found: list[tuple[str, list[dict[str, Any]]]] = []
+    if isinstance(value, dict):
+        for key, child in value.items():
+            path = f"{prefix}.{key}" if prefix else key
+            found.extend(_object_lists(child, path))
+    elif isinstance(value, list):
+        if value and all(isinstance(item, dict) for item in value):
+            found.append((prefix, value))
+        for index, child in enumerate(value):
+            found.extend(_object_lists(child, f"{prefix}.{index}" if prefix else str(index)))
+    return found
+
+
+@dataclass(frozen=True)
+class DataExtractor:
+    records_path: str | None = None
+    fields: dict[str, str] = field(default_factory=dict)
+
+    def extract(self, payload: Any) -> list[Any]:
+        if self.records_path is not None:
+            records = dotted_get(payload, self.records_path)
+        elif isinstance(payload, list):
+            records = payload
+        else:
+            candidates = _object_lists(payload)
+            records = max(candidates, key=lambda pair: len(pair[1]))[1] if candidates else [payload]
+        if not isinstance(records, list):
+            records = [records]
+        if not self.fields:
+            return records
+        output: list[dict[str, Any]] = []
+        for record in records:
+            output.append({name: dotted_get(record, path, None) for name, path in self.fields.items()})
+        return output

+"""Passive API surface discovery; no auth bypass or browser stealth."""
+from __future__ import annotations
+
+import json
+import re
+from html.parser import HTMLParser
+from urllib.parse import parse_qsl, urlencode, urljoin, urlsplit, urlunsplit
+
+from .models import ApiEndpoint, DiscoveryReport
+
+_URL_RE = re.compile(r"(?P<quote>[\"'])(?P<url>(?:https?|wss?)://[^\"']+|/[^\"']+|\./[^\"']+)(?P=quote)")
+_FETCH_RE = re.compile(r"\bfetch\s*\(\s*(['\"])(?P<url>[^'\"]+)", re.I)
+_AXIOS_RE = re.compile(r"\baxios\.(?P<method>get|post|put|patch|delete)\s*\(\s*(['\"])(?P<url>[^'\"]+)", re.I)
+_WS_RE = re.compile(r"\b(?:new\s+WebSocket|WebSocket)\s*\(\s*(['\"])(?P<url>wss?://[^'\"]+)", re.I)
+_GRAPHQL_RE = re.compile(r"(?:graphql|\bquery\s+[A-Za-z_]*)", re.I)
+_SENSITIVE = {"token", "access_token", "api_key", "apikey", "key", "signature", "sig", "password", "secret"}
+
+
+def _safe_url(base: str, raw: str) -> str:
+    absolute = urljoin(base, raw)
+    parts = urlsplit(absolute)
+    query = [(k, "REDACTED" if k.lower() in _SENSITIVE else v) for k, v in parse_qsl(parts.query, keep_blank_values=True)]
+    return urlunsplit((parts.scheme, parts.netloc, parts.path, urlencode(query), ""))
+
+
+class _MarkupParser(HTMLParser):
+    def __init__(self, base: str) -> None:
+        super().__init__()
+        self.base = base
+        self.forms: list[tuple[str, str]] = []
+        self.scripts: list[str] = []
+        self.inline: list[str] = []
+
+    def handle_starttag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
+        values = dict(attrs)
+        if tag.lower() == "form":
+            self.forms.append((values.get("method", "GET").upper(), _safe_url(self.base, values.get("action", self.base))))
+        elif tag.lower() == "script":
+            src = values.get("src")
+            if src:
+                self.scripts.append(_safe_url(self.base, src))
+
+    def handle_data(self, data: str) -> None:
+        if data.strip():
+            self.inline.append(data)
+
+
+class ApiDiscovery:
+    def scan_html(self, source_url: str, html: str) -> DiscoveryReport:
+        parser = _MarkupParser(source_url)
+        parser.feed(html)
+        endpoints: list[ApiEndpoint] = []
+        seen: set[tuple[str, str, str]] = set()
+
+        def add(method: str, raw: str, kind: str, confidence: float) -> None:
+            url = _safe_url(source_url, raw)
+            key = (method.upper(), url, kind)

+# async-api-crawler
+
+Asyncio-native crawler for authorized collection and passive API-surface mapping.
+
+## Capabilities
+
+- Shared `aiohttp.ClientSession` and bounded global concurrency.
+- Per-host token-bucket throttling, Retry-After support, exponential full-jitter backoff.
+- Retry classification for timeouts, connection errors, 408/425/429, and 5xx responses.
+- Response body-size and total-time budgets.
+- SSRF-safe URL policy: HTTP(S) only, private/loopback/link-local seed blocking by default, and independent redirect-hop blocking.
+- Passive HTML analysis: forms, `fetch`, Axios, GraphQL hints, WebSockets, external scripts.
+- OpenAPI 3 path extraction.
+- JSON record extraction with dotted paths and field mapping.
+- Partial-failure isolation through `gather(..., return_exceptions=True)`.
+
+## Installation
+
+```bash
+python3 -m venv .venv
+. .venv/bin/activate
+python -m pip install -e . pytest pytest-asyncio pytest-cov
+```
+
+The current host already has Python 3.12, aiohttp, pytest, and pytest-asyncio.
+
+## Usage
+
+Fetch URL metadata as JSONL:
+
+```bash
+python -m async_api_crawler.cli crawl https://example.com --output out/results.jsonl
+```
+
+Passive inspect an HTML page (this fetches HTML only; it does not execute JavaScript):
+
+```bash
+python -m async_api_crawler.cli discover https://example.com
+```
+
+Inspect a local OpenAPI document:
+
+```bash
+python -m async_api_crawler.cli discover openapi.json --openapi
+```
+
+Extract records:
+
+```bash
+python -m async_api_crawler.cli extract response.json --path data.items --field id=id --field name=profile.name
+```
+
+Local/private targets are denied by default. For an explicitly authorized local test server only:
+
+```bash
+python -m async_api_crawler.cli crawl http://127.0.0.1:8080/health --allow-private
