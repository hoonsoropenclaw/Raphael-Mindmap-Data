+def build_config(args: argparse.Namespace) -> BotConfig:
+    """組裝 BotConfig：CLI > env > 預設值"""
+    token = None
+    if args.token_file:
+        token = load_token_from_file(args.token_file)
+    elif os.environ.get("TELEGRAM_BOT_TOKEN"):
+        token = os.environ["TELEGRAM_BOT_TOKEN"].strip()
+    elif os.environ.get("TELEGRAM_BOT_TOKEN_FILE"):
+        token = load_token_from_file(os.environ["TELEGRAM_BOT_TOKEN_FILE"])
+    if not token:
+        raise SystemExit(
+            "❌ 沒拿到 Telegram Bot Token。請用 --token-file /path/to/token "
+            "或設 TELEGRAM_BOT_TOKEN / TELEGRAM_BOT_TOKEN_FILE 環境變數"
+        )
+
+    allowed = []
+    if args.allowed_users:
+        allowed = [int(x) for x in args.allowed_users.split(",") if x.strip()]
+    elif os.environ.get("ALLOWED_USERS_FILE"):
+        allowed = load_allowed_users(os.environ["ALLOWED_USERS_FILE"])
+
+    return BotConfig(
+        token=token,
+        allowed_users=allowed,
+        ocr_default_lang=args.lang,
+        ocr_aggressive=args.aggressive,
+    )
