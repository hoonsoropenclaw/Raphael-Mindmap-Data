+"""Token bucket rate limiter — 非同步、安全。
+
+設計要點(來自 senior architect feedback):
+- asyncio.Lock per bucket:不要 per-request lock,會成為瓶頸
+- Lock 持鎖時間只有算術,不是 I/O — 100 個 coroutine 同時打也很短
+- 桶子狀態變動:try_consume() 在持鎖下原子地計算 + 扣減
+- 支援 replace Redis 的擴展點:同一個介面背後可以換實作
+- 429 帶 Retry-After header
+"""
+from __future__ import annotations
+
+import asyncio
+import time
+from dataclasses import dataclass
+from typing import Awaitable, Callable
+
+from fastapi import HTTPException, Request, status
+
+
+@dataclass
+class BucketSnapshot:
+    """外部觀察桶子狀態的 snapshot — test 與監控用。"""
+
+    capacity: int
+    available: float
+    refill_per_sec: float
+
+
+class TokenBucket:
+    """非同步 token bucket — 桶子容量 C、每秒補充 R 個 token。
+
+    行為:
+    - 啟動時桶子是滿的(C tokens)
+    - 每次成功 acquire 就扣 1,空了就拒絕
+    - 隨時間流逝,被動補充最高到 C
+    """
+
+    def __init__(self, capacity: int, refill_per_sec: float) -> None:
+        if capacity <= 0:
+            raise ValueError("capacity must be > 0")
+        if refill_per_sec <= 0:
+            raise ValueError("refill_per_sec must be > 0")
+        self.capacity = capacity
+        self.refill_per_sec = refill_per_sec
+        self._tokens: float = float(capacity)
+        self._last_refill: float = time.monotonic()
+        # 持鎖範圍只包算術 — 短小到飛快
+        self._lock = asyncio.Lock()
+
+    def _refill_locked(self) -> None:
+        """持鎖才呼叫 — 補滿用被動流逝算式。"""
+        now = time.monotonic()
+        elapsed = now - self._last_refill
+        if elapsed > 0:
+            self._tokens = min(
+                self.capacity,
+                self._tokens + elapsed * self.refill_per_sec,
+            )
+            self._last_refill = now
+
+    async def try_consume(self, n: int = 1) -> bool:
+        """嘗試拿 n 個 token;成功 True,失敗 False。"""
+        async with self._lock:
+            self._refill_locked()
+            if self._tokens >= n:
+                self._tokens -= n
+                return True
+            return False
+
+    async def snapshot(self) -> BucketSnapshot:
+        """目前桶子狀態 — 鎖內計算避免 race。"""
+        async with self._lock:
+            self._refill_locked()
+            return BucketSnapshot(
+                capacity=self.capacity,
+                available=self._tokens,
+                refill_per_sec=self.refill_per_sec,
+            )

+# Todo API — FastAPI + SQLite + Token-Bucket Rate Limit
+
+Production-shape 簡單待辦 API。7 endpoints,SQLite 持久化,記憶體 token bucket
+rate limit,完整單元測試(91% 覆蓋率)。
+
+## 快速啟動
+
+```bash
+# 1. 建虛擬環境
+python3 -m venv .venv
+source .venv/bin/activate
+
+# 2. 裝依賴
+pip install -r requirements.txt
+
+# 3. 跑 server (預設 capacity=10, refill=1 token/sec)
+uvicorn app.main:app --reload --port 8000
+
+# 4. 跑測試
+pytest
+# 或加 coverage
+coverage run --source=app -m pytest
+coverage report -m
+```
+
+## Endpoints
+
+| Method | Path                | 說明                | 是否 rate-limited |
+|--------|---------------------|---------------------|-------------------|
+| GET    | `/health`           | 健康檢查            | No                |
+| GET    | `/rate-limit`       | 桶子狀態快照        | No                |
+| GET    | `/todos`            | 列表(新→舊,可過濾) | Yes               |
+| POST   | `/todos`            | 建立                | Yes               |
+| GET    | `/todos/{id}`       | 讀單筆             | Yes               |
+| PUT    | `/todos/{id}`       | 更新(partial)      | Yes               |
+| DELETE | `/todos/{id}`       | 刪除 → 204         | Yes               |
+
+OpenAPI 文件:`http://localhost:8000/docs`
+
+## 設計重點
+
+### 1. 同步 SQLAlchemy,不要 async
+Async session 在 lifecycle 容易漏 `close`/`refresh`,會有 3 個以上坑(def endpoints
++ `sessionmaker` + `db.execute()` 已足夠)。
+
+### 2. Token bucket + asyncio.Lock per bucket
+持鎖範圍只有算術,極短;100 個 coroutine 同時打也很短。同一個介面背後
+可以換 Redis 實作(見 `rate_limit.py: TokenBucket`)。
+
+### 3. WAL + umask 0o077
+SQLite 開 WAL,WAL/SHM sidecar 經 `os.umask(0o077)` 鎖到 0o600,目錄 0o700。
+防 CWE-276。
+
+### 4. Pydantic v2 驗證
+`Field(min_length=1, max_length=255)` — 空 title 直接 422 + 結構化 detail。
+
+### 5. 429 + Retry-After
+`HTTPException(detail=..., headers={"Retry-After": "1"})`,符合 RFC 7231。
+
+### 6. 測試隔離
+- StaticPool in-memory SQLite,跨 session 共用同一個 DB
+- `test_db_setup` fixture 透過 `app.dependency_overrides` 替換 `get_db`
+- `fresh_limiter` / `small_limiter` fixture 在 lifespan 後注入 `app.state.rate_limiter`
+
+## 環境變數
+
+| 變數 | 預設 | 說明 |
+|------|------|------|
+| `TODO_DB_URL` | `sqlite:///./data/todos.db` | SQLAlchemy 連線字串 |
+| `TODO_RATE_CAPACITY` | `10` | Token bucket 容量 |
+| `TODO_RATE_REFILL` | `1.0` | 每秒補充速率 |
+
+## 已知限制
+
+- **Process-local limiter** — 單 instance 才準;多 instance 需換 Redis 後端
+- **無認證** — 可加 `Authorization` dependency + per-user limiter key
+- **SQLite single-writer** — 對「簡單 todo API」足夠;要 scale 換 Postgres
