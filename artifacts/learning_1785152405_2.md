+"""Database session management.
+
+Three things live here:
+  * `make_engine(database_url, ...)` — the engine factory. Handles the two
+    SQLite-specific gotchas (check_same_thread for the threadpool,
+    StaticPool for the in-memory variant). Other DBs are unaffected.
+  * Module-level `engine` and `SessionLocal` bound to the default DB URL,
+    used by the production app.
+  * `get_db()` — FastAPI dependency that yields a session and guarantees
+    it gets closed. Tests override this via `app.dependency_overrides`.
+"""
+from collections.abc import Generator
+
+from sqlalchemy import create_engine
+from sqlalchemy.engine import Engine
+from sqlalchemy.orm import Session, sessionmaker

+def make_engine(database_url: str, *, echo: bool = False) -> Engine:
+    """Build a SQLAlchemy engine with SQLite-safe defaults.
+
+    Gotchas handled (internalized from Round 5 — see `IMPLEMENTATION.md`):
+      * `check_same_thread=False` — required because FastAPI's threadpool
+        hands the same connection across worker threads. SQLite's default
+        is `check_same_thread=True`, which would refuse the checkout.
+      * `poolclass=StaticPool` for `:memory:` — without it, each new
+        connection gets its own private in-memory database, and the tables
