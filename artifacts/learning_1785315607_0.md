+# PRD — Policy Atlas Dynamic Permission Dashboard
+
+## 1. Goal
+Build a production-quality, executable Next.js dashboard that lets administrators understand and safely modify role-based access control (RBAC) relationships through an interactive React Flow graph.
+
+## 2. Target users
+- Security administrator: reviews effective permissions and policy risk.
+- Application owner: creates roles and grants/revokes direct permissions.
+- Auditor: inspects changes and why a permission is effective.
+
+## 3. Required modules
+1. Next.js App Router shell and responsive dashboard layout.
+2. React Flow role → permission relationship graph.
+3. Pure RBAC policy engine for direct and inherited permission calculation.
+4. Role selector and effective-permission matrix.
+5. Role creation and permission grant/revoke controls with validation.
+6. Risk indicators for wildcard/high-impact permissions.
+7. Audit trail with actor, action, target, and timestamp.
+8. Local persistence and reset-to-demo capability.
+9. Standalone `index.html` review artifact in the project root.
+10. Automated tests, production build, and browser E2E verification.
+
+## 4. Functional requirements
+- FR-1: Selecting a role highlights the role, its inherited roles, and effective permissions.
+- FR-2: A role's effective permissions are the de-duplicated union of direct permissions and recursively inherited permissions.
+- FR-3: Role inheritance cycles must be rejected.
+- FR-4: Every permission mutation creates an audit event.
+- FR-5: Viewer cannot mutate policy; editor can grant non-critical permissions; administrator can perform all demo mutations.
+- FR-6: Search filters roles and permissions without changing stored policy.
+- FR-7: Revoke controls distinguish direct grants from inherited access.
+- FR-8: Reset restores a known, deterministic demo policy.
+- FR-9: Reload retains mutations in localStorage.
+- FR-10: UI shows summary metrics and risk exposure.
+
+## 5. Non-functional requirements
+- Accessible keyboard focus, labels, semantic controls, and reduced-motion support.
+- No secrets, backend credentials, or unsafe HTML rendering.
+- Pure functions for policy decisions; UI is not the source of authorization truth.
+- Responsive at desktop/tablet/mobile widths.
+- Distinct industrial/editorial visual direction, avoiding generic SaaS gradients.
+
+## 6. Demo role model
+- Platform administrator: all permissions, including wildcard security controls.
+- Security analyst: reads audit, users, policies and exports audit.
+- Operations editor: reads/updates projects and deploys.
+- Support viewer: reads users and projects only.
+
+## 7. Acceptance criteria
+- `npm test` passes policy-engine unit tests.
+- `npm run test:coverage` meets 80% thresholds for the policy engine.
+- `npm run lint` returns zero errors.
+- `npm run build` creates a successful Next.js static export.
+- Browser E2E verifies role switching, search, add role, grant, revoke, viewer denial, reset, and no console errors.
+- Root `index.html` exists and is usable as a standalone review artifact.

+  Permission,
+  PermissionExplanation,
+  Role,

+export function effectivePermissionIds(roleId: string, roles: Role[]): string[] {
+  const roleById = new Map(roles.map((role) => [role.id, role]));
+  const visited = new Set<string>();
+  const permissions = new Set<string>();
+
+  function visit(currentRoleId: string) {
+    if (visited.has(currentRoleId)) return;
+    visited.add(currentRoleId);
+
+    const role = roleById.get(currentRoleId);
+    if (!role) return;
+
+    role.permissionIds.forEach((permissionId) => permissions.add(permissionId));
+    role.inherits.forEach(visit);
+  }
+
+  visit(roleId);
+  return [...permissions].sort();
+}
+
+export function explainPermission(
+  roleId: string,
+  permissionId: string,
+  roles: Role[],
+): PermissionExplanation | null {
+  const roleById = new Map(roles.map((role) => [role.id, role]));
+  const visited = new Set<string>();
+
+  function search(currentRoleId: string, path: string[]): PermissionExplanation | null {
+    if (visited.has(currentRoleId)) return null;
+    visited.add(currentRoleId);
+
+    const role = roleById.get(currentRoleId);
+    if (!role) return null;
+
+    const nextPath = [...path, currentRoleId];
+    if (role.permissionIds.includes(permissionId)) {
+      return {
+        permissionId,
+        sourceRoleId: currentRoleId,
+        path: nextPath,
+        direct: currentRoleId === roleId,
+      };
+    }
+
+    for (const inheritedRoleId of role.inherits) {
+      const found = search(inheritedRoleId, nextPath);
+      if (found) return found;
+    }
+
+    return null;
+  }
+
+  return search(roleId, []);
+}
