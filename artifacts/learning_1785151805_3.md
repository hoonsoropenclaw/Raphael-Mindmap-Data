+const STORAGE_KEY = 'orbit-command-center.tasks.v1'

+const wait = (ms: number) => new Promise<void>((resolve) => window.setTimeout(resolve, ms))
+
+export async function loadTasks(): Promise<Task[]> {
+  await wait(720)
+  try {
+    const raw = window.localStorage.getItem(STORAGE_KEY)
+    if (raw) {
+      const parsed: unknown = JSON.parse(raw)
+      if (Array.isArray(parsed)) return parsed as Task[]
+    }
+  } catch {
+    throw new Error('Local workspace could not be read.')
+  }
+  return []
+}

+export async function saveTasks(tasks: Task[]): Promise<void> {
+  await wait(320)
+  try {
+    window.localStorage.setItem(STORAGE_KEY, JSON.stringify(tasks))
+  } catch (error) {
+    if (error instanceof DOMException && error.name === 'QuotaExceededError') {
+      throw new Error('Storage is full. Archive older tasks to continue saving.')
+    }
+    throw new Error('Workspace changes could not be saved.')
+  }

+# Orbit Task Command Center — 實作思路
+
+## 產品方向
+
+這是一個「個人作業系統」式的 React task command center，視覺主題為 editorial utility：用大比例留白、紙張色、單一珊瑚紅強調色與 orbit 圖形，讓介面更像一個可長期使用的工作儀表板，而不是一般 SaaS 卡片牆。
+
+## 架構
+
+- `src/types.ts`：集中定義 `Task`、`AppState`、`Action` discriminated union，讓狀態轉移可被 TypeScript 完整檢查。
+- `src/reducer.ts`：純 reducer 管理 task、篩選、搜尋、專案、view、theme、modal、儲存與錯誤狀態；default branch 使用 `never` 做 exhaustive check。
+- `src/storage.ts`：localStorage adapter，包含延遲模擬、JSON 解析保護與 `QuotaExceededError` 轉譯。
+- `src/useTaskCommandCenter.ts`：side effects 與 domain state 的邊界。負責 hydrate、debounced persistence、theme attribute、toast timeout、鍵盤快捷鍵與 derived selectors。
+- `src/components/`：將表單、篩選列、任務列、統計列、loading/empty states 拆成可測試的 leaf components。
+- `src/App.tsx`：只負責頁面組裝與事件串接，不把 reducer 邏輯塞進 JSX。
+- `src/App.css`：純 CSS variable design system，沒有 utility framework；light/dark theme 共用同一套 token。
+
+## 複雜狀態管理
+
+1. 初始載入顯示 skeleton，從 localStorage hydrate；若沒有資料則載入 seed tasks。
+2. 任務 toggle 與 subtasks toggle 都是 reducer action，避免局部 state 互相不同步。
+3. 任務新增後立即更新 UI，接著透過 420ms debounce 儲存，避免快速連續操作造成高頻寫入。

+6. Command Palette 支援 `Cmd/Ctrl + K`、Escape 關閉，以及建立任務、切換 theme、切換清單等命令。
+7. theme 使用 `document.documentElement.dataset.theme` 驅動 CSS variables，不需重複渲染整棵 UI。
