+TELEGRAM_API = "https://api.telegram.org/bot{token}/{method}"
+DEFAULT_WEATHER_API = "https://wttr.in/{location}"
+RUNNING = True
+
+@dataclass(frozen=True)
+class Config:
+    token: str
+    allowed_users: frozenset[int]
+    weather_api_url: str = DEFAULT_WEATHER_API
+    timeout: int = 20
+
+
+def _env_users() -> frozenset[int]:
+    raw = os.getenv("TELEGRAM_ALLOWED_USERS", "")
+    users: set[int] = set()
+    for item in raw.split(","):
+        item = item.strip()
+        if item:
+            try:
+                users.add(int(item))
+            except ValueError:
+                raise ValueError("TELEGRAM_ALLOWED_USERS must contain numeric IDs")
+    return frozenset(users)
+
+
+def load_config() -> Config:
+    token = os.getenv("TELEGRAM_BOT_TOKEN", "").strip()
+    if not token:
+        raise RuntimeError("TELEGRAM_BOT_TOKEN is required")
+    return Config(token=token, allowed_users=_env_users(),
+                  weather_api_url=os.getenv("WEATHER_API_URL", DEFAULT_WEATHER_API))
+
+
+def http_json(url: str, *, method: str = "GET", payload: dict[str, Any] | None = None,
+              timeout: int = 20) -> dict[str, Any]:
+    body = None if payload is None else json.dumps(payload).encode()
+    headers = {"User-Agent": "telegram-weather-bot/1.0", "Content-Type": "application/json"}
+    request = Request(url, data=body, headers=headers, method=method)
+    with urlopen(request, timeout=timeout) as response:
+        return json.loads(response.read().decode("utf-8"))

+                     method="POST", payload=payload, timeout=config.timeout)

+# Telegram 天氣機器人
+
+## 啟動
+
+1. 從 BotFather 取得 token。
+2. 設定環境變數：
+   `export TELEGRAM_BOT_TOKEN='你的 token'`
+3. 可選：限制使用者：
+   `export TELEGRAM_ALLOWED_USERS='123456789,987654321'`
+4. 執行：`./run.sh`
+
+直接傳送地點文字（例如「台北」）即可查詢 wttr.in 天氣；`/start` 和 `/help` 顯示說明。
+
+## 架構與安全決策
+
+- 使用 Python standard library 的 `urllib`，避免為三步驟訊息流程引入第三方依賴。
+- `telegram_weather_bot.py` 同時封裝 Telegram long polling、wttr.in 查詢、訊息路由與 graceful shutdown。
+- Long polling 適合無公開 HTTPS endpoint 的主機；每次更新使用 `offset = update_id + 1` 避免重複處理。
+- token 只從環境變數讀取，不寫入檔案或程式碼。
+- `TELEGRAM_ALLOWED_USERS` 是基本 allowlist；空值代表不限制，生產環境建議設定。
+- 地點經過空白正規化與 100 字元長度限制，並以 URL encoding 傳送。
+- API 錯誤只回傳一般錯誤訊息，詳細例外寫 stderr，避免洩漏內部資訊。
+- 以 SIGINT/SIGTERM 設定停止旗標，讓目前 polling 週期自然結束。
