+// =============================================================================
+// Server Actions — 寫入操作後自動觸發 ISR 重新驗證
+// 這是 Next.js 16 ISR 的「精細控制」武器: 不是傻等 60 秒,而是「資料變了」就立刻失效快取
+// =============================================================================
+"use server";
+
+import { revalidatePath, revalidateTag } from "next/cache";
+import { redirect } from "next/navigation";
+import {
+  createPost,
+  updatePostViews,
+  updateProductInventory,
+} from "@/lib/data";
+import { TAGS } from "@/lib/cache";
+
+// ── Action 1: 發布新文章 → revalidatePath('/blog') + revalidateTag('posts') ──
+export async function publishPostAction(formData: FormData) {
+  const title = String(formData.get("title") ?? "").trim();
+  const category = String(formData.get("category") ?? "engineering") as
+    | "engineering"
+    | "product"
+    | "ops";
+
+  if (!title) {
+    return { ok: false, error: "標題必填" };
+  }
+
+  const post = await createPost({ title, category });
+
+  // 三種 revalidation 手法組合 (展示完整 ISR 武器庫):
+  revalidatePath("/blog"); // 1. 讓 /blog 列表下次請求立刻重新生成
+  revalidatePath(`/blog/${post.slug}`); // 2. 該文章頁面立即失效
+  revalidateTag(TAGS.post(post.slug)); // 3. tag 細粒度: 只有這篇文章的 fetch cache 失效
+  revalidateTag(TAGS.category(category)); //    連帶 category 列表也更新
+
+  redirect(`/blog/${post.slug}`);
+}
+
+// ── Action 2: 模擬文章瀏覽數 +1 → 只 revalidate 該文章 ──
+export async function incrementViewsAction(slug: string) {
+  const post = await getCachedPostForAction(slug);
+  if (!post) return { ok: false };
+  await updatePostViews(slug, post.views + 1);
+  revalidateTag(TAGS.post(slug));
+  revalidatePath(`/blog/${slug}`);
+  return { ok: true, views: post.views + 1 };
+}
+
+// ── Action 3: 商品庫存調整 → revalidate product tag ──
+export async function adjustInventoryAction(slug: string, delta: number) {
+  await updateProductInventory(slug, delta);
+  revalidateTag(TAGS.product(slug));
+  revalidateTag(TAGS.allProducts);
+  revalidatePath("/products");
+  revalidatePath(`/products/${slug}`);
+}
+
+// ── 內部 helper: Server Action 不能直接拿頁面 props,要用 unstable_cache ──
+import { unstable_cache } from "next/cache";
+import { getPostBySlug } from "@/lib/data";

+  unstable_cache(
+    async () => getPostBySlug(slug),
+    [`post-${slug}-v1`],
+    { revalidate: 60, tags: [TAGS.allPosts, TAGS.post(slug)] },
+  )();

+// =============================================================================
+// /api/revalidate — Token-gated on-demand revalidation API
+// 這是 Pages Router 時代 res.revalidate() 的 App Router 對應做法
+// 用途: 外部 CMS / webhook / cron job 可以打這個 endpoint 強制重新生成特定路由
+// =============================================================================
+import { NextRequest, NextResponse } from "next/server";
+import { revalidatePath, revalidateTag } from "next/cache";
+
+export const runtime = "nodejs"; // ISR 需要 Node runtime
+
+export async function POST(req: NextRequest) {
+  // ── 1. Token 驗證 (防外部任意觸發) ──
+  const { searchParams } = new URL(req.url);
+  const secret = searchParams.get("secret");
+  const expected = process.env.REVALIDATE_SECRET ?? "dev-secret";
+
+  if (secret !== expected) {
+    return NextResponse.json(
+      { revalidated: false, error: "Invalid token" },
+      { status: 401 },
+    );
+  }
+
+  // ── 2. 解析要重新驗證的目標 ──
+  const path = searchParams.get("path");
+  const tag = searchParams.get("tag");
+
+  if (!path && !tag) {
+    return NextResponse.json(
+      {
+        revalidated: false,
+        error: "必須提供 path 或 tag 其中之一",
+        usage: {
+          by_path: "POST /api/revalidate?secret=xxx&path=/blog",
+          by_tag: "POST /api/revalidate?secret=xxx&tag=posts",
+          both: "POST /api/revalidate?secret=xxx&path=/blog&tag=posts",
+        },
+      },
+      { status: 400 },
+    );
+  }
+
+  try {
+    const now = new Date().toISOString();
+    if (path) revalidatePath(path);
+    if (tag) revalidateTag(tag);
+
+    return NextResponse.json({
+      revalidated: true,
+      path: path ?? null,

    ⭐ 核心學習要點
    ═══════════════════════════════════════════════
    
    1. stale-while-revalidate 才是 ISR 的精髓 — 不是「60 秒後強制更新」,而是「60 秒後第一個請求仍回舊版(秒回)+背景觸發重生」,所以使用者永遠感受不到延遲。
    
    2. unstable_cache + tags 是 Server Action 時代的 ISR 主力 — Pages Router 時代只有 res.revalidate(path),App Router 多了 tag 維度,讓「更新單篇文章瀏覽數」可以只失效那篇文章的 cache,完全不打擾其他頁。
    
    3. dynamicParams = true 是大型內容站的標配 — 電商 10 萬 SKU 不能 build time 全生成,只預建熱門 + 其餘首次訪問 on-demand 生成並永久快取。
    
    4. Server Actions 是 revalidate 的最自然觸發點 — 在 mutation 函式末尾呼叫 revalidatePath/Tag,寫入與快取失效是同一個 transaction,沒有 race condition。
    
    5. force-dynamic 是 ISR 的逃生出口 — 像 /admin 這種「永遠要最新資料」的場景,明確關掉快取比錯誤配置 revalidate=0 更清晰。
