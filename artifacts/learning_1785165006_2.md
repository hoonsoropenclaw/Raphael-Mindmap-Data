+#!/usr/bin/env python3
+"""Fetch GitHub trending repositories and render a Markdown table.
+
+Uses GitHub's public search API, so no third-party dependencies or token are
+required. Set GITHUB_TOKEN for higher API limits.
+"""
+from __future__ import annotations
+
+import argparse
+import json
+import os
+import sys
+from datetime import date, timedelta
+from urllib.error import HTTPError, URLError
+from urllib.request import Request, urlopen
+
+
+def fetch_trending(language: str | None, since: str, limit: int, timeout: int) -> list[dict]:
+    query = f"created:>={since}"
+    if language:
+        query += f" language:{language}"
+    params = f"q={__import__('urllib.parse', fromlist=['urlencode']).quote(query)}&sort=stars&order=desc&per_page={limit}"
+    req = Request(
+        f"https://api.github.com/search/repositories?{params}",
+        headers={
+            "Accept": "application/vnd.github+json",
+            "User-Agent": "github-trending-markdown/1.0",
+            **({"Authorization": f"Bearer {os.environ['GITHUB_TOKEN'].strip()}"} if os.environ.get("GITHUB_TOKEN") else {}),
+        },
+    )
+    with urlopen(req, timeout=timeout) as response:
+        payload = json.load(response)
+    return payload.get("items", [])
+
+
+def escape_cell(value: object) -> str:
+    return str(value or "").replace("|", "\\|").replace("\n", " ").strip()
+
+
+def render(items: list[dict], generated: str, language: str | None, since: str) -> str:
+    scope = language or "all languages"
+    lines = [
+        "# GitHub Trending Repositories",
+        "",
+        f"Generated: {generated}  |  Scope: {scope}  |  Created since: {since}",
+        "",
+        "| # | Repository | Description | Stars | Forks | Language |",
+        "|---:|---|---|---:|---:|---|",
+    ]
+    for index, repo in enumerate(items, 1):
+        lines.append(
+            f"| {index} | [{escape_cell(repo.get('full_name'))}]({repo.get('html_url', '')}) "
+            f"| {escape_cell(repo.get('description'))} | {repo.get('stargazers_count', 0):,} "
+            f"| {repo.get('forks_count', 0):,} | {escape_cell(repo.get('language'))} |"
+        )
+    return "\n".join(lines) + "\n"
+
+
+def main() -> int:
+    parser = argparse.ArgumentParser(description=__doc__)
+    parser.add_argument("-o", "--output", default="github-trending.md")
+    parser.add_argument("-l", "--language", help="GitHub language filter, e.g. Python")
+    parser.add_argument("-d", "--days", type=int, default=7)
+    parser.add_argument("-n", "--limit", type=int, default=25)
+    parser.add_argument("--timeout", type=int, default=20)
+    args = parser.parse_args()
+    if not 1 <= args.limit <= 100 or args.days < 0:
+        parser.error("limit must be 1..100 and days must be non-negative")
+    since = (date.today() - timedelta(days=args.days)).isoformat()
+    try:
+        items = fetch_trending(args.language, since, args.limit, args.timeout)
+        output = render(items, date.today().isoformat(), args.language, since)
+        if args.output == "-":
+            sys.stdout.write(output)
+        else:
+            with open(args.output, "w", encoding="utf-8") as handle:
+                handle.write(output)
+            print(f"Wrote {len(items)} repositories to {args.output}")

+# GitHub Trending Markdown — 實作思路與審查紀錄
+
+## 架構
+
+- 使用 Python 3 標準函式庫 `urllib` 呼叫 GitHub REST Search API，避免第三方依賴。
+- 以 `created:>=YYYY-MM-DD` + 可選 `language:` 建立近期熱門 repository 查詢。
+- 預設依 stars 遞減排序，輸出最多 25 筆；可用 CLI 參數調整天數、語言、數量、逾時與輸出檔。
+- 若設定 `GITHUB_TOKEN`，以 Bearer token 發送，提升 API rate limit；未設定時仍可使用公開 API。
+- Markdown 欄位包含排名、repository 連結、描述、stars、forks、主要語言。
+
+## 防護與決策
+
+1. 使用 `User-Agent` 與 GitHub 建議的 media type header。
+2. 對 Markdown 描述與名稱進行 `|`、換行逸出，避免破壞表格。
+3. 限制 `limit` 為 1..100、拒絕負數 days。
+4. 捕捉 HTTP、URL、timeout、JSON 解析錯誤，輸出 stderr 並回傳非零 exit code。
+5. `--output -` 可直接輸出 stdout，方便 cron 或 pipe；預設寫入 `github-trending.md`。
+
+## 使用方式
+
+```bash
+python3 github_trending.py
+python3 github_trending.py --days 1 --limit 10 --language Python -o python-trending.md
+python3 github_trending.py --output -
+GITHUB_TOKEN=... python3 github_trending.py
+```
