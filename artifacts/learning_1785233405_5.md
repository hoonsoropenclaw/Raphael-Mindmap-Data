+"""Per-IP token-bucket rate limiter implemented as a FastAPI dependency.
+
+Design notes (cross-checked with trial-and-error skill:
+  - fastapi-sync-vs-async-pattern-20260727.md §"Uvicorn Proxy Trust Is a Server-Layer Boundary"
+  - We key ONLY on request.client.host (ASGI scope), not on raw X-Forwarded-For.

+    doing it in app code would bypass Uvicorn's trust boundary.
+  - Default: 10 requests per 60s sliding window per IP. Configurable via
+    RATE_LIMIT_MAX / RATE_LIMIT_WINDOW_SECONDS env vars so tests can crank it.
+  - Storage is in-process (dict). Sufficient for single-process uvicorn.
+    For multi-worker deploys, swap for Redis. Out of scope for this exercise.
+"""
+from __future__ import annotations
+
+import os
+import threading
+import time
+from collections import defaultdict, deque
+from collections.abc import Callable, Iterator
+
+from fastapi import Depends, HTTPException, Request, status
+
+
+def _settings() -> tuple[int, float]:
+    """Return (max_requests, window_seconds) from env with sane defaults."""
+    max_requests = int(os.environ.get("RATE_LIMIT_MAX", "10"))
+    window_seconds = float(os.environ.get("RATE_LIMIT_WINDOW_SECONDS", "60"))
+    return max_requests, window_seconds
+
+
+class RateLimiter:
+    """Thread-safe sliding-window rate limiter keyed by client IP."""
+
+    def __init__(self, max_requests: int | None = None, window_seconds: float | None = None) -> None:
+        m, w = _settings()
+        self.max_requests = max_requests if max_requests is not None else m
+        self.window_seconds = window_seconds if window_seconds is not None else w
+        self._hits: dict[str, deque[float]] = defaultdict(deque)
+        self._lock = threading.Lock()
+
+    def check(self, key: str) -> tuple[bool, int, float]:
+        """Record a hit and return (allowed, remaining, retry_after_seconds).
+
+        Sliding-window: drop entries older than window_seconds, then check length.
+        """
+        now = time.monotonic()
+        cutoff = now - self.window_seconds
+        with self._lock:
+            bucket = self._hits[key]
+            while bucket and bucket[0] < cutoff:
+                bucket.popleft()
+            if len(bucket) >= self.max_requests:
+                # Retry-After = how long until the oldest entry falls out of the window
+                retry_after = max(0.0, bucket[0] + self.window_seconds - now)
+                return False, 0, retry_after
+            bucket.append(now)
+            remaining = self.max_requests - len(bucket)
+            return True, remaining, 0.0
+
+    def reset(self) -> None:
+        """Clear all buckets. Used by tests."""
+        with self._lock:
+            self._hits.clear()
