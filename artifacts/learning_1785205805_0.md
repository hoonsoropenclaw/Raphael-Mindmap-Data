+def check_gh_auth() -> tuple[bool, str]:
+    """確認 gh CLI 已登入。回傳 (ok, message)。"""
+    try:
+        result = subprocess.run(
+            ["gh", "auth", "status"],
+            capture_output=True, text=True, timeout=15,
+        )
+    except FileNotFoundError:
+        return False, "找不到 gh CLI。請先安裝：https://cli.github.com/"
+    except subprocess.TimeoutExpired:
+        return False, "gh auth status 逾時"
+
+    output = (result.stdout + result.stderr).strip()
+    if "Logged in to" in output and "Active account: true" in output:
+        # 抽出 active account
+        m = re.search(r"Logged in to github\.com account (\S+)", output)
+        account = m.group(1) if m else "unknown"
+        return True, f"已認證帳號: {account}"
+    return False, f"gh 尚未登入或帳號未啟用:\n{output}"
+
