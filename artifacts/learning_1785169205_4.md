+# Todo API
+
+A simple Todo REST API built with FastAPI, backed by SQLite, protected by rate limiting, and covered by unit tests.
+
+## Features
+
+- CRUD endpoints for todos (`/todos`)

+- Pytest suite with isolated in-memory DB per test

+"""Database engine + session factory.
+
+Keeps persistence concerns out of the route handlers. The engine is created
+once at startup; callers obtain sessions via the `get_session` dependency.
+"""
+from __future__ import annotations
+
+from contextlib import contextmanager
+from typing import Iterator, Optional
+from sqlmodel import SQLModel, Session, create_engine
+
+# Engine is set by `init_db`. Imported by routes via `get_session`.
+_engine = None
+
+
+def init_db(database_url: str = "sqlite:///./todos.db") -> None:
+    """Create the global engine and ensure tables exist.
+
+    Args:
+        database_url: SQLAlchemy-compatible URL. Defaults to a file-based
+            SQLite DB next to the working directory. Use ``sqlite:///:memory:``
+            in tests for an isolated in-memory store.
+    """
+    global _engine
+    connect_args = {"check_same_thread": False} if database_url.startswith("sqlite") else {}
+    _engine = create_engine(database_url, connect_args=connect_args, echo=False)
+    # Import models so SQLModel.metadata is populated before create_all.
+    from todo_api import models  # noqa: F401
+    SQLModel.metadata.create_all(_engine)
+
+
+def get_engine():
+    if _engine is None:
+        raise RuntimeError("Database not initialized. Call init_db() first.")
+    return _engine
+
+
+def get_session() -> Iterator[Session]:
+    """FastAPI dependency: yields a session, commits on success, rolls back on error."""
+    engine = get_engine()
+    with Session(engine) as session:
+        try:
+            yield session
+            session.commit()
+        except Exception:
+            session.rollback()
+            raise
+
+
+@contextmanager
+def session_scope() -> Iterator[Session]:
+    """Imperative context manager (useful in scripts/tests outside FastAPI)."""
+    engine = get_engine()
+    with Session(engine) as session:
+        try:
+            yield session
+            session.commit()
+        except Exception:
+            session.rollback()
+            raise
+

+"""FastAPI application factory + entrypoint.
+
+`create_app()` is the factory; module-level `app` is the ASGI target that
+uvicorn imports. Splitting the two lets tests spin up isolated app instances
+with overridden settings (e.g. in-memory DB, looser rate limits).
+"""
+from __future__ import annotations
+
+import os
+from contextlib import asynccontextmanager
+
+from fastapi import FastAPI, Request
+from fastapi.responses import JSONResponse
+from slowapi.errors import RateLimitExceeded
+from slowapi.middleware import SlowAPIMiddleware
+
+from todo_api import __version__
+from todo_api.database import init_db
+from todo_api.rate_limit import limiter
+from todo_api.routers import todos as todos_router
+
+
+DATABASE_URL = os.getenv("DATABASE_URL", "sqlite:///./todos.db")
+
+
+@asynccontextmanager
+async def lifespan(app: FastAPI):
+    init_db(DATABASE_URL)
+    yield
+
+
+def create_app(database_url: str | None = None) -> FastAPI:
+    """Build a fresh app. Used by tests to inject a different DB URL."""
+    app = FastAPI(
+        title="Todo API",
+        version=__version__,
+        description="Simple Todo REST API with rate limiting and SQLite persistence.",
+        lifespan=lifespan,
+    )
+
+    # Bind the limiter to app.state so SlowAPIMiddleware + decorators agree.
+    app.state.limiter = limiter
+    app.add_middleware(SlowAPIMiddleware)
+
+    # Health check — useful for uptime probes.
+    @app.get("/health", tags=["meta"])
+    def health() -> dict:
+        return {"status": "ok", "version": __version__}
+
+    # Surface rate-limit violations as structured JSON, not HTML.
+    @app.exception_handler(RateLimitExceeded)
+    async def rate_limit_handler(request: Request, exc: RateLimitExceeded):
+        return JSONResponse(
+            status_code=429,
+            content={
+                "error": "rate_limited",
+                "detail": f"Limit {exc.detail} exceeded for {request.client.host if request.client else 'unknown'}",
+            },
+        )
+
+    app.include_router(todos_router.router)
+
+    # Initialize DB eagerly for tests that don't drive the lifespan event.
+    init_db(database_url or DATABASE_URL)
+    return app
+
+
+app = create_app()
+
+
+if __name__ == "__main__":
+    import uvicorn
+
+    uvicorn.run("todo_api.main:app", host="127.0.0.1", port=8000, reload=True)

+
+> 本檔記錄本次「Python FastAPI 待辦事項 API + rate limit + SQLite + 單元測試」實作的完整思路、踩坑、教訓。給未來的自己 / Senior Architect 審查用。
+
+---
+
+## 1. 任務範圍
+
+- FastAPI HTTP API，提供 todo CRUD
+- 每個 todo 欄位：`id`, `title`, `done`, `created_at`, `updated_at`
+- 用 SQLite 持久化（檔案 `todos.db`）
+- 用 `slowapi` 做 IP 級 rate limiting
+- 寫 pytest 單元測試覆蓋 service + API 兩層
+- 可用 `uvicorn todo_api.main:app` 直接跑起來
+
+## 2. 架構決策
+
+採**分層 + 介面分離**架構，方便單元測試不必每次都跑 FastAPI TestClient：
+
+```
+src/todo_api/
+├── __init__.py
+├── main.py            # FastAPI app factory + lifespan + 429 handler
+├── database.py        # SQLAlchemy engine、init_db()、get_session() 依賴
+├── models.py          # SQLModel ORM + Pydantic DTO (Create/Update/Read)
+├── rate_limit.py      # slowapi Limiter 設定 + reset_for_tests()
+├── services.py        # 純函式 CRUD：create/get/list/update/delete
+└── routers/
+    └── todos.py       # HTTP 路由層：thin wrapper → services
+```
+
+**為什麼分層**：
+- `services.py` 是純函式，吃 `Session` 吐 ORM 物件 → 可離線單測
+- `routers/todos.py` 只做 HTTP 進出、錯誤轉 404、Pydantic 驗證 → 可專注 HTTP 行為
+- `models.py` 把 ORM 跟 DTO 用 SQLModel 合體，但 Create/Update/Read 仍分離，避免 client 傳 `id` 篡改主鍵
+- `database.py` 集中所有 persistence 邏輯（engine、session、init_db），測試時只換這層的 URL
+
+**App 用 factory pattern**（`create_app()`），而非頂層直接 `app = FastAPI()`：
+- 讓測試可以用 `create_app(database_url=":memory:")` 注入隔離 DB
+- 模組匯入時不會偷偷 init 一個檔案 DB 出來（這是踩坑 #1 的根因）
+
+## 3. Rate Limit 設計
+
+- 全域 default limit：**60/minute/IP**（可由 `RATE_LIMIT_DEFAULT` env var 覆寫）
+- 每個路由**個別上更緊的 limit**：
+  - `GET` 系列：`100/minute`
+  - `POST` / `PATCH` / `DELETE`：`20/minute`（寫操作比較貴）
+- key 函數用 `slowapi.util.get_remote_address`（per-IP 簡單實用）
+- `RateLimitExceeded` exception handler 回 **JSON `{"error": "rate_limited", ...}`** 而非預設的 HTML
+- 測試時 `reset_for_tests("1000/minute")` 把上限拉高，避免干擾其他單元測試
+
