① Async handler bot.py

async def handle_voice(update, ctx):
    voice = update.message.voice
    buf = io.BytesIO()
    await (await ctx.bot.get_file(voice.file_id)) \
        .download_to_memory(buf)
    await ctx.bot.send_chat_action(update.effective_chat.id,
                                   ChatAction.TYPING)
    result = await ctx.application.bot_data["whisper"] \
        .transcribe(buf.getvalue(),
                    filename="voice.ogg",
                    hint_lang=resolve_whisper_code(lang))
    await update.message.reply_text(
        t(lang, "transcription_done",
          detected=result.detected_language,
          text=result.text))

    # 1. resolve user's preferred language

    if voice.duration > SETTINGS.max_voice_seconds:
        return await update.message.reply_text(
            t(lang, "voice_too_long", ...))

    # 3. download › whisper › reply
    result = await whisper.transcribe(...)

            != resolve_whisper_code(lang):

🗄️ user_prefs.py SQLite WAL

@asynccontextmanager
async def _connect():
    conn = await aiosqlite.connect(DB_PATH)
    try:
        await conn.execute("PRAGMA journal_mode=WAL")
        await conn.execute("PRAGMA synchronous=NORMAL")
        yield conn
    finally:
        await conn.commit()
        await conn.close()

async def set_lang(user_id, lang):
    await conn.execute("""
        INSERT INTO user_prefs (user_id, lang, translate)
        VALUES (?, ?, ?)
        ON CONFLICT(user_id) DO UPDATE
            SET lang = excluded.lang
    """, (user_id, lang, 1))
