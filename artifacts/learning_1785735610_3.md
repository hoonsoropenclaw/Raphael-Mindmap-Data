+"""Schema-driven extraction with BeautifulSoup.
+
+Implements the design notes from micro-skill
+`data_export_and_schema_driven_extraction.md`: selectors + per-field
+optionality + per-record canonicalisation.
+"""
+from __future__ import annotations
+
+import logging
+from dataclasses import dataclass
+from typing import Iterable, List, Optional
+
+from bs4 import BeautifulSoup
+
+from .config import ExtractionSchema, FieldSpec
+
+
+log = logging.getLogger(__name__)
+
+
+@dataclass
+class ExtractionResult:
+    records: List[dict]
+    skipped: int       # items where a required field was missing
+
+
+def _value_from(node, spec: FieldSpec) -> Optional[str]:
+    if spec.attr:
+        v = node.get(spec.attr)
+        if isinstance(v, list):                # class="a b" -> ["a","b"]
+            v = " ".join(v)
+        return (v or "").strip() or None
+    text = node.get_text(" ", strip=True)
+    return text or None
+
+
+def extract(soup: BeautifulSoup, schema: ExtractionSchema) -> ExtractionResult:
+    records: List[dict] = []
+    skipped = 0
+    for node in soup.select(schema.item_selector):
+        rec: dict = {}
+        missing_required = False
+        for f in schema.fields:
+            v = _value_from(node, f)
+            if v is None:
+                if f.required:
+                    missing_required = True
+                    break
+                continue
+            rec[f.name] = v
+        if missing_required:
+            skipped += 1
+            continue
+        # don't keep records that are completely empty
+        if rec:
+            records.append(rec)
+    return ExtractionResult(records=records, skipped=skipped)
+
+
+def extract_many(html_iter: Iterable[tuple[str, bytes]],
+                 schema: ExtractionSchema) -> dict[str, ExtractionResult]:
+    """Convenience batch helper used by tests."""
+    out: dict[str, ExtractionResult] = {}
+    for url, body in html_iter:
+        soup = BeautifulSoup(body, "html.parser")
+        out[url] = extract(soup, schema)

+## 架構決策（為什麼這樣拆）
+
+### 1. 四層去重（defence in depth）
+- **URL UNIQUE on `pages.url`** — 同一 URL 永遠不重複插入
+- **Content-hash INDEX on `pages.content_hash`** — 同一份 body 就算來自不同 URL 也不重複
+- **Record-key UNIQUE on `(page_id, record_key)`** — 同一頁的同一筆紀錄不重複（record_key 是 `sha256(canonical_json)`）
+- **Source UNIQUE on `sources.url`** — 起點 URL 不重複
+
+任何一層失守還有其他層擋，符合「defense in depth」原則。
+
+### 2. SQLite WAL + thread-safe RLock
+照 `sqlite3_thread_safe.md` 微技能的 pattern：`check_same_thread=False` + 單一 RLock 守護所有寫入。WAL 模式讓讀寫並行不死鎖。`PRAGMA synchronous=NORMAL` 比 FULL 快 5x、斷電仍安全。
+
+### 3. Light-first, heavy-later
+照 `automated_testing_and_scraping.md` 微技能的策略：永遠先 `requests + bs4`，只有靜態 HTML 缺 schema marker 時才降級到 Selenium headless Chrome。`_looks_static()` 用「首個 tag + 首個 class 都在 HTML 裡」當啟發式。對 server-rendered 站（quotes.toscrape.com / books.toscrape.com）永遠走 static 零瀏覽器開銷。
+
+### 4. Pagination 知識持久化
+第 2 次跑時若 root URL 已被 dedup，從 `outgoing_links` 表撈回上次存的 pagination 連結，不用重新 fetch 就知道後續分頁。**只存 schema 指定的 pagination selector**，不存全部 anchor（否則 tag/author 連結會炸 queue）。
+
+### 5. Budget 三種語意
+- `max_pages=N` (N > 0) — 本次最多 insert N 個新頁，已知頁照樣算進 dup
+- `max_pages=0` — 純 dedup audit：掃過已知 URL 算 dup，不 fetch 任何新頁
+- 這比「永遠 budget=DB 總頁數」清楚，避免使用者困惑「為什麼第 2 次跑還是抓到新頁」
+
+### 6. Schema-driven extraction
+照 `data_export_and_schema_driven_extraction.md`：JSON 定義 `item_selector` + `fields[]`，CLI 透過 `--schema` 載入。內建 `presets.py` 提供 quotes / books 兩個常用網站的 preset，不用每次寫 selector。

+## 遇到的真實問題與修復
