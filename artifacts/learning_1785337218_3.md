+def request_json(
+    url: str,
+    *,
+    method: str = "GET",
+    headers: Mapping[str, str] | None = None,
+    params: Mapping[str, Any] | None = None,
+    body: Any = None,
+    timeout: float = 30,
+    retries: int = 3,
+) -> Any:
+    if params:
+        clean = {k: v for k, v in params.items() if v is not None}
+        url += ("&" if "?" in url else "?") + urllib.parse.urlencode(clean, doseq=True)
+    payload = None if body is None else json.dumps(body).encode("utf-8")

+        req = urllib.request.Request(url, data=payload, headers=all_headers, method=method)
+        try:
+            with urllib.request.urlopen(req, timeout=timeout) as response:
+                raw = response.read()

+        except urllib.error.HTTPError as exc:
+            try:
+                detail = exc.read().decode("utf-8", "replace")[:500]

+from __future__ import annotations
+
+import secrets
+import sqlite3
+import threading
+import time
+from pathlib import Path
+
+
+class StateStore:
+    def __init__(self, path: Path):
+        path.parent.mkdir(parents=True, exist_ok=True)
+        self.db = sqlite3.connect(path, check_same_thread=False)
+        self.db.row_factory = sqlite3.Row
+        self.lock = threading.RLock()
+        with self.db:
+            self.db.execute("PRAGMA journal_mode=WAL")
+            self.db.executescript("""
+            CREATE TABLE IF NOT EXISTS kv (key TEXT PRIMARY KEY, value TEXT NOT NULL);
+            CREATE TABLE IF NOT EXISTS subscriptions (
+              chat_id INTEGER NOT NULL, user_id INTEGER NOT NULL,
+              PRIMARY KEY(chat_id, user_id)
+            );
+            CREATE TABLE IF NOT EXISTS event_map (
+              event_id TEXT PRIMARY KEY, chat_id INTEGER NOT NULL, user_id INTEGER NOT NULL,
+              created_by_bot INTEGER NOT NULL DEFAULT 0, updated_at REAL NOT NULL
+            );
+            CREATE TABLE IF NOT EXISTS pending_delete (
+              code TEXT PRIMARY KEY, event_id TEXT NOT NULL, chat_id INTEGER NOT NULL,
+              user_id INTEGER NOT NULL, expires_at REAL NOT NULL
+            );
+            """)
+
+    def close(self) -> None:
+        self.db.close()
+
+    def get(self, key: str) -> str | None:
+        with self.lock:
+            row = self.db.execute("SELECT value FROM kv WHERE key=?", (key,)).fetchone()
+            return row[0] if row else None
+
+    def set(self, key: str, value: str) -> None:
+        with self.lock, self.db:
+            self.db.execute("INSERT INTO kv(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value", (key, value))
+
+    def subscribe(self, chat_id: int, user_id: int) -> None:
+        with self.lock, self.db:
+            self.db.execute("INSERT OR IGNORE INTO subscriptions VALUES(?,?)", (chat_id, user_id))
+
+    def subscribers(self) -> list[tuple[int, int]]:
+        with self.lock:
+            return [(r[0], r[1]) for r in self.db.execute("SELECT chat_id,user_id FROM subscriptions")]
+
+    def map_event(self, event_id: str, chat_id: int, user_id: int, created_by_bot: bool) -> None:
+        with self.lock, self.db:
+            self.db.execute(
+                "INSERT INTO event_map VALUES(?,?,?,?,?) ON CONFLICT(event_id) DO UPDATE SET chat_id=excluded.chat_id,user_id=excluded.user_id,created_by_bot=excluded.created_by_bot,updated_at=excluded.updated_at",
+                (event_id, chat_id, user_id, int(created_by_bot), time.time()),
