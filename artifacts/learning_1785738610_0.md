+import { test, expect } from '@playwright/test';

+test.describe('cross-browser smoke and visual regression', () => {
+  test('loads matrix and preserves visual baseline', async ({ page }) => {
+    await page.goto('/web_output.html');
+    await expect(page).toHaveTitle('Playwright Cross-Browser Visual Lab');
+    await expect(page.locator('[data-testid^="browser-"]')).toHaveCount(3);
+    await expect(page.locator('[data-testid="visual-preview"]')).toBeVisible();
+    await expect(page).toHaveScreenshot('matrix-desktop.png', { fullPage: true });
+  });
+
+  test('runs smoke action and updates live result', async ({ page }) => {
+    await page.goto('/web_output.html');
+    await page.getByTestId('run-test').click();
+    await expect(page.getByTestId('test-result')).toHaveText(/Smoke test completed/);
+  });
+
+  test('toggles contrast without breaking anchors', async ({ page }) => {
+    await page.goto('/web_output.html');
+    await page.getByTestId('toggle-mode').click();
+    await expect(page.locator('body')).toHaveClass(/contrast/);
+    await expect(page.getByTestId('run-test')).toBeEnabled();
+  });
+
+  test('fits mobile viewport without horizontal overflow', async ({ page }) => {
+    await page.setViewportSize({ width: 390, height: 844 });
+    await page.goto('/web_output.html');
+    const overflow = await page.evaluate(() => document.documentElement.scrollWidth > window.innerWidth);
+    expect(overflow).toBe(false);
+    await expect(page).toHaveScreenshot('matrix-mobile.png', { fullPage: true });
+  });

+npm test              # compare current output with baselines
