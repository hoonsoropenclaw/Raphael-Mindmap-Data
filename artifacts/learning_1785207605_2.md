+const STORAGE_KEY = 'react-todo:state:v1'

+function loadFromStorage() {
+  try {
+    const raw = localStorage.getItem(STORAGE_KEY)
+    if (!raw) return null
+    const parsed = JSON.parse(raw)
+    // selectedIds is a Set — restore from array
+    return {
+      items: Array.isArray(parsed.items) ? parsed.items : [],
+      filter: parsed.filter ?? 'all',
+      query: parsed.query ?? '',
+      selectedIds: new Set(parsed.selectedIds ?? []),
+    }
+  } catch (e) {
+    console.warn('[TodoContext] failed to load state:', e)
+    return null
+  }

+function saveToStorage(state) {
+  try {

+      items: state.items,
+      filter: state.filter,
+      query: state.query,
+      selectedIds: [...state.selectedIds],
+    }
+    localStorage.setItem(STORAGE_KEY, JSON.stringify(serializable))
+  } catch (e) {

+# React Todo · Complex State Demo
+
+A single-page Todo app that demonstrates **non-trivial React state management** without
+any third-party state library. Built with Vite + React 18.
+
+## Complex state features
+
+| Feature | How it's modeled |
+|---|---|
+| CRUD on items | `useReducer` with action types: `add` / `toggle` / `edit` / `remove` |
+| Filter (all / active / done) | Reducer stores `filter`; selector derives visible list |
+| Search query | Reducer stores `query`; selector combines filter + query |
+| Multi-select | `selectedIds: Set<string>` in reducer (note: serialized via array on persist) |
+| Bulk actions | `bulkDone` / `bulkRemove` operate over `selectedIds` (or passed `ids`) |
+| Priority (low/normal/high) | Item-level field, rendered as colored tag + emphasis on high |
+| Persistence | `useEffect` mirrors state to `localStorage`; `TodoProvider` hydrates on mount |
+| Derived state | Selectors `selectVisible` / `selectCounts` kept outside the reducer |
+

+
+```bash
+cd react-todo
+npm install
+npm run dev          # http://localhost:5173
+npm run build        # production bundle in dist/
+npm test             # vitest unit tests for reducer + selectors

+## Architectural notes
+
+- **Why `useReducer` over multiple `useState`**: filter / query / items / selectedIds
+  interact — toggling an item while a filter is active should not require parent
+  re-wiring. Reducer centralizes the transition rules.
+- **Why Context for state dispatch**: only the dispatch (stable identity) and a memoized
+  state object flow through context. Selectors are computed in components via
+  `useMemo` to avoid re-running filter logic on unrelated re-renders.
+- **Why `Set` for `selectedIds`**: O(1) lookup on bulk operations; serialized to an
+  array on persist and rehydrated as a Set on boot.
+- **Why selectors outside the reducer**: keeps the reducer a pure function of
+  (state, action). `selectVisible` is recomputed only when its inputs change.
