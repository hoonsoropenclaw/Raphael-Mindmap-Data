+export const STORAGE_KEY = 'incident-console:v1'

+function readStoredIncidents() {
+  try {
+    const value = window.localStorage.getItem(STORAGE_KEY)
+    return value ? JSON.parse(value) : []
+  } catch (error) {
+    console.warn('Unable to restore incidents; using an empty console.', error)
+    return []
+  }
+}
+
+export function useIncidentConsole() {
+  const [state, dispatch] = useReducer(incidentReducer, undefined, () => buildInitialState(readStoredIncidents()))
+  const searchRef = useRef(null)
+
+  useEffect(() => {
+    try {
+      window.localStorage.setItem(STORAGE_KEY, JSON.stringify(state.incidents))
+    } catch (error) {
+      console.warn('Unable to persist incidents.', error)
+    }
+  }, [state.incidents])
+
+  useEffect(() => {
+    function onKeyDown(event) {
+      const target = event.target
+      const isTyping = target instanceof HTMLInputElement || target instanceof HTMLTextAreaElement || target instanceof HTMLSelectElement
+      if (event.key === '/' && !isTyping) {
+        event.preventDefault()
+        searchRef.current?.focus()
+      }
+      if (event.key === 'Escape') {
+        dispatch({ type: 'CANCEL_DELETE' })

+        event.preventDefault()
+        dispatch({ type: event.shiftKey ? 'REDO' : 'UNDO' })
+      }
+    }
