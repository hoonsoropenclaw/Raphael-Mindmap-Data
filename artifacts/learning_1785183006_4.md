@dataclass
class WeatherReport:
    """單一地點的天氣查詢結果"""

    location: str
    area_name: str = ""
    region: str = ""
    country: str = ""
    condition: str = ""
    temp_c: Optional[int] = None
    feels_like_c: Optional[int] = None
    humidity: Optional[int] = None
    cloud_cover: Optional[int] = None
    wind_kmph: Optional[int] = None
    wind_dir: str = ""
    pressure_hpa: Optional[int] = None
    visibility_km: Optional[int] = None
    uv_index: Optional[int] = None
    observation_time_utc: str = ""
    error: str = ""
    raw: dict[str, Any] = field(default_factory=dict)

    def format_for_chat(self) -> str:
        """輸出 Telegram-ready 多行訊息 (Markdown)"""
        if self.error:
            return f"⚠️ {self.error}"

        # location_label: 顯示使用者輸入 + wttr.in 對應的最近地區
        if self.area_name and self.area_name != self.location:
            loc = f"{self.location} ({self.area_name})"
        else:
            loc = self.area_name or self.location

        lines = [f"📍 *{loc}* — {self.condition}"]

        if self.temp_c is not None:
            feels = f"  (體感 {self.feels_like_c}°C)" if self.feels_like_c is not None else ""
            lines.append(f"🌡️  溫度: {self.temp_c}°C{feels}")

        if self.humidity is not None:
            cloud = f"  雲量: {self.cloud_cover}%" if self.cloud_cover is not None else ""
            lines.append(f"💧  濕度: {self.humidity}%{cloud}")

        if self.wind_kmph is not None:
            lines.append(f"🌬️  風: {self.wind_dir} {self.wind_kmph} km/h")

        extras: list[str] = []
        if self.pressure_hpa is not None:
            extras.append(f"氣壓: {self.pressure_hpa} hPa")
        if self.visibility_km is not None:
            extras.append(f"可見度: {self.visibility_km} km")
        if extras:
            lines.append(f"📑  {'  '.join(extras)}")

        if self.uv_index is not None or self.observation_time_utc:
            uv = f"UV: {self.uv_index}" if self.uv_index is not None else ""
            obs = f"觀測時間: {self.observation_time_utc} UTC" if self.observation_time_utc else ""
            sep = "  " if uv and obs else ""
            lines.append(f"☀️  {uv}{sep}{obs}")

        from datetime import datetime, timezone
        ts = datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M UTC")
        lines.append(f"_查詢時間_: {ts}")

        return "\\n".join(lines)


def _sanitize(text: str) -> str:
    """剝除 lone surrogates,避免 Telegram API 噴 UnicodeEncodeError。
    wttr.in 偶爾會傳回不完整的 UTF-16 編碼字串,Python 把它當 CPython 內部表示
    (surrogate pair) 後送 Telegram 會爆炸。
    """
    return text.encode("utf-16", errors="surrogatepass").decode("utf-16", errors="ignore")


def _normalize_location(loc: str) -> str:
    """清洗使用者輸入的地點字串"""
    loc = loc.strip()
    if not loc:
        raise ValueError("地點不可為空")
    # 移除 Telegram Markdown 特殊字元 (避免破壞回傳格式)
    for ch in ("*", "_", "`", "["):
        loc = loc.replace(ch, "")
    return loc[:100]


def _http_get_json(url: str) -> dict[str, Any]:
    """GET JSON,通用 helper"""
    req = urllib.request.Request(url, headers={"User-Agent": DEFAULT_USER_AGENT})
    with urllib.request.urlopen(req, timeout=WTTR_TIMEOUT) as resp:
        body = resp.read().decode("utf-8", errors="replace")
    return json.loads(body)


def _parse_observation_time(obs_raw: str) -> str:
    """從 wttr.in 的 localObsDateTime 抽出 HH:MM。

    輸入可能為 "0", "143", "abc", "9999", "1430" 等 garbage 格式。
    改為 try/except + 範圍檢查,避免 Cycle 540 教的 `02::2:00Z` 慘案。
    """
    if not obs_raw:
        return ""
    digits = "".join(c for c in str(obs_raw) if c.isdigit())
    if len(digits) < 3:
        return ""
    try:
        # wttr.in 格式可能是 "0", "1430", "1430 PM" → 取前 4 碼
        h = int(digits[:2])
        m = int(digits[2:4]) if len(digits) >= 4 else 0
        if not (0 <= h <= 23 and 0 <= m <= 59):
            return ""
        return f"{h:02d}:{m:02d}"
    except (ValueError, IndexError):
        return ""


def _parse_payload(payload: dict[str, Any], location: str) -> WeatherReport:
    """從 wttr.in `format=j1` JSON 抽出 WeatherReport。
    Fail-loud 在 missing fields (Senior Architect 教訓: silent default 會讓 stale API 假裝成功)。
    """
    try:
        current = payload["current_condition"][0]
        area = payload.get("nearest_area", [{}])[0]
        area_name = (area.get("areaName") or [{}])[0].get("value", "") if area.get("areaName") else ""
        region = (area.get("region") or [{}])[0].get("value", "") if area.get("region") else ""
        country = (area.get("country") or [{}])[0].get("value", "") if area.get("country") else ""

        weather_code = current["weatherCode"]  # raise KeyError if missing
        cond_zh = CONDITION_ZH.get(weather_code, f"未知 ({weather_code})")

        obs_raw = current.get("localObsDateTime", "")
        obs_time = _parse_observation_time(obs_raw)

        return WeatherReport(
            location=location,
            area_name=_sanitize(area_name),
            region=_sanitize(region),
            country=_sanitize(country),
            condition=cond_zh,
            temp_c=int(current["temp_C"]),
            feels_like_c=int(current["FeelsLikeC"]),
            humidity=int(current["humidity"]),
            cloud_cover=int(current["cloudcover"]),
            wind_kmph=int(current["windspeedKmph"]),
            wind_dir=_sanitize(current.get("winddir16Point", "")),
            pressure_hpa=int(current["pressure"]),
            visibility_km=int(current["visibility"]),
            uv_index=int(current["uvIndex"]),
            observation_time_utc=obs_time,
            raw=payload,
        )
    except (KeyError, IndexError, ValueError, TypeError) as e:
        return WeatherReport(
            location=location,
            error=f"wttr.in 回傳格式異常: {type(e).__name__}: {e}",
        )


def fetch_full(location: str) -> WeatherReport:
    """查詢完整天氣資訊 (format=j1)"""
    try:
        loc = _normalize_location(location)
    except ValueError as e:
        return WeatherReport(location=location, error=str(e))

    url = f"{WTTR_BASE}/{urllib.parse.quote(loc)}?format=j1&lang=zh"
    try:
        payload = _http_get_json(url)
    except urllib.error.HTTPError as e:
        return WeatherReport(location=loc, error=f"找不到地區「{loc}」(HTTP {e.code})")
    except urllib.error.URLError as e:
        return WeatherReport(location=loc, error=f"網路錯誤: {e.reason}")
    except json.JSONDecodeError as e:
        return WeatherReport(location=loc, error=f"wttr.in 回傳非 JSON: {e}")

    return _parse_payload(payload, loc)


def fetch_one_line(location: str) -> str:
    """查詢單行摘要 (format=3)"""
    try:
        loc = _normalize_location(location)
    except ValueError as e:
        return f"⚠️ {e}"

    url = f"{WTTR_BASE}/{urllib.parse.quote(loc)}?format=3"
    req = urllib.request.Request(url, headers={"User-Agent": DEFAULT_USER_AGENT})
    try:
        with urllib.request.urlopen(req, timeout=WTTR_TIMEOUT) as resp:
            return _sanitize(resp.read().decode("utf-8", errors="replace").strip())
    except urllib.error.HTTPError as e:
        return f"⚠️ 找不到地區「{loc}」(HTTP {e.code})"
    except urllib.error.URLError as e:
        return f"⚠️ 網路錯誤: {e.reason}"


def _cli() -> int:
    """CLI 入口: 接受 1+ 地點參數,輸出 one-line 摘要"""
    if len(sys.argv) < 2:
        print("用法: python3 wttr_weather.py <地點1> [地點2] ...", file=sys.stderr)
        return 2
    rc = 0
    for loc in sys.argv[1:]:
        line = fetch_one_line(loc)
        print(f"📍 {line}")
        if line.startswith("⚠️"):
            rc = 1
    return rc


if __name__ == "__main__":
    sys.exit(_cli())

TELEGRAM_API_BASE = "https://api.telegram.org/bot{token}/{method}"

# === Layer 1: Allowlist ===

ALLOWED_USERS: set[int] = set()


def _load_allowed_users() -> None:
    """從 env 讀取 TELEGRAM_ALLOWED_USERS (逗號分隔的 user_id 清單)"""
    raw = os.environ.get("TELEGRAM_ALLOWED_USERS", "").strip()
    if not raw:
        log.warning("TELEGRAM_ALLOWED_USERS not set — bot is PUBLIC (anyone can use)")
        return
    for tok in raw.split(","):
        tok = tok.strip()
        if tok.isdigit():
            ALLOWED_USERS.add(int(tok))
    log.info("loaded %d allowed users", len(ALLOWED_USERS))


def is_allowed(user_id: int) -> bool:
    """L1 allowlist 檢查。空清單 = 公開模式 (log warning, 仍放行)"""
    if not ALLOWED_USERS:
        return True
    return user_id in ALLOWED_USERS


# === Layer 2: Rate-limit (Token Bucket) ===

class RateLimiter:
    """每使用者 3 秒 cooldown 的 in-memory token bucket

    教訓 1 (Senior Architect): "Token Bucket 與 SQLite 併發寫入可能暴露出鎖競爭"
    解法: 整個 `_check_and_update` 包在 `threading.Lock` 內,保證 atomic check-and-update。
    教訓 2 (Senior Architect): "辨識 ithreading Lock 仍不夠,真實高併發仍可能 race"
    解法: 額外用 `asyncio.gather` 微型壓測驗證 (test_load.py)。
    """

    COOLDOWN_SEC = 3.0

    def __init__(self) -> None:
        import threading
        self._lock = threading.Lock()
        self._last_call: dict[int, float] = {}

    def allow(self, user_id: int) -> bool:
        """True = 允許這次呼叫,False = 還在 cooldown"""
        with self._lock:
            now = time.monotonic()
            last = self._last_call.get(user_id, 0.0)
            if now - last < self.COOLDOWN_SEC:
                return False
            self._last_call[user_id] = now
            return True


# === Telegram HTTP helper ===

def _http_post(token: str, method: str, payload: dict[str, Any]) -> dict[str, Any]:
    """Telegram Bot API POST helper

    教訓 35 (Cycle 532): JSON body 內含 user-controlled 文字 → 一定要 json.dumps() 後
    設 Content-Type: application/json,不要 form-encoded。
    """
    url = TELEGRAM_API_BASE.format(token=token, method=method)
    body = json.dumps(payload).encode("utf-8")
    req = urllib.request.Request(
        url,
        data=body,
        headers={"Content-Type": "application/json"},
        method="POST",
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        return json.loads(resp.read().decode("utf-8"))


def _http_get(token: str, method: str, params: dict[str, Any]) -> dict[str, Any]:
    """Telegram Bot API GET helper (用於 getUpdates)"""
    url = TELEGRAM_API_BASE.format(token=token, method=method)
    if params:
        url += "?" + urllib.parse.urlencode(params)
    req = urllib.request.Request(url)
    with urllib.request.urlopen(req, timeout=POLL_TIMEOUT + 10) as resp:
        return json.loads(resp.read().decode("utf-8"))

HELP_TEXT = """🌍 *天氣查詢機器人*

可用指令:
• /start  — 自我介紹
• /help   — 顯示本說明
• /now <地點> — 單行摘要

def handle_message(text: str, user_id: int) -> Optional[str]:
    """解析使用者訊息,回傳要送給 Telegram 的回覆字串 (None = 不回覆)"""
    if not is_allowed(user_id):
        log.info("denied user_id=%s", user_id)
        return None

    text = (text or "").strip()
    if not text:
        return None

    # 指令分流
    if text in ("/start", "/help"):
        return HELP_TEXT

    if text.startswith("/now"):
        location = text[4:].strip()
        if not location:
            return "⚠️ 用法: /now <地點>\\n例如: /now Tokyo"
        return wttr_weather.fetch_full(location).format_for_chat()

    # 預設:直接當地點查詢
    return wttr_weather.fetch_full(text).format_for_chat()


def send_reply(token: str, chat_id: int, text: str) -> bool:
    """送訊息給 Telegram,_sanitize 防 surrogate pair 噴錯"""
    if not text:
        return True
    safe_text = wttr_weather._sanitize(text)
    try:
        _http_post(token, "sendMessage", {
            "chat_id": chat_id,
            "text": safe_text,
            "parse_mode": "Markdown",
            "disable_web_page_preview": True,
        })
        return True
    except urllib.error.HTTPError as e:
        # 409 conflict = 兩個 bot 搶同一 token;短暫退讓
        if e.code == 409:
            log.warning("telegram 409 conflict — another bot using same token?")
            time.sleep(15)
        else:
            log.error("sendMessage HTTP %s: %s", e.code, e.read().decode("utf-8", errors="replace"))
        return False
    except (urllib.error.URLError, json.JSONDecodeError) as e:
        log.error("sendMessage network/JSON error: %s", e)
        return False


# === Long-polling 主迴圈 ===

class TgBot:
    def __init__(self, token: str) -> None:
        self.token = token
        self.rate_limiter = RateLimiter()
        self._running = True
        self._offset: Optional[int] = None
        self._last_heartbeat = time.monotonic()

    def stop(self, *_: Any) -> None:
        """Graceful shutdown — 旗標 + 讓主迴圈自然結束"""
        log.info("stop signal received, finishing current poll...")
        self._running = False

    def _get_updates(self) -> list[dict[str, Any]]:
        """GET /getUpdates with long-polling

        教訓 4 (cycle 538): 用絕對 offset = update_id + 1,不要 += 1
        """
        try:
            params: dict[str, Any] = {"timeout": POLL_TIMEOUT, "allowed_updates": ["message"]}
            if self._offset is not None:
                params["offset"] = self._offset
            resp = _http_get(self.token, "getUpdates", params)
            if not resp.get("ok"):
                log.error("getUpdates not ok: %s", resp)
                return []
            return resp.get("result", [])
        except urllib.error.HTTPError as e:
            if e.code == 409:
                log.warning("getUpdates 409 conflict — sleeping 15s")
                time.sleep(15)
            else:
                log.error("getUpdates HTTP %s", e.code)
            return []
        except (urllib.error.URLError, json.JSONDecodeError) as e:
            log.error("getUpdates network/JSON error: %s — retry in %ss", e, POLL_RETRY_SLEEP)
            time.sleep(POLL_RETRY_SLEEP)
            return []

    def _process_update(self, update: dict[str, Any]) -> None:
        message = update.get("message") or {}
        user = message.get("from") or {}
        user_id = user.get("id", 0)
        text = message.get("text", "")
        chat_id = message.get("chat", {}).get("id")

        if not chat_id or not user_id:
            log.warning("update missing chat_id or user_id: %s", update)
            return

        if not is_allowed(user_id):
            log.info("denied user_id=%s", user_id)
            return

        if not self.rate_limiter.allow(user_id):
            log.info("rate-limited user_id=%s", user_id)
            send_reply(self.token, chat_id, "⏳ 請稍等 3 秒再查詢下一筆")
            return

        reply = handle_message(text, user_id)
        if reply is not None:
            send_reply(self.token, chat_id, reply)

    def run(self) -> int:
        log.info("long-polling started (timeout=%ss, allowed=%d)", POLL_TIMEOUT, len(ALLOWED_USERS))
        while self._running:
            updates = self._get_updates()
            for update in updates:
                self._process_update(update)
                # 絕對 offset (教訓 4)
                self._offset = update["update_id"] + 1
            now = time.monotonic()
            if now - self._last_heartbeat > HEARTBEAT_INTERVAL:
                log.info("heartbeat: alive, offset=%s", self._offset)
                self._last_heartbeat = now
        log.info("graceful shutdown complete")
        return 0

    rl.COOLDOWN_SEC = 60.0  # 拉長 cooldown 確保 100 個都在 cooldown 內
