+# ---------- Time helpers -----------------------------------------------------

+def _parse_iso(s: str) -> datetime:
+    """Parse an ISO 8601 string to a timezone-aware UTC datetime.
+
+    Accepts both 'YYYY-MM-DD' (all-day) and full datetimes with or without
+    offsets / trailing 'Z'. Naive datetimes are returned as-is so the caller
+    can decide on timezone assignment.
+    """
+    if "T" not in s and len(s) == 10:  # pure date
+        d = datetime.strptime(s, "%Y-%m-%d")
+        return d.replace(tzinfo=timezone.utc)
+    # Python 3.11+ fromisoformat handles 'Z' natively; be defensive anyway.
+    s_norm = s.replace("Z", "+00:00") if s.endswith("Z") else s
+    try:
+        dt = datetime.fromisoformat(s_norm)
+    except ValueError as e:
+        raise ValueError(f"Invalid ISO datetime: {s!r}") from e
+    if dt.tzinfo is None:
+        return dt.replace(tzinfo=timezone.utc)
+    return dt.astimezone(timezone.utc)
+
+
+def _iso(dt: datetime) -> str:
+    """Render a datetime as ISO 8601 with explicit UTC offset."""
+    return dt.astimezone(timezone.utc).isoformat()

+# FastAPI × Google Calendar — Sync & Reminders
+
+Event synchronization and reminder management built on FastAPI with a pluggable
+calendar provider backend. Falls back to an in-memory provider when no Google
+credentials are configured (so the service still starts on a stock machine),
+and switches to the real Google Calendar API when `GOOGLE_CLIENT_SECRET_PATH`
+is set.
+
+## Quick start

+```bash
+# Install (already present on this machine)

+# Run (in-memory backend)
+python -m app.main
+# → [main] provider = inmemory
+# → INFO  Uvicorn running on http://127.0.0.1:8000

+# Open the dashboard

+| GET    | `/reminders?within_minutes=N` | —                 | 200    | reminders firing in the next N minutes        |
