+
+Module-level pragma listener (architect feedback: no nested listeners —
+they allocate a new function per engine creation).
+"""
+from __future__ import annotations
+
+import os
+import stat
+from pathlib import Path
+
+from sqlalchemy import create_engine, event

+def _set_sqlite_pragmas(dbapi_connection, connection_record):
+    """Enable WAL + foreign keys. Module-level (no closure issues)."""
+    cursor = dbapi_connection.cursor()
+    cursor.execute("PRAGMA journal_mode=WAL")
+    cursor.execute("PRAGMA foreign_keys=ON")
+    cursor.execute("PRAGMA synchronous=NORMAL")
+    cursor.close()
+
+
+def _enforce_db_file_mode(db_path: Path) -> None:
+    """SQLite creates files at 0o644. Tighten to 0o600 + handle WAL/SHM sidecars.
+
+    We only chmod a file we just observed being created in our private dir,
+    never an existing shared path. We check stat.S_ISREG to avoid chmod-ing
+    a FIFO, socket, or device.
+    """
+    parent_mode = db_path.parent.stat().st_mode
+    if not parent_mode & stat.S_IRWXU:  # noqa — sanity check parent is private
+        raise PermissionError(
+            f"Parent dir {db_path.parent} is not 0o700. Refusing to open DB."
+        )
+
+    for side in (db_path, db_path.with_suffix(db_path.suffix + "-wal"),
+                 db_path.with_suffix(db_path.suffix + "-shm")):
+        if side.exists():
+            st = side.stat()
+            if not stat.S_ISREG(st.st_mode):
+                continue
+            current = stat.S_IMODE(st.st_mode)
+            if current & 0o077:
+                os.chmod(side, 0o600)
+
+
+def build_engine(db_path: Path | None = None) -> Engine:
+    target = Path(db_path or settings.db_path)
+    ensure_private_dir(target)
+    engine = create_engine(
+        f"sqlite:///{target}",
+        connect_args={"check_same_thread": False},
+        future=True,
+    )
+    # Tighter file mode BEFORE first connect writes anything.
+    _enforce_db_file_mode(target)
+    event.listen(engine, "connect", _set_sqlite_pragmas)
+    return engine

+from dataclasses import dataclass, field

+@dataclass
+class SlidingWindowLimiter:
+    """Per-key fixed-window limiter.
+
+    Each key keeps a deque[int] of recent unix timestamps. A request is allowed
+    iff fewer than `max_requests` timestamps fall inside the [now - window, now]
+    interval. Memory bound: each key stores at most max_requests timestamps,
+    and TTL eviction removes idle keys entirely.
+    """
+
+    max_requests: int
+    window_seconds: int
+    ttl_seconds: int = 300  # bucket evicted after 5 minutes idle
+    _buckets: dict[str, deque[float]] = field(default_factory=dict)
+    _last_seen: dict[str, float] = field(default_factory=dict)
+    _lock: asyncio.Lock = field(default_factory=asyncio.Lock)
+
+    async def check(self, key: str, *, now: float | None = None) -> tuple[bool, int]:
+        """Return (allowed, retry_after_seconds). Caller passes the same `now`
+        to many keys if they want clock-injection behaviour in tests."""
+        ts = time.time() if now is None else now
+        async with self._lock:
+            bucket = self._buckets.get(key)
+            if bucket is None:
+                # Architect §1: explicit init. We use the SAME config as caller
+                # requested — never a "default-then-correct" trap.
+                bucket = deque()
+                self._buckets[key] = bucket
+            self._last_seen[key] = ts
+
+            # Drop timestamps outside the window.
+            cutoff = ts - self.window_seconds
+            while bucket and bucket[0] < cutoff:
+                bucket.popleft()
+
+            if len(bucket) < self.max_requests:
+                bucket.append(ts)
+                return True, 0
+
+            # Full — retry_after is (oldest_in_window + window) - now
+            retry = max(0, int(bucket[0] + self.window_seconds - ts) + 1)
+            return False, retry
+
+    async def sweep(self, *, now: float | None = None) -> int:
+        """Evict buckets idle for more than ttl_seconds. Returns count evicted."""
+        ts = time.time() if now is None else now
+        async with self._lock:
+            stale = [k for k, t in self._last_seen.items() if ts - t > self.ttl_seconds]
+            for k in stale:
+                self._buckets.pop(k, None)
+                self._last_seen.pop(k, None)
+            return len(stale)
+

+        return len(self._buckets)
