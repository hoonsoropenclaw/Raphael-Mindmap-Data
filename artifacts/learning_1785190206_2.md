+"""Rate limiting setup using slowapi.
+
+Uses a custom key function that prefers X-Forwarded-For (when behind a proxy)
+but falls back to client.host for direct connections. Shared in-memory storage
+is fine for a single-process dev/test app; in production swap for Redis.
+"""
+from __future__ import annotations
+
+from fastapi import Request
+from slowapi import Limiter
+from slowapi.errors import RateLimitExceeded
+from slowapi.util import get_remote_address
+from starlette.responses import JSONResponse
+
+
+def _key_func(request: Request) -> str:
+    """Identify the client for rate-limiting purposes.
+
+    Prefers X-Forwarded-For first hop (set by nginx/cloudflare); otherwise
+    the direct connection IP. Slowapi's get_remote_address already inspects
+    X-Forwarded-For so we just delegate.
+    """
+    return get_remote_address(request)
+
+
+# Single shared Limiter instance; importable by routers via app.state.limiter.
+# default_limits applies to every route that doesn't override. 60/minute is a
+# sane default for an internal todo API; tighten per-route as needed.
+limiter = Limiter(
+    key_func=_key_func,
+    default_limits=["60/minute"],
+    headers_enabled=True,
+)
+
+
+def rate_limit_exceeded_handler(
+    request: Request, exc: RateLimitExceeded
+) -> JSONResponse:

+        content={
+            "error": "rate_limit_exceeded",

+            "retry_after_seconds": 60,

+"""Rate-limit tests.
+
+We bump past the per-route limit and assert the 429 response shape. To make
+the test fast we temporarily override the limit to a small number using
+limiter.limit() — slowapi lets us stack limits via decorator replacement, but
+the cleanest path is to hammer the endpoint beyond its default and verify
+behaviour. With the route-level default of 30/min on POST, 31 requests in a
+row triggers the limit.
+
+This is slow but deterministic. If it ever becomes flaky, swap to a custom
+endpoint with a 2/minute limit.
+"""
+from __future__ import annotations
+
+
+def test_rate_limit_triggers_on_burst(client):
+    """POSTing beyond the route limit returns 429."""
+    # Per-route limit on POST is 30/minute — issue 31 rapid requests
+    last_status = None
+    last_body = None
+    for i in range(31):
+        resp = client.post("/todos", json={"title": f"burst {i}"})
+        last_status = resp.status_code
+        last_body = resp.json() if resp.status_code == 429 else None
+        if resp.status_code == 429:
+            break
+
+    assert last_status == 429, "expected 429 once burst exceeded 30/min"
+    assert last_body["error"] == "rate_limit_exceeded"
+    assert "retry_after_seconds" in last_body
+
+
+def test_rate_limit_handler_returns_json(client):
+    """The 429 response is JSON, not FastAPI's default HTML."""
+    for i in range(31):
+        resp = client.post("/todos", json={"title": f"x{i}"})
+        if resp.status_code == 429:
+            assert resp.headers["content-type"].startswith("application/json")
+            return
+    # If we got here we never hit the limit — fail loudly
+    raise AssertionError("did not observe 429 within 31 requests")

+# Todo API
+
+A minimal FastAPI todo service with SQLite storage and per-endpoint rate limiting.
+
+## Quick start
+
+```bash
+pip install -r requirements.txt
+uvicorn app.main:app --reload
+```
+
+Open `http://localhost:8000/docs` for the interactive Swagger UI.
+
+## Endpoints
+
+| Method | Path | Description | Rate limit |
+|--------|------|-------------|------------|
+| `GET`  | `/health` | Liveness probe | 60/min (default) |
+| `POST` | `/todos` | Create a todo | 30/min |
+| `GET`  | `/todos` | List todos (paginated, filter by `completed`) | 60/min |
+| `GET`  | `/todos/{id}` | Fetch one todo | 120/min |
+| `PUT`  | `/todos/{id}` | Partial update | 30/min |
+| `DELETE` | `/todos/{id}` | Delete a todo | 30/min |
+
+### Example payload
+
+```bash
+curl -X POST http://localhost:8000/todos \
+  -H "Content-Type: application/json" \
+  -d '{"title":"Buy milk","description":"2L semi-skimmed"}'
+```
+
+## Configuration
+
+| Env var | Default | Description |
+|---------|---------|-------------|
+| `TODO_DATABASE_URL` | `sqlite:///./todos.db` | SQLAlchemy DB URL |
+
+## Tests
+
+```bash
+pytest -v
+```
+
+Tests use an in-memory SQLite (`StaticPool`) and override `get_db`, so they
+never touch the real `todos.db`. Rate-limit counters are reset between tests.

+### 3.1 分層：`models` ↔ `schemas` 分離
