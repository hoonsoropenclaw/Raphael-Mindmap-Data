+"""自製 TokenBucket rate limiter（per-IP）。
+
+Why 自製而不是 slowapi？
+- 純 Python dataclass，沒綁 FastAPI → 易 unit test
+- asyncio.Lock 粒度是 per bucket，100 並發 acquire 不會同時通過
+- 可換 backend：Redis 只要實作 acquire(key) 介面
+
+Senior 原則 #2：asyncio.gather 真正並行。
+"""
+
+import asyncio
+import time
+from dataclasses import dataclass, field
+from typing import Dict
+
+
+class RateLimitExceeded(Exception):
+    """acquire() 失敗時 raise，由 FastAPI dependency 轉 429。"""
+
+    def __init__(self, retry_after: float):
+        self.retry_after = retry_after
+        super().__init__(f"rate limit exceeded, retry after {retry_after:.1f}s")
+
+
+@dataclass
+class _Bucket:
+    """單一 IP 的 token bucket 狀態。"""
+    tokens: float
+    capacity: float
+    refill_per_second: float
+    last_refill: float = field(default_factory=time.monotonic)
+
+    def refill(self, now: float) -> None:
+        """按時間流逝補 token。"""
+        elapsed = now - self.last_refill
+        if elapsed > 0:
+            self.tokens = min(self.capacity, self.tokens + elapsed * self.refill_per_second)
+            self.last_refill = now
+
+
+class TokenBucket:
+    """Per-IP token bucket，併發安全。
+
+    內部結構：
+    - _buckets: Dict[ip, _Bucket] — lazy 建立
+    - _lock: per-instance — 保護 _buckets 字典本身
+    - 每個 bucket 內部操作不是 atomic，但 acquire() 在 lock 內做完整 refill+consume，
+      所以跨 IP 也安全
+    """
+
+    def __init__(
+        self,
+        capacity: float,
+        refill_per_second: float,
+        burst_capacity: float = 5.0,
+    ):
+        # burst_capacity 是 abstraction 安全閥：effective_capacity = min(set, burst)
+        self._capacity = min(capacity, burst_capacity)
+        self._refill_per_second = refill_per_second
+        self._buckets: Dict[str, _Bucket] = {}
+        self._lock = asyncio.Lock()
+
+    def _get_or_create(self, key: str) -> _Bucket:
+        b = self._buckets.get(key)

+                capacity=self._capacity,
+                refill_per_second=self._refill_per_second,
+            )
+            self._buckets[key] = b
