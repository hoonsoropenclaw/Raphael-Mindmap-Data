+#!/usr/bin/env python3
+"""Small, dependency-light PNG visual comparator used by the Playwright runner."""
+from __future__ import annotations
+
+import argparse
+import json
+import sys
+from pathlib import Path
+
+from PIL import Image, ImageChops, ImageEnhance
+
+
+def compare(baseline: Path, current: Path, diff_path: Path, channel_threshold: int, max_ratio: float) -> dict:
+    if not baseline.exists():
+        return {"status": "missing-baseline", "pass": False, "reason": str(baseline)}
+    if not current.exists():
+        return {"status": "missing-current", "pass": False, "reason": str(current)}
+
+    base = Image.open(baseline).convert("RGBA")
+    now = Image.open(current).convert("RGBA")
+    if base.size != now.size:
+        return {
+            "status": "dimension-mismatch",
+            "pass": False,
+            "baseline_size": list(base.size),
+            "current_size": list(now.size),
+        }
+
+    diff = ImageChops.difference(base, now)
+    rgb = diff.convert("RGB")
+    pixels = list(rgb.getdata())
+    changed = sum(1 for pixel in pixels if max(pixel) > channel_threshold)
+    total = len(pixels)
+    ratio = changed / total if total else 0.0
+    extrema = rgb.getextrema()
+    max_channel = max(high for _, high in extrema)
+    passed = ratio <= max_ratio
+
+    if not passed:
+        # Make changed regions easy to inspect without adding a third-party Node module.
+        amplified = ImageEnhance.Contrast(rgb).enhance(4.0)
+        amplified.save(diff_path)
+    else:
+        diff_path.unlink(missing_ok=True)
+
+    return {
+        "status": "pass" if passed else "regression",
+        "pass": passed,
+        "baseline_size": list(base.size),
+        "current_size": list(now.size),
+        "changed_pixels": changed,
+        "total_pixels": total,
+        "changed_ratio": round(ratio, 8),
+        "max_channel_delta": max_channel,
+        "channel_threshold": channel_threshold,
+        "max_allowed_ratio": max_ratio,
+        "diff_path": str(diff_path) if not passed else None,
+    }
+
+
+def main() -> int:
+    parser = argparse.ArgumentParser()
+    parser.add_argument("--baseline", required=True, type=Path)
+    parser.add_argument("--current", required=True, type=Path)
+    parser.add_argument("--diff", required=True, type=Path)

+# Cross-browser Lab
+
+This directory contains a self-contained Playwright fixture and a runnable visual-regression harness.
+
+## Deliverables
+
+- `web_output.html` — deterministic, accessible HTML fixture with theme, search, keyboard, copy, and run-state interactions.
+- `tests/run-cross-browser.cjs` — custom runner that executes the same contract in Chromium, Firefox, and WebKit at desktop and mobile viewports, then compares per-engine PNG baselines.
+- `tools/visual_diff.py` — Pillow-based comparator that emits JSON metrics and a highlighted diff image on regression.
+- `playwright.config.cjs` + `tests/cross-browser.spec.cjs` — optional native Playwright Test matrix and `toHaveScreenshot` checkpoint.
+- `artifacts/` — generated baselines, current captures, diffs, and JSON/Markdown reports (created on first run).
+
+## Run the custom matrix
+
+The installed Playwright package and browser binaries are already available in this Hermes environment. From this directory:
+
+```bash
+npm run test:baseline   # first run: create isolated baselines
+npm run test:visual     # later runs: compare current captures to baselines
+```
+
+Useful overrides:
+
+```bash
+node tests/run-cross-browser.cjs --browser=chromium
+node tests/run-cross-browser.cjs --browser=chromium,firefox --viewport=mobile
+node tests/run-cross-browser.cjs --headed
+BASE_URL=http://127.0.0.1:4173 node tests/run-cross-browser.cjs --skip-server
+```
+
+The custom runner chooses an ephemeral local port by default, so it does not touch existing services on ports 8080 or 5000. Each browser has an independent baseline because font rasterization and layout engines legitimately differ.
+
+## Run the native Playwright Test suite
+
+```bash
+npm run test:pw
+npm run test:pw -- --update-snapshots
+```
+
+The native suite uses `playwright.config.cjs`, a fixed local server port `4173`, and the three desktop browser projects. The custom runner is the authoritative report because it also covers mobile and dark-theme captures.
+
+## Regression policy
+
+- First run creates a baseline only when `--update-baseline` is supplied (or a snapshot is absent).
+- Later runs fail with a non-zero exit code when the changed-pixel ratio exceeds `0.002` at a channel threshold of `16`.
+- A failing comparison writes `artifacts/visual-diffs/<browser>/<viewport>/<state>.png` and includes the metrics in `artifacts/report.json`.
+- Update baselines only for an intentional UI change, then rerun the complete matrix.
