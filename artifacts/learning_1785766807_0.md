+"""
+End-to-End Orchestrator
+- Run cross-browser → visual regression → HTML report in one go
+- Flags:
+  --update-baseline  Create/refresh baselines from current screenshots
+"""
+from __future__ import annotations
+
+import argparse
+import subprocess
+import sys
+import time
+from pathlib import Path
+
+ROOT = Path(__file__).resolve().parent.parent
+SCRIPTS = ROOT / "scripts"
+
+
+def run(label: str, cmd: list[str]) -> int:
+    print(f"\n=== {label} ===")
+    print(" ".join(cmd), flush=True)
+    rc = subprocess.call(cmd)
+    print(f"--- {label} exit={rc} ---")
+    return rc
+
+
+def main() -> int:
+    p = argparse.ArgumentParser()
+    p.add_argument("--update-baseline", action="store_true",
+                   help="Treat current screenshots as new baselines")
+    p.add_argument("--skip-cross-browser", action="store_true",
+                   help="Skip cross-browser capture (reuse existing current/*.png)")
+    args = p.parse_args()
+
+    started = time.time()
+
+    if not args.skip_cross_browser:
+        rc = run("Step 1/3 — Cross-Browser Capture", ["python3", str(SCRIPTS / "run_cross_browser.py")])
+        if rc != 0:
+            print(f"[fatal] cross-browser step failed (rc={rc})")
+            return rc
+
+    vargs = ["python3", str(SCRIPTS / "run_visual_regression.py")]
+    if args.update_baseline:
+        vargs.append("--update-baseline")
+    rc = run("Step 2/3 — Visual Regression Diff", vargs)
+    diff_rc = rc  # visual-regression rc=1 means at least one diff fail (kept as data)
+
+    rc = run("Step 3/3 — HTML Report", ["python3", str(SCRIPTS / "generate_html_report.py")])
+    if rc != 0:
+        print("[fatal] report step failed")
+        return rc
+
+    elapsed = time.time() - started
+    print(f"\nAll done in {elapsed:.2f}s")
+    print(f"Report: {ROOT / 'reports' / 'report.html'}")
+    return diff_rc

+
+if __name__ == "__main__":

+# Playwright 跨瀏覽器視覺回歸測試實驗室
+
+整合 `Advanced Cross-Browser Automation and Testing with Playwright`、`Comprehensive Visual Regression and Cross-Browser Testing`、`Comprehensive Visual Regression and Automated Testing` 三個微技能的成品。
+
+## 結構
+
+```
+project/
+├── scripts/
+│   ├── run_cross_browser.py        # Playwright 跨瀏覽器 + viewport 截圖 + functional E2E
+│   ├── run_visual_regression.py    # pixelmatch 像素比對 + ignore-region 中性漆
+│   ├── generate_html_report.py     # Glassmorphism 自含 HTML 報告(base64 內嵌)
+│   └── run_all.py                  # 一鍵整合 (cross-browser → diff → report)
+├── tests/
+│   ├── playwright.config.js        # 多瀏覽器 × 多 viewport project 設定
+│   ├── demo.spec.js                # 官方 Playwright Test spec-style 範例
+│   └── package.json                # @playwright/test + pixelmatch deps
+├── baselines/                      # 基準截圖 (baseline)
+├── current/                        # 本次執行截圖
+├── diffs/                          # pixelmatch diff overlay PNG
+└── reports/report.html             # 自含 HTML 報告
+```
+
+## 使用
+
+### Python 一鍵整合
+
+```bash
+# 首次跑 → 自動建立 baseline
+python3 scripts/run_all.py --update-baseline
+
+# 之後跑 → 對 baseline 比對
+python3 scripts/run_all.py
+```
+
+### 單獨跑某一步
+
+```bash
+python3 scripts/run_cross_browser.py        # 跑 5 個 profile
+python3 scripts/run_visual_regression.py    # 像素比對 + ignore-region
+python3 scripts/generate_html_report.py     # 產出 reports/report.html
+```
+
+### Node 端官方 Playwright Test
+
+```bash
+cd tests
+npm install
+npx playwright install        # 已裝可跳過
+npm test                       # 跑 spec,自帶 html reporter
+npm run test:probe             # 跑視覺漂移探測 (RUN_VISUAL_PROBE=1)
+```
