+"""Analytics Sink — DynamoDB Streams consumer → batched Firehose put.
+
+Streams give us ordering guarantees per partition key.
+Firehose buffers + batches → cheap S3 writes.
+"""
+from __future__ import annotations
+
+import base64
+import json
+import os
+from typing import Any, Dict, List
+
+import boto3
+from aws_lambda_powertools import Logger, Tracer, Metrics
+
+logger = Logger(service="analytics-sink")
+tracer = Tracer(service="analytics-sink")
+metrics = Metrics(namespace="OrderPipeline", service="analytics-sink")
+
+_firehose = boto3.client("firehose")
+FIREHOSE_STREAM = os.environ["FIREHOSE_STREAM"]
+
+
+def _decode(d: Dict[str, Any]) -> str:
+    if "NewImage" in d:
+        return json.dumps(d["NewImage"], default=str, separators=(",", ":"))
+    return json.dumps(d, default=str, separators=(",", ":"))
+
+
+@tracer.capture_lambda_handler
+@logger.inject_lambda_context
+@metrics.log_metrics
+def handler(event, context):
+    records = event.get("Records", [])
+    batch: List[Dict[str, str]] = []
+    for r in records:
+        if r.get("eventName") in {"INSERT", "MODIFY"}:
+            batch.append({"Data": (_decode(r["dynamodb"]) + "\n").encode("utf-8")})
+    if not batch:
+        return {"sent": 0}
+
+    failed: List[Dict[str, Any]] = []
+    chunk = 500      # Firehose PutRecordBatch limit
+    for i in range(0, len(batch), chunk):
+        resp = _firehose.put_record_batch(
+            DeliveryStreamName=FIREHOSE_STREAM,
+            Records=batch[i:i + chunk],
+        )
+        failed.extend(resp.get("Failed", []))
+
+    metrics.add_metric(name="AnalyticsRecordsSent", unit="Count", value=len(batch))
+    metrics.add_metric(name="AnalyticsFailures", unit="Count", value=len(failed))
+    if failed:
+        # surface partial-batch failures back to Streams retry loop
+        raise RuntimeError(f"{len(failed)} firehose put failures")
+    return {"sent": len(batch)}
