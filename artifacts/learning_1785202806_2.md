+        connect_args["check_same_thread"] = False
+        if ":memory:" in db_url or db_url.endswith(":memory:"):
+            from sqlalchemy.pool import StaticPool
+            engine_kwargs["poolclass"] = StaticPool
+            engine_kwargs["connect_args"] = connect_args

+"""In-memory sliding-window rate limiter.
+
+Design notes
+------------
+* **Why in-memory**: keeps the project dependency-free (no Redis). One process,
+  one limiter. The interface is small (``.check(key)`` + ``.reset()``) so
+  swapping in a Redis-backed limiter later only touches this file.
+* **Why sliding window (not fixed window or token bucket)**:
+    - Smooths bursts better than fixed window.
+    - Simpler than token bucket for a "N requests per W seconds" rule.
+* **Thread safety**: a single ``threading.Lock`` guards the deque mutation +
+  ``pop`` loop. A single lock is sufficient because the deque per key is the
+  hot resource and the critical section is microsecond-scale.
+* **Eviction**: when a key's window slides fully past, all of its entries are
+  popped in one pass. To prevent unbounded memory growth from one-off client
+  IPs, we also evict any key whose deque is empty after sliding.
+"""
+from __future__ import annotations
+
+import threading
+import time
+from collections import deque
+from collections.abc import Iterable
+
+
+class RateLimiter:
+    """Sliding-window per-key rate limiter.
+
+    Parameters
+    ----------
+    max_requests:
+        Number of allowed requests per ``window_seconds``.
+    window_seconds:
+        Sliding window length in seconds.
+    """
+
+    def __init__(self, max_requests: int, window_seconds: float):
+        if max_requests <= 0:
+            raise ValueError("max_requests must be positive")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be positive")
+        self._max = max_requests
+        self._window = float(window_seconds)
+        self._buckets: dict[str, deque[float]] = {}
+        self._lock = threading.Lock()
+
+    @property
+    def max_requests(self) -> int:
+        return self._max
+
+    @property
+    def window_seconds(self) -> float:
+        return self._window
+
+    def _evict(self, key: str, now: float) -> None:
+        bucket = self._buckets.get(key)
+        if not bucket:
+            return
+        cutoff = now - self._window
+        while bucket and bucket[0] <= cutoff:
+            bucket.popleft()
+        if not bucket:
+            # Avoid leaking empty buckets from one-off client keys.
+            self._buckets.pop(key, None)
+
+    def check(self, key: str, now: float | None = None) -> tuple[bool, int]:
+        """Record a hit for ``key`` and return ``(allowed, remaining)``.
+
+        ``allowed`` is True if the request is within the limit. ``remaining``
+        is the number of additional requests allowed in the current window
+        after this one (0 if blocked, >=0 otherwise).
+        """
+        ts = time.monotonic() if now is None else now
+        with self._lock:
+            self._evict(key, ts)
+            bucket = self._buckets.setdefault(key, deque())
+            if len(bucket) >= self._max:
+                return False, 0

+# Todo API
+
+A minimal Todo REST API built with **FastAPI + SQLAlchemy 2.0 (sync) + SQLite**, with an in-process sliding-window rate limiter and a full pytest suite (30 tests).

+
