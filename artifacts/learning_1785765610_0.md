+"""
+glassmorphism_verifier.py - Glassmorphism 跨瀏覽器視覺一致性驗證模組
+
+職責：
+  1. 從執行中的 Playwright page 抽取 computed style 來驗證 backdrop-filter
+     在不同瀏覽器引擎（Chromium / WebKit / Firefox）下是否都正確渲染
+  2. 統計頁面上玻璃面板（.glass-panel / .glass-card）的數量、樣式屬性
+  3. 比對 baseline 的玻璃樣式 fingerprint（用於「結構層級」的視覺回歸 —
+     pixel 比對是底層、computed style 比對是中層、文字 + 結構是上層）
+  4. 偵測 backdrop-filter 是否真的被瀏覽器套用（避免 Safari 舊版/某些 headless
+     模式把 backdrop-filter 當成無支援屬性直接忽略掉）
+
+設計原則：
+  - 純函式 + dataclass，可單獨 import 測試
+  - 不依賴 Playwright screenshot（雖然 cross_browser_runner 會用）
+  - 回傳結構化 GlassmorphismReport 給 report_generator 跟報告看板
+"""
+from __future__ import annotations

+import json
+import re

+from pathlib import Path
+from typing import Optional

+

+# 預期樣式指紋（與 target_site/index.html 的 CSS 一致）

+EXPECTED_BACKDROP_BLUR_PX = 14          # .glass-panel 的預期 blur 半徑
+EXPECTED_BACKDROP_BLUR_STRONG_PX = 22   # .glass-panel.strong
+EXPECTED_BACKDROP_BLUR_WEAK_PX = 8      # .glass-panel.weak / .glass-card
+EXPECTED_GLASS_PANELS_MIN = 5           # 主頁面至少 5 個 glass-panel
+EXPECTED_GLASS_CARDS_MIN = 4            # 至少 4 個 glass-card
+EXPECTED_ALPHA_RANGE = (0.05, 0.50)     # glass-bg 的 alpha 範圍（避免 0/1）
+EXPECTED_SATURATE_PRESENT = True        # saturate(180%) 應該被保留
+
+# 已知的「跨瀏覽器字串差異」
+#   - Chromium / Firefox 新版：backdrop-filter
+#   - 舊 Safari / iOS WebKit 仍可能回 -webkit-backdrop-filter
+#   - Firefox 70+ 才支援 backdrop-filter，舊版會回 "none"

+    selector: str

+    background_color: str
+    border_radius: str
+    box_shadow: str
+    has_blur: bool
+    blur_radius_px: Optional[float]
+    has_saturate: bool
+    alpha: Optional[float]
