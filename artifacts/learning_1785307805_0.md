+"""SQLite database helper for the todo app.
+
+Pure stdlib — keeps the dependency surface tiny and makes the layer
+trivial to swap (Postgres / MySQL) without touching the FastAPI code.
+"""
+
+from __future__ import annotations
+
+import os
+import sqlite3
+from contextlib import contextmanager
+from pathlib import Path
+from typing import Iterator
+
+# Default DB location; overridable via env for tests / deployment.
+DEFAULT_DB_PATH = Path(os.environ.get("TODO_DB_PATH", "todos.db"))
+
+
+SCHEMA = """
+CREATE TABLE IF NOT EXISTS todos (
+    id          INTEGER PRIMARY KEY AUTOINCREMENT,
+    title       TEXT    NOT NULL,
+    completed   INTEGER NOT NULL DEFAULT 0,
+    created_at  TEXT    NOT NULL DEFAULT CURRENT_TIMESTAMP
+);
+CREATE INDEX IF NOT EXISTS idx_todos_completed ON todos(completed);
+"""
+
+
+def connect(db_path: Path | str = DEFAULT_DB_PATH) -> sqlite3.Connection:
+    """Open a connection with sensible defaults for a small web app."""
+    conn = sqlite3.connect(db_path, check_same_thread=False, isolation_level=None)
+    conn.row_factory = sqlite3.Row
+    conn.execute("PRAGMA foreign_keys = ON")
+    conn.execute("PRAGMA journal_mode = WAL")
+    return conn
+
+
+def init_db(db_path: Path | str = DEFAULT_DB_PATH) -> None:
+    """Create tables if they don't exist. Safe to call on every startup."""
+    conn = connect(db_path)
+    try:
+        conn.executescript(SCHEMA)
+    finally:
+        conn.close()
+
+
+@contextmanager
+def get_conn(db_path: Path | str = DEFAULT_DB_PATH) -> Iterator[sqlite3.Connection]:
+    """Context-managed connection. Auto-commits on success, rolls back on error."""
+    conn = connect(db_path)
+    try:
+        yield conn
+    except Exception:
+        conn.execute("ROLLBACK")
+        raise
+    else:
+        # We use autocommit (isolation_level=None) so no explicit COMMIT needed;
+        # but keep the hook here in case someone re-enables transactions.
+        pass
+    finally:
+        conn.close()

+## 架構決策
+
+1. **無 ORM**：直接用內建 `sqlite3`，避免 SQLAlchemy 帶來的非必要複雜度。Todo 是 4 個欄位的單純 table，ORM 投資報酬率太低。換 Postgres 時只需改 `db.py` 的 `connect()`，business logic 不動。
+2. **slowapi（不是自刻）**：官方推薦的 FastAPI 限流方案，底層是 limits 庫，支援 IP / 自訂 key / 多種時間窗。`@limiter.limit("10/minute")` 寫法比 middleware 乾淨，且與 OpenAPI 文件整合。
+3. **限流分層**：default 120/min 全域 + 寫入端點 10/min + 讀取端點 60/min — 阻止寫入濫用但不影響正常使用。
+4. **lifespan 取代 on_event**：FastAPI 0.110+ `on_event` deprecated，改用 `lifespan` context manager（消警告，符合最新風格）。
+5. **測試隔離**：`TODO_DB_PATH` env var + pytest session-scoped temp file，確保測試不污染真實 DB。rate limit 在大部分測試關閉（設 `Limiter.enabled = False`），單一專屬測試再打開驗證 429。

+## 除錯紀錄（4 個 bug，全在第一輪 pytest 抓出）
+
+### Bug #1 — `Limiter._rate_limits` 不存在
+**症狀**：`AttributeError: type object 'Limiter' has no attribute '_rate_limits'`
+
+**根因**：原打算 `Limiter._rate_limits.clear()` 拿掉所有規則。slowapi 0.1.10 沒有這個 class attr — 規則是存在 instance 而非 class。
+
+**修法**：觀察 Limiter 原始碼，發現 `_check_request_limit` 第一個檢查就是 `if not self.enabled: return`。而 `enabled` 是 instance 的普通 bool attribute，可以直接設為 False。所有測試 disable，單一專屬測試再 re-enable。
+
+### Bug #2 — passthrough 沒有履行 contract
+**症狀**：試圖把 `_check_request_limit` 整個替換成 `return True` 的 passthrough，結果 `AttributeError: 'State' object has no attribute 'view_rate_limit'`
+
+**根因**：slowapi 的 `sync_wrapper` 在裝飾器跑完後還會讀 `request.state.view_rate_limit` 來判斷要不要丟 RateLimitExceeded。完全短路會讓 wrapper 以為有違規 → AttributeError。
+
+**修法**：放棄 monkeypatch method，改用 `Limiter.enabled = False` — 走正常路徑但跳過所有檢查，wrapper 拿到的 `view_rate_limit` 是合理的空物件，沒有副作用。
+
+### Bug #3 — sqlite autocommit 模式下 ROLLBACK 報錯
+**症狀**：`test_update_missing_todo_returns_404` 失敗：`sqlite3.OperationalError: cannot rollback - no transaction is active`
+
+**根因**：`get_conn()` 在 except 區塊盲目執行 `conn.execute("ROLLBACK")`，但 `connect()` 設了 `isolation_level=None`（autocommit），沒有 transaction 可以 rollback。這個錯誤只會在「故意 raise 404」這種路徑觸發，因為 happy path 不會進 except。
+
+**修法**：刪掉 except 區塊的 ROLLBACK。autocommit 模式下單一 statement 自動 commit，不需要 rollback；註解說明若未來改回 transaction mode 要改用 BEGIN/COMMIT。

+### Bug #4 — 模組載入順序問題

+**修法**：fixture 內 `globals()["_original_check"] = _orig`，顯式寫入模組 namespace，確保專屬測試讀到真實的 original method。
