+"""
+Rate limiting via slowapi.
+
+slowapi is a thin port of Flask-Limiter to Starlette/FastAPI. It uses an
+in-memory storage by default — fine for a single-process dev/test setup.
+For production multi-process, swap storage_uri to Redis.
+
+Strategy chosen here:
+- Limit per remote IP (key_func=get_remote_address).
+- Default limit applied app-wide; specific routes can override.
+- The 429 response is JSON-shaped via a custom handler so the API contract
+  stays consistent.
+"""
+
+from __future__ import annotations
+
+from fastapi import Request
+from fastapi.responses import JSONResponse
+from slowapi import Limiter
+from slowapi.errors import RateLimitExceeded
+from slowapi.util import get_remote_address
+
+from app.config import settings
+
+# Default limit pulled from settings so tests can lower it.
+_default_limit = f"{settings.rate_limit_per_minute}/minute"
+
+limiter = Limiter(
+    key_func=get_remote_address,
+    default_limits=[_default_limit],
+    headers_enabled=True,  # adds X-RateLimit-* headers (handy for clients)
+)
+
+
+async def rate_limit_exceeded_handler(
+    request: Request, exc: RateLimitExceeded
+) -> JSONResponse:
+    """Convert RateLimitExceeded into a structured 429 JSON response."""
+    # exc.detail is the string form of the limit ("60 per 1 minute").
+    return JSONResponse(
+        status_code=429,
+        content={
+            "detail": f"Rate limit exceeded: {exc.detail}",
+            "limit": exc.detail,
+        },
+        headers=getattr(exc, "headers", {}) or {},
+    )
