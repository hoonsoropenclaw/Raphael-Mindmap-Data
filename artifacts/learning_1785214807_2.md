+// React context + hook for the tasks reducer.
+// Keeps dispatch wiring in one place, so components don't need to know
+// whether they're talking to a reducer, a remote API, or a fixture.
+
+import { createContext, useContext, useEffect, useReducer, useRef } from 'react';
+import {
+  tasksReducer,
+  initialState,
+} from './tasksReducer.js';
+
+const TasksStateCtx = createContext(null);
+const TasksDispatchCtx = createContext(null);
+
+export function TasksProvider({ children }) {
+  const [state, dispatch] = useReducer(tasksReducer, initialState);
+
+  // Simulated server tick — every 2.5s bump progress on 'running' tasks.
+  // In a real app this would be subscribeToWebsocket() or SWR mutation.
+  const tickRef = useRef(null);
+  useEffect(() => {
+    tickRef.current = setInterval(() => {
+      dispatch({ type: 'ADVANCE_PROGRESS' });
+    }, 2500);
+    return () => clearInterval(tickRef.current);
+  }, []);
+
+  // Keyboard shortcuts: ⌘Z / ⌘⇧Z (and Ctrl variants)
+  useEffect(() => {
+    const onKey = (e) => {
+      const meta = e.metaKey || e.ctrlKey;
+      if (!meta) return;
+      const tag = (e.target.tagName || '').toLowerCase();
+      if (tag === 'input' || tag === 'textarea') return;
+      if (e.key === 'z' && !e.shiftKey) {
+        e.preventDefault();
+        dispatch({ type: 'UNDO' });
+      } else if ((e.key === 'z' && e.shiftKey) || e.key === 'y') {
+        e.preventDefault();
+        dispatch({ type: 'REDO' });
+      }
+    };
+    window.addEventListener('keydown', onKey);
+    return () => window.removeEventListener('keydown', onKey);
+  }, []);
+
+  return (
+    <TasksStateCtx.Provider value={state}>
