+// 共享 helper：視覺回歸比對（pixelmatch + pngjs）
+// 套用自 skill #9 comprehensive-visual-regression-testing
+//
+// 設計要點：
+//  - 純 Node、無需外部二進位（pixelmatch + pngjs 即可）
+//  - threshold 與 maxRatio 對應 skill #9 推薦值（element 0.005 / page 0.01）
+//  - diff_buf = bytearray（修掉 skill #9 提到的 "diff_buf type" 踩坑）
+//  - 第二次跑一定產出 diff 圖（紅色高亮），方便人工 review
+
+const fs = require('fs');
+const path = require('path');
+const { PNG } = require('pngjs');
+const pixelmatch = require('pixelmatch').default || require('pixelmatch');
+
+/**
+ * 比較兩張 PNG，回傳 { mismatched, total, ratio, diffBuffer }。
+ * @param {string} baselinePath
+ * @param {string} actualPath
+ * @param {string} diffPath
+ * @param {{threshold?: number, maxRatio?: number}} options
+ */
+function diffPng(baselinePath, actualPath, diffPath, options = {}) {
+  const threshold = options.threshold ?? 0.01; // skill #9 預設 0.01
+  const maxRatio = options.maxRatio ?? 0.01;
+
+  const baseline = PNG.sync.read(fs.readFileSync(baselinePath));
+  const actual = PNG.sync.read(fs.readFileSync(actualPath));
+
+  if (baseline.width !== actual.width || baseline.height !== actual.height) {
+    throw new Error(
+      `Size mismatch: baseline ${baseline.width}x${baseline.height} vs actual ${actual.width}x${actual.height}. ` +
+      `重新建立 baseline 或確認 viewport 一致。`
+    );
+  }
+
+  const { width, height } = baseline;
+  const diff = new PNG({ width, height });
+  // skill #9 提示：diff_buf 必須是 bytearray（PNG instance 內部使用 Uint8Array，可直接傳）
+  const mismatched = pixelmatch(
+    baseline.data,
+    actual.data,
+    diff.data,
+    width,
+    height,
+    { threshold }
+  );
+  const total = width * height;
+  const ratio = total === 0 ? 0 : mismatched / total;
+
+  fs.writeFileSync(diffPath, PNG.sync.write(diff));
+
+  return { mismatched, total, ratio, maxRatio, width, height };
+}
+
+/**
+ * 確保 baseline 目錄存在
+ */
+function ensureDir(p) {
+  fs.mkdirSync(p, { recursive: true });
+}
+
+/**
+ * 組合路徑 helper
+ */
+function baselinePath(name, browserName) {
+  return path.join('baselines', name, `${browserName}.png`);
+}
+function actualPath(name, browserName, testInfo) {
+  return path.join('reports', 'actual', testInfo.titlePath().join('__').replace(/\s+/g, '_'), `${browserName}.png`);
+}
+function diffOutputPath(name, browserName, testInfo) {
+  return path.join('diffs', name, `${browserName}.png`);
+}

+# 學習任務執行筆記：Playwright 跨瀏覽器 + 視覺回歸
+
+## 套用的微技能
+
+1. **#11 cross-browser-smoke-testing** — 提供 projects 矩陣骨架（chromium-desktop + firefox-tablet + webkit-mobile）、webServer 設定、reporter 模式
+2. **#9 comprehensive-visual-regression-testing** — 提供 diffPng + element/page 兩層比對 + threshold 經驗值、baseline 自動建立流程
+
+兩個 skill 互補：#11 解決「跨瀏覽器怎麼組織」，#9 解決「像素怎麼比」。把它們合在同個 Playwright 專案跑。
+
+## 實作面臨的 5 個坑 + 解法
+
+### 1. `testInfo.titlePath()` 不是 fixture 上的方法
+**症狀**：`TypeError: testInfo.titlePath is not a function`
+**原因**：我誤用 test annotation 上的 `testInfo.titlePath()`，但測試函數收到的是 `testInfo` fixture object，**它只有 `title` 屬性**（`titlePath` 是 `TestInfo` annotation method）
+**解法**：用 `testInfo.title` + slug fallback
+
+### 2. `reuseExistingServer: true` 抓到別人的 server
+**症狀**：第一次跑 smoke 全 timeout，title 接收到 `Lumen Glass · UI Component Library`（不是我寫的）
+**原因**：N100 上有別的 hermes session 佔著 4173 port serve 別的目錄，Playwright 預設 `reuseExistingServer: true` 直接複用、不自己起
+**解法**：
+- `reuseExistingServer: false`
+- `cwd: __dirname` 確保 server 從本專案目錄起
+- 不過保險起見，啟動前先 `lsof -ti:4173 | xargs -r kill -9`
+
+### 3. `webServer.outputFolder` 不寫入磁碟
+**症狀**：跑了測試後 `reports/html/` 沒出現
+**原因**：`testsDir: './'` + `outputFolder: 'reports/html'` + `outputDir: 'reports/test-artifacts'` 三個路徑衝突？實測後來 `outputFolder: 'reports/playwright-html'` 還是不行，最後用 `playwright-report/`（在 project root 單獨）才產
+**解法**：report folder 放 `playwright-report/`（project root），跟 `test-artifacts` 分層
+
+### 4. baseline 文字小改但視覺沒失敗
+**症狀**：把 `<h1>` 加 3 個字視覺回歸還 PASS
+**原因**：background-flush 文字小變化在 1440px 寬螢幕上佔 < 0.5% 像素，低於 `maxRatio: 0.01` 容忍線
+**解法**：要驗失敗路徑必須改**大視覺區塊**（背景色、整個 layout 移位），不能用文字小改
+
+### 5. `python3 -m http.server` 預設目錄列 index
+**症狀**：serve 父目錄時會列所有 learning_* 子目錄
+**原因**：http.server 沒指定 `DirectoryIndex` 就會列目錄
+**解法**：本次沒踩到（因為我有 web_output.html），但若 deploy 沒 index 會列目錄——實務要建空 index.html 或用 nginx 設定 deny
+
+## 環境事實（記下備用）
+
+- Playwright 1.62.1 已內建 Node CLI（`./node_modules/.bin/playwright`）
+- Chromium 預設已下載（`~/.cache/ms-playwright/chromium-1234` 多版本）
+- Firefox/WebKit 需 `npx playwright install firefox webkit` 下載（首次 ~150MB）
+- playwright-core + pixelmatch + pngjs 純 Node，皆可在 N100 headless 跑

+## 驗證流程
+
+1. **建 baseline**：跑 1 次 → 全部 visual 測試 SKIP、12 張 baseline 圖產出
+2. **比對**：跑 2 次 → 全部 PASS（actual == baseline）
+3. **失敗路徑**：故意改 counter-card 背景紅 → 2 個 visual 測試 FAIL（ratio 0.98）、diff 圖附在 test report
+4. **跨瀏覽器**：3 個矩陣都通（chromium-desktop / firefox-tablet / webkit-mobile）

+## 設計取捨

+1. **為什麼把 visual 放 `tests-visual/` 而不是 `tests/`** — config 預設 `testDir: './tests'`，要涵蓋多目錄就用 `testDir: './'` + `testMatch` pattern
