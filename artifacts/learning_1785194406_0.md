+"""SQLite engine + session factory.
+
+Kept intentionally small: a single engine, a sessionmaker, and a `get_db`
+FastAPI dependency that yields a Session and closes it after the request.
+The schema lives in `models.py` so Base.metadata is the single source of truth.
+"""
+
+from __future__ import annotations
+
+from typing import Iterator
+
+from sqlalchemy import create_engine
+from sqlalchemy.engine import Engine
+from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker
+
+
+class Base(DeclarativeBase):
+    """Declarative base — every ORM model inherits from this."""
+
+
+def make_engine(database_url: str) -> Engine:
+    """Create a SQLAlchemy engine.
+
+    For SQLite we need `check_same_thread=False` so FastAPI's threadpool
+    can reuse connections across requests.
+    """
+    connect_args = {}
+    if database_url.startswith("sqlite"):
+        connect_args["check_same_thread"] = False
+    return create_engine(database_url, connect_args=connect_args, future=True)
+
+
+def make_session_factory(engine: Engine) -> sessionmaker[Session]:
+    """Wrap the engine in a sessionmaker that does NOT autocommit/autoflush.
+
+    FastAPI dependencies commit explicitly via the dependency itself, and
+    tests want to assert against un-flushed state when needed.
+    """
+    return sessionmaker(bind=engine, autoflush=False, autocommit=False, future=True)
+
+
+def get_db(session_factory: sessionmaker[Session]) -> Iterator[Session]:
+    """FastAPI dependency: yield a session, close it after the request."""
+    db = session_factory()
+    try:
+        yield db
+    finally:
+        db.close()

+"""In-process sliding-window rate limiter.
+
+Why in-process? For a single-instance demo / homework API this is enough
+and avoids a Redis dependency. For multi-instance deployments, swap
+`RateLimiter.allow()` for a Redis-backed implementation.
+
+Design:
+- `RateLimiter` holds a {client_ip: deque[timestamp]} dict.
+- `allow(client_id, limit, window_seconds)` returns True/False.
+- A periodic `_evict_old` is NOT run here — the deque trims itself when
+  it exceeds the window, so memory stays bounded per client.
+- Time is `time.monotonic()` to be immune to wall-clock adjustments.
+"""
+
+from __future__ import annotations
+
+import threading
+import time
+from collections import defaultdict, deque
+from typing import Deque, DefaultDict
+
+
+class RateLimiter:
+    def __init__(self) -> None:
+        self._hits: DefaultDict[str, Deque[float]] = defaultdict(deque)
+        self._lock = threading.Lock()
+
+    def allow(self, client_id: str, limit: int, window_seconds: float) -> bool:
+        """Return True if the request fits under the rate limit, else False."""
+        now = time.monotonic()
+        cutoff = now - window_seconds
+        with self._lock:
+            q = self._hits[client_id]
+            # Drop entries older than the window.
+            while q and q[0] <= cutoff:
+                q.popleft()
+            if len(q) >= limit:
+                return False
+            q.append(now)
+            return True
+
+    def reset(self, client_id: str | None = None) -> None:
+        """Clear state — used by tests between cases."""
+        with self._lock:
+            if client_id is None:
+                self._hits.clear()
+            else:
+                self._hits.pop(client_id, None)
+
+
+# A module-level default so the FastAPI dependency can just import this.
+default_limiter = RateLimiter()
