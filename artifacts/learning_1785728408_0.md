+async function assertPage(page, browserName, viewportName) {
+  await page.waitForLoadState('domcontentloaded');
+  await page.waitForFunction(() => document.readyState === 'complete', null, { timeout: 15000 });
+  const title = await page.title();
+  if (!title.includes('Raphael')) throw new Error(`${browserName}/${viewportName}: unexpected title: ${title}`);
+  const bodyText = await page.locator('body').innerText();
+  if (!bodyText.trim()) throw new Error(`${browserName}/${viewportName}: empty body`);
+  const overflow = await page.evaluate(() => ({
+    scrollWidth: document.documentElement.scrollWidth,
+    clientWidth: document.documentElement.clientWidth,
+    hasHorizontalOverflow: document.documentElement.scrollWidth > document.documentElement.clientWidth + 1,
+  }));
+  if (overflow.hasHorizontalOverflow) {
+    throw new Error(`${browserName}/${viewportName}: horizontal overflow ${JSON.stringify(overflow)}`);
+  }
+  return { title, bodyChars: bodyText.length, overflow };
+}

+# Playwright 跨瀏覽器與視覺回歸測試
+
+## 內容
+- `web_output.html`：從目前可用的本機網頁服務抓取的完整 HTML 交付物。
+- `tests/cross-browser-visual.js`：Playwright 測試腳本。
+  - Chromium、Firefox、WebKit
+  - Desktop 1280×900、Mobile 390×844
+  - 標題、非空內容、水平 overflow 檢查
+  - PNG baseline capture / byte-level visual comparison
+- `tests/visual-baselines/`：六組基準截圖。
+- `tests/visual-results/`：最近一次測試截圖與 JSON 報告。
+
+## 執行
+從工作目錄執行：
+
+```bash
+node tests/cross-browser-visual.js
+```
+
+預設會用目前工作目錄的 `web_output.html` 以暫時 HTTP server `http://127.0.0.1:18081/web_output.html` 測試。
+
+只做視覺比較：
+
+```bash
+VISUAL_MODE=compare node tests/cross-browser-visual.js
+```
+
+指定其他頁面：
+
+```bash
+TARGET_URL=http://127.0.0.1:8080/ VISUAL_MODE=compare node tests/cross-browser-visual.js
+```
+
+注意：跨瀏覽器視覺 baseline 應分瀏覽器與 viewport 儲存；不能把 Chromium 圖片拿去和 Firefox 比，否則字型 rasterization 差異會產生大量誤報。
