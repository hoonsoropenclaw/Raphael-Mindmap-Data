+// Architect Round 1 feedback explicitly called these out:
+//   - debounce or throttle state updates
+//   - gracefully catch QuotaExceededError
+//
+// Strategy:
+//   - In-memory write buffer coalesces a burst of changes into one write.
+//   - Writes are flushed 400ms after the latest call (good UX for typing).
+//   - On flush failure (quota, disabled storage, etc.), we surface the error
+//     via the provided onError callback so the UI can show a banner.
+
+import type { Todo } from "./types";
+
+const STORAGE_KEY = "todo-dashboard:v1";
+const DEBOUNCE_MS = 400;
+
+export interface PersistOptions {
+  // Called when a write fails. Failure is non-fatal — in-memory state survives.
+  onError?: (message: string) => void;
+}
+
+export class Persistence {
+  private timer: ReturnType<typeof setTimeout> | null = null;
+  private pending: Todo[] | null = null;
+  private onError: (m: string) => void;
+
+  constructor(opts: PersistOptions = {}) {
+    this.onError = opts.onError ?? (() => {});
+  }
+
+  // Schedule a debounced write. Subsequent calls within DEBOUNCE_MS coalesce.
+  schedule(todos: Todo[]): void {
+    this.pending = todos;
+    if (this.timer) clearTimeout(this.timer);
+    this.timer = setTimeout(() => {
+      this.flushNow();
+    }, DEBOUNCE_MS);
+  }
+
+  // Force a synchronous flush — used on page unload via beforeunload.
+  flushNow(): void {
+    if (this.timer) {
+      clearTimeout(this.timer);
+      this.timer = null;
+    }

+    this.pending = null;

+      // Some browsers (private mode) throw on the very first setItem call.
+      localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));

### 架構決策（與 Round 1 feedback 對齊）

| Round 1 建議 | 本輪實作 |
|---|---|
| `useReducer` for interrelated state slices | 單一 `todoReducer`，14 種 action 全部走 discriminated union；`never` 強制 exhaustive switch |

| `QuotaExceededError` 安全 | onError callback 把錯誤推入 reducer → `ErrorBanner` |

- **Optimistic update + rollback**：toggle 立即本地更新，失敗時用 `UPDATE` 把 completed 改回去，UI 跟 server 最終一致
- **Hydrate path**：先看 localStorage，沒有才走 mock api，hydrate 階段加 `cancelled` flag 避免 unmount 後 setState
- **錯誤路徑統一**：mock api 錯誤 / storage 錯誤都走 `SET_ERROR` → `ErrorBanner` 提供 retry + dismiss
- **A11y**：所有按鈕都有 `aria-label`，checkbox 包在 `LabelText`，progress bar 有 `aria-label`
