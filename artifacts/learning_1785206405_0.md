+
+    def __post_init__(self) -> None:
+        parsed = urlsplit(self.api_base)
+        loopback = parsed.hostname in {"127.0.0.1", "localhost", "::1"}
+        if parsed.scheme != "https" and not (parsed.scheme == "http" and loopback):
+            raise ValueError("Telegram API 必須使用 HTTPS（本機 loopback 測試除外）。")
+
+    def call(self, method: str, params: dict | None = None, timeout: float = 35.0) -> dict:
+        data = urllib.parse.urlencode(params or {}).encode()
+        url = f"{self.api_base.rstrip('/')}/bot{self.token}/{method}"
+        request = urllib.request.Request(url, data=data, headers={"User-Agent": "telegram-weather-bot/1.0"})
+        try:
+            with urllib.request.urlopen(request, timeout=timeout) as response:
+                payload = json.load(response)
+        except (urllib.error.HTTPError, urllib.error.URLError, socket.timeout, json.JSONDecodeError):
+            raise TelegramError(f"Telegram {method} 呼叫失敗。") from None
+        if not payload.get("ok"):
+            raise TelegramError(f"Telegram {method} 拒絕請求。")
+        return payload.get("result")
