📄 middleware.ts (根目錄)

// middleware.ts — Next.js 14 Edge Middleware
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

// RBAC 路由矩陣
const ROLE_ROUTES: Record<string, string[]> = {
  guest:  ['/', '/login'],
  viewer: ['/', '/login', '/dashboard', '/articles'],
  editor: ['/', '/login', '/dashboard', '/articles', '/articles/new', '/articles/:id/edit'],
  admin:  ['/', '/login', '/dashboard', '/articles', '/articles/new', '/articles/:id/edit',
           '/admin/users', '/admin/settings']
};

const ROLE_LEVEL: Record<string, number> = { guest: 0, viewer: 1, editor: 2, admin: 3 };
const PUBLIC_PATHS = ['/', '/login', '/403'];

function matchRoute(pattern: string, pathname: string): Record<string, string> | null {
  const keys: string[] = [];
  const regexStr = '^' + pattern.replace(/:[a-zA-Z]+/g, (m) => {
    keys.push(m.slice(1));
    return '([^/]+)';
  }) + '$';
  const m = pathname.match(new RegExp(regexStr));
  if (!m) return null;
  const params: Record<string, string> = {};
  keys.forEach((k, i) => params[k] = m[i + 1]);
  return params;
}

function getUserRole(req: NextRequest): string {
  // 實際應用：解析 JWT / session cookie
  // 這裡示範：從 cookie 讀 role
  return req.cookies.get('user_role')?.value || 'guest';
}

function isAuthorized(role: string, pathname: string): boolean {
  const allowed = ROLE_ROUTES[role] || [];
  if (allowed.includes(pathname)) return true;
  return allowed.some(p => p.includes(':') && matchRoute(p, pathname) !== null);
}

export function middleware(req: NextRequest) {
  const { pathname } = req.nextUrl;

  // 公開路由直通
  if (PUBLIC_PATHS.includes(pathname)) return NextResponse.next();

  const role = getUserRole(req);

  if (!isAuthorized(role, pathname)) {
    const url = req.nextUrl.clone();
    url.pathname = '/403';
    url.searchParams.set('from', pathname);
    return NextResponse.redirect(url);
  }

  return NextResponse.next();
}

export const config = {
  // 匹配所有路徑（但排除 _next, api, static）
  matcher: ['/((?!_next|api|favicon.ico).*)']
};

📄 lib/rbac.ts (角色判斷工具)

// lib/rbac.ts — 共用 RBAC 函式
export type Role = 'guest' | 'viewer' | 'editor' | 'admin';

export const ROLE_LEVEL: Record<Role, number> = {
  guest: 0, viewer: 1, editor: 2, admin: 3
};

export function hasRole(userRole: Role, requiredRole: Role): boolean {
  return ROLE_LEVEL[userRole] >= ROLE_LEVEL[requiredRole];
}

export function getCurrentUser(cookies: { get: (k: string) => { value: string } | undefined }) {
  const role = (cookies.get('user_role')?.value as Role) || 'guest';
  return { id: 1, name: 'Demo User', role };
}

📄 app/articles/[id]/edit/page.tsx (巢狀動態路由)

// app/articles/[id]/edit/page.tsx — 巢狀動態路由
import { cookies } from 'next/headers';
import { redirect } from 'next/navigation';
import { getCurrentUser, hasRole } from '../../../../lib/rbac';

interface Props { params: { id: string } }

export default function EditArticlePage({ params }: Props) {
  const user = getCurrentUser(cookies());

  // middleware 已擋，這裡做業務層級二次檢查
  if (!hasRole(user.role, 'editor')) {
    redirect('/403');
  }

  return (
    <div>
      <h1>編輯文章 #{params.id}</h1>
      <form action={`/api/articles/${params.id}`} method="POST">
        <input type="text" name="title" />
        <button type="submit">儲存</button>
      </form>
    </div>
  );
}

✅ 實作要點總結

  • middleware.ts 統一在 Edge 層攔截，比每個 page 都寫判斷更乾淨
  • matcher 設定 排除 _next/api/static，效能最佳化
  • 動態路由 [id] 透過 params 物件傳遞參數；巢狀用 [...slug] catch-all
  • 二次檢查 middleware 擋外部存取，page 內仍要做業務邏輯檢查（防 race condition）
  • cookie/session 實務上用 NextAuth.js 或 jose 解析 JWT，不要自己手刻
