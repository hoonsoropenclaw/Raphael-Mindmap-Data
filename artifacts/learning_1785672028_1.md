+/**
+ * lib/form-schema.ts — Dynamic form schema with field-level RBAC.
+ *
+ * A schema is a JSON object describing fields. Each field can declare:
+ *   - type: text | number | select | textarea | date | checkbox | file
+ *   - required: boolean
+ *   - visibility: which role can SEE this field
+ *   - editable:  which role can EDIT this field
+ *   - validate: inline zod-like constraints
+ *
+ * The renderer (FormRenderer.tsx) takes the schema + current principal and
+ * dynamically renders the form. Fields the user can't see are stripped; fields
+ * they can't edit are read-only.
+ */
+

+  | "text"
+  | "number"
+  | "select"
+  | "textarea"
+  | "date"
