+"""In-memory rate limiter。
+
+設計：
+- 用「固定窗口計數器」(fixed-window counter)。每個 client IP 對應一個窗口，窗口期內超過 N 次就 429。
+- 簡單、可預測、O(1) 記憶體／每次請求。
+- 用 thread-safe 的 dict + Lock 保護，避免 race condition。
+- 用 dependency 形式注入到 router，unit test 可以直接換實作。
+
+限制：
+- In-memory 表示每個 process 獨立計數；多 process 部署需改 Redis。
+- 沒有清理過期 client 的機制（dict 會無限增長），在大量 IP 場景應加 LRU；對 demo 規模 OK。
+"""
+from fastapi import Request, HTTPException, status
+from collections import defaultdict, deque
+from threading import Lock
+from time import time
+from typing import Deque, Dict
+
+
+class RateLimiter:
+    """固定窗口滑動變形：以最後 N 次請求的 timestamp deque 判斷。"""
+
+    def __init__(self, max_requests: int, window_seconds: float):
+        self.max_requests = max_requests
+        self.window_seconds = window_seconds
+        # client_key -> deque of timestamps
+        self._buckets: Dict[str, Deque[float]] = defaultdict(deque)
+        self._lock = Lock()
+
+    def check(self, client_key: str) -> tuple[bool, int, float]:
+        """檢查 client 是否在限額內。
+
+        Returns:
+            (allowed, remaining, retry_after_seconds)
+        """
+        now = time()
+        cutoff = now - self.window_seconds
+
+        with self._lock:
+            bucket = self._buckets[client_key]
+            # 移除窗口外的舊紀錄
+            while bucket and bucket[0] <= cutoff:
+                bucket.popleft()
+
+            if len(bucket) >= self.max_requests:
+                # retry_after = 直到最舊紀錄過期為止
+                retry_after = max(0.0, bucket[0] + self.window_seconds - now)
+                return False, 0, retry_after
+
+            bucket.append(now)
+            remaining = self.max_requests - len(bucket)
+            return True, remaining, 0.0
+
+    def reset(self, client_key: str | None = None) -> None:
+        """清空計數（測試用）。"""
+        with self._lock:
+            if client_key is None:
+                self._buckets.clear()
+            else:
+                self._buckets.pop(client_key, None)
+
+
