"""
Config — bot 設定與環境變數覆寫入口。

設計原則 (吸收自 feedback):
- 模組分離: config 獨立,測試可控 state
- 環境變數注入: 工廠函式 + 環境變數 fallback (跟 FastAPI Todo API 一致)
- 所有路徑/threshold 可 ENV 覆寫 (跟 logwatcher / sysmon-quickcheck 一致)
- Dataclass 而不是 dict: 型別安全、IDE autocomplete 可用
"""

from __future__ import annotations

import os
from dataclasses import dataclass, field
from typing import Optional


def _env(name: str, default: Optional[str] = None) -> Optional[str]:
    """讀環境變數,空字串當 None 處理 (避免 TELEGRAM_BOT_TOKEN= 變成空 token)。"""
    raw = os.environ.get(name, default)
    if raw is None:
        return None
    raw = raw.strip()
    return raw if raw else None


def _env_int(name: str, default: int) -> int:
    raw = _env(name)
    if raw is None:
        return default
    try:
        return int(raw)
    except ValueError as exc:
        raise ValueError(f"{name} 必須是整數, 得到 {raw!r}") from exc


def _env_float(name: str, default: float) -> float:
    raw = _env(name)
    if raw is None:
        return default
    try:
        return float(raw)
    except ValueError as exc:
        raise ValueError(f"{name} 必須是浮點數, 得到 {raw!r}") from exc


@dataclass(frozen=True)
class WeatherConfig:
    """Open-Meteo API 設定。

    Open-Meteo 是免費天氣 API, 不需要 API key:
    - Geocoding: https://geocoding-api.open-meteo.com/v1/search
    - Forecast:  https://api.open-meteo.com/v1/forecast
    """

    geocoding_url: str = "https://geocoding-api.open-meteo.com/v1/search"
    forecast_url: str = "https://api.open-meteo.com/v1/forecast"
    user_agent: str = "telegram-weather-bot/0.1 (contact: local-dev)"
    request_timeout_sec: float = 10.0
    forecast_days: int = 3  # 當前 + 未來 2 天
    timezone: str = "auto"  # 讓 API 用城市當地時區


@dataclass(frozen=True)
class BotConfig:
    """Telegram bot 行為設定。"""

    rate_limit_window_sec: int = 5
    rate_limit_max_per_window: int = 3
    weather_cache_ttl_sec: int = 600


@dataclass(frozen=True)
class AppConfig:
    """整個 app 的 root config — 由工廠函式組合, 測試可直接注入 fake。"""

    telegram_token: str
    weather: WeatherConfig = field(default_factory=WeatherConfig)
    bot: BotConfig = field(default_factory=BotConfig)

    def __post_init__(self) -> None:
        if not self.telegram_token:
            raise ValueError(
                "TELEGRAM_BOT_TOKEN 不能為空。"
                "請從 @BotFather 申請 token 並 export 到環境變數。"
            )


def load_config(
    token: Optional[str] = None,
    weather: Optional[WeatherConfig] = None,
    bot: Optional[BotConfig] = None,
) -> AppConfig:
    """App factory — 從環境變數讀設定, 測試可直接傳入 fake config。

    所有 ENV 都是 opt-in, 沒設就用預設值 (跟 logwatcher 一致)。
    """
    if token is not None:
        resolved_token = token
    else:
        resolved_token = _env("TELEGRAM_BOT_TOKEN") or ""

    weather_cfg = weather or WeatherConfig(
        request_timeout_sec=_env_float("WEATHER_REQUEST_TIMEOUT_SEC", 10.0),
        forecast_days=_env_int("WEATHER_FORECAST_DAYS", 3),
    )

    bot_cfg = bot or BotConfig(
        rate_limit_window_sec=_env_int("BOT_RATE_LIMIT_WINDOW_SEC", 5),
        rate_limit_max_per_window=_env_int("BOT_RATE_LIMIT_MAX", 3),
        weather_cache_ttl_sec=_env_int("BOT_WEATHER_CACHE_TTL_SEC", 600),
    )

    return AppConfig(
        telegram_token=resolved_token,
        weather=weather_cfg,
        bot=bot_cfg,
    )

+# Telegram Weather Bot

+## 特性
+

+- **錯誤防呆**:城市找不到、API 失敗、網路逾時都有 user-friendly 訊息
+- **rate limit + cache**:每個 chat 滑動窗口 + 同城市 10 分鐘快取,不會 spam Open-Meteo

+### 1.1 為什麼選 Open-Meteo 而非其他天氣 API
+
+| 候選 | API key | 免費 quota | 多語系 |
+|------|---------|-----------|--------|
+| **Open-Meteo** | ❌ 不要 | ✅ 商業使用也免費 | ✅ 簡中 (lang=zh) |
+| OpenWeatherMap | ✅ 要註冊 | 1000/day | ✅ |
+| WeatherAPI | ✅ 要註冊 | 1M/月 | ✅ |

+
+**選 Open-Meteo 的原因**:
+1. **零摩擦** — 不需要註冊、不需要藏 token 到 .env、不需要負擔 API key rotation
+2. **N100 home server 友善** — 永久免費 quota,不會哪天超量斷了
+3. **Geocoding 也是免費的** — OpenWeatherMap 的 geocoding 要另外付費
+4. **明文 HTTPS** — 不需要 OAuth dance
+

+
+| 框架 | 維護 | async | 大小 |
+|------|------|-------|------|
+| **python-telegram-bot** | ✅ 活躍 | ✅ 原生 | 大但全 |
+| pyTelegramBotAPI (telebot) | ✅ | ❌ polling thread | 小 |
+| aiogram | ✅ | ✅ | 中 |
+| Telethon | (MTProto) | n/a | (用戶端用) |
+
+**選 python-telegram-bot 的原因**:
+1. **已經裝了** — `pip list` 顯示 v22.8 (v22+ 是純 asyncio)
+2. **官方 library** — Telegram Bot API 的 reference impl
+3. **DI-friendly** — ApplicationBuilder + bot_data 機制對 DI 友善
+4. **Future-proof** — 跟得上 Bot API 新功能
+
+(代價: package 大(會拉一堆 transitive deps),但反正我們已經裝了)
+
