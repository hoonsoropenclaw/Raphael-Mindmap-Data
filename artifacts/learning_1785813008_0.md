+"""
+visual_diff.py — pixel-accurate visual regression diff in pure-Pillow + numpy.
+
+Why not pixelmatch?
+  The micro-skill `advanced_browser_testing_with_playwright.md` recommends
+  pixelmatch, but that's a Node.js library. We don't have Node/npm in this
+  minimal env, only Pillow + numpy. So we re-implement the same primitives:
+
+    * load two PNGs as RGB(A)
+    * check dimensions match
+    * per-pixel delta with adjustable threshold (default 0.2 ≈ 12% / 255 luminance)
+    * produce a side-by-side diff PNG highlighting mismatches in red
+    * report pixel-mismatch ratio + max-delta
+
+This is the "pure stdlib fallback" pattern — skill-derived, not blindly copied.
+"""
+
+from __future__ import annotations
+
+from dataclasses import dataclass
+from pathlib import Path
+
+import numpy as np
+from PIL import Image
+
+
+@dataclass
+class DiffResult:
+    width: int
+    height: int
+    total_pixels: int
+    mismatched_pixels: int
+    mismatch_ratio: float  # 0.0..1.0
+    max_channel_delta: int  # 0..255
+    diff_path: Path | None  # path to written diff PNG, if output_path was given
+
+    @property
+    def passed(self) -> bool:
+        return self.mismatch_ratio <= 0.001  # 0.1% tolerance
+
+
+def _to_rgb_array(png_path: Path) -> np.ndarray:
+    img = Image.open(png_path)
+    if img.mode != "RGB":
+        img = img.convert("RGB")
+    arr = np.asarray(img, dtype=np.int16)  # int16 to avoid overflow on subtract
