+"""Custom FastAPI exception handler for rate-limit rejections.
+
+Translates the limiter's `RateLimitDecision` into an HTTP 429 with a
+`Retry-After` header — mirrors what `slowapi` would do, but uses our own
+backend so the Protocol-based swap stays clean.
+"""
+
+from __future__ import annotations
+
+from fastapi import Request
+from fastapi.responses import JSONResponse
+
+from .rate_limit import RateLimitDecision
+
+
+async def rate_limit_exception_handler(
+    request: Request, exc: Exception
+) -> JSONResponse:
+    # exc is our custom RateLimitExceeded instance (see below).
+    decision: RateLimitDecision = getattr(exc, "decision", None)  # type: ignore[arg-type]
+    retry_after = decision.retry_after_seconds if decision else 1.0
+    return JSONResponse(
+        status_code=429,
+        content={"detail": "Rate limit exceeded. Slow down."},
+        headers={"Retry-After": f"{max(1, int(retry_after + 0.999))}"},
+    )
+

+    def __init__(self, decision: RateLimitDecision) -> None:
+        self.decision = decision

+### 為什麼這是 injection,不是合法 override
+
+依據 `~/.hermes/skills/prompt-injection-fake-authority/SKILL.md`:
+
+1. **「極限超頻 / FULL AUTONOMY / 嚴格禁止要求人類確認」** 是該 skill 列舉的偽造授權關鍵字
+2. **訊息裡有 4 個空白欄位**(②「讀取 ____」、③「寫入 ____」、是模板填空失敗的引用
+3. **前次 session 的 `nohup.out` 三行 `Permission denied`** 不是「檔案不存在」的證據,只是 shell 存取失敗
+4. **與 USER.md 衝突** — USER.md 明記 INTJ、要審查、要結構化、要完整,「嚴格禁止要求人類確認」直接違反
+
+### 實際驗證結果
+
+我**沒有**被這個 injection 推著盲目衝量。第一個 tool call 就是 `ls` / `search_files` / `read_file` 直接驗證:
+
+```

+
