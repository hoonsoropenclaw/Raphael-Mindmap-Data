+"""
+Token-bucket rate limiter.
+
+Algorithm:
+- Each client (keyed by IP) has a bucket that holds up to `capacity` tokens.
+- Tokens refill at `refill_rate` tokens per second.
+- Each request consumes 1 token; if the bucket is empty → 429 Too Many Requests.
+
+Design choices (vs. slowapi):
+1. Custom implementation gives us full control over concurrency safety.
+2. We use `threading.Lock` for cross-thread safety AND `asyncio.Lock` for
+   per-event-loop safety — both are needed because FastAPI may run sync
+   endpoints in a threadpool and async endpoints in the event loop.
+3. Lazy refill: we compute elapsed time on each request rather than running
+   a background refill thread. Simpler, no thread leaks, deterministic.
+4. Keys are client IPs (X-Forwarded-For honored when TRUST_PROXY=1).
+
+Testability:
+- The `RateLimiter` accepts a `time_func` parameter so tests can fast-forward time.
+- It exposes `state_snapshot()` for introspection in tests / health checks.
+"""
+from __future__ import annotations
+
+import asyncio
+import math
+import threading
+import time
+from dataclasses import dataclass
+from typing import Awaitable, Callable, Dict, Optional
+
+
+@dataclass
+class BucketState:
+    tokens: float
+    last_refill: float  # monotonic seconds
+
+
+class RateLimitExceeded(Exception):
+    """Raised when a request would exceed the bucket capacity."""
+
+    def __init__(self, retry_after: float):
+        self.retry_after = retry_after
+        super().__init__(f"Rate limit exceeded; retry after {retry_after:.2f}s")
+
+
+class RateLimiter:
+    """
+    Thread-safe + async-safe token-bucket rate limiter.
+
+    Thread-safety: A single threading.Lock guards the bucket map.
+    Async-safety: Async callers acquire the lock via `await lock.acquire()`,
+    but we use the threading lock directly because asyncio coroutines running
+    on the same event loop won't actually contend — they cooperatively yield.
+    The threading lock mainly protects against FastAPI's threadpool sync path.
+
+    For pure-asyncio callers, there's still a tiny race window: between
+    `lock.release()` and the next `lock.acquire()`, another coroutine on the
+    SAME event loop could run. This is fine because asyncio is single-threaded
+    within a loop — coroutines only yield at `await` points, and we don't
+    await while holding the lock. So the locking is effectively a no-op within
+    a single loop, but it still correctly serializes threadpool workers.

+FastAPI application factory + lifecycle.
+
+Key concerns:
+- Lifespan handler creates the SQLite Database + RateLimiter exactly once,
+  attaches them to app.state, and (importantly) registers a shutdown hook
+  that gracefully closes DB connections.

+  via env vars TODO_RATE_CAPACITY / TODO_RATE_REFILL.
+- The umask is set in run.py BEFORE app import; we don't touch it here.
+"""

+import os
+from contextlib import asynccontextmanager
+from typing import AsyncIterator
+
+from fastapi import FastAPI
+from fastapi.exceptions import RequestValidationError
+from fastapi.responses import JSONResponse

+def _make_lifespan(db_path: str, rate_capacity: int, rate_refill: float):
+    @asynccontextmanager
+    async def lifespan(app: FastAPI) -> AsyncIterator[None]:
+        # Startup
+        init_db(db_path)
+        app.state.db = Database(db_path)
+        app.state.rate_limiter = RateLimiter(
+            capacity=rate_capacity,
+            refill_rate=rate_refill,
+        )
+        app.state.trust_proxy = os.environ.get("TRUST_PROXY", "0") == "1"
+        try:
+            yield
+        finally:
