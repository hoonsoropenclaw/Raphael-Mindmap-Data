+"""Minimal WSGI service used to exercise the Kubernetes delivery pipeline."""
+
+from __future__ import annotations
+
+import json
+import os
+from collections.abc import Callable, Iterable
+from typing import Any
+
+StartResponse = Callable[[str, list[tuple[str, str]]], Any]
+Environ = dict[str, Any]
+
+_SECURITY_HEADERS = [
+    ("Content-Type", "application/json"),
+    ("Cache-Control", "no-store"),
+    ("X-Content-Type-Options", "nosniff"),
+    ("X-Frame-Options", "DENY"),
+    ("Content-Security-Policy", "default-src 'none'"),
+]
+
+
+def _response(
+    start_response: StartResponse,
+    status: str,
+    payload: dict[str, object],
+) -> Iterable[bytes]:
+    body = json.dumps(payload, separators=(",", ":")).encode("utf-8")
+    headers = [*_SECURITY_HEADERS, ("Content-Length", str(len(body)))]
+    start_response(status, headers)
+    return [body]
+
+
+def application(environ: Environ, start_response: StartResponse) -> Iterable[bytes]:
+    """Serve metadata and probe endpoints without external dependencies."""
+    if environ.get("REQUEST_METHOD") != "GET":
+        return _response(start_response, "405 Method Not Allowed", {"error": "method_not_allowed"})
+
+    path = environ.get("PATH_INFO", "/")
+    if path == "/":
+        return _response(
+            start_response,
+            "200 OK",
+            {
+                "service": "k8s-cd-demo",
+                "version": os.getenv("APP_VERSION", "dev"),
+                "environment": os.getenv("APP_ENV", "local"),
+            },
+        )
+    if path == "/healthz":
+        return _response(start_response, "200 OK", {"status": "ok"})
+    if path == "/readyz":
+        return _response(start_response, "200 OK", {"status": "ready"})
+    return _response(start_response, "404 Not Found", {"error": "not_found"})

+    content = path.read_text(encoding="utf-8")

+## 任務解讀

+## 架構決策
+
+### 1. Pull-based GitOps，而非 CI push 到 production
+
+CI 不持有 kubeconfig，不直接對真實 cluster 執行 `kubectl apply`。Argo CD 讀 Git desired state，開啟 prune/self-heal。好處：
+
+- cluster credential 不進 GitHub Actions
+- Git diff 是完整審計軌跡
+- drift 自動修復
+- rollback 可用 Git revision/revert
+
+### 2. Build once, promote by digest
+
+main 只建置一次 image，附 SBOM/provenance 並 Cosign keyless-sign。staging 先用 immutable SHA tag；production workflow 接收 staging 已驗證 digest，驗證簽署 identity 後才建立 promotion PR，不重建 image。
+
+### 3. Promotion 採 PR，而非 bot 直接寫 main
+
+第一次版本曾設計成 workflow 直接 commit main。審查後判定這會繞過正常 PR review/branch protection，而且 `[skip ci]` 可能讓未驗證的 desired-state change 直接進 main。因此重構為：
+
+- source PR：CI + Kind smoke
+- main delivery：重跑 reusable quality gate，才 build/sign
+- staging/production promotion：都由 bot 開 PR
+- reviewer merge 後 Argo CD 才 reconcile
+
+### 4. Kubernetes hardening
+
+manifest 包含非 root UID/GID、read-only root filesystem、drop ALL capabilities、RuntimeDefault seccomp、不掛 ServiceAccount token、resource requests/limits、三種 probes、rolling update、PDB、HPA、NetworkPolicy 與 restricted Pod Security labels。
