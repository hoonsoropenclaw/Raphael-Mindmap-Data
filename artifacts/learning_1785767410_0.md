+  await page.goto(startUrl, { waitUntil: 'load' });

+  // 案例 2:四個瀏覽器 badge 都在

+def sha256_of(path):
+    h = hashlib.sha256()
+    with open(path, 'rb') as f:
+        for chunk in iter(lambda: f.read(65536), b''):
+            h.update(chunk)

+def compare_one(baseline_path, current_path, diff_path, threshold):
+    """比對一張截圖,回傳 (regression, diff_ratio, total_pixels, diff_pixels, diff_mask)"""
+    if not os.path.exists(baseline_path):

+        b = b.resize(a.size)
+
+    diff = ImageChops.difference(a, b)

+    if not bbox:
+        return False, 0.0, a.size[0] * a.size[1], 0, None

+    diff_mask = None
+    diff_pixels = 0
+    try:
+        import numpy as np
+        arr_a = np.array(a)
+        arr_b = np.array(b)
+        diff_mask = (arr_a != arr_b).any(axis=2)
+        diff_pixels = int(diff_mask.sum())
+    except ImportError:
