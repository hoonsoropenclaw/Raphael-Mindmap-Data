+"""SQLite database setup using SQLAlchemy 2.x sync API.
+
+We use the synchronous API on purpose:
+- Simpler dependency injection for FastAPI (no async engine overhead)
+- SQLite is single-file and single-writer — async doesn't help much
+- Tests can call CRUD functions directly without asyncio.run boilerplate
+"""
+from __future__ import annotations
+
+import os
+from collections.abc import Generator
+from pathlib import Path
+
+from sqlalchemy import create_engine
+from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker
+
+# Resolve DB path relative to project root, not the CWD, so `pytest` from any
+# directory produces consistent results.
+PROJECT_ROOT = Path(__file__).resolve().parent.parent
+DEFAULT_DB_PATH = PROJECT_ROOT / "todos.db"
+
+
+def get_database_url() -> str:
+    """Return the active database URL.
+
+    Override via the `TODO_DATABASE_URL` env var. Tests inject a per-session
+    sqlite:///:memory: URL via monkeypatch.
+    """
+    return os.environ.get("TODO_DATABASE_URL", f"sqlite:///{DEFAULT_DB_PATH}")
+
+
+class Base(DeclarativeBase):
+    """Declarative base for all ORM models."""
+
+
+# `check_same_thread=False` is required for SQLite when used across threads
+# (FastAPI dependency injection can run on a thread pool).
+engine = create_engine(
+    get_database_url(),
+    connect_args={"check_same_thread": False} if get_database_url().startswith("sqlite") else {},
+)
+
+SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
+
+
+def init_db() -> None:
+    """Create all tables. Idempotent — safe to call on every startup."""
+    # Import models so they register with Base.metadata before create_all.
+    from app import models  # noqa: F401
+
+    Base.metadata.create_all(bind=engine)
+
+
+def get_db() -> Generator[Session, None, None]:
+    """FastAPI dependency that yields a request-scoped Session.
+
+    Always closes — even on exception — to avoid leaking connections.
+    """
+    db = SessionLocal()
+    try:
+        yield db
+    finally:
+        db.close()

+"""Pure CRUD functions. No FastAPI imports here — easy to unit test.
+
+Each function takes a SQLAlchemy `Session` so the caller controls the
+transaction boundary. We `flush()` (not `commit()`) to push writes to the
+DB inside the current transaction; the caller commits.
+"""
+from __future__ import annotations
+
+from sqlalchemy import select
+from sqlalchemy.orm import Session
+
+from app import models, schemas
+
+
+def create_todo(db: Session, payload: schemas.TodoCreate) -> models.Todo:
+    """Insert a new todo. `done` is always False on creation."""
+    todo = models.Todo(
+        title=payload.title,
+        description=payload.description,
+        done=False,
+    )
+    db.add(todo)
+    db.flush()
+    db.refresh(todo)
+    return todo
+
+
+def get_todo(db: Session, todo_id: int) -> models.Todo | None:
+    """Fetch a single todo by id, or None if not found."""
+    return db.get(models.Todo, todo_id)
+
+
+def list_todos(db: Session, *, done: bool | None = None, limit: int = 100, offset: int = 0) -> list[models.Todo]:
+    """List todos, newest first.
+
+    `done=None` returns all todos. `done=True/False` filters by completion.
+    Pagination defaults are conservative; clients can override.
+    """
+    stmt = select(models.Todo)
+    if done is not None:
+        stmt = stmt.where(models.Todo.done == done)
+    stmt = stmt.order_by(models.Todo.created_at.desc()).limit(limit).offset(offset)
+    return list(db.scalars(stmt).all())
+
+
+def update_todo(db: Session, todo: models.Todo, patch: schemas.TodoUpdate) -> models.Todo:
+    """Apply a partial update. Only fields explicitly set in `patch` are touched.
+
+    Using `model_fields_set` is the Pydantic-v2 way to tell "client sent null"
+    apart from "client didn't send the field at all". A None for `title` would
+    be rejected by the schema's min_length=1 anyway, so this is safe.
+    """
+    data = patch.model_dump(exclude_unset=True)
+    for field, value in data.items():
