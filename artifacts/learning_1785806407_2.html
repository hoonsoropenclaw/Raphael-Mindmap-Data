設計原則
  • 拒絕為預設 — 所有動作預設 deny,只在策略明確 allow 時才放行 (fail-closed)。
  • 策略與機制分離 — policy.ts 集中定義; authorize() 為唯一執行點。
  • RBAC + ABAC 雙軌 — 角色給予基底能力,屬性條件處理細節 (owner、時間窗、IP、地域)。
  • 可審計 — 每次決策都寫入 audit log (who / what / when / why),有助合規追查。

policy.ts

策略層 (Single Source of Truth)

所有授權規則集中於 lib/rbac/policy.ts,由 authorize() 讀取。改規則不必碰 middleware / API / UI。

// lib/rbac/policy.ts
import { rolePermissions, ABAC_RULES } from './roles';
import type { Session, Action, Resource, DecisionTrace } from './types';

export interface AuthorizeInput {
  session: Session;
  resource: Resource;     // { type: 'document' | 'project' | 'billing' | ..., ownerId: string }
  action: Action;         // 'read' | 'write' | 'delete' | 'share' | 'export'
  context?: {
    ip?: string;
    now?: Date;
    sameOrg?: boolean;
  };
}

export interface AuthorizeResult {
  allowed: boolean;
  reason: string;
  trace: DecisionTrace[];
}

export function authorize(input: AuthorizeInput): AuthorizeResult {
  const trace: DecisionTrace[] = [];
  const { session, resource, action, context = {} } = input;
  const basePerms = rolePermissions[session.role] || [];

  // Step 1 — 角色基底檢查
  const baseKey = `${resource.type}:${action}`;
  const baseDecision = basePerms.includes(baseKey)
    || basePerms.includes(`${resource.type}:*`)
    || basePerms.includes(`*:${action}`);
  trace.push({ step: 'base-role', allow: baseDecision, detail: \`\${session.role} → \${baseKey} = \${baseDecision}\` });
  if (!baseDecision) {
    return { allowed: false, reason: 'role-mismatch', trace };
  }

  // Step 2 — ABAC 屬性條件 (deny-overrides)
  const rules = ABAC_RULES[resource.type] || [];
  for (const rule of rules) {
    const result = rule({ session, resource, action, context, now: context.now ?? new Date() });
    trace.push({ step: \`abac:\${rule.name}\`, allow: result.allow, detail: result.detail });
    if (!result.allow) {
      return { allowed: false, reason: \`abac:\${rule.name}\`, trace };
    }
  }

  // Step 3 — Owner-only override
  if (resource.ownerOnly && resource.ownerId !== session.userId && !session.isAdmin) {
    trace.push({ step: 'owner-check', allow: false, detail: 'non-owner' });
    return { allowed: false, reason: 'owner-only', trace };
  }

  return { allowed: true, reason: 'granted', trace };
}
// lib/rbac/roles.ts
export const ROLE_HIERARCHY = ['guest', 'user', 'manager', 'admin', 'superadmin'] as const;
export type Role = typeof ROLE_HIERARCHY[number];

export const rolePermissions: Record = {
  guest: [
    'home:read', 'public:read',
  ],
  user: [
    'home:read', 'public:read',
    'document:read', 'document:write', 'document:share',
    'project:read',
    'profile:read', 'profile:write',
  ],
  manager: [
    'home:read', 'public:read',
    'document:read', 'document:write', 'document:share',
    'project:read', 'project:write',
    'profile:read', 'profile:write',
    'team:read', 'team:write',
    'billing:read',
  ],
  admin: [
    '*:*',  // admin 全部資源動作皆可(仍受 owner / 時間窗 ABAC 約束)
  ],
  superadmin: [
    '*:*',
    'audit:read', 'audit:export',
    'system:rotate-key',
  ],
};

// lib/rbac/abac.ts
import type { Session, Resource, Action } from './types';
interface Ctx { session: Session; resource: Resource; action: Action; context: any; now: Date; }
type ABAC = (c: Ctx) => { allow: boolean; detail: string };

export const ABAC_RULES: Record = {
  billing: [
    ({ session, context }) => ({
      allow: context.ip !== undefined && !context.ip.startsWith('192.168.'),
      detail: 'billing action restricted to non-internal IPs',
    }),
  ],
  document: [
    ({ session, resource, context }) => ({
      allow: context.now.getHours() >= 8 && context.now.getHours() < 20,
      detail: 'document write restricted to business hours 08:00–20:00',
    }),
  ],
  audit: [
    ({ session }) => ({
      allow: session.role === 'superadmin',
      detail: 'audit log export requires superadmin',
    }),
  ],
};
// lib/rbac/types.ts
export interface Session {
  userId: string;
  role: 'guest' | 'user' | 'manager' | 'admin' | 'superadmin';
  orgId?: string;
  isAdmin?: boolean;
}
export type Action = 'read' | 'write' | 'delete' | 'share' | 'export' | 'rotate';
export type ResourceType = 'home' | 'public' | 'document' | 'project' | 'profile' | 'team' | 'billing' | 'audit' | 'system';

export interface Resource {
  type: ResourceType;
  id: string;
  ownerId: string;
  ownerOnly?: boolean;
}

export interface DecisionTrace {
  step: string;
  allow: boolean;
  detail: string;
}
middleware.ts

邊緣中介層

在 Edge Runtime 透過 getToken() 解析 session,直接擋掉未授權路由,連 API handler 都不用啟動。

// middleware.ts
import { NextResponse, type NextRequest } from 'next/server';
import { getToken } from 'next-auth/jwt';
import { ROLE_HIERARCHY } from './lib/rbac/roles';

const ROUTE_RULES: Array<{ prefix: string; minRole: string }> = [
  { prefix: '/admin',     minRole: 'admin' },
  { prefix: '/manager',   minRole: 'manager' },
  { prefix: '/dashboard', minRole: 'user' },
  { prefix: '/profile',   minRole: 'user' },
  // 公開路由 guest 也可
];

const ROLE_RANK: Record<string, number> = Object.fromEntries(
  ROLE_HIERARCHY.map((r, i) => [r, i])
);

export async function middleware(req: NextRequest) {
  const { pathname } = req.nextUrl;
  const rule = ROUTE_RULES.find(r => pathname.startsWith(r.prefix));
  if (!rule) return NextResponse.next();

  const token = await getToken({ req, secret: process.env.NEXTAUTH_SECRET });
  if (!token) {
    const url = req.nextUrl.clone();
    url.pathname = '/login';
    url.searchParams.set('next', pathname);
    return NextResponse.redirect(url);
  }

  const userRank = ROLE_RANK[token.role as string] ?? -1;
  const minRank  = ROLE_RANK[rule.minRole] ?? Infinity;
  if (userRank < minRank) {
    return NextResponse.json(
      { error: 'forbidden', required: rule.minRole, actual: token.role },
      { status: 403 }
    );
  }

  return NextResponse.next();
}

export const config = {
  matcher: ['/admin/:path*', '/manager/:path*', '/dashboard/:path*', '/profile/:path*'],
};
邊緣層只能做「粗粒度角色檢查」;真正的資源級別 (例如「使用者 A 可刪除文件 123 但不能刪除文件 456」) 必須在 API Route 內透過 authorize() 執行,以避免在 Edge 把整個 policy 邏輯搬進來造成 bundle size 暴增。
app/api/documents/[id]/route.ts

API Route 守衛

每個 handler 在第一行呼叫 requireAuth() + authorize()。任何環節不通過即回 403,並寫 audit log。

// app/api/documents/[id]/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { authorize } from '@/lib/rbac/policy';
import { loadDocument } from '@/lib/db';
import { audit } from '@/lib/audit';

export async function DELETE(req: NextRequest, { params }: { params: { id: string } }) {
  const session = await getServerSession(authOptions);
  if (!session) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });

  const doc = await loadDocument(params.id);
  if (!doc) return NextResponse.json({ error: 'not-found' }, { status: 404 });

  const decision = authorize({
    session,
    resource: { type: 'document', id: doc.id, ownerId: doc.ownerId, ownerOnly: true },
    action: 'delete',
    context: { ip: req.headers.get('x-forwarded-for') ?? undefined, sameOrg: doc.orgId === session.orgId },
  });

  audit({
    actor: session.userId,
    action: 'document:delete',
    resource: \`document:\${doc.id}\`,
    allowed: decision.allowed,
    reason: decision.reason,
    trace: decision.trace,
  });

  if (!decision.allowed) {
    return NextResponse.json({ error: 'forbidden', reason: decision.reason }, { status: 403 });
  }

  await deleteDocument(params.id);
  return NextResponse.json({ ok: true });
}

export async function PATCH(req: NextRequest, { params }: { params: { id: string } }) {
  const session = await getServerSession(authOptions);
  if (!session) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });

  const doc = await loadDocument(params.id);
  if (!doc) return NextResponse.json({ error: 'not-found' }, { status: 404 });

  const decision = authorize({
    session,
    resource: { type: 'document', id: doc.id, ownerId: doc.ownerId, ownerOnly: true },
    action: 'write',
    context: { ip: req.headers.get('x-forwarded-for') ?? undefined },
  });

  if (!decision.allowed) {
    audit({ actor: session.userId, action: 'document:write', resource: \`document:\${doc.id}\`, allowed: false, reason: decision.reason });
    return NextResponse.json({ error: 'forbidden' }, { status: 403 });
  }

  // ... update document ...
  audit({ actor: session.userId, action: 'document:write', resource: \`document:\${doc.id}\`, allowed: true });
  return NextResponse.json({ ok: true });
}
注意: requireAuth() 只確認「有登入」,真正檢查權限的是 authorize()。兩者分離,讓 401 (未登入) 與 403 (已登入但沒權限) 的語意正確。
components/Can.tsx + hooks/useCan.ts

React Hook & 元件

前端透過 useSession() 取得身份,再以 useCan() 詢問策略層。但 UI 永遠不能作為唯一防線 — 真正的保護在 API。

// hooks/useCan.ts
'use client';
import { useSession } from 'next-auth/react';
import { authorize } from '@/lib/rbac/policy';
import type { Action, Resource } from '@/lib/rbac/types';

export function useCan(resource: Resource, action: Action, context?: Record<string, any>) {
  const { data: session, status } = useSession();
  if (status !== 'authenticated' || !session) return { allowed: false, reason: 'unauthenticated' };
  return authorize({ session, resource, action, context });
}

// components/Can.tsx
'use client';
import { useCan } from '@/hooks/useCan';
import type { Action, Resource } from '@/lib/rbac/types';

interface Props {
  resource: Resource;
  action: Action;
  fallback?: React.ReactNode;
  children: React.ReactNode;
}

export function Can({ resource, action, fallback = null, children }: Props) {
  const { allowed } = useCan(resource, action);
  return <>{allowed ? children : fallback}</>;
}

// usage example
function DocumentActions({ doc }: { doc: Document }) {
  return (
    <div className="actions">
      <Can resource={{ type: 'document', id: doc.id, ownerId: doc.ownerId }} action="write">
        <Button>編輯</Button>
      </Can>
      <Can resource={{ type: 'document', id: doc.id, ownerId: doc.ownerId, ownerOnly: true }} action="delete">
        <Button variant="danger">刪除</Button>
      </Can>
    </div>
  );
}
SSR / Hydration: useCan() 在 server component 不適用;若需要 SSR 渲染決定,可在 server component 直接呼叫 authorize()(它無副作用、可安全在 RSC 跑)。
