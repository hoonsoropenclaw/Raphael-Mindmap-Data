🛠 API 範例

# 批次派送
curl -X POST http://localhost:8000/notify \
  -H 'Content-Type: application/json' \
  -d '{"message":"Hello!","chat_ids":["123","456"],"kind":"markdown"}'

# 單則
curl -X POST http://localhost:8000/notify/single \
  -H 'Content-Type: application/json' \
  -d '{"chat_id":"123","message":"hi","kind":"markdown"}'

# 健康檢查 / 統計 / 歷史
curl http://localhost:8000/health
curl http://localhost:8000/stats
curl 'http://localhost:8000/history?limit=20'

# 直接呼叫 Telegram Bot API（繞過 kind 邏輯）
curl -X POST http://localhost:8000/notify/direct \
  -H 'Content-Type: application/json' \
  -d '{"chat_id":"123","text":"*Bold*","parse_mode":"MarkdownV2"}'

# OpenAPI docs
open http://localhost:8000/docs
