+def _fmt_int(n: int | None) -> str:
+    """Format int with thousand separators, '-' for missing."""
+    return f"{n:,}" if n is not None else "-"
+
+
+def _esc(text: str | None) -> str:
+    """Escape characters that break a markdown table cell."""
+    if text is None:
+        return ""
+    # Pipes break tables; newlines collapse the row.
+    return (
+        text.replace("|", "\\|")
+            .replace("\r\n", " ")
+            .replace("\n", " ")
+    )

+    """Render a sorted-by-period-stars markdown table.
+

+# GitHub Trending → Markdown

+A small, stdlib-only tool that scrapes `github.com/trending` and prints
+the result as a markdown table.
+
+## Why this exists

+weather bot (Senior Architect Round 3 feedback):
+1. **Modular boundary** — fetcher / parser / renderer / CLI are separate
+   modules, each independently testable.
+2. **No third-party deps** — `urllib` + `re` + `html` from stdlib only.
+3. **Mockable HTTP boundary** — `fetch_html()` takes an injected `fetcher`
+   so tests run in ~50 ms with zero network.
+
