+"""Page Object Model base class for the visual-regression framework.
+
+Conventions:
+- Subclasses must assign `path` (relative URL).
+- `navigate()` calls `goto()` with `wait_until='domcontentloaded'`.
+- Helpers: `wait_idle()`, `stabilize()`, `add_visual_styling()`.

+
+import re
+from pathlib import Path
+from typing import Optional
+
+from playwright.sync_api import Page, expect
+
+
+class BasePage:
+    path: str = "/"
+
+    def __init__(self, page: Page, base_url: str = "file:///home/hoonsoropenclaw/.hermes/projects/learning_1785789608_1"):
+        self.page = page
+        self.base_url = base_url
+
+    # ------------------------------------------------------------------ nav
+    def navigate(self) -> "BasePage":
+        self.page.goto(self.full_url, wait_until="domcontentloaded")
+        return self
+
+    @property
+    def full_url(self) -> str:
+        # For file:// URLs we tolerate absolute paths.
+        if self.base_url.startswith("file://"):
+            return f"{self.base_url.rstrip('/')}/{self.path.lstrip('/')}" if self.path else self.base_url
+        return f"{self.base_url.rstrip('/')}{self.path}"
+
+    # -------------------------------------------------------------- elements
+    def locator(self, selector: str):
+        return self.page.locator(selector)
+
+    def title(self) -> str:
+        return self.page.title()
+
+    # -------------------------------------------------------------- helpers
+    def wait_idle(self, timeout: int = 3000) -> None:
+        """Block until network is idle or the timeout elapses.
+
+        Used after navigation and before screenshotting, to avoid
+        capturing in-flight resources. Mirrors the auto-wait pattern
+        from `playwright_automation_and_verification` micro-skill.
+        """
+        try:
+            self.page.wait_for_load_state("networkidle", timeout=timeout)
+        except Exception:
+            # networkidle can be slow on local file:// pages; that's OK.
+            pass
+

+        """Tiny fixed settle used right before screenshots so font/

+"""Visual regression core — pixel-level diff + baseline manager.
+
+Combines:
+- A PIL/RGB pixel-by-pixel diff (no external native deps)
+- A baseline-name strategy: ``vr_baseline_<browser>_<viewport>.png``
+- A "raw" diff PNG for the report
+- A doubly-protected threshold: pixel count budget AND ratio threshold
+"""
+from __future__ import annotations
+
+import io
+from dataclasses import dataclass
+from pathlib import Path
+from typing import Optional, Tuple
+
+from PIL import Image, ImageChops
+
+
+@dataclass
+class DiffResult:
+    ok: bool
+    diff_pixels: int
+    diff_ratio: float
+    baseline_path: Path
+    current_path: Path
+    diff_image_path: Optional[Path] = None
+    reason: str = ""
+
+
+def _load_rgb(path: Path) -> Image.Image:
+    img = Image.open(path).convert("RGB")
+    return img
+
+
+def diff_images(baseline_path: Path, current_path: Path, diff_image_dst: Optional[Path] = None,
+                 pixel_budget: int = 1500, pixel_threshold: float = 0.002,
+                 ignore_regions: Optional[list[Tuple[int, int, int, int]]] = None) -> DiffResult:
+    """Compare two PNGs.
+
+    Strategy: convert both to RGB; if sizes differ, resize current to baseline
+    (we treat size mismatch as a hard fail). Then ``ImageChops.difference``
+    pixel-by-pixel, accumulate the count of non-zero pixels, and also
+    compute the diff ratio (= diff_pixels / total_pixels).
+
+    Returns a DiffResult; the call site decides whether the test passes
+    (diff_pixels <= budget) OR (diff_ratio <= threshold). Both must be
+    satisfied to call OK — that's the "doubly-protected" pattern from

+# Cross-Browser Visual Regression Framework
+
+A pytest-based framework that combines **Page Object Model**, **cross-browser
+automation**, and **pixel-level visual regression** for a static demo
+page (`web_output.html`).
+
+## What's inside
+
+| Layer | Path | Purpose |
+|-------|------|---------|
+| Page Objects | `pages/` | `BasePage`, `LoginPage`, `DashboardPage`, `ComponentPage` |
+| Visual regression core | `visual/` | `DiffResult`, pixel-by-pixel diff, `BaselineManager` |
+| Browser/viewport config | `framework/` | `config.py`, `report.py` (HTML report) |
+| Test suites | `tests/` | smoke, login E2E, dashboard E2E, visual regression |
+| Fixtures & entry | `conftest.py`, `run_tests.py`, `pytest.ini` | parametrized browser×viewport matrix |
+| Demo under test | `web_output.html` | 4 pages: home, users, components (DnD), login |
+| Baselines | `baselines/` | PNG baselines, named `vr_baseline_<browser>_<viewport>_<scenario>.png` |
+| Artifacts | `artifacts/` | current screenshots + diff PNGs |
+| Reports | `reports/run-<ts>/` | `report.html` + `results.jsonl` |
+
+## Quick start
+
+```bash
+# 1. Install deps (only Pillow + pytest + playwright needed)
+pip install pillow pytest playwright
+python -m playwright install chromium firefox webkit
+
+# 2. Create baselines (one-shot, then never run again unless UI changes):
+python run_tests.py --update-baseline --smoke
+python run_tests.py --update-baseline --visual
+
+# 3. Run the full matrix (3 browsers × 3 viewports = 9 combos)
+python run_tests.py
+
+# 4. Filter
+python run_tests.py --browser-filter chromium --viewport-filter desktop
+python run_tests.py --visual --pixel-budget 5000
+```
+
+CLI flags:
+
+| Flag | Effect |
+|------|--------|
+| `--smoke` | Only the `test_smoke.py` suite |
+| `--visual` | Only the `test_visual_regression.py` suite |
+| `--e2e` | Only `test_login_flow.py` + `test_dashboard_rows.py` |
+| `--update-baseline` | Set `UPDATE_BASELINE=1` — baselines are re-created from current screenshots |
+| `--browser-filter chromium,firefox` | Restrict the matrix |
+| `--viewport-filter desktop,mobile` | Restrict the matrix |
+| `--pixel-budget N` | Per-test diff-pixel budget (default 2500) |
+| `--pixel-threshold X` | Per-test diff-ratio cap (default 0.005) |
+| `--list-browsers` | Print matrix and exit |
+
+## How visual regression works
+
+For every (browser, viewport) pair, each visual test:
+
+1. Navigates the browser to the demo page.
+2. Switches to a specific view (`view-home`, `view-users`, `view-components`).
+3. Injects anti-alias/settle CSS + waits 150 ms.
+4. Captures a `full_page` PNG → `artifacts/vr_current_<browser>_<viewport>_<scenario>.png`.
+5. If a baseline doesn't exist (or `UPDATE_BASELINE=1`), promotes the current PNG to baseline.
+6. Otherwise runs `visual/diff.py`:
+   - pixel-by-pixel RGB difference, threshold `max-channel-diff > 8` counts as one diff pixel
+   - Returns a `DiffResult(diff_pixels, diff_ratio)`
+   - **Doubly-protected assertion:** the test passes if *either* `diff_pixels ≤ pixel_budget` *or* `diff_ratio ≤ pixel_threshold`.
+   - A `vr_diff_<...>.png` is dropped into `artifacts/` for inspection.
+
+## How cross-browser works
+
+`framework/config.py` declares `BrowserSpec(name, label, launch_args)` and
+`Viewport(name, w, h)`. The fixture `browser_page` is parametrized with
+indirect parameters so every test that asks for `browser_page` receives one
+fresh `(browser context, page)` per matrix cell.
+
+Chromium is launched with `--no-sandbox` because of the headless N100
+container environment (lesson from micro-skill `playwright_automation_and_verification` §6.3).
