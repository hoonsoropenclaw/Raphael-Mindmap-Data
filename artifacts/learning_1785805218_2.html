⚡ StaticCrawler urllib + BeautifulSoup

純 stdlib urllib + BeautifulSoup 解析;無 Chromium 開銷,單次抓取 ~1.4s。 適用於 SSR 網頁,如 Wikipedia、新聞網站、靜態論壇。 

# 1. 發起 GET 請求,自帶 User-Agent
def _http_get(self, url: str) -> tuple[int, str, str]:
    req = urllib.request.Request(url, headers={
        "User-Agent": USER_AGENT,
        "Accept-Language": "en-US,en;q=0.9,zh-TW;q=0.8",
    })
    with urllib.request.urlopen(req, timeout=self.timeout_s) as resp:
        body = resp.read()
        charset = resp.headers.get_content_charset() or "utf-8"
        html = body.decode(charset, errors="replace")
        return resp.status, html, resp.geturl()

# 2. BeautifulSoup 解析並產出統一 CrawlResult
soup = BeautifulSoup(html, "lxml")
return CrawlResult(
    url=url, mode="static", status="ok",
    http_code=code, title=soup.title.string.strip(),
    html=html, text=soup.get_text(" ", strip=True),
    metrics={"html_len": len(html), "tag_count": len(soup.find_all()), ...},
)

⚡ AsyncDynamicCrawler Playwright async + asyncio.Semaphore

          

            一個 browser、多個 context、asyncio.Semaphore 控制並行度。比「每頁開一個 browser」省 80% 資源。
          

          
# 共用單一 browser;每個 URL 一個隔離 context(各自 cookie / localStorage)
async def _fetch_one(self, browser, url, semaphore):
    async with semaphore:   # 限制並行數
        ctx = await browser.new_context(user_agent=USER_AGENT, ...)
        try:
            page = await ctx.new_page()
            await page.goto(url, wait_until="domcontentloaded")
            await page.wait_for_load_state("networkidle")
            html = await page.content()
            ...
        finally:
            await ctx.close()

# 入口:asyncio.gather 同時觸發 N 個任務
async def fetch_many(self, urls):
    async with async_playwright() as p:
        browser = await p.chromium.launch(...)
        tasks = [self._fetch_one(browser, u, sem) for u in urls]
        return await asyncio.gather(*tasks)

↻ RetryPolicy 指數退避 + Jitter

          

            對應《總工程師審查回饋 v58》:「爬蟲模組必須設計得夠強健,具備重試機制 (Retry/Backoff)」。
            delay = min(cap, base × 2^attempt) ± jitter_ratio × delay
          

          
@dataclass
class RetryPolicy:
    max_attempts: int = 3
    base_delay_s: float = 0.4
    cap_delay_s: float = 4.0
    jitter_ratio: float = 0.25

    def sleep_for(self, attempt: int) -> float:
        # attempt 是第幾次失敗 (1-based)
        delay = min(self.cap_delay_s,
                   self.base_delay_s * (2 ** (attempt - 1)))
        jitter = delay * self.jitter_ratio
        return max(0.0, delay + random.uniform(-jitter, jitter))

    def run(self, fn: Callable[[], CrawlResult]) -> CrawlResult:
        for attempt in range(1, self.max_attempts + 1):
            result = fn()
            result.attempt = attempt
            if result.status == "ok": return result
            if attempt < self.max_attempts:
                time.sleep(self.sleep_for(attempt))
        return result  # 全部 attempts 都 fail,回最後一次
