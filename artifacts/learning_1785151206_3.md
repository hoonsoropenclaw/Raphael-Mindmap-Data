+# --- Engine factory ---------------------------------------------------------

+def make_engine(database_url: str, *, echo: bool = False) -> Engine:
+    """Build a SQLAlchemy engine.
+
+    SQLite needs `check_same_thread=False` so FastAPI's threadpool can share
+    the connection. For other DBs the flag is ignored.
+    """
+    connect_args = {}
+    if database_url.startswith("sqlite"):
+        connect_args["check_same_thread"] = False
+    return create_engine(database_url, echo=echo, future=True, connect_args=connect_args)

+    """Create all tables. Idempotent — safe to call on every startup."""

+        super().__init__(f"Todo with id={todo_id} not found")
+        self.todo_id = todo_id

+
+class TodoService:
+    """Stateless service wrapping Todo CRUD operations."""
+
+    def __init__(self, db: Session):
+        self.db = db
+
+    # ---- queries ----------------------------------------------------------
+    def list(self, *, skip: int = 0, limit: int = 100) -> list[Todo]:
+        """Return todos ordered by id, paginated."""
+        return (
+            self.db.query(Todo)
+            .order_by(Todo.id.asc())
+            .offset(skip)
+            .limit(limit)
+            .all()
+        )
+
+    def get(self, todo_id: int) -> Todo:
+        """Return a single todo or raise TodoNotFoundError."""
+        todo = self.db.get(Todo, todo_id)
+        if todo is None:
+            raise TodoNotFoundError(todo_id)
+        return todo
+
+    # ---- mutations --------------------------------------------------------
+    def create(self, payload: TodoCreate) -> Todo:
+        """Insert a new todo and return the persisted row."""
+        todo = Todo(title=payload.title, done=payload.done)

+the *production* engine's tables (from the prior test's drop_all) are gone.
