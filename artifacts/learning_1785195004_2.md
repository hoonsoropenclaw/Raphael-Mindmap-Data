+/**
+ * Apply filter + search + sort to a list. Stateless.
+ */
+export function applyView(tasks, { filter, sort, search }) {
+  const q = search.trim().toLowerCase();
+  let out = tasks;
+
+  if (filter !== FILTERS.ALL) {
+    out = out.filter((t) =>
+      filter === FILTERS.COMPLETED ? t.completed : !t.completed
+    );
+  }
+
+  if (q) {
+    out = out.filter(
+      (t) =>
+        t.title.toLowerCase().includes(q) ||
+        (t.notes || '').toLowerCase().includes(q) ||
+        (t.tags || []).some((tag) => tag.toLowerCase().includes(q))
+    );
+  }
+
+  out = [...out];
+  switch (sort) {
+    case SORTS.CREATED_ASC:
+      out.sort((a, b) => a.createdAt - b.createdAt);
+      break;
+    case SORTS.CREATED_DESC:
+      out.sort((a, b) => b.createdAt - a.createdAt);
+      break;
+    case SORTS.PRIORITY_DESC:
+      out.sort((a, b) => {
+        const rank = PRIORITY_RANK[b.priority] - PRIORITY_RANK[a.priority];
+        return rank !== 0 ? rank : b.createdAt - a.createdAt;
+      });
+      break;

+# Task Dashboard
+
+A complete React 18 UI showcasing **complex state management** without external
+state libraries — just `useReducer` + Context + useMemo + localStorage.
+
+## Architecture
+
+```
+                ┌──────────────┐
+                │   main.jsx   │
+                │  <Provider>  │
+                └──────┬───────┘
+                       │
+              ┌────────▼─────────┐
+              │  TaskProvider    │  ← useReducer + Context + useEffect
+              │   (Context)      │     (hydrates from / writes to localStorage)
+              └────────┬─────────┘
+                       │
+        ┌──────────────┼──────────────┐
+        ▼              ▼              ▼
+   TaskForm       TaskList       StatsBar / Toolbar
+        │              │
+        │       ┌──────▼──────┐
+        │       │  TaskItem   │  (edit-in-place)
+        │       └─────────────┘
+        │
+        └── dispatches ACTIONS.X ──► taskReducer ──► new state ──► re-render
+```
+
+### State shape
+
+```js
+{
+  tasks:  Task[]   // {id,title,notes,priority,completed,tags,createdAt,updatedAt}
+  filter: 'all' | 'active' | 'completed'
+  sort:   'created_desc' | 'created_asc' | 'priority_desc' | 'alphabetical'
+  search: string
+  hydrated: boolean
+}
+```
+
+### Why useReducer (not Redux/Zustand)?
+
+- Self-contained — zero runtime deps
+- Type-friendly — single action enum in `utils/constants.js`
+- Easy to test — pure reducer, no mock providers
+- Sufficient — single store, no cross-slice subscriptions needed
+
+## Run
+
+```bash
+npm install
+npm run dev        # http://localhost:5173
+npm test           # vitest unit tests (reducer + utils)
+npm run build      # production bundle in ./dist
+```
+
+## Features
+
+- ✅ Add / edit / delete / toggle tasks
+- ✅ Priority levels (low / medium / high) with color-coded borders
+- ✅ Inline editing (no modals, no portal overhead)
+- ✅ Filter chips with live counts (all / active / completed)
+- ✅ Sorting (newest / oldest / priority / alphabetical)
+- ✅ Full-text search across title, notes, tags
+- ✅ Stats bar (total / active / done / completion %)
+- ✅ Persistence to `localStorage` (key: `task-dashboard:v1`)
+- ✅ JSON export / import for backup
+- ✅ Bulk clear completed
