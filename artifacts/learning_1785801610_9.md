+def _parse_query(url: str) -> dict[str, str]:
+    parsed = urlparse(url)
+    out: dict[str, str] = {}
+    for k, v in parse_qs(parsed.query, keep_blank_values=True).items():
+        out[k] = v[0] if v else ""
+    return out
+
+
+def _decode_body(post_data: Optional[dict[str, Any]]) -> Optional[Any]:
+    """HAR postData.text 可能是表單編碼字串或 JSON。"""
+    if not post_data:
+        return None
+    text = post_data.get("text")
+    if text is None:
+        return None
+    mime = post_data.get("mimeType", "").lower()
+    if "json" in mime:
+        try:
+            return json.loads(text)
+        except (json.JSONDecodeError, TypeError):
+            return text
+    if "x-www-form-urlencoded" in mime or "form" in mime:
+        try:
+            return dict(parse_qs(text, keep_blank_values=True))
+        except Exception:
+            return text
+    return text
+
+
+def _decode_response_body(content: Optional[dict[str, Any]]) -> Optional[Any]:
+    if not content:
+        return None

+    if text is None:
+        return None

+# Hermes Scraper API
+
+基於 FastAPI 的 RESTful API 逆向工程與數據抓取服務。
+
+## 功能
+
+1. **HAR 解析** — 解析 Chrome DevTools / Firefox / Charles 匯出的 HAR 檔，萃取完整 request/response metadata
+2. **Endpoint 自動發現** — 從 HAR 或 HTML/JS 原始碼自動歸納 API endpoint 結構（path template `{id}`、query params、headers）
+3. **異步抓取** — 內建 `httpx`（純 HTTP）與 `playwright`（執行 JS）兩種模式，可選填 headers / cookies / wait_selector
+4. **持久化** — SQLite WAL 模式儲存 scrape job、endpoint catalog、webhook 訂閱
+5. **Webhook** — 預留事件訂閱（`scrape.completed` / `discovery.completed`）
+6. **Web Dashboard** — `web_output.html` 提供 GUI 直接操作
