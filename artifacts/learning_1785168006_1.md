+// useLocalStorage.js
+// 通用 hook：把任意 state 雙向同步到 localStorage
+// 解法要點：lazy initializer 讀一次、寫入 debounce 用 useEffect 自動同步、JSON parse 失敗時 fallback
+import { useState, useEffect } from 'react'
+
+export function useLocalStorage(key, initialValue) {
+  // 1) lazy init：只在第一次 render 讀 localStorage（避免每次 render 都序列化）
+  const [value, setValue] = useState(() => {
+    try {
+      const stored = window.localStorage.getItem(key)
+      return stored !== null ? JSON.parse(stored) : initialValue
+    } catch (err) {
+      console.warn(`[useLocalStorage] parse failed for key=${key}:`, err)
+      return initialValue
+    }
+  })
+
+  // 2) value 一變就寫回（debounce 不必要，因為 React 已 batch）
+  useEffect(() => {
+    try {
+      window.localStorage.setItem(key, JSON.stringify(value))
+    } catch (err) {
+      console.error(`[useLocalStorage] write failed for key=${key}:`, err)
+    }
+  }, [key, value])
+
+  return [value, setValue]

+## 架構決策
+
+### 狀態分層
+
+| 狀態 | 存放 | 為什麼 |
+|------|------|--------|
+| `todos[]` | `useTodos` hook + useReducer | 多種動作、需要 undo |
+| `filter` / `search` | App 本地 useState | view-only、不需持久化 |
+| `theme` | useLocalStorage | 跨 session 記住偏好 |
+| `todos` 持久化 | App 層 useEffect 寫 localStorage | reducer 不直接綁 storage（避免 SSR/SSR-like 環境出問題） |
+
+### 為什麼 useReducer 而不是 useState
+- 動作種類 6+ → 多個 setState 容易出錯
+- reducer 是純函式 → 易測試
+- snapshot/undo 需要「上一個穩定狀態」，reducer 集中管理更乾淨
+
+### 為什麼 undo stack 用 ref
+- undo history 不該觸發 re-render（否則每次 push 都會 re-render 整個 tree）
+- ref 存 mutation-only 資料剛好
+
+### 為什麼 衍生狀態用 useMemo
+- filter + search + stats 都是純計算、依賴 todos/filter/search
+- 不入 reducer（reducer 只管主狀態，不管衍生）
+
+### CSS 架構
+- CSS 自訂屬性（`--bg / --text / --primary` 等）做主題切換
+- `[data-theme='dark']` 覆寫所有變數 → 一鍵切換
+- 不引入 Tailwind/styled-components → 純 CSS + BEM 命名 + 自訂屬性，符合「第一次就做對」的最小依賴原則
+- 響應式用 `@media (max-width: 540px)`
