+def _read_increment(path: Path, previous: dict, max_bytes: int) -> tuple[list[str], dict]:
+    stat = path.stat()
+    inode = int(stat.st_ino)
+    previous_inode = previous.get("inode")
+    offset = int(previous.get("offset", 0))
+    if previous_inode != inode or stat.st_size < offset:
+        offset = 0
+    start = max(offset, stat.st_size - max_bytes)
+    truncated = start > offset
+    with path.open("rb") as handle:
+        handle.seek(start)
+        if truncated and start > 0:

+        data = handle.read(max_bytes)
+        new_offset = handle.tell()

+def load_state(path: str) -> dict:
+    file = Path(path)
+    if not file.exists():
+        return {}
+    try:
+        text = file.read_text(encoding="utf-8").strip()
+        if not text:
+            return {}
+        data = json.loads(text)
+        if not isinstance(data, dict):
+            raise ValueError("root is not an object")
+        return data
+    except (OSError, ValueError, json.JSONDecodeError) as exc:
+        raise SinkError(f"cannot load state {path}: {exc}") from None
+

+def atomic_write_json(path: str, data: dict) -> None:
+    target = Path(path)
+    target.parent.mkdir(parents=True, exist_ok=True)
+    fd, temp_name = tempfile.mkstemp(dir=target.parent, prefix=f".{target.name}.", suffix=".tmp")
+    try:
+        with os.fdopen(fd, "w", encoding="utf-8") as handle:
+            json.dump(data, handle, ensure_ascii=False, sort_keys=True)
+            handle.write("\n")
+            handle.flush()
+            os.fsync(handle.fileno())
+        os.chmod(temp_name, 0o600)
+        os.replace(temp_name, target)
+        os.chmod(target, 0o600)
+    except Exception:
+        with suppress(OSError):
+            os.unlink(temp_name)
+        raise
+

+    target = Path(path)

+def deduplicate(alerts: list[Alert], state: dict, now_epoch: float, repeat_seconds: float) -> tuple[list[Alert], dict]:
