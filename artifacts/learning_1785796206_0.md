+function loadPng(p) {
+  return PNG.sync.read(readFileSync(p));

+function diffPng(actualPath, baselinePath, diffPath, { threshold = 0.1, maxRatio = 0.001 } = {}) {
+  const baseline = loadPng(baselinePath);
+  const actual = loadPng(actualPath);
+  if (baseline.width !== actual.width || baseline.height !== actual.height) {
+    throw new Error(
+      `size mismatch: baseline=${baseline.width}x${baseline.height} actual=${actual.width}x${actual.height}`
+    );
+  }
+  const diff = new PNG({ width: baseline.width, height: baseline.height });
+  const mismatched = pixelmatch(
+    baseline.data,
+    actual.data,
+    diff.data,
+    baseline.width,
+    baseline.height,
+    { threshold }
+  );
+  writeFileSync(diffPath, PNG.sync.write(diff));
+  const ratio = mismatched / (baseline.width * baseline.height);
+  return { mismatched, ratio, exceeds: ratio > maxRatio };
+}

+# Playwright Cross-Browser + Visual Regression Lab

+| `comprehensive_visual_regression_testing` | element- and page-level `pixelmatch` comparison |
+| `advanced_web_ui_integration` | Glassmorphism card + Tailwind CDN injection pattern |

+## How the visual regression works
+
+1. **First run** — no baseline exists → the actual screenshot is **promoted to baseline** (test passes).
+2. **Subsequent runs** — actual is diffed against baseline with `pixelmatch` (`threshold=0.005..0.01`). If the ratio of mismatched pixels exceeds `maxRatio`, the test fails and a `*.diff.png` highlighting changed pixels is written.
+3. **Intentional UI changes** — run `npm run update-baselines` to refresh baselines; commit the new PNGs.
