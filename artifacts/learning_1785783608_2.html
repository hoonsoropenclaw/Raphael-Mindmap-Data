# 直接拿來用
python3 -c "
import asyncio
from async_crawler import Crawler, AsyncHTTPClient, ClientConfig, CrawlerConfig, RetryPolicy

async def main():
    async with AsyncHTTPClient(ClientConfig()) as http:
        crawler = Crawler(
            http,
            CrawlerConfig(max_pages=50, max_depth=2, concurrency=8),
            RetryPolicy(max_attempts=5, initial_wait=0.5, multiplier=2.0, jitter=0.2),
        )
        results = await crawler.crawl(['https://example.com'])
        for r in results:
            print(r.status, r.url, r.attempts)

asyncio.run(main())
"
