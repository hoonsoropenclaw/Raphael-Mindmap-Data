+from __future__ import annotations

+from typing import Dict, Optional
+
+
+TRACEPARENT_HEADER = "traceparent"
+TRACESTATE_HEADER = "tracestate"

+@dataclass
+class TraceContext:
+    """Active trace context for the current async task."""
+    trace_id: str
+    span_id: str  # current span id (will become parent for next child)
+    flags: str = "01"  # 01 = sampled
+    tracestate: str = ""
+
+    def to_traceparent(self) -> str:
+        """Serialize to W3C traceparent header value."""
+        return f"00-{self.trace_id}-{self.span_id}-{self.flags}"
+
+    @classmethod
+    def from_traceparent(cls, header: str) -> Optional["TraceContext"]:
+        """Parse W3C traceparent header. Returns None if malformed."""
+        if not header:
+            return None
+        parts = header.strip().split("-")
+        if len(parts) != 4:
+            return None
+        version, trace_id, parent_span_id, flags = parts
+        if version != "00":
+            return None  # only spec version 00 supported
+        if len(trace_id) != 32 or len(parent_span_id) != 16:
+            return None
+        return cls(
+            trace_id=trace_id,
+            span_id=parent_span_id,
+            flags=flags or "01",
+        )

+# Hermes Distributed Tracing System
+
+A self-contained, end-to-end **distributed tracing system for FastAPI microservices** that lets you visualize request flow across services with full span trees, service call graphs, and per-trace flame graphs.
+
+Built from scratch — no OpenTelemetry, no Jaeger, no external services — using only FastAPI, httpx, and Python's standard library.
+
+---
+
+## What it does
+
+When a client calls `POST /orders` on the Order Service, the request flow:
+
+```
+client → Order Service → Inventory Service (reserve stock)
+                       → Payment Service   (fraud check → card charge)
+```
+
+is captured as a single `trace_id` that spans **all three services**. Every internal operation (DB lookup, fraud check, card processor call) becomes a span linked to its parent, so you get a full hierarchical tree with timings.
+
+A web dashboard at `http://127.0.0.1:8001/` lets you browse all recent traces, view per-trace flame graphs, and inspect cross-service call patterns.
+
+---
+
+## Architecture
+
+```
+┌─────────────┐    POST /orders    ┌──────────────┐
+│   client    │ ─────────────────► │ order-service│
+└─────────────┘                    └──────┬───────┘
+                                          │ (auto-instrumented httpx)
+                  ┌───────────────────────┼──────────────────────┐
+                  ▼                       ▼                      ▼
+          ┌──────────────┐         ┌──────────────┐         (none)
+          │inventory-svc │         │ payment-svc  │
+          └──────┬───────┘         └──────┬───────┘
+                 │                        │
+                 │  POST /api/v1/spans    │
+                 └────────────┬───────────┘
+                              ▼
+                    ┌──────────────────┐
+                    │  Trace Collector │
+                    │     (port 8001)  │
+                    └────────┬─────────┘
+                             │ query
+                             ▼
+                    ┌──────────────────┐
+                    │   Dashboard      │
+                    │   (HTML/JS)      │
+                    └──────────────────┘
+```
+
+Each service uses the **`tracing_sdk`** package, which provides:
+- `tracer.start_span(...)` — manual span creation
+- `instrument_app(app)` — auto-creates SERVER spans for inbound FastAPI requests
+- `instrument_httpx()` — auto-creates CLIENT spans for outbound httpx calls + injects W3C `traceparent` header
+
+The Collector receives span batches, builds in-memory trace trees, and exposes them via REST + the dashboard.
