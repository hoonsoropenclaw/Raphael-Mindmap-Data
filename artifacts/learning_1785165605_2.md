+"""Environment-backed application settings.
+
+The app factory accepts explicit arguments for tests and embedding. Environment
+variables provide convenient defaults for a local/Uvicorn deployment without
+making the rest of the application depend directly on ``os.environ``.
+"""
+
+from __future__ import annotations
+
+import os
+from dataclasses import dataclass
+
+
+def _positive_int(name: str, default: int) -> int:
+    raw = os.getenv(name)
+    if raw is None:
+        return default
+    try:
+        value = int(raw)
+    except ValueError as exc:
+        raise ValueError(f"{name} must be an integer") from exc
+    if value < 1:
+        raise ValueError(f"{name} must be at least 1")
+    return value
+
+
+def _positive_float(name: str, default: float) -> float:
+    raw = os.getenv(name)
+    if raw is None:
+        return default
+    try:
+        value = float(raw)
+    except ValueError as exc:
+        raise ValueError(f"{name} must be a number") from exc
+    if value <= 0:
+        raise ValueError(f"{name} must be greater than 0")
+    return value
+
+
+def _boolean(name: str, default: bool) -> bool:
+    raw = os.getenv(name)
+    if raw is None:
+        return default
+    normalized = raw.strip().lower()
+    if normalized in {"1", "true", "yes", "on"}:
+        return True
+    if normalized in {"0", "false", "no", "off"}:
+        return False
+    raise ValueError(f"{name} must be a boolean")
+
+
+@dataclass(frozen=True)
+class Settings:
+    database_url: str = "sqlite:///./todos.db"
+    rate_limit_max_requests: int = 60
+    rate_limit_window_seconds: float = 60.0
+    trust_proxy_headers: bool = False
+

+        return cls(

+
+from __future__ import annotations
+
+from collections import deque
+from dataclasses import dataclass
+from threading import Lock
+from time import monotonic
+from typing import Callable
+
+
+@dataclass(frozen=True)
+class RateLimitDecision:
+    allowed: bool
+    remaining: int
+    retry_after: int = 0
+
+
+class RateLimiter:
+    """Allow at most ``max_requests`` per key in a rolling time window.
+
+    ``monotonic`` time is used so NTP/system-clock changes cannot extend or
+    shorten a client's window unexpectedly. State is deliberately process-local;
+    a multi-worker deployment should provide a shared Redis-backed equivalent.
+    """
+
+    def __init__(
+        self,
+        max_requests: int = 60,
+        window_seconds: float = 60.0,
+        clock: Callable[[], float] = monotonic,
+    ) -> None:
+        if max_requests < 1:
+            raise ValueError("max_requests must be at least 1")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be greater than 0")
+        self.max_requests = max_requests
+        self.window_seconds = float(window_seconds)
+        self._clock = clock
+        self._requests: dict[str, deque[float]] = {}
+        self._lock = Lock()
+
+    def check(self, key: str) -> RateLimitDecision:
+        now = self._clock()
+        cutoff = now - self.window_seconds
+        with self._lock:
+            timestamps = self._requests.setdefault(key, deque())
+            while timestamps and timestamps[0] <= cutoff:
+                timestamps.popleft()
+
+            if len(timestamps) >= self.max_requests:
+                wait = timestamps[0] + self.window_seconds - now
+                retry_after = max(1, int(wait + 0.999999))
+                return RateLimitDecision(False, 0, retry_after)
+
+            timestamps.append(now)
+            return RateLimitDecision(True, self.max_requests - len(timestamps))
+
+    def reset(self) -> None:
+        """Clear all counters; useful for controlled tests and maintenance."""
+        with self._lock:
+            self._requests.clear()

+## 1. Scope and accepted architecture feedback

+2. Make external/stateful boundaries independently testable.
+3. Use a rate limiter to prevent abuse rather than documenting the risk only.
+4. Keep the design proportional to the workload and explicitly document when a
+   process-local mechanism stops being safe.
+5. Run real tests and a real HTTP server smoke test; do not treat compilation as
+   runtime verification.
+
+The FastAPI/SQLite lessons from prior projects were also applied: synchronous
+SQLite operations live in ordinary `def` endpoints, so FastAPI moves them to its
+thread pool. A blocking sync repository is not called directly from an

+### 2.1 App factory
+

+- `database_url`: each test receives a temporary SQLite file;
+- `RateLimiter`: tests can choose limits and inject a fake monotonic clock;
+- `Settings`: proxy-header trust can be enabled only in a controlled test or a
+  deployment behind a trusted proxy.
+
+`app.main:app` remains a conventional Uvicorn import target.
+
+### 2.2 SQLite repository
+
+`TodoRepository` is the persistence boundary and uses parameterized SQL for all
+values. The table stores:
+
+- `id`: auto-increment integer primary key;
+- `title`: non-empty `TEXT`, guarded both by Pydantic and a database `CHECK`;
+- `completed`: constrained to `0`/`1`;
+- `created_at` and `updated_at`: UTC ISO-8601 strings.
+
+The repository owns one lazily-created connection. A `threading.Lock` protects
+complete database operations because synchronous FastAPI routes may execute in
+different worker threads. The update method intentionally performs read,
+update, and read-back under one lock, avoiding a time-of-check/time-of-use race.
+File-backed SQLite enables WAL, normal synchronous mode, foreign keys, and a
+10-second busy timeout.

+### 2.3 HTTP contract
+
+- `GET /health`

+- `GET /todos/{id}`
+- `PATCH /todos/{id}` → partial update
+- `DELETE /todos/{id}` → `204`

+Blank/oversized titles, invalid pagination, and empty PATCH bodies return `422`.
+Missing resources return `404`.

+### Persistence

+non-blank titles and boolean completion at the database layer. One repository
+connection is guarded by a lock; complete update/read-back operations are
+atomic relative to other repository operations. File-backed databases enable
+WAL and a busy timeout.
+
+### API
+
+- `GET /health`
+- `POST /todos` (`201`)
+- `GET /todos?skip=0&limit=100`
+- `GET /todos/{id}`
+- `PATCH /todos/{id}`
+- `DELETE /todos/{id}` (`204`)
+
+Pydantic rejects blank/oversized titles; empty PATCH and invalid pagination
+return `422`; missing resources return `404`.
+
+### Rate limiter
+
+A thread-safe sliding-window limiter tracks request timestamps per client using
+a monotonic clock. Responses include `X-RateLimit-Limit` and
+`X-RateLimit-Remaining`; rejected requests return `429` and `Retry-After`.
+`X-Forwarded-For` is ignored unless proxy trust is explicitly enabled, avoiding
+header-spoofing bypasses.
