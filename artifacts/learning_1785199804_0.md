+// localStorage persistence — 用 custom hook 包起來，讓 reducer 邏輯保持純淨。
+// 只持久化 todos 列表（不要把 UI 狀態也序列化，使用者切到別的 tab 重新打開時應該乾淨）。
+const STORAGE_KEY = 'react-todo:v1'
+
+export function useTodoPersistence(state, dispatch) {
+  const isFirstRender = useRef(true)
+
+  // 載入
+  useEffect(() => {
+    try {
+      const raw = localStorage.getItem(STORAGE_KEY)
+      if (raw) {
+        const todos = JSON.parse(raw)
+        if (Array.isArray(todos)) {
+          dispatch({ type: 'LOAD_FROM_STORAGE', todos })
+        }
+      }
+    } catch (err) {
+      console.warn('[todo] 載入 localStorage 失敗:', err)
+    }
+  }, [dispatch])
+
+  // 寫入 — 跳過第一次渲染避免覆蓋剛載入的資料
+  useEffect(() => {
+    if (isFirstRender.current) {
+      isFirstRender.current = false
+      return
+    }
+    try {
+      localStorage.setItem(STORAGE_KEY, JSON.stringify(state.todos))
+    } catch (err) {
+      console.warn('[todo] 寫入 localStorage 失敗:', err)
+    }
+  }, [state.todos])

+  useEffect(() => {

+          onKeyDown={onKeyDown}

+        />
+      </li>
+    )
+  }

+  return (

+## 三、本次交付的技術決策
+
+### 1. 狀態管理選 `useReducer` 而非 `useState` / Redux
+
+Todo 的狀態是 **多個相互關聯的子狀態**（todo 列表 + filter + editingId + bulkMode + selectedIds + inputDraft）。用 `useState` 散落各處會引發 cross-field bug：例如「批次刪除時 editingId 應該清空」「filter 切換不影響 selectedIds 統計」這類規則很難表達。
+
+`useReducer` 一次描述所有 transition rule，把 cross-field invariant 集中在 reducer 裡。
+
+### 2. 為什麼不用 Redux / Zustand
+
+這個規模的 state tree 用 `useReducer` 就夠，引入 store 是 over-engineering。**但 reducer 形狀刻意保持可序列化** → 之後想升級成 Redux/Zustand 時只改 dispatch wrapper，business logic 不動。
+
+### 3. localStorage 為什麼用 custom hook 分開
+
+- Reducer 保持純函式（純函式 → 易測試、易 time-travel debug）
+- persistence 副作用獨立在 `useTodoPersistence` hook 裡
+- 用 `isFirstRender` ref 避免「剛載入 → 馬上回寫」的死循環

+### 4. 為什麼只持久化 `todos`，不持久化 UI 狀態
+
+使用者跨分頁切換時，UI 狀態（filter / bulkMode / editingId）應該是「當下頁面的工作狀態」，不是「資料」。持久化 UI 狀態會造成「我前天選的 filter 還在鎖著」這種 bug。
