+from __future__ import annotations

+import asyncio
+import time
+from collections import defaultdict
+from dataclasses import dataclass
+from pathlib import Path
+from threading import Lock
+
+
+@dataclass
+class Bucket:
+    started_at: float
+    count: int = 0
+
+
+class RateLimiter:
+    """Process-local fixed-window limiter; replaceable with Redis at scale."""
+
+    def __init__(self, limit: int, window_seconds: float = 1.0) -> None:
+        if limit < 1:
+            raise ValueError('limit must be positive')
+        self.limit = limit
+        self.window_seconds = window_seconds
+        self._buckets: dict[str, Bucket] = {}
+        self._lock = Lock()
+
+    def allow(self, key: str) -> bool:
+        now = time.monotonic()
+        with self._lock:
+            bucket = self._buckets.get(key)
+            if bucket is None or now - bucket.started_at >= self.window_seconds:
+                self._buckets[key] = Bucket(started_at=now, count=1)
+                return True
+            if bucket.count >= self.limit:
+                return False
+            bucket.count += 1
+            return True

+def _secure_parent(path: Path) -> None:
+    parent = path.parent
+    if not parent.exists():
+        old_umask = os.umask(0o077)
+        try:
+            parent.mkdir(parents=True, mode=0o700)
+        finally:
+            os.umask(old_umask)
+    mode = parent.stat().st_mode & 0o777
+    if mode != 0o700:
+        raise PermissionError(f'database directory must be mode 0700: {parent} is {mode:04o}')
+    parent.chmod(0o700)
+
+
+def prepare_database(path: str) -> None:
+    db_path = Path(path).expanduser().resolve()
+    _secure_parent(db_path)
+    old_umask = os.umask(0o077)
+    try:
+        with sqlite3.connect(db_path) as connection:
+            connection.execute('PRAGMA journal_mode=WAL')
+            connection.execute(SCHEMA)
+    finally:
+        os.umask(old_umask)
+    for sidecar in (db_path, Path(f'{db_path}-wal'), Path(f'{db_path}-shm')):
+        if sidecar.exists():
+            sidecar.chmod(0o600)

+def connect(path: str) -> Iterator[sqlite3.Connection]:
+    connection = sqlite3.connect(path)

+# Todo API
+
+A small FastAPI CRUD service backed by SQLite.
+
+## Run
+
+```bash
+python3 -m uvicorn app.main:app --host 127.0.0.1 --port 8000

+## Endpoints
+
+- `GET /health`
+- `GET /todos`
+- `POST /todos` with `{"title": "..."}`
+- `GET /todos/{id}`

+- `DELETE /todos/{id}`

For multi-instance deployments, replace the limiter implementation with a shared Redis-backed adapter.

+## Implemented architecture
+
+- FastAPI application factory: `app.main:create_app`.
+- Synchronous SQLite access with standard-library `sqlite3` and `def` route handlers. This intentionally follows the proven sync pattern and avoids mixing `async def` with blocking SQLAlchemy/database work.
+- SQLite schema uses parameterized SQL, a `todos` table, integer primary key, bounded title, boolean completion flag, and creation timestamp.
+- CRUD endpoints: health, list, create, get-one, update, delete.
+- Pydantic v2 request models enforce title length 1–255 and structured 422 responses.
+- Process-local fixed-window rate limiter keyed only by `request.client.host` from the ASGI scope. It uses a lock around each bucket mutation and returns `429` plus `Retry-After: 1`.
+- SQLite file hardening: newly-created data parents use umask `0077` and mode `0700`; database and sidecars are reasserted to `0600`. Existing parents with a broader mode fail closed rather than being silently chmodded.
+
