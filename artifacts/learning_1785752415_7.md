+class_name GameRules
+extends RefCounted
+
+static func launch_velocity(origin: Vector2, target: Vector2, power: float) -> Vector2:
+    var delta := target - origin
+    if delta.is_zero_approx():
+        return Vector2.ZERO
+    return delta.normalized() * power
+
+static func hit_score(speed: float) -> int:
+    return 10 + mini(int(maxf(speed, 0.0) / 20.0), 40)
+
+static func consume_ammo(ammo: int) -> int:
+    return maxi(ammo - 1, 0)
