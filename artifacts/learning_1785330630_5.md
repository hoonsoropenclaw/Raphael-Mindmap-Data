+extends Node
+class_name ObjectPool
+## Generic object pool — keeps a free list of pre-instantiated nodes so we never
+## call instantiate()/queue_free() during gameplay (key for bullet spam).
+##
+## Usage:
+##   var pool := ObjectPool.new()
+##   pool.scene = preload("res://scenes/bullet.tscn")
+##   pool.parent = $Bullets
+##   pool.prewarm(20)
+##   add_child(pool)
+##   var b := pool.acquire() as Node2D
+##   ...
+##   pool.release(b)
+##
+## Micro-skill alignment: §2 Object Pooling (data_repo/skills/godot_engine_advanced_development.md).
+
+@export var scene: PackedScene
+@export var parent_path: NodePath
+@export var initial_size: int = 16
+
+var _free: Array[Node] = []
+var _parent: Node = null
+
+
+func _ready() -> void:
+	_parent = get_node_or_null(parent_path) if parent_path != NodePath() else get_parent()
+	if scene == null:
+		push_warning("ObjectPool has no scene assigned — pool will be inert.")
+		return
+	prewarm(initial_size)
+
+
+func prewarm(n: int) -> void:
+	for i in n:
+		var obj := _instantiate_inactive()
+		_free.append(obj)
+
+
+func acquire() -> Node:
+	if _free.is_empty():
+		# Pool exhausted — grow on demand (better than crashing, but log it).
+		var obj := _instantiate_inactive()
+		return obj
+	var obj: Node = _free.pop_back()
+	if is_instance_valid(obj):
+		obj.set_process(true)
+		obj.set_physics_process(true)
+		obj.set_process_input(true)
+		obj.visible = true
+		# Reset common Node2D state.
+		if obj is Node2D:
+			(obj as Node2D).rotation = 0.0
+			(obj as Node2D).scale = Vector2.ONE
+	return obj
+
+
+func release(obj: Node) -> void:
+	if obj == null or not is_instance_valid(obj):
+		return
+	if obj.get_parent() != null:
+		obj.get_parent().remove_child(obj)
+	# Deactivate so it costs nothing in the physics tree.
+	obj.set_process(false)
+	obj.set_physics_process(false)
+	obj.set_process_input(false)
