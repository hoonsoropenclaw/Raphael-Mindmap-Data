+@dataclass

+    weather_api_url: str = "https://wttr.in"
+    poll_timeout: int = 30
+    allowed_users: frozenset[int] | None = None
+
+    @classmethod
+    def from_env(cls) -> "Config":
+        token = os.environ.get("TELEGRAM_BOT_TOKEN", "").strip()
+        if not token:
+            raise RuntimeError("TELEGRAM_BOT_TOKEN is required")
+        raw_users = os.environ.get("TELEGRAM_ALLOWED_USERS", "").strip()
+        users: frozenset[int] | None = None
+        if raw_users:
+            try:
+                users = frozenset(int(value.strip()) for value in raw_users.split(",") if value.strip())
+            except ValueError as exc:
+                raise RuntimeError("TELEGRAM_ALLOWED_USERS must contain numeric IDs") from exc
+        try:
+            poll_timeout = max(1, min(50, int(os.environ.get("POLL_TIMEOUT", "30"))))
+        except ValueError as exc:
+            raise RuntimeError("POLL_TIMEOUT must be an integer") from exc
+        return cls(
+            token=token,
+            weather_api_url=os.environ.get("WEATHER_API_URL", "https://wttr.in").rstrip("/"),
+            poll_timeout=poll_timeout,
+            allowed_users=users,
+        )

+# Telegram Weather Bot
+
+## 啟動
+
+```bash
+export TELEGRAM_BOT_TOKEN='從 BotFather 取得的 token'
+# 可選：限制可使用的 Telegram user ID
+export TELEGRAM_ALLOWED_USERS='123456789,987654321'
+./run.sh
+```
+
+## 使用
+
+- `/start` 或 `/help`
+- `/weather Taipei`
+- 直接傳送 `Tokyo`
+
+## 架構與防護
+
+- Python standard library only：`urllib`、`json`、`signal`、`dataclass`
+- Telegram Bot API long polling，使用 `offset = update_id + 1` 避免重複更新
+- wttr.in `format=j1` 查詢天氣
+- Telegram user allowlist（設定 `TELEGRAM_ALLOWED_USERS` 後啟用）
+- 每位使用者 3 秒冷卻，避免濫用
+- 城市輸入清理與長度限制
+- 所有外部 HTTP 呼叫都有 timeout；輪詢失敗會延遲重試
+- SIGINT/SIGTERM 可優雅停止
