+class CalculatorError(ValueError):
+    """所有計算器錯誤的基底類別。測試可用 except CalculatorError 統一處理。"""
+
+
+class InvalidNumberError(CalculatorError):
+    """傳入非數字 (例如字串、None) 時拋出。"""
+
+
+class DivisionByZeroError(CalculatorError):
+    """除數為 0 時拋出,與一般 ZeroDivisionError 區分以便測試斷言。"""
+
+
+def _coerce_number(value, name: str) -> float:
+    """把 value 強制轉成 float,失敗時拋出 InvalidNumberError。
+
+    使用 bool 的 isinstance 判斷會把 True/False 視為 int 1/0,這裡
+    明確排除以避免「測試計算沒過但其實傳了布林」這種 bug 被掩蓋。
+    """
+    if isinstance(value, bool):
+        raise InvalidNumberError(f"{name} 不可為布林值")
+    if value is None:
+        raise InvalidNumberError(f"{name} 不可為 None")
+    if isinstance(value, (int, float)):
+        return float(value)
+    raise InvalidNumberError(f"{name} 必須是數字,得到 {type(value).__name__}")
+
+
+def add(a, b) -> float:
+    return _coerce_number(a, "a") + _coerce_number(b, "b")
+
+
+def subtract(a, b) -> float:
+    return _coerce_number(a, "a") - _coerce_number(b, "b")
+
+
+def multiply(a, b) -> float:
+    return _coerce_number(a, "a") * _coerce_number(b, "b")
