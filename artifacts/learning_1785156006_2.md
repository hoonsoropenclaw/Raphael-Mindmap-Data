+from fastapi import APIRouter, Depends, HTTPException, Request, Response, status
+from sqlalchemy import select
+from sqlalchemy.ext.asyncio import AsyncSession
+from slowapi import Limiter
+
+from .database import get_session
+from .models import Todo
+from .rate_limit import limiter
+from .schemas import TodoCreate, TodoRead, TodoUpdate
+
+router = APIRouter(prefix="/todos", tags=["todos"])
+
+
+@router.get("", response_model=list[TodoRead])
+@limiter.limit("30/minute")
+async def list_todos(request: Request, session: AsyncSession = Depends(get_session)):
+    result = await session.execute(select(Todo).order_by(Todo.id))
+    return list(result.scalars().all())
+
+
+@router.post("", response_model=TodoRead, status_code=status.HTTP_201_CREATED)
+@limiter.limit("20/minute")
+async def create_todo(request: Request, payload: TodoCreate, session: AsyncSession = Depends(get_session)):
+    todo = Todo(title=payload.title, description=payload.description)
+    session.add(todo)
+    await session.commit()
+    await session.refresh(todo)
+    return todo
+
+
+@router.get("/{todo_id}", response_model=TodoRead)
+@limiter.limit("30/minute")
+async def get_todo(request: Request, todo_id: int, session: AsyncSession = Depends(get_session)):
+    todo = await session.get(Todo, todo_id)
+    if todo is None:
+        raise HTTPException(status_code=404, detail="Todo not found")
+    return todo
+
+
+@router.patch("/{todo_id}", response_model=TodoRead)
+@limiter.limit("20/minute")
+async def update_todo(request: Request, todo_id: int, payload: TodoUpdate, session: AsyncSession = Depends(get_session)):
+    todo = await session.get(Todo, todo_id)
+    if todo is None:
+        raise HTTPException(status_code=404, detail="Todo not found")
+    for field, value in payload.model_dump(exclude_unset=True).items():
+        setattr(todo, field, value)
+    await session.commit()
+    await session.refresh(todo)
+    return todo
+
+
+@router.delete("/{todo_id}", status_code=status.HTTP_204_NO_CONTENT)
+@limiter.limit("20/minute")
+async def delete_todo(request: Request, todo_id: int, session: AsyncSession = Depends(get_session)):
+    todo = await session.get(Todo, todo_id)
+    if todo is None:
+        raise HTTPException(status_code=404, detail="Todo not found")
+    await session.delete(todo)
+    await session.commit()
+    return Response(status_code=status.HTTP_204_NO_CONTENT)

+# Implementation Review
+
+## Architecture
+
+The project is intentionally small and separated into five concerns:
+
+1. `app/main.py` owns FastAPI application startup/shutdown, health checks, and rate-limit middleware.
+2. `app/database.py` owns the async SQLAlchemy engine, session factory, schema initialization, and dependency injection.
+3. `app/models.py` defines the SQLite-backed `Todo` table.
+4. `app/schemas.py` defines strict request validation and response serialization with Pydantic.
+5. `app/routes.py` implements CRUD endpoints and uses async SQLAlchemy operations throughout.
+
+## Key Decisions
+
+- SQLite was selected for zero-configuration local persistence.
+- `sqlite+aiosqlite` and `async_sessionmaker(expire_on_commit=False)` prevent blocking the event loop and avoid post-commit expiration surprises.
+- Every write commits and explicitly refreshes the entity before returning it, so response data is current.
+- `slowapi` provides IP-based rate limiting through FastAPI middleware. Limits are intentionally stricter on mutations.
+- Tests use an in-memory SQLite database with `StaticPool`, ensuring all async sessions share one connection and the test schema/data remain visible.
+- FastAPI dependency overrides isolate tests from the production database file.
+
+## Validation Coverage
+
+- Full create/list/update/delete lifecycle.
+- 404 behavior for missing resources.
+- Pydantic validation for empty titles.
+- Rate-limit enforcement and expected HTTP 429 response.
+
+## Operational Notes
+
+The production default database is `todo.db` in the project root. For deployment, configure a durable volume or replace the database URL with a managed database. The default rate limiter is in-memory, so a multi-worker or multi-instance deployment should use a shared storage backend such as Redis.
