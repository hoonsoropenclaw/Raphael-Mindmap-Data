+    """建立連線，啟用必要的 PRAGMA。"""

+    conn.execute("PRAGMA journal_mode = WAL")
+    return conn

+    """交易 context manager。自動 commit / rollback。"""

+        conn.rollback()

+
+設計選擇：
+- 用 in-memory dict（單機部署適用），key = (client_ip, route_signature)
+- sliding window：每次請求計算 [now - window, now] 區間內的請求數
+- 超過 limit 回 429 + Retry-After header
+- 自動清理過期 entry 避免記憶體無界增長
+"""
+from __future__ import annotations
+
+import time
+from collections import deque
+from threading import Lock
+from typing import Deque, Dict, Tuple
+
+
+class SlidingWindowRateLimiter:
+    """Thread-safe sliding window rate limiter。
+
+    演算法：每個 (key) 維護一個 timestamp deque。
+    每次 request：先 pop 所有 < now - window 的舊 timestamp，再看 len(deque) 是否 < limit。
+    通過則 append 當前 timestamp；不通過則拒絕。
+    """
+

+        if window_seconds <= 0:
+            raise ValueError("window_seconds 必須 > 0")
+        self.max_requests = max_requests
+        self.window_seconds = window_seconds
+        self._buckets: Dict[str, Deque[float]] = {}
+        self._lock = Lock()
+
+    def is_allowed(self, key: str, now: float | None = None) -> Tuple[bool, int]:
+        """檢查是否允許請求。
+
+        Returns:
+            (allowed, retry_after_seconds)
+            - allowed = True: 允許
+            - allowed = False: 拒絕，retry_after_seconds 為建議等待秒數
+        """
+        if now is None:
+            now = time.monotonic()

+
+        with self._lock:
+            bucket = self._buckets.get(key)
+            if bucket is None:

+# Todo API
+
+簡單的待辦事項 REST API。FastAPI + SQLite + 自製 sliding window rate limiter + pytest 測試。
+
+## 快速開始
+
+```bash
+# 1. 安裝依賴（系統已有就不必）
+pip install -r requirements.txt
+
+# 2. 啟動 server
+uvicorn app.main:app --reload --host 0.0.0.0 --port 8000
+
+# 3. 開瀏覽器看 OpenAPI docs
+#    http://localhost:8000/docs
+```
+
+## 環境變數
+
+| 變數 | 預設 | 說明 |
+|------|------|------|
+| `TODO_DB_PATH` | `/tmp/todo_api.db` | SQLite 檔案位置 |
+| `TODO_RATE_LIMIT` | `60` | 每 window 秒最多 N 個請求 |
+| `TODO_RATE_WINDOW` | `60` | rate limit 滑動視窗（秒） |
+
+## API Endpoints
+
+| Method | Path | 說明 |
+|--------|------|------|
+| GET | `/health` | 健康檢查（不限流）|
+| POST | `/todos` | 建立 todo |
+| GET | `/todos` | 列出 todo（支援 `?completed=true&limit=20&offset=0`）|
+| GET | `/todos/{id}` | 取得單一 todo |
+| PATCH | `/todos/{id}` | 部分更新 todo |
+| DELETE | `/todos/{id}` | 刪除 todo |
+
+## 跑測試
+
+```bash
+pytest                          # 跑全部
+pytest --cov=app                # 跑全部 + 覆蓋率
+pytest tests/test_api.py -v     # 只跑 API 測試
+```
+
+## 架構
+
+```
+todo_api/
+├── app/
+│   ├── __init__.py
+│   ├── main.py           # FastAPI app factory + endpoints
+│   ├── database.py       # SQLite connection / schema
+│   ├── repository.py     # CRUD 操作（純資料層）

+# Todo API 實作思路（學習記錄）
+
+> 給 Senior Architect review + 給未來的自己回憶設計取捨用。
+
+## 1. 任務範圍
+
+FastAPI + SQLite + rate limit + 單元測試的最小可執行待辦事項 API。
+
+## 2. 架構分層
+
+```
+HTTP request
+   ↓
+FastAPI endpoint (main.py)        ← 處理 HTTP/驗證/限流
+   ↓
+Repository (repository.py)        ← 純資料操作（無 HTTP 概念）
+   ↓
+SQLite (database.py)              ← 連線管理 + schema
+```
+
+**為什麼抽 repository 層？** 兩個理由：
+1. 測試可以直接測資料層、不用每次走 HTTP
+2. 想換 SQLAlchemy / 別的 ORM 時只動 repository.py
+
+**為什麼不用 SQLAlchemy？** 題目說「簡單」，sqlite3 標準庫就夠了。多 ORM = 多依賴、多複雜度、多 bug surface。
+
+## 3. 關鍵設計決策
+
+### 3.1 Rate limit：sliding window
+
+選 sliding window 而非 fixed window 或 token bucket：
+- **fixed window**：在 window 邊界會 burst（59 秒打 60 個 + 第 60 秒再打 60 個 = 1 秒內 120 個）
+- **token bucket**：需要設定 refill rate,對簡單 API overkill
+- **sliding window**：每個請求算「過去 60 秒內有幾次」,最公平,程式碼也簡單
+
+實作細節：
+- 用 deque 存每個 key 的 timestamp
+- 每次請求先 pop 過期的 (< now - window)
+- 通過則 append,不通過算 retry_after = 最早一次 timestamp + window - now
+- **key 設計**：`(client_ip, path_prefix)` 而非 `client_ip` — 避免 /todos/* 跟 /health 共享額度
+
+### 3.2 連線管理：transaction() context manager
+
+**踩到的坑**（值得記下）：最初用 `with _conn(request) as conn:` 包 INSERT,結果 INSERT 沒 commit 就 conn.close() = **資料庫沒寫入**但 API 回 201 + lastrowid。
+
+SQLite 的語意：
+- 沒顯式 commit → close 時觸發 implicit rollback
+- 必須顯式 commit 或包在 transaction context 內
+
+修法：寫入操作統一用 `transaction(db_path)` context manager,讀取才用單純 get_connection + close。
+
+### 3.3 App factory + lifespan 互動
+
+**另一個踩到的坑**：
+- `create_app(db_path=temp_db_path)` 設定 `app.state.db_path = temp_db_path`
+- 但 `TestClient(app) as c` 進入時觸發 lifespan,**lifespan 內**又跑 `app.state.db_path = _get_db_path()`（從 env 讀 = 預設 /tmp/todo_api.db）
+- 結果測試中所有 endpoint 都讀寫 /tmp/todo_api.db,互相污染
+
+修法：lifespan 加 idempotent 檢查 — `if not getattr(app.state, "db_path", None):` 才初始化。
+
+**為什麼是這個設計？** 給兩種用法：
+- **uvicorn 直接啟動**：`app = create_app()` → lifespan 從 env 讀 db_path
+- **測試注入**：`create_app(db_path="/tmp/xxx.db")` → 跳過 env,直接用傳入的
+
+### 3.4 環境變數驅動設定
+
+`TODO_DB_PATH`、`TODO_RATE_LIMIT`、`TODO_RATE_WINDOW` 都從 env 讀,不寫死在程式裡。理由：
+- production 可以不改程式碼就調設定
+- 測試可以用 fixture 暫時改設定（low_rate_limit_client 就是）
+
+### 3.5 PATCH 語意（部分更新）
+
+`TodoUpdate` 所有欄位 optional,實作：
+1. `data.model_dump(exclude_unset=True)` 只拿使用者真的有送的欄位
+2. 沒送任何欄位 = noop,直接回現況（不報錯,符合 PATCH 慣例）
+3. SQL 動態組 `SET col1 = ?, col2 = ?` 然後追加 `, updated_at = datetime('now')`
