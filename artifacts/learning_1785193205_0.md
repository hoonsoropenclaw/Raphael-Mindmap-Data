+"""SQLite database layer.
+
+設計重點：
+- 使用 sqlite3 標準庫（不額外依賴 SQLAlchemy，符合「簡單」需求）
+- connection 是 thread-safe 的 SQLite 連線，使用 check_same_thread=False
+- 每個 request 透過 FastAPI dependency 拿到自己的 connection
+- 用 context manager 確保連線正確關閉
+- Schema migration 用 IF NOT EXISTS，啟動時自動建表
+"""
+from __future__ import annotations
+
+import sqlite3
+from contextlib import contextmanager
+from pathlib import Path
+from typing import Iterator
+
+DEFAULT_DB_PATH = "todos.db"
+
+
+def _connect(db_path: str) -> sqlite3.Connection:
+    """建立一個 SQLite 連線，設定 Row factory 讓結果可以像 dict 一樣取值。"""
+    conn = sqlite3.connect(db_path, check_same_thread=False)
+    conn.row_factory = sqlite3.Row
+    conn.execute("PRAGMA foreign_keys = ON")
+    return conn
+
+
+SCHEMA = """
+CREATE TABLE IF NOT EXISTS todos (
+    id         INTEGER PRIMARY KEY AUTOINCREMENT,
+    title      TEXT    NOT NULL,
+    completed  INTEGER NOT NULL DEFAULT 0,
+    created_at TEXT    NOT NULL DEFAULT (datetime('now'))
+);
+"""
+
+
+def init_db(db_path: str = DEFAULT_DB_PATH) -> None:
+    """初始化資料庫，建立 todos 表格。"""
+    Path(db_path).parent.mkdir(parents=True, exist_ok=True)
+    with _connect(db_path) as conn:
+        conn.executescript(SCHEMA)
+        conn.commit()
+
+
+@contextmanager
+def get_connection(db_path: str = DEFAULT_DB_PATH) -> Iterator[sqlite3.Connection]:
+    """取得一個 SQLite connection，離開 with 時自動 commit/close。"""
+    conn = _connect(db_path)
+    try:
+        yield conn
+        conn.commit()
+    except Exception:
+        conn.rollback()
+        raise
+    finally:
+        conn.close()

+### 坑 1:default argument 在 import time 就被 freeze
+**問題**:`def init_db(db_path: str = DEFAULT_DB_PATH)` 看起來很自然,但 `DEFAULT_DB_PATH` 在 module 載入時就 capture 了。測試用 `monkeypatch.setattr("app.db.DEFAULT_DB_PATH", ...)` 改了 module 屬性,但函式內讀到的還是舊值。
+
+**解法**:把 default 改成 `None`,在函式內動態讀 module-level 值:
+```python
+def init_db(db_path: str | None = None) -> None:

+        db_path = DEFAULT_DB_PATH
+    ...
+```
