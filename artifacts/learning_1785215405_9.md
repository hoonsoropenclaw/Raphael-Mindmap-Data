+"""FastAPI application factory.
+
+Why a factory instead of a module-level `app = FastAPI()`:
+- Tests construct their own app with an in-memory SQLite DB.
+- Production can pass different settings (CORS, rate-limit backend, etc.)
+  without polluting import-time globals.
+"""
+from __future__ import annotations
+
+from contextlib import asynccontextmanager
+
+from fastapi import FastAPI, Request
+from fastapi.responses import JSONResponse
+from slowapi.errors import RateLimitExceeded
+
+from app.database import init_db
+from app.rate_limit import limiter
+from app.routers import todos
+
+
+@asynccontextmanager
+async def lifespan(app: FastAPI):
+    """Create tables on startup. Idempotent."""
+    init_db()
+    yield
+
+
+def create_app() -> FastAPI:
+    app = FastAPI(
+        title="Todo API",
+        version="1.0.0",
+        description="Simple Todo API with SQLite persistence and rate limiting.",
+        lifespan=lifespan,
+    )
+
+    # slowapi requires the limiter to live on app.state so the
+    # `@limiter.limit(...)` decorators can find it via request.app.state.
+    app.state.limiter = limiter
+    app.add_exception_handler(
+        RateLimitExceeded,
+        _rate_limit_handler,
+    )
+
+    app.include_router(todos.router, prefix="/api/v1")
+
+    @app.get("/health", tags=["meta"])
+    def health() -> dict[str, str]:
+        return {"status": "ok"}
+
+    return app
+
+
+async def _rate_limit_handler(request: Request, exc: RateLimitExceeded):
+    """Return a structured JSON 429 instead of slowapi's default HTML."""
+    return JSONResponse(
+        status_code=429,
+        content={
+            "error": "rate_limit_exceeded",
+            "detail": f"Rate limit exceeded: {exc.detail}",
+        },
+    )
+
+
+# Module-level instance for `uvicorn app.main:app` workflow.
+app = create_app()
