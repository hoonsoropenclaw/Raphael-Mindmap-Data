+"""
+verify_weather.py — 不依賴 Telegram library,純驗證 weather.py 邏輯。
+
+用 stdlib 的 urllib 抓 wttr.in(零憑證)直接驗證整個 query → format 流程。
+可以確認:
+- JSON 解析沒壞
+- 統一 schema 沒爆
+- format_weather() 沒有 None 拋 NoneType
+
+要驗 OWM 走法:OWM_API_KEY=xxx python3 verify_weather.py owm
+"""
+from __future__ import annotations
+
+import os
+import sys
+import time
+
+import weather
+
+
+def verify_wttr(query: str) -> None:
+    print(f"[verify] wttr query={query!r} ...")
+    t0 = time.monotonic()
+    info = weather.query(query, provider="wttr", owm_api_key=None, timeout=10.0)
+    dt = time.monotonic() - t0
+    text = weather.format_weather(info)
+    print(f"[verify] OK in {dt:.2f}s")
+    print("---- formatted output ----")
+    print(text)
+    print("--------------------------")
+    # sanity checks
+    assert info.temp_c != 0 or True, "temp 可能是 0,不該硬 assert"  # noop
+    assert isinstance(info.humidity, int)
+    assert info.provider == "wttr.in"
+    print("[verify] wttr: sanity checks passed")
+
+
+def verify_owm(query: str, api_key: str) -> None:
+    print(f"[verify] owm  query={query!r} ...")
+    info = weather.query(query, provider="openweathermap", owm_api_key=api_key, timeout=10.0)
+    text = weather.format_weather(info)
+    print(f"[verify] OK")
+    print("---- formatted output ----")
+    print(text)
+    print("--------------------------")
+    assert info.provider == "openweathermap"
+    print("[verify] owm:  sanity checks passed")
+
+
+def main() -> None:
+    if len(sys.argv) > 1 and sys.argv[1] == "owm":
+        key = os.environ.get("OWM_API_KEY", "").strip()

+## 功能
