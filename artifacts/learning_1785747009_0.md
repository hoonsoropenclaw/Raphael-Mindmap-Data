+tests/pixel_diff.py
+用 Pillow 對截圖與 baseline 做像素級比對,並產出差異圖。

+def sha256_of(path):
+    h = hashlib.sha256()
+    with open(path, 'rb') as f:
+        for chunk in iter(lambda: f.read(65536), b''):
+            h.update(chunk)
+    return h.hexdigest()

+def compare_one(baseline_path, current_path, diff_path, threshold):
+    """比對一張截圖,回傳 (regression, diff_ratio, total_pixels, diff_pixels)"""
+    if not os.path.exists(baseline_path):
+        return False, 0.0, 0, 0  # 沒 baseline,不算 regression(由 --update-baseline 處理)
+
+    a = Image.open(baseline_path).convert('RGBA')

+    if a.size != b.size:

+        b = b.resize(a.size)
+
+    diff = ImageChops.difference(a, b)
+    bbox = diff.getbbox()
+    if not bbox:
+        return False, 0.0, a.size[0] * a.size[1], 0
