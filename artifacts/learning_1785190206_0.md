+import json

+import time
+from dataclasses import dataclass, field
+from datetime import datetime, timezone

+from typing import Callable, Optional
+
+
+@dataclass
+class AlertEvent:
+    timestamp: str               # ISO8601 UTC, e.g. "20260728T061500Z"
+    check_name: str              # "cpu", "ram", ...
+    level: str                   # "warn" | "crit" | "error"
+    message: str
+    detail: dict = field(default_factory=dict)
+    value: Optional[float] = None
+
+
+# Level 優先順序 (給 sort 用)
+LEVEL_ORDER = {"ok": 0, "warn": 1, "crit": 2, "error": 3, "skip": 0}
+
+
+# ───────────────────────── Channels ─────────────────────────
+def stderr_channel(evt: AlertEvent) -> bool:
+    """永遠 dispatch 到 stderr, 加顏色標記 (簡單 ASCII)"""
+    icon = {"warn": "⚠️ ", "crit": "🔥", "error": "💥"}.get(evt.level, "•")
+    line = (f"{icon} [{evt.level.upper():5}] {evt.check_name}: {evt.message}"
+            f"  (value={evt.value}, ts={evt.timestamp})")
+    sys.stderr.write(line + "\n")
+    sys.stderr.flush()
+    return True
+

+                    "message": evt.message,
+                    "value": evt.value,
+                    "detail": evt.detail,
+                }, ensure_ascii=False) + "\n")

+2. **多層級閾值** — warn (黃) vs crit (紅), 不只 on/off
+3. **可設定** — 不能寫死, 使用者要能改
+4. **可重用** — 路由/告警邏輯拆出來, 不是「一次性 script」
+5. **可測試** — pytest 驗證, 不能只是「能跑」
+
+## 二、為什麼做 6 個檢查器 (不是 5 也不是 10)
+
+| 檢查器      | 必要性 | 為什麼不做其他 X |
+|------------|--------|------------------|
+| cpu        | ✅ 必要 | 不做 per-CPU (太細, load avg 已夠) |
+| ram        | ✅ 必要 | 不做 per-process (oom-killer 會留下 dmesg, 從 journal 抓) |
+| disk       | ✅ 必要 | 不做 inode (那是 ext4 特定; btrfs / zfs 沒這問題) |
+| journal    | ✅ 必要 | 不做 dmesg 直接讀 (systemd journal 已聚合) |
+| services   | ✅ 必要 | 不做 zombie process (v1 已有; 我併入 services 也行但降低關注度) |
+| thermal    | ✅ 必要 (N100) | 不做 fan speed / voltage (需 IPMI / 特定 hw, 不通用) |
+| ~~network~~ | ❌ 不做 | ifconfig 數字變化太快, 抓不住「真的出問題」的瞬間; netstat/ss 留給 netalert 之類專門工具 |
+| ~~users~~   | ❌ 不做 | `who` 對單機監控意義不大 |
+| ~~docker~~  | ❌ 不做 | 不該綁特定 runtime |
+
+## 三、為什麼拆 alert_router.py (獨立檔)
+
+3 個原因:
+
+1. **可重用**: 任何 Python 程式都可能想發告警, 不該被迫 import sys_ops_health 整套
+2. **可獨立測試**: AlertRouter 跟 subprocess / 系統資源完全無關, 純函數測試 5ms 跑完
+3. **可獨立擴充**: 未來加 Telegram / webhook / email channel 只改這個檔, 主程式 0 改動
+
+**替代方案**:
+- ❌ 寫成 sys_ops_health.py 內部 class — 不可重用, 測試要啟動 subprocess
+- ❌ 寫成 3 個獨立 tool (slack_alert.py, log_alert.py, stderr_alert.py) — 使用者要自己串接
+
+## 四、為什麼 stdlib-only (不裝 psutil / requests / pyyaml)
+
+**stdlib-only 的 4 個理由**:
+
+1. **N100 cron 環境**: `pip install -r requirements.txt` 在 cron job 失敗率 > 5%, 我們已經看過太多次
+2. **零升級風險**: psutil API 在 5.x → 6.x 改過, 2023 年後從 df 改回 os.statvfs; 鎖版本不是解
+3. **可審查性**: 一個 Python script 沒有 requirements.txt, 審計員一眼看完
+4. **教練價值**: 證明 stdlib 也能寫出 production-grade 工具, 後續任務可參考
+
+**損失**:
+- YAML 解析只支援 2 層 + numeric/bool (沒 list, 沒 nested, 沒 multiline)
+- 但對「閾值設定」這種 flat config 完全夠用
+
+## 五、為什麼需要 Throttle (避免 Alert Storm)
