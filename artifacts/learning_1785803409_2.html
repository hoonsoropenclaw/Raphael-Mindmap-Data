Design tokens

The single source of truth — every component reads from these.

/* Glassmorphism core — from frontend_ui_system §1.1 */
:root {
  --glass:        rgba(16, 43, 36, 0.46);
  --glass-strong: rgba(12, 36, 30, 0.78);
  --glass-soft:   rgba(20, 50, 42, 0.32);
  --border:       rgba(94, 234, 212, 0.18);
  --bd:           blur(18px) saturate(140%);
}

/* Surface */
.glass        { background: var(--glass);        backdrop-filter: var(--bd); }
.glass-strong { background: var(--glass-strong); backdrop-filter: blur(28px) saturate(160%); }
.glass-soft   { background: var(--glass-soft);   backdrop-filter: blur(10px) saturate(120%); }
