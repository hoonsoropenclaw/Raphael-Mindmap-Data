+    db: Session,

+    """Return todos, optionally filtered by completion status.

+"""Rate limiting setup using slowapi.
+
+slowapi is the FastAPI-native port of Flask-Limiter. We key on the
+client IP (X-Forwarded-For aware) and store counters in-memory; that
+is sufficient for a single-process demo. For multi-process / multi-host
+production you would swap the storage to Redis — see
+``Limiter(storage_uri="redis://...")`` in slowapi docs.
+
+Why slowapi over a hand-rolled middleware:
+  * Battle-tested token-bucket semantics
+  * Plays nicely with FastAPI's dependency injection
+  * Standard `Retry-After` headers on 429
+  * Test override via `app.state.limiter` lets us disable limits
+"""
+from __future__ import annotations
+
+from fastapi import Request
+from fastapi.responses import JSONResponse
+from slowapi import Limiter
+from slowapi.errors import RateLimitExceeded
+from slowapi.util import get_remote_address
+
+from app.core.config import settings
+
+
+def _client_key(request: Request) -> str:
+    """Use X-Forwarded-For first hop if present, else the raw remote.
+
+    FastAPI's `Request.client.host` already reads through `uvicorn`'s
+    forwarded-header handling, but `get_remote_address` is hard-coded
+    to use `request.client.host`, so we provide a thin wrapper that
+    also accepts X-Forwarded-For — useful when running behind nginx.
+    """
+    xff = request.headers.get("x-forwarded-for")
+    if xff:
+        # first IP in the comma-separated list is the originating client
+        return xff.split(",")[0].strip()
+    return get_remote_address(request) or "anonymous"
+
+
+# Single global limiter, attached to app.state.limiter in `create_app`.
+limiter = Limiter(
+    key_func=_client_key,
+    default_limits=[settings.rate_limit_default],
+)
+
+
+async def rate_limit_exceeded_handler(
+    request: Request, exc: RateLimitExceeded
+) -> JSONResponse:
+    """Return a structured 429 body (slowapi default is text/plain)."""
+    response = JSONResponse(
+        status_code=429,
+        content={
+            "detail": "Rate limit exceeded",
+            "limit": str(exc.detail),
+        },
+    )
+    # Mirror the Retry-After header so well-behaved clients back off.
+    if hasattr(exc, "retry_after") and exc.retry_after is not None:
+        response.headers["Retry-After"] = str(int(exc.retry_after))
+    return response
