+// Filter a single task against current filter + search.
+export function matchesFilter(task, { filter, search }) {
+  // Filter: all → show everything; active → !completed; completed → completed.
+  if (filter === 'active' && task.completed) return false;
+  if (filter === 'completed' && !task.completed) return false;
+
+  // Search: case-insensitive across title, description, tags.
+  if (search && search.trim().length > 0) {
+    const q = search.trim().toLowerCase();
+    const haystack = [
+      task.title || '',
+      task.description || '',
+      ...(task.tags || []).map((t) => (typeof t === 'string' ? t : t.label)),
+    ].join(' ').toLowerCase();
+    if (!haystack.includes(q)) return false;
+  }
+  return true;
+}

+import { STORAGE_KEY } from '../utils/constants.js';

+// Persist a slice of state to localStorage. Skips the hydration write so we
+// don't immediately overwrite stored data with the empty default.
+// Returns nothing — the component doesn't need read access (state comes from reducer).
+export function useLocalStoragePersist(state) {
+  const hydratedRef = useRef(false);
+
+  useEffect(() => {
+    if (!state._hydrated) return; // skip the first render — default state
+    hydratedRef.current = true;
+  }, [state._hydrated]);
+
+  useEffect(() => {
+    if (!hydratedRef.current) return;
+    try {
+      // Don't persist transient flags
+      const { _hydrated, ...persistable } = state;
+      window.localStorage.setItem(STORAGE_KEY, JSON.stringify(persistable));
+    } catch (err) {
+      // Quota exceeded or storage disabled — fail silent rather than throwing
+      // inside an effect (React would warn about uncaught effect errors).
+      // The app remains fully usable; only reload-survival degrades.
+      console.warn('[kanban] failed to persist state:', err.message);
+    }
+  }, [state]);

+    // Sanity-check: must be an object with `tasks` map and `columnOrder` keys.
+    if (!parsed || typeof parsed !== 'object' || !parsed.tasks || !parsed.columnOrder) {
+      return null;
+    }

+  }

+export function KanbanProvider({ children }) {
+  const [state, dispatch] = useReducer(kanbanReducer, initialState);
+
+  // One-shot hydration on mount — runs before user actions, so initial paint
+  // already shows persisted columns. We guard against double-dispatch.
+  useEffect(() => {
+    const stored = readStoredState();
+    if (stored) {
+      dispatch(loadState(stored));
+    } else {
+      // No stored data — mark hydrated so persistence starts firing.
+      dispatch(loadState({ ...initialState }));
+    }
+  }, []);

+  useLocalStoragePersist(state);
+
+  // Memoize the context value so consumers don't re-render unnecessarily.
+  const value = useMemo(() => ({ state, dispatch }), [state]);
+

+## 2. State 設計決策 — Normalized 結構
+
+### 為什麼不把 task 直接放進 `columns[id].tasks[]` 陣列？
+
+採用 **two-index normal form**：
+
+```js
+state = {
+  tasks:        { [id]: taskObj },      // 1. normalized map
+  columnOrder:  { [status]: id[] },     // 2. 順序索引（per-column）
+  filter, search, _hydrated
+}
+```
+
+理由：
+- **O(1) lookup** — 用 task 物件做 stable identity，React memo 不會誤判重複渲染
+- **沒有選擇更新風險** — 拖曳時只需 splice 兩個 columnOrder 陣列，不必 deep clone
+- **刪除原子化** — 從 map 拿掉 key，從 columnOrder filter 掉 id，兩步獨立
+- **反序列化容易** — localStorage 結構即 JSON，無循環參考、無特殊型別
+
+### Reducer 設計 4 個關鍵原則
+
+1. **純函式**（Pure）— 不在 reducer 內呼叫 localStorage / Date.now()，所有 side effect 走 useEffect；這樣 reducer 可在 worker、test、SSR 重用
+2. **Discriminated union switch** — `default` 拋錯，未實作的 action type 在「執行時」立刻炸出來（而不是悄悄靜默錯誤）
+3. **Out-of-bounds 防護** — `moveTask` 的 `targetIndex` 用 `Math.max/Math.min` 夾住；錯誤 status 字串丟 `ADD_TASK` 直接 `throw`
+4. **Status 變更自動 re-bucket** — `UPDATE_TASK` 偵測到 patch.status 變更時，從舊 column 移除、新 column 尾端追加；同一個 dispatch 完成跨欄搬遷
+
+## 3. Drag-and-Drop — 不裝 react-dnd / dnd-kit
+
+決定自製，因為：
+
+| 考量 | 自製 HTML5 DnD | 第三方 lib |
+|------|---------------|----------|
+| Bundle size | 0 KB (200 行內) | 30-50 KB+ |
+| 學習展示性 | 完整暴露原始事件流程 | 抽象黑盒 |
+| 客製自由度 | 任意插 marker / 動畫 | 受 API 限制 |
+| 維護負擔 | 自己寫，自己 debug | 依賴升級安全 |
+

+```json

+```
