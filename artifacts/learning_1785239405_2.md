+"""In-memory sliding-window rate limiter.
+
+Algorithm: per-key sliding window log.
+For each key we keep a deque of recent request timestamps.
+On each request we drop expired entries (older than window),
+then reject if len(remaining) >= max_requests.
+
+Trade-offs vs token bucket:
+- Exact (no burst smoothing artifacts) — every request in the window counts.
+- O(N) memory per key where N = max_requests (bounded, small).
+- Thread-safe via a single lock (sufficient for SQLite + single-process uvicorn).
+
+Production alternative: redis-backed limiter (not done here — extra dependency).
+"""
+import threading
+import time
+from collections import deque
+from dataclasses import dataclass
+from typing import Deque, Dict
+
+
+@dataclass
+class RateLimitConfig:
+    """Configuration for a single rate limit policy."""
+    max_requests: int          # allowed requests per window
+    window_seconds: float      # window length in seconds
+
+
+class RateLimiter:
+    """Thread-safe sliding-window rate limiter keyed by client identifier."""
+
+    def __init__(self, config: RateLimitConfig) -> None:
+        if config.max_requests <= 0:
+            raise ValueError("max_requests must be positive")
+        if config.window_seconds <= 0:
+            raise ValueError("window_seconds must be positive")
+        self.config = config
+        self._hits: Dict[str, Deque[float]] = {}
+        self._lock = threading.Lock()
+
+    def _now(self) -> float:
+        return time.monotonic()
+
+    def is_allowed(self, key: str) -> tuple[bool, int, float]:
+        """Check & record a hit.
+
+        Returns:
+            (allowed, remaining, retry_after_seconds)
+            - allowed: True if the request may proceed
+            - remaining: how many requests remain in the current window
+            - retry_after_seconds: how long until oldest hit expires (0 if allowed)
+        """
+        now = self._now()
+        cutoff = now - self.config.window_seconds
+        with self._lock:
+            dq = self._hits.get(key)
+            if dq is None:
+                dq = deque()
+                self._hits[key] = dq
+
+            # Drop expired entries
+            while dq and dq[0] <= cutoff:
+                dq.popleft()
+
+            if len(dq) >= self.config.max_requests:
+                # Compute retry_after = time until oldest hit leaves the window
+                retry_after = max(0.0, dq[0] + self.config.window_seconds - now)
+                return False, 0, retry_after
+
+            dq.append(now)
+            remaining = self.config.max_requests - len(dq)
+            return True, remaining, 0.0

+## 特色

+- **Rate limiting**：sliding-window 演算法，thread-safe，per-client-IP
+- **可設定**：`create_app(db_path, rate_max_requests, rate_window_seconds)` 工廠函式方便測試

+## 🧠 實作思路（為什麼這樣設計）
+
+### 1. App factory pattern（不是 top-level singleton）
+
+```python
+def create_app(db_path=..., rate_max_requests=..., rate_window_seconds=...) -> FastAPI
+```
+
+**理由**：測試時需要每個 test 一個全新 app（避免 rate limiter state 殘留、DB state 污染）。工廠函式讓注入設定變 trivial；`uvicorn app.main:app` 仍然能用因為最底下有 `app = create_app()` 給 ASGI loader 用。
+
+### 2. Repository 分離 SQL 與 HTTP
+
+Route handlers 只處理：parse request → 呼叫 repo → 回傳 Pydantic model。
+所有 `INSERT/SELECT/UPDATE/DELETE` 都在 `repository.py`。
+
+**好處**：
+- `test_repository.py` 可以純 unit test（不經 HTTP），跑更快、訊號更純
+- 換 DB（Postgres、DuckDB）只需要改 repository，routes 動都不用動
+- 單元測試可以驗證 SQL 邏輯（filter、pagination、partial update）獨立於 HTTP 框架
+
+### 3. Sliding-window log（不是 token bucket）
+
+每個 client IP 維護一個 `deque[float]`，記錄過去 N 秒內的 timestamp。
+- 允許：len(deque) < max_requests
+- 拒絕：len(deque) >= max_requests，回傳 retry_after = oldest + window - now
+
+**為什麼選 sliding window 而非 token bucket**：
+- 精確：每一個 request 都會計入窗口，沒有 burst smoothing artifact
+- 簡單：~30 行 code（含 thread lock），零外部依賴
+- 易測試：`test_window_slides` 直接 sleep 1 秒驗證窗口滑動
+
+**權衡**：O(max_requests) 記憶體 per key（小型 API OK）。Production 多實例部署要改 Redis-backed（目前用 in-memory dict，跨 process 不共享）。
+
