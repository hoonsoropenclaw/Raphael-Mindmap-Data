🏗️ 三層抓取架構

1

Layer 1 — 列表頁（Selenium 等 JS）

driver.get() 載入 /index.html，WebDriverWait 等 .card[data-id] 出現（SPA 渲染完成）， 抓取 page_source，用 BeautifulSoup 解析 6 張 .card 元素，抽出 id / title / price / category / detail_url。

2

Layer 2 — 詳情頁（Selenium × 6 進詳情）

對每張卡片的 detail_url 再 driver.get()，等 .review 出現（評論 JS 載入完），用 BeautifulSoup 解析 dl/dt/dd 規格表 與 .review 評論清單，計算平均評分。

3

Layer 3 — 聚合層（dataclass + JSON）

把兩層結果彙整進 ScrapeResult dataclass（含 base_url / scraped_at / duration_sec / errors），序列化為巢狀 JSON 輸出到 scraped_products.json。

💻 完整實作

#!/usr/bin/env python3
"""
multi_layer_scraper.py
======================

整合 **Selenium** + **BeautifulSoup** 進行多層次網頁數據抓取與解析。

「多層次」設計：
  Layer 1（列表頁）— 用 Selenium 驅動 headless Chrome，等 JS 渲染後，
                    把整份 page_source 交給 BeautifulSoup 抽出商品卡片清單。
  Layer 2（詳情頁）— 對每張卡片擷取 URL，再驅動 Selenium 進入每個詳情頁，
                    再次用 BeautifulSoup 解析規格 (dl/dt/dd) 與評論。
  Layer 3（聚合層）— 把兩層結果彙整為巢狀 JSON，做基本驗證 / 統計。

設計重點：
  • **Selenium 解 JS 渲染瓶頸**：SPA / 動態內容是 requests 抓不到的，
    Selenium 等於給 BeautifulSoup 一份「完整的 DOM 快照」。
  • **BeautifulSoup 解解析瓶頸**：Selenium 的 find_element 太脆弱，
    CSS selector + BS4 的 find_all 才能穩定處理巢狀 HTML。
  • **顯式等待 (WebDriverWait)**：避免 race condition（JS 還沒跑就抓）。
  • **headless Chrome**：容器 / 伺服器無 GUI 環境必備。
  • **defensive parse**：欄位缺失不 crash，回 None 或空清單。

使用方式：
    python3 multi_layer_scraper.py                   # 抓 mock 站 (預設 http://127.0.0.1:8765)
    python3 multi_layer_scraper.py --url <SITE_URL>  # 自訂目標
    python3 multi_layer_scraper.py --out result.json # 自訂輸出檔
"""

from __future__ import annotations

import argparse
import json
import logging
import re
import sys
import time
from dataclasses import dataclass, field, asdict
from pathlib import Path
from typing import Any
from urllib.parse import urljoin

from bs4 import BeautifulSoup
from selenium import webdriver
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.chrome.service import Service
from selenium.webdriver.common.by import By
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.support.ui import WebDriverWait


# ───────────────────────────────────────────────────────────────────────────
# Logging
# ───────────────────────────────────────────────────────────────────────────
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    datefmt="%H:%M:%S",
)
log = logging.getLogger("multi_layer_scraper")


# ───────────────────────────────────────────────────────────────────────────
# Data model — 用 dataclass 表徵三層結構
# ───────────────────────────────────────────────────────────────────────────
@dataclass
class Review:
    user: str | None = None
    rating: int | None = None
    text: str | None = None


@dataclass
class ProductDetail:
    product_id: str
    title: str | None = None
    price_ntd: int | None = None
    category: str | None = None
    description: str | None = None
    specs: dict[str, str] = field(default_factory=dict)
    reviews: list[Review] = field(default_factory=list)
    avg_rating: float | None = None
    source_url: str | None = None


@dataclass
class ScrapeResult:
    base_url: str
    scraped_at: float
    duration_sec: float
    total_products: int
    total_reviews: int
    products: list[ProductDetail] = field(default_factory=list)
    errors: list[str] = field(default_factory=list)


# ───────────────────────────────────────────────────────────────────────────
# Selenium driver factory
# ───────────────────────────────────────────────────────────────────────────
def build_driver(headless: bool = True) -> webdriver.Chrome:
    """建立 headless Chrome driver。

    容器 / 無 GUI 環境必須：
      • --headless=new（新 headless 模式，行為接近有 GUI）
      • --no-sandbox（容器內以 root 跑需關閉 sandbox）
      • --disable-dev-shm-usage（容器內 /dev/shm 太小會 crash）
      • --disable-gpu（相容性）
    """
    opts = Options()
    if headless:
        opts.add_argument("--headless=new")
    opts.add_argument("--no-sandbox")
    opts.add_argument("--disable-dev-shm-usage")
    opts.add_argument("--disable-gpu")
    opts.add_argument("--window-size=1280,900")
    opts.add_argument(
        "--user-agent=Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 "
        "(KHTML, like Gecko) Chrome/151.0.0.0 Safari/537.36"
    )
    # 抑制 selenium-manager 的多餘 log
    opts.add_argument("--log-level=3")
    service = Service(log_output="/dev/null")
    log.info("啟動 headless Chrome …")
    driver = webdriver.Chrome(options=opts, service=service)
    driver.set_page_load_timeout(30)
    return driver


# ───────────────────────────────────────────────────────────────────────────
# Layer 1 — 列表頁解析（Selenium + BeautifulSoup）
# ───────────────────────────────────────────────────────────────────────────
def fetch_rendered_html(driver: webdriver.Chrome, url: str, wait_css: str | None = None,
                        extra_wait_ms: int = 0) -> str:
    """Selenium 載入頁面、等 JS 渲染完成、回傳完整 page_source。"""
    log.info("GET %s", url)
    driver.get(url)
    if wait_css:
        try:
            WebDriverWait(driver, 15).until(
                EC.presence_of_element_located((By.CSS_SELECTOR, wait_css))
            )
            log.info("  ✔ 已等到 selector: %s", wait_css)
        except Exception as exc:
            log.warning("  ✘ 等待 selector %s 失敗：%s", wait_css, exc)
    if extra_wait_ms > 0:
        time.sleep(extra_wait_ms / 1000)
    return driver.page_source


def parse_price(text: str) -> int | None:
    """從 'NT$ 4,980' 抽出 4980 (int)。"""
    if not text:
        return None
    m = re.search(r"([\d,]+)", text)
    return int(m.group(1).replace(",", "")) if m else None


def parse_product_list(html: str, base_url: str) -> list[dict[str, Any]]:
    """Layer 1：用 BS4 解析商品卡片清單，回傳 [{id, title, price, cat, detail_url} …]。"""
    soup = BeautifulSoup(html, "html.parser")
    cards = soup.select(".card[data-id]")
    log.info("Layer 1 找到 %d 張商品卡", len(cards))

    products: list[dict[str, Any]] = []
    for card in cards:
        pid = card.get("data-id")
        cat = card.get("data-cat")
        stock = card.get("data-stock")
        title_el = card.select_one("h3")
        price_el = card.select_one(".price")
        link_el = card.select_one("a.btn[href]")
        if not (pid and link_el):
            log.warning("  跳過無 id/link 的卡片: %s", title_el.get_text(strip=True) if title_el else "?")
            continue
        products.append({
            "id": pid,
            "title": title_el.get_text(strip=True) if title_el else None,
            "price_ntd": parse_price(price_el.get_text() if price_el else ""),
            "category": cat,
            "stock": stock,
            "detail_url": urljoin(base_url, link_el["href"]),
        })
    return products


# ───────────────────────────────────────────────────────────────────────────
# Layer 2 — 詳情頁解析（Selenium + BeautifulSoup）
# ───────────────────────────────────────────────────────────────────────────
def parse_product_detail(html: str, source_url: str) -> ProductDetail:
    """Layer 2：解析詳情頁 — 標題 / 價格 / 規格 dl / 評論 .review。"""
    soup = BeautifulSoup(html, "html.parser")

    title_el = soup.select_one("h1")
    price_el = soup.select_one(".price")
    pid_el = soup.select_one(".spec dd")  # 第一個 dd 是產品編號

    # 規格表
    specs: dict[str, str] = {}
    spec_pairs = soup.select(".spec dt")
    for dt in spec_pairs:
        key = dt.get_text(strip=True)
        dd = dt.find_next_sibling("dd")
        if dd:
            specs[key] = dd.get_text(strip=True)

    # 評論 — JS 動態載入後才存在
    reviews: list[Review] = []
    for div in soup.select(".review"):
        b = div.select_one("b")
        # 評分從 stars 文字 "★ 5/5" 抽出
        stars_text = div.select_one(".stars")
        rating_match = re.search(r"(\d+)\s*/\s*5", stars_text.get_text() if stars_text else "")
        # 評論文字：拿掉 user / stars 後剩的純文字（用 contents 拆更穩）
        review_text = ""
        for child in div.children:
            if child in (b, stars_text):
                continue
            review_text += child.get_text(" ", strip=True) if hasattr(child, "get_text") else str(child).strip()
        review_text = review_text.lstrip("·\u00b7 ").strip()
        reviews.append(Review(
            user=b.get_text(strip=True) if b else None,
            rating=int(rating_match.group(1)) if rating_match else None,
            text=review_text or None,
        ))

    # 平均評分（從 JS 渲染後的 avgRating span 拿）
    avg_el = soup.select_one("#avgRating")
    avg_match = re.search(r"([\d.]+)", avg_el.get_text() if avg_el else "")
    avg_rating = float(avg_match.group(1)) if avg_match else None

    # 產品編號 fallback：從 specs 拿
    pid = (specs.get("產品編號") or "").strip() or (pid_el.get_text(strip=True) if pid_el else "")

    # 分類
    category = specs.get("分類")

    # 描述：規格區以外的段落
    desc_el = soup.select_one(".wrap > p")
    description = desc_el.get_text(strip=True) if desc_el else None

    return ProductDetail(
        product_id=pid or "UNKNOWN",
        title=title_el.get_text(strip=True) if title_el else None,
        price_ntd=parse_price(price_el.get_text() if price_el else ""),
        category=category,
        description=description,
        specs=specs,
        reviews=reviews,
        avg_rating=avg_rating,
        source_url=source_url,
    )


# ───────────────────────────────────────────────────────────────────────────
# Orchestrator
# ───────────────────────────────────────────────────────────────────────────
def scrape(base_url: str, headless: bool = True) -> ScrapeResult:
    """三層抓取主流程：driver → Layer1 列表 → 逐個 Layer2 詳情 → 聚合。"""
    started = time.time()
    driver = build_driver(headless=headless)
    result = ScrapeResult(
        base_url=base_url,
        scraped_at=started,
        duration_sec=0.0,
        total_products=0,
        total_reviews=0,
    )
    try:
        list_url = urljoin(base_url, "/index.html")
        # Layer 1
        html = fetch_rendered_html(
            driver, list_url,
            wait_css=".card[data-id]",  # 等 SPA 渲染完
            extra_wait_ms=200,
        )
        items = parse_product_list(html, base_url)
        result.total_products = len(items)

        # Layer 2 — 對每張卡片的 detail_url 進詳情頁
        for idx, item in enumerate(items, 1):
            try:
                log.info("Layer 2 [%d/%d] %s %s", idx, len(items), item["id"], item["title"])
                detail_html = fetch_rendered_html(
                    driver, item["detail_url"],
                    wait_css=".review",      # 等評論 JS 載入
                    extra_wait_ms=100,
                )
                # 404 / Error response 偵測
                # python http.server 預設 404 頁 title 含 "Error"、URL 不變
                if ("Not Found" in detail_html[:1000]
                        or "<title>Error" in detail_html
                        or "404" in driver.title):
                    raise RuntimeError(f"詳情頁 404 ({driver.current_url})")
                detail = parse_product_detail(detail_html, item["detail_url"])
                # 補上 list 層的 category 與 stock（詳情頁 spec 有更精確分類）
                if not detail.category:
                    detail.category = item.get("category")
                result.products.append(detail)
                result.total_reviews += len(detail.reviews)
            except Exception as exc:
                msg = f"{item['id']}: {exc}"
                log.error("  ✘ Layer 2 失敗 %s", msg)
                result.errors.append(msg)
    finally:
        try:
            driver.quit()
            log.info("已關閉 Chrome driver")
        except Exception:
            pass
    result.duration_sec = round(time.time() - started, 2)
    return result


# ───────────────────────────────────────────────────────────────────────────
# CLI
# ───────────────────────────────────────────────────────────────────────────
def main() -> int:
    p = argparse.ArgumentParser(description="Selenium + BeautifulSoup 多層次爬蟲")
    p.add_argument("--url", default="http://127.0.0.1:8765",
                   help="目標站 base URL（預設 mock 站）")
    p.add_argument("--out", default="scraped_products.json",
                   help="輸出 JSON 檔（預設 scraped_products.json）")
    p.add_argument("--no-headless", action="store_true",
                   help="關閉 headless（除錯用）")
    args = p.parse_args()

    result = scrape(args.url, headless=not args.no_headless)

    # dataclass → dict
    payload = asdict(result)
    out_path = Path(args.out)
    out_path.write_text(json.dumps(payload, ensure_ascii=False, indent=2))
    log.info("✔ 完成：%d 件商品 / %d 則評論 / %d 個錯誤 / 耗時 %.2fs",
             result.total_products, result.total_reviews,
             len(result.errors), result.duration_sec)
    log.info("輸出檔：%s", out_path.resolve())

    # Console preview
    print("\n" + "=" * 60)
    print(f"📦 {result.total_products} 件商品，{result.total_reviews} 則評論，"
          f"{len(result.errors)} 個錯誤，耗時 {result.duration_sec}s")
    print("=" * 60)
    for p_ in result.products:
        rating = f"★{p_.avg_rating}" if p_.avg_rating else "n/a"
        print(f"  · {p_.product_id} {p_.title}  NT${p_.price_ntd}  {rating}  ({len(p_.reviews)} 評論)")
    if result.errors:
        print("\n[ERRORS]")
        for e in result.errors:
            print("  -", e)
    return 0


if __name__ == "__main__":
    sys.exit(main())

🎯 設計要點

  • 為什麼 Selenium + BeautifulSoup？ Selenium 解決「JS 渲染後才能看到的 DOM」問題（requests 抓不到 SPA），BeautifulSoup 解決「在 HTML 裡穩定找元素」問題（Selenium 的 find_element 太脆弱）。互補而非重複。
  • 為什麼 WebDriverWait？ 避免 race condition：JS 還沒跑就抓，會抓到空 grid 或「載入中...」。等特定 selector 出現後才取 page_source。
  • 為什麼 headless=new？ 新 headless 模式更接近真實 Chrome、支援完整 JS API。容器 / 無 GUI 環境必備。
  • 為什麼 dataclass + asdict？ 比裸 dict 更易讀、易擴展欄位、序列化成 JSON 一行搞定。
  • 為什麼 defensive parse？ 真實網站欄位常缺漏，optional 型別 + try/except 才能 graceful degrade。
