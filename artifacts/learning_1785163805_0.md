+def telegram_call(token: str, method: str, params: dict, timeout: int = 40) -> dict:
+    body = urlencode(params).encode()
+    request = Request(API.format(token=token, method=method), data=body,
+                      headers={"Content-Type": "application/x-www-form-urlencoded"})
+    with urlopen(request, timeout=timeout) as response:
+        payload = json.load(response)
+    if not payload.get("ok"):
+        raise RuntimeError(payload.get("description", "Telegram API error"))
+    return payload["result"]
+

+@dataclass
+class RateLimiter:
+    cooldown: float = 3.0
+
+    def __post_init__(self) -> None:
+        self.last_seen: dict[int, float] = {}
+
+    def allowed(self, user_id: int) -> bool:
+        now = time.monotonic()
+        previous = self.last_seen.get(user_id, 0.0)
+        if now - previous < self.cooldown:
+            return False
+        self.last_seen[user_id] = now
+        return True
+
+
+def parse_allowed_users() -> set[int] | None:
+    raw = os.getenv("TELEGRAM_ALLOWED_USERS", "").strip()

+# Telegram 天氣機器人實作思路
+
+## 架構
+- `telegram_bot.py`：Telegram Bot API 長輪詢、訊號處理、權限 allowlist、使用者 rate limit、錯誤處理。
+- `wttr_weather.py`：獨立的 wttr.in 天氣 API client，可單獨以 CLI 測試；使用 Python standard library，無第三方套件依賴。
+- `run.sh`：檢查必要環境變數後啟動機器人。
+
+## 核心流程
+1. 從 `TELEGRAM_BOT_TOKEN` 讀取 Bot token。
+2. 以 `getUpdates` 長輪詢 Telegram，使用 `offset = update_id + 1` 避免重複處理。
+3. 收到文字後，將城市名稱做控制字元清理與長度限制。
+4. 呼叫 `https://wttr.in/<city>?format=j1`，解析目前天氣資料。
+5. 以本地字典把常見英文天氣描述翻譯成中文，再透過 `sendMessage` 回覆。
+6. 相同城市快取 10 分鐘，降低 wttr.in 請求量。
+7. SIGINT/SIGTERM 設定停止旗標，讓輪詢正常結束。
+
+## 防護
+- 設定 `TELEGRAM_ALLOWED_USERS` 後，只允許逗號分隔的 Telegram user ID 使用。
+- 每位使用者預設 3 秒冷卻，避免濫用天氣 API。
+- 網路錯誤、API 回應錯誤、城市資料格式錯誤都會轉成安全的使用者訊息，不讓程式直接退出。
+- 不把 token 寫入檔案或程式碼。
+
+## 架構回饋採納
+- 長輪詢適合沒有公開 HTTPS endpoint 的主機；若日後部署到 Vercel/serverless，應改用 webhook。
+- 增加 10 分鐘城市快取，以降低上游 API 429/rate-limit 風險。
+
+## 啟動
+```bash
+export TELEGRAM_BOT_TOKEN='從 BotFather 取得的 token'
+export TELEGRAM_ALLOWED_USERS='123456789,987654321'  # 可選
+./run.sh
+```
