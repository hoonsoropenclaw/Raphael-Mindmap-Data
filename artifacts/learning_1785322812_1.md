+import { createContext, useCallback, useContext, useEffect, useState } from "react";
+
+type Theme = "light" | "dark" | "corporate" | "business" | "emerald" | "autumn";
+
+type ThemeContextValue = {
+  theme: Theme;
+  setTheme: (t: Theme) => void;
+  cycleTheme: () => void;
+  themes: Theme[];
+};
+
+const ThemeContext = createContext<ThemeContextValue | null>(null);
+
+const THEMES: Theme[] = ["light", "dark", "corporate", "business", "emerald", "autumn"];
+const STORAGE_KEY = "hermes-theme";
+
+export function ThemeProvider({ children }: { children: React.ReactNode }) {
+  const [theme, setThemeState] = useState<Theme>("light");
+
+  // Read from localStorage on mount (avoid SSR mismatch)
+  useEffect(() => {
+    try {
+      const stored = localStorage.getItem(STORAGE_KEY) as Theme | null;
+      if (stored && THEMES.includes(stored)) {
+        setThemeState(stored);
+      }
+    } catch {
+      // ignore
+    }
+  }, []);
+
+  // Apply theme to <html data-theme>
+  useEffect(() => {
+    document.documentElement.setAttribute("data-theme", theme);
+  }, [theme]);
+
+  const setTheme = useCallback((t: Theme) => {
+    setThemeState(t);
+    try {
+      localStorage.setItem(STORAGE_KEY, t);
+    } catch {
+      // ignore
+    }
+  }, []);
+
+  const cycleTheme = useCallback(() => {
+    setThemeState((current) => {
+      const idx = THEMES.indexOf(current);
+      const next = THEMES[(idx + 1) % THEMES.length];
+      try {
+        localStorage.setItem(STORAGE_KEY, next);
+      } catch {
+        // ignore
+      }
+      return next;
+    });
+  }, []);
+
+  return (
+    <ThemeContext.Provider value={{ theme, setTheme, cycleTheme, themes: THEMES }}>
+      {children}
+    </ThemeContext.Provider>
+  );

+# Hermes Stack — Next.js + Tailwind v4 + DaisyUI 5
+
+A full-stack Next.js 16 application styled with Tailwind CSS v4 and DaisyUI 5. Server-rendered,
+responsive, theme-switchable, API-ready — built as a production-grade starter.
+
+## Stack
+
+| Layer            | Version | Role                                                |
+| ---------------- | ------- | --------------------------------------------------- |
+| Next.js          | 15.5.22 | App Router + RSC + Turbopack                        |
+| React            | 19.2.8  | Server / client components                          |
+| TypeScript       | 5.x     | Strict mode across the stack                        |
+| Tailwind CSS     | 4.3.3   | `@theme`-based design system, Oxide engine          |
+| DaisyUI          | 5.7.4   | 6 themes, semantic component classes                |
+| PostCSS          | 8.4.x   | Tailwind v4 plugin pipeline                         |
+
+## Routes
+
+| Method | Path                | Type    | Description                                  |
+| ------ | ------------------- | ------- | -------------------------------------------- |
+| GET    | `/`                 | Static  | Hero, features, code preview, CTA            |
+| GET    | `/dashboard`        | Static  | Stats grid, user table, activity feed        |
+| GET    | `/docs`             | Static  | Quick-start, API reference, theme showcase   |
+| GET    | `/contact`          | Static  | Contact form (client island for state)       |
+| GET    | `/api/health`       | Dynamic | Liveness probe with runtime metadata         |
+| GET    | `/api/users`        | Dynamic | User roster; supports `?status=` and `?role=` |
+| POST   | `/api/users`        | Dynamic | Echoes a new user payload (422 on missing)   |
+| —      | `*`                 | —       | Custom 404 via `not-found.tsx`               |
+
+## Features
+
+- **Six DaisyUI themes** wired in `globals.css`: `light` (default), `dark`, `corporate`, `business`, `emerald`, `autumn`. Click the sun icon in the navbar to cycle; selection persists in `localStorage` via the `ThemeProvider` context.
+- **Server-first rendering**: only the navbar theme toggle, mobile menu, and contact form carry `"use client"`. Everything else is RSC.
+- **Strict TypeScript**: `tsconfig.json` has `strict: true` + `noEmit: true`, and `next build` runs the type check as part of the build step.
+- **Responsive layout**: 4 breakpoints (`sm` / `md` / `lg` / `xl`) with `max-w-7xl` containers and grid stacks that collapse cleanly.
+- **Accessibility**: focus-visible rings, `aria-label` on icon buttons, `prefers-reduced-motion` short-circuits all keyframe animations.
+- **OG + metadata** wired through `metadata` export in `app/layout.tsx` (title template + description + OG tags).
+
