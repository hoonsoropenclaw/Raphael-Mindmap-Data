Code (DaisyUI mockup-code)

# install the design system
$ npm install @frosted/ui

# tailwind.config.js
import { frosted } from '@frosted/ui'

export default {
  plugins: [frosted],
  daisyui: {
    themes: ['frosted', 'frosted-dark'],
  },
}
