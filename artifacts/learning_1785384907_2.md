from html.parser import HTMLParser

class V(HTMLParser):
    def __init__(self):
        super().__init__()
        self.stack = []
        self.errors = []
        self.void = {'area','base','br','col','embed','hr','img','input','link','meta','param','source','track','wbr','path','circle','rect','line','polyline','polygon','ellipse','use','stop'}
    def handle_starttag(self, tag, attrs):
        if tag not in self.void:
            self.stack.append(tag)
    def handle_endtag(self, tag):
        if tag in self.void: return
        if not self.stack:
            self.errors.append(f"Closing </{tag}> with empty stack")
            return
        if self.stack[-1] == tag:
            self.stack.pop()
        else:
            self.errors.append(f"Mismatch: expected </{self.stack[-1]}>, got </{tag}>")
