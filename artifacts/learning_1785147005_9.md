+# ────────── CPU (兩次讀 /proc/stat 取 delta) ──────────
+sample_cpu() {
+    local interval="${1:-1}"
+    # 第一次讀
+    local r1 idle1 total1
+    read -r _ _ _ _ idle1 _ _ _ _ _ total1 < <(head -1 /proc/stat)
+    total1=$((total1))
+    idle1=$((idle1))
+    sleep "$interval"
+    # 第二次讀
+    local idle2 total2
+    read -r _ _ _ _ idle2 _ _ _ _ _ total2 < <(head -1 /proc/stat)
+    total2=$((total2))
+    idle2=$((idle2))
+    # delta
+    local d_total=$((total2 - total1))
+    local d_idle=$((idle2 - idle1))
+    if (( d_total <= 0 )); then
+        return 1
+    fi
+    # bash 浮點用 awk 算
+    local pct
+    pct=$(awk -v d_total="$d_total" -v d_idle="$d_idle" 'BEGIN{printf "%.1f", (d_total-d_idle)*100.0/d_total}')
+    printf 'CPU %s\n' "$pct"
+}

+# ────────── 記憶體 (MemAvailable 語意) ──────────
+sample_memory() {
+    if [[ ! -r /proc/meminfo ]]; then return 1; fi
+    local mem_total mem_avail
+    mem_total=$(awk '/^MemTotal:/ {print $2}' /proc/meminfo)
+    mem_avail=$(awk '/^MemAvailable:/ {print $2}' /proc/meminfo)
+    if [[ -z "$mem_total" || -z "$mem_avail" || "$mem_total" -le 0 ]]; then
+        return 1
+    fi
+    local used_pct
+    used_pct=$(awk -v t="$mem_total" -v a="$mem_avail" 'BEGIN{printf "%.1f", (t-a)*100.0/t}')
+    printf 'MEM %s %d %d\n' "$used_pct" "$mem_avail" "$mem_total"
+}

+    if [[ ! -r /proc/meminfo ]]; then return 1; fi

+    used_kb=$((swap_total - swap_free))
+    printf 'SWAP %s %d %d\n' "$used_pct" "$used_kb" "$swap_total"
+}
