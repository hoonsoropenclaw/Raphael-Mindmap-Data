+    const id = state.tasks.at(-1).id;
+    expect(state.tasks.at(-1).title).toBe('Ship review');
+
+    state = reduce(state, { type: 'UPDATE_TASK', payload: { id, changes: { status: 'active' } } });
+    expect(state.tasks.find((task) => task.id === id).status).toBe('active');
+
+    state = reduce(state, { type: 'DELETE_TASK', payload: id });
+    expect(state.tasks.some((task) => task.id === id)).toBe(false);
+
+    state = reduce(state, { type: 'UNDO' });
+    expect(state.tasks.some((task) => task.id === id)).toBe(true);
+    state = reduce(state, { type: 'REDO' });
+    expect(state.tasks.some((task) => task.id === id)).toBe(false);
+  });
+
+  it('toggles selections and applies batch status updates', () => {
+    let state = reduce(initialState, { type: 'TOGGLE_SELECT', payload: 'task-1' });
+    state = reduce(state, { type: 'TOGGLE_SELECT', payload: 'task-2' });
+    expect(state.selected).toEqual(['task-1', 'task-2']);
+    state = reduce(state, { type: 'BATCH_STATUS', payload: 'done' });
+    expect(state.tasks.filter((task) => ['task-1', 'task-2'].includes(task.id)).every((task) => task.status === 'done')).toBe(true);

+## 3. 關鍵架構決策
+
+### 3.1 為什麼是「任務編排工作台」而不是 Todo ？
+訊息沒指定主題；挑這個元件類別因為：
+- Todo 太常見、無法展示「複雜」狀態管理
+- 需要能展示 reducer + 多條件篩選 + 排序 + 批次 + Undo/Redo + 持久化 + Modal
+- 「工作台」主題給了合理空間放 metrics / progress / batch bar / 視覺層級
+
+### 3.2 為什麼用 `useReducer` + `Context` 而不是多個 `useState` 或 Zustand
+- 跨欄位聯動（selection 決定 batch bar、filters 決定 visibleTasks、history 決定 undo 鈕 disabled）
+- 多個 `useState` 容易產生競態（`setQuery` 後 `setFilters` 仍舊渲染舊資料）
+- Zustand 對這個規模 overkill、且會引入額外依賴
+- Context 解決 prop drilling、但**不**對外暴露 `dispatch`（避免子元件亂 dispatch）
+
+### 3.3 為什麼 Undo 範圍限定在「資料變更」
+- 把 query / filters 寫進歷史 → 一次 Undo 把 UI 跳回陌生狀態，反而是 bug
+- 實務上使用者按 Undo 是「我剛剛改錯資料」
+- 測試 5「keeps history bounded」明確驗證 25 次 SET_QUERY 後 past ≤ 20
+
+### 3.4 為什麼 `selectVisibleTasks` 與 `getMetrics` 是純函式
+- 邏輯與 UI 分離 → 能在不渲染 React 樹下被測試
+- 這也是測試 4「sorts without mutating source」能存在的原因
+- 100% pure → 容易擴展成 server-side filter / 動態載入
+
+### 3.5 為什麼 lucide-react + tree-shake
+- 200KB 級圖示包是 React 專案 LCP 最大殺手
+- lucide-react 是按需 import 的純 ESM
+- 5 個 icon 只用 65KB JS bundle（含 React 本身）→ acceptable
+
+### 3.6 為什麼 history 上限 20
+- 心理學上 Undo 超過 20 步幾乎不會被使用
+- 1 個 task snapshot ≈ 0.2-0.5KB JSON，20 筆 ≈ 10KB localStorage 預算，可接受
+- 防止大量批次操作後 JSON 體積爆炸
