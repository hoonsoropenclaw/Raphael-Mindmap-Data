+#!/usr/bin/env bash
+set -euo pipefail
+ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
+GODOT="$ROOT/.tools/godot/godot"
+if [[ ! -x "$GODOT" ]]; then
+  printf 'Godot binary missing: %s\n' "$GODOT" >&2
+  exit 1
+fi
+mode="${1:-gui}"
+case "$mode" in
+  gui)
+    if [[ -n "${DISPLAY:-}" || -n "${WAYLAND_DISPLAY:-}" ]]; then
+      exec "$GODOT" --path "$ROOT"
+    elif command -v xvfb-run >/dev/null 2>&1; then
+      exec xvfb-run -a "$GODOT" --path "$ROOT" --quit-after 180
+    else
+      printf 'No graphical display. Use: %s --cli [workflow]\n' "$0" >&2
+      exit 2
+    fi
+    ;;
+  --cli|cli)
+    WORKFLOW="${2:-workflows/health_check.json}"
+    RESULT="${3:-automation-output/result.json}"
+    exec "$GODOT" --headless --path "$ROOT" -- --cli "--workflow=$WORKFLOW" "--result=$RESULT"
+    ;;
+  *)
+    printf 'Usage: %s [gui|--cli [workflow.json] [result.json]]\n' "$0" >&2
+    exit 64
+    ;;
+esac

+# Native Automation Studio
+
+A real Godot-native desktop application that executes safe, portable JSON automation workflows. It supports a graphical desktop UI and a deterministic headless CLI mode.
+
+## Included modules
+
+- Native Godot 4 desktop UI
+- JSON workflow loader and validator
+- Safe project-scoped filesystem automation
+- Platform/runtime introspection
+- Machine-readable result reports
+- Headless CI/automation mode
+- Success and recovery-path integration tests
+
+Shell command execution is intentionally excluded. Workflow paths are constrained to this project or `user://`.
+
+## Run
+
+```bash
+chmod +x run.sh tests/integration.sh
+./run.sh gui
+```
+
+On a headless Linux machine, `run.sh gui` uses `xvfb-run` for a short smoke session.
+
+CLI automation:
+
+```bash
+./run.sh --cli workflows/health_check.json automation-output/result.json
+```
+
+Test everything:
+
+```bash
+./tests/integration.sh
+```
+
+## Workflow schema
+
+```json
+{
+  "name": "example",
+  "steps": [
+    {"id": "platform", "type": "platform_info"},
+    {"id": "mkdir", "type": "make_dir", "path": "automation-output/sample"},
+    {"id": "write", "type": "write_file", "path": "automation-output/sample/message.txt", "content": "hello"},
+    {"id": "read", "type": "read_file", "path": "automation-output/sample/message.txt"},
+    {"id": "wait", "type": "delay", "milliseconds": 25},
+    {"id": "list", "type": "list_dir", "path": "automation-output/sample"}
+  ]
+}
+```
+
+Add `"continue_on_error": true` to a step when a failure should be recorded without stopping later steps.
+
+## Architecture
+
+- `src/main.gd`: UI, CLI parser, workflow engine, safe native integrations
+- `main.tscn`: root native scene
+- `workflows/`: reusable automation recipes
+- `run.sh`: GUI/headless launcher
+- `tests/integration.sh`: real Godot integration verification
+- `.tools/godot/`: project-local Godot 4.7.1 runtime and fingerprint
