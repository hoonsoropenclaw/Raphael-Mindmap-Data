+"""Thread-safe token bucket rate limiter (single-process, in-memory)."""
+from __future__ import annotations
+
+import math
+import threading
+import time
+from dataclasses import dataclass
+
+
+@dataclass(slots=True)
+class _Bucket:
+    tokens: float
+    last_refill: float
+
+
+class TokenBucket:
+    """Per-key token bucket with refill-on-demand semantics.
+
+    Thread safe: all state mutations occur under a single mutex so concurrent
+    coroutines or threads always see a consistent (tokens, last_refill) pair.
+    """
+
+    def __init__(self, capacity: int, refill_rate: float) -> None:
+        if capacity <= 0:
+            raise ValueError("capacity must be positive")
+        if refill_rate < 0:
+            raise ValueError("refill_rate must be non-negative")
+        self._capacity = float(capacity)
+        self._refill_rate = float(refill_rate)
+        self._buckets: dict[str, _Bucket] = {}
+        self._lock = threading.Lock()
+
+    @property
+    def capacity(self) -> int:
+        return int(self._capacity)
+
+    def consume(self, key: str, tokens: float = 1.0) -> tuple[bool, int, int]:
+        """Try to consume ``tokens`` from the bucket for ``key``.
+
+        Returns ``(allowed, remaining, retry_after_seconds)``.  ``retry_after``
+        is 0 when allowed and >= 1 (rounded up) when denied.
+        """
+
+        if tokens <= 0:
+            raise ValueError("tokens must be positive")
+
+        now = time.monotonic()
+        with self._lock:
+            bucket = self._buckets.get(key)
+            if bucket is None:
+                bucket = _Bucket(tokens=self._capacity, last_refill=now)
+                self._buckets[key] = bucket
+
+            elapsed = now - bucket.last_refill
+            if elapsed > 0 and self._refill_rate > 0:
+                bucket.tokens = min(self._capacity, bucket.tokens + elapsed * self._refill_rate)
+            bucket.last_refill = now
+
+            if bucket.tokens >= tokens:
+                bucket.tokens -= tokens
+                remaining = int(math.floor(bucket.tokens))
+                return True, remaining, 0
+
+            deficit = tokens - bucket.tokens
+            if self._refill_rate > 0:
+                retry_after = max(1, math.ceil(deficit / self._refill_rate))
+            else:
+                retry_after = 1
+            remaining = int(math.floor(bucket.tokens))
+            return False, remaining, retry_after
+
+    def reset(self) -> None:
+        with self._lock:
+            self._buckets.clear()
