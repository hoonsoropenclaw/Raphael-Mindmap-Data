+// Pure functions for RBAC: cycle detection, permission resolution, validation.
+// Exported via ESM so both app.js (UI) and tests can import the same module.
+//
+// Permission model:
+//   tri-state per (role, resource): 'allow' | 'deny' | 'inherit'
+//   inheritance forms a DAG of roles; edges = "inherits from"
+//   resolution priority (highest first): deny > explicit allow > inherited allow > implicit deny
+//
+// All exports are pure functions; no DOM, no globals.

+export function detectCycle(edges) {
+  // Returns true if adding (src -> dst) to edges would form a cycle.
+  // Algorithm: walk forward from dst; if we can reach src, it's a cycle.
+  const adj = new Map();
+  for (const e of edges) {
+    if (!adj.has(e.source)) adj.set(e.source, []);
+    adj.get(e.source).push(e.target);
+  }
+  // DFS with white/gray/black coloring
+  const WHITE = 0, GRAY = 1, BLACK = 2;
+  const color = new Map();
+  for (const e of edges) {
+    if (!color.has(e.source)) color.set(e.source, WHITE);
+    if (!color.has(e.target)) color.set(e.target, WHITE);
+  }
+  function dfs(u) {
+    color.set(u, GRAY);
+    const next = adj.get(u) || [];
+    for (const v of next) {
+      const c = color.get(v) ?? WHITE;
+      if (c === GRAY) return true;       // back edge → cycle
+      if (c === WHITE && dfs(v)) return true;
+    }
+    color.set(u, BLACK);
+    return false;
+  }
+  for (const u of color.keys()) {
+    if (color.get(u) === WHITE && dfs(u)) return true;
+  }
+  return false;
