+"""Rate limiter primitives.
+
+Implements a fixed-window counter keyed by client IP. Each window allows
+``limit`` requests; further requests receive 429 with a Retry-After hint.
+A single ``asyncio.Lock`` guards the counter map so that concurrent
+requests on the same bucket cannot over-grant.
+"""
+
+from __future__ import annotations
+
+import asyncio
+import time
+from dataclasses import dataclass, field
+from typing import Dict, Tuple
+
+
+@dataclass
+class FixedWindowLimiter:
+    limit: int
+    window_seconds: float
+    _buckets: Dict[str, Tuple[int, float]] = field(default_factory=dict)
+    _lock: asyncio.Lock = field(default_factory=asyncio.Lock)
+
+    async def check(self, key: str) -> Tuple[bool, int]:
+        """Return ``(allowed, retry_after_seconds)``.
+
+        ``retry_after_seconds`` is only meaningful when ``allowed`` is
+        ``False`` and is always at least 1 to satisfy HTTP ``Retry-After``.
+        """
+        now = time.monotonic()
+        window = max(self.window_seconds, 0.001)
+        async with self._lock:
+            count, window_start = self._buckets.get(key, (0, now))
+            elapsed = now - window_start
+            if elapsed >= window:
+                count = 0
+                window_start = now
+            if count >= self.limit:
+                retry_after = max(1, int(window - elapsed))
+                self._buckets[key] = (count, window_start)
+                return False, retry_after
+            count += 1
+            self._buckets[key] = (count, window_start)
+        return True, 0

+# Todo API
+
+A small FastAPI service that manages a list of todos, backed by SQLite and
+protected by a per-client fixed-window rate limiter.
+
+## Endpoints
+
+| Method | Path                | Description                                 |
+| ------ | ------------------- | ------------------------------------------- |
+| GET    | `/health`           | Liveness probe (not rate limited)           |
+| GET    | `/todos`            | List todos, newest first                    |
+| POST   | `/todos`            | Create a todo (`{"title": "..."}`)          |
+| GET    | `/todos/{id}`       | Read a single todo                          |
+| PUT    | `/todos/{id}`       | Patch title / completed flag                |
+| DELETE | `/todos/{id}`       | Remove a todo                               |
+
+Validation rules (returns `422`):
+
+- `title` is required, non-blank, ≤ 255 characters after stripping
+- `completed` is optional, defaults to `false`
+
+Rate limiting (returns `429` with a `Retry-After` header in seconds):
+
+- Default: 60 requests per 60-second window, keyed by ASGI `client.host`
+- Configurable via `create_app(rate_limit=..., rate_window_seconds=...)`
+- `/health` is **not** rate limited so external monitors stay free
+
+## Run
+
+```bash
+pip install -r requirements.txt
+uvicorn todo_api.main:app --host 0.0.0.0 --port 8000
+```
