+"""
+storage.py — Pluggable result sinks.
+
+Three sinks:
+    JsonlSink   - append-only JSONL (best for large crawls)
+    CsvSink     - one CSV file
+    SqliteSink  - SQLite with a single results table (queryable later)
+
+The sink is selected via config.output_format.
+"""
+from __future__ import annotations
+
+import csv
+import json
+import sqlite3
+import threading
+import time
+from pathlib import Path
+from typing import Any, Iterable, Optional
+
+
+class _BaseSink:
+    def write(self, record: dict) -> None: ...
+    def write_many(self, records: Iterable[dict]) -> None:
+        for r in records:
+            self.write(r)
+    def close(self) -> None: ...
+
+
+class JsonlSink(_BaseSink):
+    def __init__(self, path: str | Path):
+        self.path = Path(path)
+        self.path.parent.mkdir(parents=True, exist_ok=True)
+        self._fh = open(self.path, "a", encoding="utf-8")
+        self._lock = threading.Lock()
+
+    def write(self, record: dict) -> None:
+        line = json.dumps(record, ensure_ascii=False, default=str)
+        with self._lock:
+            self._fh.write(line + "\n")
+            self._fh.flush()
+
+    def close(self) -> None:
+        with self._lock:
+            self._fh.close()
+
+
+class CsvSink(_BaseSink):
+    def __init__(self, path: str | Path):
+        self.path = Path(path)
+        self.path.parent.mkdir(parents=True, exist_ok=True)
+        # We discover columns on the first write
+        self._fh = open(self.path, "a", encoding="utf-8", newline="")
+        self._writer: Optional[csv.DictWriter] = None
+        self._lock = threading.Lock()
+        self._columns: Optional[list[str]] = None
+
+    def write(self, record: dict) -> None:
+        # Flatten one level for CSV
+        flat = {k: _flatten(v) for k, v in record.items()}
+        with self._lock:
+            if self._writer is None:
+                self._columns = list(flat.keys())
+                self._writer = csv.DictWriter(self._fh, fieldnames=self._columns)
+                # Only write header if file is empty
+                if self._fh.tell() == 0:
+                    self._writer.writeheader()
+            # Add missing columns on subsequent rows
+            extra = [k for k in flat if k not in self._columns]
+            if extra:
+                self._columns.extend(extra)
+                self._writer.fieldnames = self._columns
+            self._writer.writerow(flat)
+            self._fh.flush()
+
+    def close(self) -> None:
+        with self._lock:
