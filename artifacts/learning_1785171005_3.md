+    # /proc/stat 第一行 "cpu" 全部 ticks

+"""Cooldown — suppress duplicate alerts within a window."""
+
+from __future__ import annotations
+
+import json
+import os
+import time
+from pathlib import Path
+
+
+class CooldownStore:
+    """File-backed cooldown: 一行 = {key, last_ts}.
+
+    Atomic write via tmp file + rename to avoid corruption on concurrent writers.
+    """
+
+    def __init__(self, path: str, cooldown_seconds: int):
+        self.path = Path(path)
+        self.cooldown_seconds = cooldown_seconds
+        self.path.parent.mkdir(parents=True, exist_ok=True)
+        if not self.path.exists():
+            self.path.touch(mode=0o600)
+
+    def _read(self) -> dict[str, float]:
+        if not self.path.exists() or self.path.stat().st_size == 0:
+            return {}
+        out = {}
+        try:
+            with self.path.open("r", encoding="utf-8") as f:
+                for line in f:
+                    line = line.strip()
+                    if not line:
+                        continue
+                    try:
+                        rec = json.loads(line)
+                        out[rec["key"]] = float(rec["last_ts"])
+                    except (json.JSONDecodeError, KeyError, ValueError):
+                        continue
+        except OSError:
+            return {}
+        return out
+
+    def _write(self, data: dict[str, float]) -> None:
+        # 寫到 tmp 再 rename (POSIX atomic)
+        tmp = self.path.with_suffix(self.path.suffix + ".tmp")
+        with tmp.open("w", encoding="utf-8") as f:
+            os.chmod(tmp, 0o600)
+            for k, ts in data.items():
+                f.write(json.dumps({"key": k, "last_ts": ts}) + "\n")
+        tmp.replace(self.path)
+
+    def is_cool(self, key: str) -> bool:
+        """Return True if `key` has NOT been alerted within cooldown window."""
+        data = self._read()
+        last = data.get(key)
+        if last is None:
+            return True
+        return (time.time() - last) > self.cooldown_seconds
+
+    def mark_alerted(self, keys: list[str]) -> None:
+        """Mark all given keys as just-alerted (update timestamp)."""
+        data = self._read()
+        now = time.time()
+        for k in keys:
+            data[k] = now
+        # 移除太舊的紀錄 (避免無限期膨脹)
+        cutoff = now - self.cooldown_seconds * 10
+        data = {k: ts for k, ts in data.items() if ts >= cutoff}
+        self._write(data)

+            self.path.unlink()

+## 設計原則
+
+### 1. **零外部依賴**
+不需 `pip install pyyaml`、`psutil`、`requests`。
+純 Python stdlib (`subprocess` / `json` / `urllib` / `dataclasses` / `pathlib`)。
+可以丟到任何乾淨的 Linux 環境就跑。
+
+### 2. **資源監控:短窗口測量**
+CPU 使用率不是讀「目前值」(瞬間跳動大),而是讀 `/proc/stat` 兩次間隔 0.5s 算 busy/total delta。
+這樣比 `top` 第一行準,也不會受短期 spike 影響。
+
+### 3. **磁碟監控:跳過虛擬檔案系統**
+`tmpfs` / `devtmpfs` / `efivarfs` / `overlay` 都不會滿,監控它們只會一直誤報。
+用 `Filesystem` 欄位 (`df -kP`) 過濾掉。
+
+### 4. **日誌掃描:兩路併行**
+- `journalctl --since="10 minutes ago" --priority=3` 收集 priority 0-3 的項目 (err 及更嚴重)
+- `/var/log/syslog` (可加更多檔) 用 regex grep 關鍵字
+兩路任何一路觸發都會出 finding。
+
+### 5. **冷卻去重:file-backed + atomic write**
+- 每個 alert 有 `key` (例 `cpu.high`、`disk.full./`、`syslog.keyword.syslog`)
+- 同 key 在 cooldown 內不重發
+- 冷卻紀錄用 `tmp + rename` 原子寫入,避免 race condition

+- webhook (自訂 HTTP POST) 失敗 → 不影響其他兩個
