+  1. 渲染任意 URL (headless)

+  # 基本抓取

+    --selector "h1" \
+    --output result.json
+
+  # 完整模式 (含 network 監聽)
+  python3.12 playwright_dynamic_scraper.py \
+    --url "https://example.com" \
+    --selector ".product" \
+    --wait-network-idle \
+    --capture-network \
+    --screenshot page.png \

+  text       元素文字
