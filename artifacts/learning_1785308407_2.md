+def _geocode_payload(name: str = "Taipei", country: str = "Taiwan", lat: float = 25.03, lon: float = 121.57) -> dict:
+    return {

+    inner = httpx.Client(transport=transport, timeout=5.0)
+    return WeatherClient(client=inner)
