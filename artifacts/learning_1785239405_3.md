+# --- CPU% via /proc/stat delta between two samples ---
+# Returns integer 0..100 (rounded).
+# Usage: cpu_percent <interval_seconds>
+cpu_percent() {
+  local interval="${1:-2}"
+  local _cpu_idle1 _cpu_total1 _cpu_idle2 _cpu_total2
+  read -r _cpu_idle1 _cpu_total1 <<<"$(awk '/^cpu / {print $5, $2+$3+$4+$5+$6+$7+$8+$9+$10}' /proc/stat 2>/dev/null)"
+  [[ -z "$_cpu_total1" || "$_cpu_total1" == "0" ]] && { echo "0"; return; }
+  sleep "$interval"
+  read -r _cpu_idle2 _cpu_total2 <<<"$(awk '/^cpu / {print $5, $2+$3+$4+$5+$6+$7+$8+$9+$10}' /proc/stat 2>/dev/null)"
+  local d_idle=$((_cpu_idle2 - _cpu_idle1))
+  local d_total=$((_cpu_total2 - _cpu_total1))
+  if (( d_total <= 0 )); then echo "0"; return; fi
+  local pct=$(( (d_total - d_idle) * 100 / d_total ))
+  echo "$pct"
+}

+# --- Count recent ERROR/CRITICAL lines from journalctl ---
+# Usage: journal_error_count <priority_range> <since_seconds>
+# priority_range e.g. "0..3"
+journal_error_count() {
+  local prio_range="$1"
+  local since_secs="$2"
+  local since_str
+  since_str="${since_secs} seconds ago"
+  if ! command -v journalctl >/dev/null 2>&1; then
+    echo "0"
+    return 0
+  fi
+  # --no-pager avoids spawning less, --quiet reduces noise
+  journalctl --no-pager -q --since "$since_str" -p "$prio_range" 2>/dev/null \
+    | wc -l | tr -d ' '
+}

+# --- Count regex matches in a file (last N lines, last N seconds) ---
+# Usage: file_pattern_count <file> <pattern> <window_seconds>
+# Returns count of matching lines whose timestamp is within the window.
+file_pattern_count() {
+  local file="$1"
+  local pattern="$2"
+  local window="$3"
+  [[ -r "$file" ]] || { echo "0"; return 0; }
+  # Get cutoff in epoch seconds
+  local cutoff
+  cutoff=$(date -d "@$(( $(date +%s) - window ))" '+%b %e %H:%M:%S' 2>/dev/null || echo "")
+  local count
+  if [[ -z "$cutoff" ]]; then
+    # Fallback: just count all matches
+    count=$(grep -cE "$pattern" "$file" 2>/dev/null || echo 0)
+  else
+    # Count lines whose month-day-time is >= cutoff (best-effort; syslog uses "Mmm DD HH:MM:SS")
+    # We use awk to parse the first 3 fields and compare.
+    count=$(awk -v pat="$pattern" -v cutoff="$cutoff" '
+      BEGIN { count=0 }
+      {
+        # date: $1=Mon $2=Day $3=Time
+        # Convert to epoch for comparison
+        ts = $1 " " $2 " " $3
+        cmd = "date -d \"" ts "\" +%s 2>/dev/null"
+        cmd | getline epoch
+        close(cmd)
+        cmd = "date -d \"" cutoff "\" +%s 2>/dev/null"
+        cmd | getline cut
+        close(cmd)

+# syswatcher

+│   ├── syswatcher.service
+│   └── syswatcher.timer
+├── logs/                        # 執行記錄
+├── state/                       # 去重狀態（cooldown 鍵值）

+### 1. 為何選 bash 而非 Python
