+"""
+Pipeline - 處理 item 的流程
+- CleanPipeline: 資料清洗（去空白、HTML 實體、編碼）
+- ValidationPipeline: 結構驗證
+- ExportPipeline: 多格式輸出
+"""
+
+from __future__ import annotations
+import abc
+import html
+import json
+import csv
+import re
+from typing import Any, Dict, List, Optional
+from pathlib import Path
+
+
+class Pipeline(abc.ABC):
+    """pipeline 抽象基底"""
+    name: str = "base"
+
+    @abc.abstractmethod
+    def process_item(self, item: Any, spider=None) -> Any:
+        ...
+
+
+class CleanPipeline(Pipeline):
+    """資料清洗 pipeline"""
+    name = "clean"
+
+    def __init__(self, strip: bool = True, unescape_html: bool = True, lowercase_keys: bool = False):
+        self.strip = strip
+        self.unescape_html = unescape_html
+        self.lowercase_keys = lowercase_keys
+
+    def _clean_value(self, v: Any) -> Any:
+        if isinstance(v, str):
+            v = html.unescape(v)
+            if self.strip:
+                v = re.sub(r"\s+", " ", v).strip()
+        elif isinstance(v, dict):
+            return {k: self._clean_value(val) for k, val in v.items()}
+        elif isinstance(v, list):
+            return [self._clean_value(x) for x in v]
+        return v
+
+    def process_item(self, item: Any, spider=None) -> Any:
+        if hasattr(item, "items"):  # Item 或 dict
+            data = {k: self._clean_value(v) for k, v in item.items()}
+            if self.lowercase_keys:
+                data = {k.lower(): v for k, v in data.items()}
+            if hasattr(item, "__setitem__"):
+                for k, v in data.items():
+                    item[k] = v
+            else:
+                for k, v in data.items():
+                    setattr(item, k, v)
+            return item
+        return item
+
+
+class ValidationPipeline(Pipeline):
+    """驗證 pipeline - 過濾掉無效 item"""
+    name = "validation"
+
+    def __init__(self, required_fields: Optional[List[str]] = None, drop_invalid: bool = True):
+        self.required_fields = required_fields or []
+        self.drop_invalid = drop_invalid
+
+    def process_item(self, item: Any, spider=None):
+        if hasattr(item, "items"):
+            missing = [f for f in self.required_fields if not item.get(f)]
+            if missing:
