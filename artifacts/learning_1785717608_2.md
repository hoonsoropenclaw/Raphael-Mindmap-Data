+"""Async semaphore-based concurrency limiter (per asyncio_semaphore_backpressure skill).
+
+We expose a small pool keyed by a name so callers can have different limits
+for e.g. outbound Google Calendar calls vs Telegram calls.
+"""
+
+from __future__ import annotations
+
+import asyncio
+from contextlib import asynccontextmanager
+from typing import AsyncIterator, Dict
+
+
+class AsyncSemaphorePool:
+    """Per-key asyncio.Semaphore pool.
+
+    Usage:
+        pool = AsyncSemaphorePool({"google": 4, "telegram": 8})
+        async with pool.acquire("google"):
+            ...
+    """
+
+    def __init__(self, limits: Dict[str, int]):
+        if not limits:
+            raise ValueError("AsyncSemaphorePool requires at least one limit")
+        for key, value in limits.items():
+            if value < 1:
+                raise ValueError(f"Concurrency for {key!r} must be >= 1, got {value}")
+        self._limits = dict(limits)
+        self._semaphores: Dict[str, asyncio.Semaphore] = {}
+
+    def _get(self, key: str) -> asyncio.Semaphore:
+        if key not in self._semaphores:
+            if key not in self._limits:
+                raise KeyError(f"Unknown concurrency key: {key!r}")
+            self._semaphores[key] = asyncio.Semaphore(self._limits[key])
+        return self._semaphores[key]
+
+    @asynccontextmanager
+    async def acquire(self, key: str) -> AsyncIterator[None]:
+        sem = self._get(key)
+        async with sem:
+            yield
+
+    def stats(self) -> Dict[str, int]:
+        """Return the configured limits (useful for /healthz dashboards)."""
+        return dict(self._limits)
