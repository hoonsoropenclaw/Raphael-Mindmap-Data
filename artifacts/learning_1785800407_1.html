核心程式碼 animate() · useTransform · useSpring

三個 Framer Motion 模式,搭配 content-visibility / will-change / rAF 合併事件,達到 60fps。

// 1) animate() 直接 API (動畫控制權最強) import { animate, inView } from "framer-motion"; // 進入 viewport 時才啟動,backdrop-filter 才掛載 inView(el, () => { el.style.willChange = "transform, opacity"; const controls = animate(el, { opacity: [0, 1], y: [60, 0], scale: [0.92, 1] }, { duration: 0.7, ease: [0.22, 1, 0.36, 1] } ); controls.then(() => el.style.willChange = "auto"); return () => controls.stop(); });
// 2) 滾動視差 (rAF 合併 pointer / scroll) let raf = null; let targetX = 0, targetY = 0; let curX = 0, curY = 0; window.addEventListener("pointermove", (e) => { targetX = (e.clientX / innerWidth - 0.5) * 40; targetY = (e.clientY / innerHeight - 0.5) * 40; if (!raf) raf = requestAnimationFrame(tick); }); function tick() { curX += (targetX - curX) * 0.12; // ease curY += (targetY - curY) * 0.12; orbs.forEach((o, i) => { const d = parseFloat(o.dataset.depth) || 0.3; o.style.transform = `translate3d(${curX * d}px, ${curY * d}px, 0)`; }); if (Math.abs(targetX - curX) > 0.1) { raf = requestAnimationFrame(tick); } else { raf = null; } }
// 3) Spring 拖曳 (鬆手回中) import { animate } from "framer-motion"; let activeX = 0, activeY = 0; let offsetX = 0, offsetY = 0; card.addEventListener("pointerdown", startDrag); window.addEventListener("pointermove", onDrag); window.addEventListener("pointerup", endDrag); function endDrag() { // 鬆手 → spring 回中心 animate(card, { x: 0, y: 0 }, { type: "spring", stiffness: 180, damping: 16, }); }
// 4) Reduced-motion 自動旁路 const rm = window.matchMedia("(prefers-reduced-motion: reduce)"); function play(target, keyframes, options = {}) { if (rm.matches) { // 直接設終態,不播動畫 Object.assign(target.style, { opacity: 1, transform: "none" }); return; } return animate(target, keyframes, { duration: 0.6, ease: [0.22, 1, 0.36, 1], ...options }); }
