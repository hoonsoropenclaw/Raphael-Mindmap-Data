+core.event — Event 物件與 Topic 主題系統
+=====================================

+Event 是一個不可變的 namedtuple-like dataclass,具備:
+    - event_id   : 全域唯一 UUID (供追蹤 / 去重)

+    - timestamp  : 事件產生的 Unix timestamp (來源端時間,不是接收端)
+    - source     : 來源識別 (例如 'binance-ws', 'coingecko-poll')
+    - schema_ver : schema 版本 (向下相容用)
+"""
+
+from __future__ import annotations
+
+import time
+import uuid
+from dataclasses import dataclass, field
+from typing import Any, FrozenSet, Tuple
+
+
+@dataclass(frozen=True)
+class Topic:
+    """階層式主題。
+
+    比對規則: 訂閱者的每一個 segment 可以是 literal 或 '*'(wildcard)。
+    例:
+        subscribe("market.*.btcusdt")  會收到 "market.binance.btcusdt"
+                                       也會收到 "market.coinbase.btcusdt"
+                                       但不會收到 "market.binance.ethusdt"
+    """
+
+    segments: Tuple[str, ...]
+
+    @classmethod
+    def parse(cls, raw: str) -> "Topic":
+        if not raw:
+            raise ValueError("Topic 字串不可為空")
+        return cls(tuple(raw.split(".")))
+
+    def matches(self, pattern: "Topic") -> bool:
+        if len(self.segments) != len(pattern.segments):
+            return False
+        return all(p == "*" or p == s for s, p in zip(self.segments, pattern.segments))
+
+    def __str__(self) -> str:
+        return ".".join(self.segments)
+
+

+core.bus — async EventBus (pub/sub)
+==================================
+
+設計:
+    - 內部用 asyncio.Queue 給每個 subscriber 緩衝,避免慢 consumer 拖慢整個 bus。

+    - 提供 publish / subscribe / start / stop 四個 API。
+    - start() 會 spawn 一個 dispatch loop: 從 in_queue 取出事件,分派給所有匹配 subscriber。
+    - 支援 graceful shutdown: stop() 後 publish 會 raise RuntimeError,讓 caller 知道。
+

+

+        self.queue = queue
+        self._closed = False

+"""
+core.backpressure — 背壓控制閘門
+================================
+
+目的: 當下遊 processor 來不及消化事件時,讓上游 source 暫停拉取,
+      避免記憶體爆掉或下游被淹死。
+
+實作: 包裝一個 asyncio.Semaphore。processor 每收到一個事件就 acquire,
+      處理完 release。source 想 publish 之前先 acquire(timeout) —
+      若在 timeout 內拿不到,就 skip 這一輪 poll (等同於丟棄這次資料)。
+"""
+
+from __future__ import annotations
+
+import asyncio
+import contextlib
+from typing import Optional
+
+
+class BackpressureGate:
+    def __init__(self, permits: int = 100):
+        self._sem = asyncio.Semaphore(permits)
+        self._permits = permits
+        self.in_flight = 0  # 觀察用
+        self._lock = asyncio.Lock()
+
+    @property
+    def permits(self) -> int:
+        return self._permits
+
+    async def acquire(self, timeout: Optional[float] = 0.0) -> bool:
+        """試著拿到一個 permit。
+
+        - timeout=None: 永遠等
+        - timeout=0:    不等
+        - timeout>0:    等最多 timeout 秒
+        """
+        try:
+            if timeout is None:
+                await self._sem.acquire()
+            else:
+                await asyncio.wait_for(self._sem.acquire(), timeout=timeout)
+        except asyncio.TimeoutError:
+            return False
+        self.in_flight += 1
+        return True
+
+    async def release(self) -> None:
+        self.in_flight = max(0, self.in_flight - 1)
+        self._sem.release()
+
+    @contextlib.asynccontextmanager
+    async def slot(self, timeout: Optional[float] = 0.0):
+        """with-statement 介面。"""
+        ok = await self.acquire(timeout=timeout)
+        if not ok:
+            yield False
+            return
+        try:
+            yield True
+        finally:
+            await self.release()
