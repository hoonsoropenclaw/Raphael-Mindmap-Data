+"""In-memory sliding-window rate limiter.
+
+Design
+------
+- Per-key deque of request timestamps; each `check()` evicts expired entries
+  then decides allow/deny based on current count vs `max_requests` per `window_seconds`.
+- Single-process only — fine for a learning-grade API. To scale horizontally,
+  swap this class for a Redis-backed one (the interface is intentionally tiny).
+- Thread-safe via `threading.Lock` even though FastAPI is async — cheap insurance
+  in case anyone ever calls `check()` from a sync context (e.g. background tasks).
+"""
+from __future__ import annotations
+
+import threading
+import time
+from collections import deque
+from collections.abc import Deque
+from dataclasses import dataclass
+
+
+@dataclass(frozen=True)
+class RateLimitDecision:
+    allowed: bool
+    remaining: int
+    retry_after: float  # seconds; 0 when allowed
+
+
+class RateLimiter:
+    def __init__(self, max_requests: int = 60, window_seconds: float = 60.0) -> None:
+        if max_requests <= 0:
+            raise ValueError("max_requests must be positive")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be positive")
+        self.max_requests = max_requests
+        self.window_seconds = window_seconds
+        self._buckets: dict[str, Deque[float]] = {}
+        self._lock = threading.Lock()
+
+    def check(self, key: str) -> RateLimitDecision:
+        """Returns whether `key` is allowed to make one more request right now."""
+        now = time.monotonic()
+        cutoff = now - self.window_seconds
+        with self._lock:
+            bucket = self._buckets.setdefault(key, deque())
+            # evict expired timestamps from the left
+            while bucket and bucket[0] <= cutoff:
+                bucket.popleft()
+            if len(bucket) >= self.max_requests:
+                # denied — retry_after is how long until oldest entry expires
+                retry_after = max(0.0, bucket[0] + self.window_seconds - now)
+                return RateLimitDecision(False, 0, retry_after)
+            bucket.append(now)
+            return RateLimitDecision(True, self.max_requests - len(bucket), 0.0)
+
+    def reset(self) -> None:
+        """Clear all state. Useful for tests."""
+        with self._lock:
+            self._buckets.clear()

+
+A minimal FastAPI todo service with SQLite persistence and per-IP rate limiting.
+
+## Layout
+
+```
+todo_api/
+├── app/
+│   ├── __init__.py
