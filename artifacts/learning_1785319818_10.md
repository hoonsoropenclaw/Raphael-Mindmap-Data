+extends CharacterBody2D
+## Player controller for a 2D platformer.
+## Demonstrates physics integration: gravity, ground snapping, coyote time, jump buffering.
+
+signal collected(value: int)
+signal died
+
+const SPEED := 220.0
+const JUMP_VELOCITY := -420.0
+const COYOTE_TIME := 0.12
+const JUMP_BUFFER := 0.12
+const MAX_FALL_SPEED := 900.0
+
+var _coyote_timer := 0.0
+var _jump_buffer_timer := 0.0
+var _alive := true
+
+@onready var sprite: ColorRect = $Sprite
+@onready var camera: Camera2D = $Camera2D
+
+func _ready() -> void:
+	add_to_group("player")
+	# Layer 2 = player (set in project.godot layer_names)
+	collision_layer = 0b0000_0010
+	collision_mask = 0b0000_0001 | 0b0000_0011  # world + enemy (so we can stomp)
+
+func _physics_process(delta: float) -> void:
+	if not _alive:
+		return
+
+	# Gravity
+	if not is_on_floor():
+		velocity.y = min(velocity.y + get_gravity().y * delta, MAX_FALL_SPEED)
+		_coyote_timer = max(0.0, _coyote_timer - delta)
+	else:
+		velocity.y = 0.0
+		_coyote_timer = COYOTE_TIME
+
+	# Horizontal input
+	var dir := Input.get_axis("move_left", "move_right")
+	velocity.x = dir * SPEED
+	# Keep coyote from drifting away when running off a ledge
+	if is_on_floor():
+		_coyote_timer = COYOTE_TIME
+
+	# Flip sprite
+	if dir != 0.0:
+		sprite.scale.x = 1.0 if dir > 0.0 else -1.0
+
+	# Jump
+	if Input.is_action_just_pressed("jump"):
+		_jump_buffer_timer = JUMP_BUFFER
+	else:
+		_jump_buffer_timer = max(0.0, _jump_buffer_timer - delta)
+
+	if _jump_buffer_timer > 0.0 and _coyote_timer > 0.0:
+		velocity.y = JUMP_VELOCITY
+		_jump_buffer_timer = 0.0
+		_coyote_timer = 0.0
+
+	move_and_slide()
+
+func _on_enemy_hit() -> void:
+	if not _alive:
+		return
+	_alive = false
+	# Death behaviour: pop up briefly then trigger signal
+	velocity = Vector2(0, -260)
+	await get_tree().create_timer(0.6).timeout
+	died.emit()
+	queue_free()
+
+func collect(value: int) -> void:
+	collected.emit(value)

+extends CharacterBody2D
+## Patrol enemy. Walks between two points; flips on wall; touches player = death.
+
+@export var patrol_distance := 120.0
+@export var speed := 80.0
+
+var _origin: float
+var _dir := 1.0
+@onready var sprite: ColorRect = $Sprite
+@onready var wall_ray: RayCast2D = $WallRay
+
+func _ready() -> void:
+	add_to_group("enemy")
+	_origin = position.x
+	# Layer 3 = enemy
+	collision_layer = 0b0000_0100
+	collision_mask = 0b0000_0001  # collide with world only
+	gravity_scale = 1.5
+
+func _physics_process(delta: float) -> void:
+	velocity.x = _dir * speed
+	if not is_on_floor():
+		velocity.y = min(velocity.y + get_gravity().y * delta, 900.0)
+	else:
+		velocity.y = 0.0
+
+	# Patrol bounds
+	if abs(position.x - _origin) > patrol_distance:
+		_dir *= -1.0
+	# Wall hit
+	if wall_ray.is_colliding():
+		_dir *= -1.0
+
+	# Update sprite facing
+	sprite.scale.x = 1.0 if _dir > 0.0 else -1.0
+
+	move_and_slide()
+
+func _on_body_entered(body: Node2D) -> void:
+	if body.is_in_group("player"):
+		if body.global_position.y < global_position.y - 8.0:
+			# Stomped from above: kill enemy
+			_die()
+		else:
+			body._on_enemy_hit()
+
+func _die() -> void:
+	# Pop animation + queue free
+	var t := create_tween()
+	t.tween_property(sprite, "scale", Vector2(sprite.scale.x, 0.0), 0.15)
+	t.tween_callback(queue_free)
