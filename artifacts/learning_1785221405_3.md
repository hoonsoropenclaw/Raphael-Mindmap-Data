+/**
+ * Pure reducer for the kanban board. All mutations go through here.
+ * Co-locates normalization (re-numbering `order` after moves).
+ */
+export function boardReducer(state: BoardState, action: BoardAction): BoardState {
+  switch (action.type) {
+    case 'ADD_TASK': {
+      const tasksInCol = tasksByColumn(state, action.task.columnId);
+      const next = { ...state.tasks, [action.task.id]: { ...action.task, order: tasksInCol.length } };
+      return { ...state, tasks: next };
+    }
+
+    case 'UPDATE_TASK': {
+      const existing = state.tasks[action.id];
+      if (!existing) return state;
+      return { ...state, tasks: { ...state.tasks, [action.id]: { ...existing, ...action.patch } } };
+    }
+
+    case 'DELETE_TASK': {
+      if (!state.tasks[action.id]) return state;
+      const { [action.id]: _, ...rest } = state.tasks;
+      return { ...state, tasks: rest };
+    }
+
+    case 'MOVE_TASK': {
+      const task = state.tasks[action.id];
+      if (!task) return state;
+      const fromCol = task.columnId;
+      const toCol = action.toColumn;
+      const newTasks = { ...state.tasks };
+
+      // Remove from source column, re-pack order.
+      const sourceList = tasksByColumn({ ...state, tasks: newTasks }, fromCol).filter(t => t.id !== action.id);
+      sourceList.forEach((t, i) => (newTasks[t.id] = { ...t, order: i }));
+
+      // Insert into destination column at desired index, re-pack order.
+      const destList = tasksByColumn({ ...state, tasks: newTasks }, toCol);
+      const insertAt = Math.max(0, Math.min(action.toIndex, destList.length));
+      const moved: Task = { ...task, columnId: toCol };
+      const without = destList.filter(t => t.id !== action.id);
+      without.splice(insertAt, 0, moved);
+      without.forEach((t, i) => (newTasks[t.id] = { ...t, columnId: toCol, order: i }));
+
+      return { ...state, tasks: newTasks };
+    }
+
+    case 'REORDER_COLUMN': {
+      // Reorder within the same column. Action payload is only meaningful when
+      // fromIndex/toIndex are valid; we re-pack order on completion.
+      const col = action.columnId;
+      const list = tasksByColumn(state, col);
+      const from = Math.max(0, Math.min(action.fromIndex, list.length - 1));
+      const to = Math.max(0, Math.min(action.toIndex, list.length - 1));
+      if (from === to) return state;
+      const next = list.slice();
+      const [picked] = next.splice(from, 1);
+      next.splice(to, 0, picked);
+      const newTasks = { ...state.tasks };
+      next.forEach((t, i) => (newTasks[t.id] = { ...t, order: i }));
+      return { ...state, tasks: newTasks };
+    }
+
+    case 'REPLACE_BOARD':
+      return action.state;
+
+    default:
+      return state;
+  }

+import type { BoardState, HistoryAction, HistoryState } from '../types';
+
+const MAX_HISTORY = 50;
+
+/**
+ * Time-travel reducer. Keeps a bounded stack of past states, the current
+ * snapshot, and a future stack. `COMMIT` with `coalesce: true` overwrites
+ * the most-recent past entry — useful for rapid edits (typing, drag) so
+ * one undo restores the prior burst, not the half-typed intermediate.
+ */
+export function historyReducer(state: HistoryState, action: HistoryAction): HistoryState {
+  switch (action.type) {
+    case 'COMMIT': {
+      if (action.coalesce) {
+        // Replace the top of the past stack with the previous present; keep
+        // the rest. Future is dropped on any new commit.
+        const top = state.past[state.past.length - 1] ?? state.present;
+        const newPast = [...state.past.slice(0, -1), top].slice(-MAX_HISTORY);
+        return {
+          past: newPast,
+          present: action.state,
+          future: [],
+          coalescing: true,
+        };
+      }
+      // First commit in a burst — push current present onto past, then advance.
+      const newPast = [...state.past, state.present].slice(-MAX_HISTORY);
+      return {
+        past: newPast,
+        present: action.state,
+        future: [],
+        coalescing: false,
+      };
+    }
+
+    case 'COALESCE_FLUSH':
+      return { ...state, coalescing: false };
+
+    case 'UNDO': {
+      if (state.past.length === 0) return state;
+      const previous = state.past[state.past.length - 1];
+      return {
+        past: state.past.slice(0, -1),
+        present: previous,
+        future: [state.present, ...state.future].slice(0, MAX_HISTORY),
+        coalescing: false,
+      };
+    }
+
+    case 'REDO': {
+      if (state.future.length === 0) return state;
+      const [next, ...rest] = state.future;
+      return {
+        past: [...state.past, state.present].slice(-MAX_HISTORY),
+        present: next as BoardState,
+        future: rest,
+        coalescing: false,
+      };
+    }
+
+    default:
+      return state;
+  }

+/**
+ * Debounce a value — return the most recent input only after it has
+ * stopped changing for `delay` ms. Used to throttle the search input
+ * so we don't re-filter the board on every keystroke.

+export function useDebounce<T>(value: T, delay = 180): T {
+  const [debounced, setDebounced] = useState(value);
+  useEffect(() => {
+    const id = window.setTimeout(() => setDebounced(value), delay);
+    return () => window.clearTimeout(id);
+  }, [value, delay]);
+  return debounced;
