+def _ensure_private_dir(path: Path) -> Path:
+    """Create the parent directory with mode 0700.
+

+    silently chmod'ing an existing shared directory (e.g. /tmp) is a
+    security anti-pattern. Reference: trial-and-error SOP "SQLite
+    directory must not chmod existing shared parents".

+    parent = path.parent
+    if parent.exists():

+        if mode & 0o077:
+            raise PermissionError(

+                f"mode {oct(mode)}. Fix the parent directory or set "
+                f"TODO_API_DB_PATH to a private location."

+        parent.mkdir(parents=True, exist_ok=True)
+        # umask-based create typically lands at 0755; tighten explicitly.
+        os.chmod(parent, 0o700)

+"""Rate limiter middleware.
+
+Sliding-window rate limiter keyed by client IP. Implementation choices
+and why:
+
+1. **Key by `request.client.host` only.** We do NOT parse X-Forwarded-For
+   or X-Real-IP in app code. Per trial-and-error SOP "Uvicorn Proxy Trust
+   Is a Server-Layer Boundary", the right place to trust proxy headers
+   is the Uvicorn runner (`--proxy-headers --forwarded-allow-ips ...`).
+   Reading headers in the app would let forged headers bypass the
+   Uvicorn allowlist.
+
+2. **In-memory, process-local.** This is a single-process API. If we
+   ever scale horizontally, swap for Redis. The state is intentionally
+   not thread-safe (we use a simple lock) because Uvicorn workers run
+   one request per worker thread at a time and the lock is cheap.
+
+3. **Sliding window via deque.** More accurate than a fixed bucket; we
+   evict timestamps older than the window when we check or record.
+
+4. **Returns Retry-After.** RFC 6585 says 429 responses SHOULD include
+   Retry-After (seconds). We compute it from the oldest timestamp in the
+   client's window.
+"""
+
+from __future__ import annotations
+
+import threading
+import time
+from collections import defaultdict, deque
+from typing import Deque, Iterable
+
+from fastapi import Request
+from fastapi.responses import JSONResponse
+from starlette.middleware.base import BaseHTTPMiddleware

+class RateLimiter:
+    """Sliding-window IP rate limiter."""
+
+    def __init__(self, max_requests: int, window_seconds: int) -> None:
+        if max_requests <= 0:
+            raise ValueError("max_requests must be positive")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be positive")
+        self.max_requests = max_requests
+        self.window_seconds = float(window_seconds)
+        self._buckets: dict[str, Deque[float]] = defaultdict(deque)
+        self._lock = threading.Lock()
+
+    def _now(self) -> float:
+        return time.monotonic()

+        """Test whether `key` may make a request right now.
+
+        Returns (allowed, retry_after_seconds). When allowed=True the
+        retry_after is 0. When allowed=False it is the number of seconds
+        the caller must wait before the oldest event in the window
+        expires.
+        """
+        now = self._now()
+        cutoff = now - self.window_seconds
+        with self._lock:
+            bucket = self._buckets[key]
+            # Evict old timestamps.

+                bucket.popleft()

+                return False, retry_after
+            return True, 0

+# Todo API — 實作思路與設計決策
+
+> 範圍：一個 FastAPI + SQLite 的待辦事項 API，含 rate limit 與完整單元測試。
+> 對象：接手維護者 / 程式審閱者 / 想知道「為什麼這樣設計」的人。
+
+## 1. 總體設計原則
+
+這個專案刻意走「同步、簡單、純 stdlib + 1 個輕量 ORM」的最小可用路徑，避免任何 async / 第三方依賴爆炸。可運行、可測試、能上 production 小流量，但不藏雷。
+
+三大設計綱領：
+
+1. **Sync SQLAlchemy** — 不引入 async session。trial-and-error SOP 已經把 async SQLAlchemy 的三個 pitfall（stale read / async-sync deadlock / `expire_on_commit=False` 漏掉）記錄得很清楚，而 todo CRUD 沒有任何 async benefit。
+2. **Fail-closed 預設** — 路徑權限、SQLite 父目錄、輸入驗證，全在啟動時就拒絕錯誤。絕不「先跑起來再說」。
+3. **測試即文件** — 41 個測試就是合約的執行版本。改 schema 的人會立刻看到哪幾個測試 break。
+
+## 2. 專案結構
+
+```
+todo-api/
+├── app/
+│   ├── __init__.py
+│   ├── config.py          # 設定 + 路徑權限校驗
+│   ├── database.py        # SQLAlchemy engine + Session + connect hook
+│   ├── models.py          # ORM 模型 (Todo)
+│   ├── schemas.py         # Pydantic 輸入/輸出模型
+│   ├── rate_limit.py      # 滑動視窗限流 + ASGI middleware
+│   ├── repository.py      # CRUD 邏輯層
+│   └── main.py            # FastAPI app factory + 路由
+├── tests/

+## 3. 各層決策
+
+### 3.1 設定層 `app/config.py`
+
+**為什麼用 dataclass(frozen=True) 而不是 pydantic-settings？**
+- 這個專案只有 3 個設定值（db_path、rate_limit、instance_id），pydantic-settings 多帶 5MB 相依跟 YAML/TOML loader，划不來。
+- `frozen=True` 讓設定物件在某個 thread 內是絕對不可變的，測試可以放心 hash 比較。
+
+**為什麼 fail-closed 不直接 `chmod` 父目錄？**
+- 試誤 SOP（trial-and-error 2026-07-28）明文警告：`/tmp` 或多人共享的目錄被 `chmod 0700` 會破壞其他租戶。
+- 正確做法：**新建**子目錄並設 0700；**拒絕**對外已是寬鬆 mode 的父目錄覆寫。

+### 3.2 資料層 `app/database.py`
+
+**為什麼選 SQLite + StaticPool？**
+- StaticPool 讓所有 thread 共用同一連線。FastAPI 預設是 thread pool，符合 SQLite 單寫者模型。
+- 避免「database is locked」與 connection pool 競爭，todos 流量與其用 QueuePool 配合重試邏輯，不如直接靜態池。
+
+**為什麼 connect hook 還要 chmod 0o600？**
+- SQLite 第一次連線才會建立檔案，預設 mode 0o644。
+- 連線 hook 是唯一能在「檔案剛被建立的那刻」插入權限檢查的地方。
+- 即使檔案已存在（0o644），也會被修正成 0o600，符合「私有資料」原則。
+- `.wal` / `.shm` sidecar 檔同處理。
+
+**為什麼 `expire_on_commit=False`？**
+- 防止 commit 後存取 lazy-loaded relationship 時 `DetachedInstanceError`。
+- 同步 session 預設即可，但顯式寫出更明確。

+### 決策 B：Rate limit 鍵只用 `request.client.host`
