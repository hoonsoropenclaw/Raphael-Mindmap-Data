+# ---------- CPU / Load ----------
+
+def loadavg() -> dict[str, float]:
+    """讀 /proc/loadavg，回傳 1/5/15 分鐘 load average 與 running/total。"""
+    try:
+        raw = Path("/proc/loadavg").read_text().strip().split()
+    except FileNotFoundError as e:
+        raise RuntimeError(f"/proc/loadavg 不存在：{e}") from e
+    # 格式: 0.45 0.27 0.21 1/362 1335860
+    if len(raw) < 3:
+        raise RuntimeError(f"/proc/loadavg 格式異常: {raw!r}")
+    running, total = (0, 0)
+    if len(raw) >= 4 and "/" in raw[3]:
+        try:
+            running, total = (int(x) for x in raw[3].split("/", 1))
+        except ValueError:
+            pass
+    return {
+        "load1": float(raw[0]),
+        "load5": float(raw[1]),
+        "load15": float(raw[2]),
+        "running_procs": running,
+        "total_procs": total,
+    }
+
+
+def cpu_count() -> int:
+    """邏輯 CPU 數（用於把 load 換算成相對值）。"""
+    try:
+        return os.cpu_count() or 1
+    except Exception:  # pragma: no cover — 容錯

+# ---------- Memory ----------
+
+def memory() -> dict[str, int]:
+    """讀 /proc/meminfo，回傳總/可用/已用 + swap（bytes）。"""
+    info: dict[str, int] = {}
+    text = Path("/proc/meminfo").read_text()
+    for line in text.splitlines():
+        m = re.match(r"^(\w+):\s+(\d+)\s*kB", line)
+        if not m:
+            continue
+        info[m.group(1)] = int(m.group(2)) * 1024  # kB → bytes
+    total = info.get("MemTotal", 0)
+    available = info.get("MemAvailable", info.get("MemFree", 0))
+    used = max(total - available, 0)
+    swap_total = info.get("SwapTotal", 0)
+    swap_free = info.get("SwapFree", 0)
+    swap_used = max(swap_total - swap_free, 0)

+        "mem_available": available,
+        "mem_used": used,
+        "mem_used_pct": round(100.0 * used / total, 1) if total else 0.0,
+        "swap_total": swap_total,

+## 1. 任務理解
+
+撰寫一個**自動化檢查 Linux 系統日誌與資源**的維運腳本，**超過閾值時發出警告**。在沒有專案既有檔案、沒有 Senior Architect 留下的 `FEEDBACK.md` 情況下，從零設計並實作。
+
+## 2. 設計決策
+
+### 2.1 為什麼選 Python（不是 bash）
+- 跨日誌格式（journalctl JSON + syslog 純文字）的健壯解析，Python 比 awk/sed 強太多
+- 跨 subprocess（journalctl、free、df、zcat）的失敗處理可以**結構化**（try/except、timeout、fallback），bash `pipefail` 在管線組合下行為不一致
+- 純函式層級的單元測試性 — metrics、logs、thresholds 三層各自可獨立驗證
+- **零依賴**：stdlib only，這台主機 PEP 668 環境也不用 pip install
+
+### 2.2 模組拆分原則
+不寫單檔 800 行怪物。每層一個職責：
+
+| 模組 | 職責 | 為什麼這樣切 |
+|---|---|---|
+| `monitor/metrics.py` | 採集資源（CPU/RAM/Disk/inode/uptime）| 純函式，可獨立 mock |
+| `monitor/logs.py` | 掃 journalctl + syslog，產生 `LogFinding` dataclass | 兩種來源互補（systemd 環境 + 純 syslog 環境）|
+| `monitor/thresholds.py` | 閾值規則 + `Alert` 產生 | 規則集中好維護 |
+| `monitor/alerter.py` | 三種 Sink：stderr / syslog / webhook | 介面乾淨（Protocol），可加新通道不動 core |
+| `monitor/report.py` | 純文字 + JSON formatter | 純函式，純輸出 |
+| `monitor/runner.py` | 串接：採集→評估→emit | CLI/daemon/cron 唯一入口 |
+| `linux_health_monitor.py` | CLI | 三模式：`once` / `daemon` / `json` |
+
+### 2.3 閾值分兩級（warning / critical）
+- 任一 critical → 整體 critical → exit 2
+- 任一 warning → 整體 warning → exit 1
+- 全 ok → exit 0
+
+cron 看到非 0 退出碼會自動寄信（預設給 root）；systemd 看到非 0 會重啟（搭配 `Restart=on-failure`）。
+
+### 2.4 防禦性設計
+- `set -Eeuo pipefail`（bash 包裝）+ try/except 包每個 subprocess（Python）
+- subprocess 一律 `timeout=` 避免卡死
+- /proc 解析錯誤不會中斷整個 metrics 採集（每個函式各自 try/except）
+- sink emit 失敗不應 crash 主流程（runner 包了 try/except）
+- Daemon 模式：signal handler 設全域 flag，sleep loop 每秒檢查，SIGTERM 不會卡滿 interval 秒
+- 任何單元失敗 → 回傳有 `error` 欄位的 dict/物件，不丟例外
