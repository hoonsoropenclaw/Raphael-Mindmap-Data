+"""Rate limiting via slowapi.
+
+Why slowapi:
+- It's the de-facto rate-limit lib for FastAPI, built on the same
+  limits library used by Flask-Limiter.
+- Uses the `Limit` decorator + a backing storage (default: in-memory),
+  which is fine for this simple API. Swap to Redis later for a cluster.
+
+The TWO non-obvious bits that trip people up:
+1. `slowapi.Limiter` must be created at import time and attached to
+   `app.state.limiter` + `app.state.exempt` in `main.py`. The `Limit`
+   decorator records the limiter by reference, so any module that uses
+   `@limiter.limit(...)` must import the SAME instance from this module.
+2. The `key_func` must produce a stable key per client. The default is
+   `get_remote_address`, which is great for a real deployment. Under
+   the test client every request looks like 127.0.0.1, which means
+   the test suite shares a quota; that's actually what we want for
+   the rate-limit *test* but means we need a WAY to bypass it for
+   other tests. We achieve this with an env-overridable key_func:
+   when `RATE_LIMIT_KEY_HEADER` is set we use a header value so tests
+   can set a unique X-Test-Client per request.
+"""
+from __future__ import annotations
+
+import os
+
+from fastapi import Request
+from slowapi import Limiter
+from slowapi.util import get_remote_address
+
+from app.config import get_settings
+
+
+def _key_func(request: Request) -> str:
+    """Default: IP. Tests can override via env var to use a header so
+    each test request gets a fresh quota.
+    """
+    header_name = os.getenv("RATE_LIMIT_KEY_HEADER")
+    if header_name:
+        value = request.headers.get(header_name)
+        if value:
+            return value
+    return get_remote_address(request) or "anonymous"
+
+
+# Single shared instance. Importing this module anywhere gives you the
+# same Limiter — that matters for the `@limiter.limit(...)` decorator.
+limiter: Limiter = Limiter(
+    key_func=_key_func,
+    default_limits=[get_settings().rate_limit_default],
+    headers_enabled=True,
+)
+
+
