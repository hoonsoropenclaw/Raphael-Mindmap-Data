+# ``sqlite+aiosqlite:///`` scheme. We translate at the edge so callers (env
+# vars, tests) can use the familiar ``sqlite:///`` form.
+def _driver_aware_url(url: str) -> str:
+    if url.startswith("sqlite:///") and "+aiosqlite" not in url:
+        return url.replace("sqlite:///", "sqlite+aiosqlite:///", 1)

+     HTTP layer.
+  2. It keeps the route handlers thin: they translate HTTP <-> service calls.
+"""
+from __future__ import annotations
+
+from sqlalchemy import select

+
+from app.models.todo import Todo
+from app.schemas.todo import TodoCreate, TodoUpdate
+
+
+class TodoNotFoundError(Exception):
+    """Raised when a todo id does not exist. Caught by the route layer and
+    mapped to a 404."""
+
+    def __init__(self, todo_id: int) -> None:
+        super().__init__(f"todo {todo_id} not found")
+        self.todo_id = todo_id

+class TodoService:
+    """Stateless. Each call uses the supplied session — the session is owned
+    by the request, not the service."""
+
+    def __init__(self, session: AsyncSession) -> None:
+        self._session = session
+
+    async def list(self, *, limit: int = 100, offset: int = 0) -> list[Todo]:
+        """Return todos newest-first, clamped by limit/offset."""
+        stmt = select(Todo).order_by(Todo.id.desc()).limit(limit).offset(offset)
+        result = await self._session.execute(stmt)
+        return list(result.scalars().all())
+
+    async def get(self, todo_id: int) -> Todo:
+        """Return a single todo, or raise ``TodoNotFoundError``."""
+        todo = await self._session.get(Todo, todo_id)
+        if todo is None:
+            raise TodoNotFoundError(todo_id)
+        return todo
+
+    async def create(self, payload: TodoCreate) -> Todo:
+        """Insert a new todo and commit."""
+        todo = Todo(
+            title=payload.title,
+            description=payload.description,
+            is_done=False,
+        )
+        self._session.add(todo)
+        await self._session.commit()

+"""FastAPI application factory + rate limiter wiring.

+Rate limiting strategy
+----------------------
+We use ``slowapi`` because it is the de-facto FastAPI rate-limit library and
+it supports the per-IP sliding window we want. The limit is set per IP to
