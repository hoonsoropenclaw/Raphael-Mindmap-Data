+  an async/sync dialect worry and complicate testing; for one table
+  hand-rolled SQL is clearer and faster.
+* A single ``sqlite3.Connection`` per Repository, guarded by ``threading.Lock``.
+  FastAPI runs sync endpoints in a threadpool, so concurrent requests
+  WILL hit the same connection; the lock keeps it safe.
+* ``check_same_thread=False`` lets us share the connection across threads
+  in the first place; the lock provides the serialization we need.

+"""In-memory sliding-window rate limiter.
+
+Design notes:
+
+* **Algorithm**: per-key sliding window with a deque of timestamps.
+  Simple, easy to reason about, easy to test. A token bucket would be
+  burstier; a fixed-window counter would be simpler but allowed 2x burst
+  at window boundaries. Sliding window is the right middle ground for
+  "request rate per minute" semantics.
+* **Clock injection**: the limiter takes a ``clock`` callable that returns
+  a monotonic-like float. Tests pass a fake clock; production passes
+  ``time.monotonic``. This avoids any ``sleep`` in tests.
+* **Thread safety**: protected by ``threading.Lock``. FastAPI sync endpoints
+  run in a threadpool; the lock makes concurrent clients safe.
+* **Process-local only**: this is a deliberate baseline. The TODO comment in
+  the class docstring calls out the multi-worker / multi-instance constraint
+  — swap to Redis (``INCR`` + ``EXPIRE``) when you go multi-instance.
+* **Standard 429 headers**: emits ``Retry-After`` (seconds) plus
+  ``X-RateLimit-Limit/Remaining/Reset`` so clients can self-throttle.
+"""
+
+from __future__ import annotations
+
+import math
+import threading
+import time
+from collections import defaultdict, deque
+from typing import Callable, Deque, Dict, Optional, Tuple
+
+
+class RateLimiter:
+    """Sliding-window rate limiter, keyed by client identifier."""
+
+    def __init__(
+        self,
+        max_requests: int = 60,
+        window_seconds: float = 60.0,
+        *,
+        clock: Callable[[], float] = time.monotonic,
+    ) -> None:
+        if max_requests <= 0:
+            raise ValueError("max_requests must be positive")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be positive")
+        self._max = max_requests
+        self._window = float(window_seconds)
+        self._clock = clock
+        self._hits: Dict[str, Deque[float]] = defaultdict(deque)
+        self._lock = threading.Lock()
+
+    @property
+    def max_requests(self) -> int:
+        return self._max
+
+    @property
+    def window_seconds(self) -> float:
+        return self._window
+
+    def check(self, key: str) -> Tuple[bool, int, float]:
+        """Try to consume one slot for ``key``.
+
+        Returns ``(allowed, remaining, retry_after_seconds)``:
+            * ``allowed`` — True if the request is allowed (and was recorded).
+            * ``remaining`` — slots left after this request (-1 if denied).
+            * ``retry_after_seconds`` — 0 if allowed; else seconds until the
+              oldest hit slides out of the window.
+        """
+        now = self._clock()
+        cutoff = now - self._window
+
+        with self._lock:
+            hits = self._hits[key]
+            # Drop expired entries from the front of the deque
+            while hits and hits[0] <= cutoff:
+                hits.popleft()
+

+"""FastAPI app factory.
+
+Why an app factory:
+
+* It lets tests construct an app with a sqlite ``:memory:`` database and a
+  tight rate-limit, then dispose of the whole thing per test.
+* Lifespan context manager owns the repository connection: open on
+  startup, close on shutdown. Module-level global state would leak
+  connections between test runs.
+
+Endpoints:
+    GET  /health       — liveness probe (unauthenticated, not rate-limited)
+    *    /todos/*      — see routers/todos.py
+"""
+
+from __future__ import annotations
+
+from contextlib import asynccontextmanager
+from typing import AsyncIterator
+
+from fastapi import FastAPI
+
+from .database import TodoRepository
+from .rate_limit import RateLimiter
+from .routers import todos as todos_router
+
+DEFAULT_DATABASE_URL = "sqlite:///./todos.db"
+DEFAULT_MAX_REQUESTS = 60
+DEFAULT_WINDOW_SECONDS = 60.0
+
+
+@asynccontextmanager
+async def lifespan(app: FastAPI) -> AsyncIterator[None]:
+    """Manage repo lifecycle. Limiter has no resources of its own."""
+    repo = app.state.repo  # type: ignore[attr-defined]
+    try:
+        yield
+    finally:
+        repo.close()
+
+
+def create_app(
+    database_url: str = DEFAULT_DATABASE_URL,
+    max_requests: int = DEFAULT_MAX_REQUESTS,
+    window_seconds: float = DEFAULT_WINDOW_SECONDS,
+) -> FastAPI:
+    """Build a FastAPI app with the given config.
+
+    All knobs default to production-friendly values; tests override them
+    via the ``app_factory`` fixture in conftest.
+    """
+    app = FastAPI(
+        title="FastAPI Todo API",
+        version="0.1.0",
+        lifespan=lifespan,
+    )
+
+    # Inject dependencies into app.state — endpoints pull them through
+    # ``Depends(get_repo)`` / ``Depends(get_limiter)``.
+    app.state.repo = TodoRepository(database_url=database_url)
+    app.state.limiter = RateLimiter(
+        max_requests=max_requests,
+        window_seconds=window_seconds,
+    )
+
+    @app.get("/health", tags=["meta"])
+    def health() -> dict:
+        return {"status": "ok"}
+
+    app.include_router(todos_router.router)
+    return app
+
+

+fastapi-todo-api
+=================
+
+Minimal FastAPI todo API. SQLite for storage. In-memory sliding-window
+rate limiter. Pydantic v2 schemas. Thread-safe. Comes with a 54-test
+unit suite covering repo, rate-limit, and HTTP layer.
+
+Why this layout
+---------------
+
+* ``app/main.py`` — ``create_app()`` factory; tests use it to inject a
+  in-memory SQLite database and a tight rate-limit per-test.
+* ``app/database.py`` — ``TodoRepository`` over stdlib ``sqlite3``,
+  guarded by ``threading.Lock``. One table, hand-rolled SQL.
+* ``app/rate_limit.py`` — sliding-window rate limiter per client key
+  (X-Forwarded-For first hop, then ``request.client.host``).
+  Clock injected for deterministic tests.
+* ``app/schemas.py`` — Pydantic v2 request/response models.
+* ``app/routers/todos.py`` — HTTP routes, sync (SQLite is sync).
+* ``tests/`` — pytest suite, ``TestClient``, fake clock for sliding-window.
+
+Run it
+------
+
+```
+pip install -e .[dev]
+uvicorn app.main:app --reload
+curl http://127.0.0.1:8000/health
+```
+
+Test it
+-------
+
+```
+pytest
+# → 54 passed in ~0.6s, coverage 95%
+```
+
+Default rate-limit is 60 req / 60 s per client.
+
+API surface
+-----------
+
+```
+GET    /health                              → {"status": "ok"}
+POST   /todos           {"title":...}      → 201 TodoRead
+GET    /todos?skip=0&limit=100             → 200 [TodoRead]
+GET    /todos/{id}                          → 200 TodoRead | 404
+PATCH  /todos/{id}      {"title"?, "completed"?}  → 200 | 404 | 422
+DELETE /todos/{id}                          → 204 | 404
+```
