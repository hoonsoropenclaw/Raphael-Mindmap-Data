+"""
+SQLite database layer for the Todo API.
+
+Design notes:
+- Pure stdlib `sqlite3` to avoid a SQLAlchemy dependency for a small project.
+- Connection-per-request via FastAPI dependency injection — keeps things thread-safe
+  under uvicorn workers without the bookkeeping of a connection pool.
+- `row_factory = sqlite3.Row` so rows behave like dicts (read by column name).
+- Schema initialization is idempotent: safe to call on every startup.
+"""
+from __future__ import annotations
+
+import sqlite3
+from contextlib import contextmanager
+from typing import Iterator
+
+# Schema kept in one place so the table is easy to evolve.
+SCHEMA = """
+CREATE TABLE IF NOT EXISTS todos (
+    id          INTEGER PRIMARY KEY AUTOINCREMENT,
+    title       TEXT    NOT NULL,
+    completed   INTEGER NOT NULL DEFAULT 0,
+    created_at  TEXT    NOT NULL DEFAULT CURRENT_TIMESTAMP
+);
+CREATE INDEX IF NOT EXISTS idx_todos_completed ON todos(completed);
+"""
+
+
+def _connect(db_path: str) -> sqlite3.Connection:
+    conn = sqlite3.connect(db_path, check_same_thread=False, isolation_level=None)
+    conn.row_factory = sqlite3.Row
+    # Foreign keys are off by default in SQLite; enable for safety even though
+    # we don't currently have FKs — future-proofs any relationship additions.
+    conn.execute("PRAGMA foreign_keys = ON")
+    conn.execute("PRAGMA journal_mode = WAL")  # better concurrency for readers
+    return conn
+
+
+def init_db(db_path: str) -> None:
+    """Create tables if they don't exist. Idempotent."""
+    with sqlite3.connect(db_path) as conn:
+        conn.executescript(SCHEMA)
+
+
+@contextmanager
+def get_connection(db_path: str) -> Iterator[sqlite3.Connection]:
+    """Context manager that yields a connection and commits/rolls back on exit."""
+    conn = _connect(db_path)
+    try:
+        yield conn
+        conn.commit()
+    except Exception:
+        conn.rollback()
+        raise
+    finally:
+        conn.close()

+"""
+In-memory sliding-window rate limiter.
+
+Design notes:
+- Sliding window (rather than fixed bucket) so a client can't burst twice at the
+  edge of a fixed window to double their quota.
+- Per-key state in a dict guarded by a single `threading.Lock` — fine for a
+  single-process uvicorn. A multi-worker deployment would need a shared store
+  (Redis); that's documented in REVIEW.md.
+- LRU-bounded: at most `max_keys` distinct keys are tracked. When the dict
+  overflows, the oldest key (by insertion order) is evicted. This prevents an
+  attacker from exhausting memory by spamming unique keys.
+- Exposes `RateLimiter` as a class so tests can inject a tiny `max_requests` /
+  `window_seconds` without monkey-patching.
+"""
+from __future__ import annotations
+
+import threading
+import time
+from collections import deque
+from dataclasses import dataclass
+
+
+class RateLimitExceeded(Exception):
+    """Raised when a key exceeds its allowed request count within the window."""
+
+    def __init__(self, retry_after: float, limit: int, window: float):
+        self.retry_after = retry_after
+        self.limit = limit
+        self.window = window
+        super().__init__(
+            f"Rate limit exceeded: {limit} requests per {window:.0f}s "
+            f"(retry after {retry_after:.2f}s)"
+        )
+
+
+@dataclass
+class _Bucket:
+    """One sliding window for a single key."""
+    timestamps: deque  # type: ignore[type-arg]
+
+
+class RateLimiter:
+    """Thread-safe sliding-window rate limiter."""
+
+    def __init__(
+        self,
+        max_requests: int = 10,
+        window_seconds: float = 60.0,
+        max_keys: int = 10_000,
+    ):
+        if max_requests < 1:
+            raise ValueError("max_requests must be >= 1")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be > 0")
+        self.max_requests = max_requests
+        self.window_seconds = window_seconds
+        self.max_keys = max_keys
+        self._buckets: dict[str, _Bucket] = {}
+        self._lock = threading.Lock()
+
+    # -- internal helpers --------------------------------------------------
+
+    def _evict_expired(self, bucket: _Bucket, now: float) -> None:
+        cutoff = now - self.window_seconds
+        ts = bucket.timestamps
+        while ts and ts[0] <= cutoff:

+## 3. 關鍵設計決策
+
+### 3.1 SQLite 用 stdlib，不用 SQLAlchemy
+**理由**：5 個表/查詢、零 relationship、零 migration 需求。SQLAlchemy 會讓依賴、import time、測試 fixture 複雜度都上去。
+**取捨**：若未來加關聯表或 migration（Alembic），需重構；目前不值得。
+
+### 3.2 連線管理：每個 request 一條連線
+透過 FastAPI `Depends(lambda: app.state.db_path)` 拿到 path，呼叫 CRUD 函式時用 `get_connection()` context manager 開連、commit/rollback、close。
+**為什麼不開 pool**：`sqlite3` 連線極輕（檔案型），開關成本 < 1ms，pool 的執行緒安全 bookkeeping 反而是負擔。
+**WAL 模式**已開：reader 不會被 writer block，對 FastAPI sync endpoint 友善。
+
+### 3.3 Rate limiter：滑動視窗 (sliding window) 而非固定桶 (fixed bucket)
+**理由**：固定桶有「邊界雙倍 burst」漏洞 — 客戶端在第 59 秒打滿 quota、第 61 秒再打滿 = 2 秒內送 2× quota。滑動視窗沒有這個問題。
+**實作**：每個 key 一個 `deque[float]`，每次 `check()` 時先 pop 掉超出 window 的舊時間戳，再判斷剩餘數量。
+**時間源用 `time.monotonic()`** — 不受系統時鐘調整影響。
+
+### 3.4 Rate limiter：per-key 記憶體上限 (max_keys)
+**理由**：攻擊者可以送 `X-Forwarded-For: 1.2.3.N`（每個 N 不同）打爆 process 記憶體。
+**實作**：`_buckets` dict 達 `max_keys` 時 evict 最舊 key（dict 保留插入順序）。預設 10k key 對 single-process uvicorn 是合理上限。
+**已知限制**：multi-worker uvicorn 各 process 獨立計數 → 配額會被 N 倍放大。生產環境需換 Redis / token bucket with shared state。
+
+### 3.5 Rate limit 用 IP，不要求登入
+題目沒要求 auth。用 `X-Forwarded-For` 第一段（相容 reverse proxy），fallback `request.client.host`。
+**已知限制**：同一 NAT 後多人共享 IP，會互搶 quota。需要精確 per-user 必須先有 auth（這也是非目標之一）。
