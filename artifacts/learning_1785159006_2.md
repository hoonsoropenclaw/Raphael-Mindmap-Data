+"""SQLite connection and schema management.
+
+Design notes
+------------
+* One connection per request, opened via FastAPI dependency injection.
+* `check_same_thread=False` is required because FastAPI/Starlette can run the
+  app across worker threads when served by uvicorn. We rely on SQLite's own
+  per-process locking; writers serialize naturally.
+* `PRAGMA foreign_keys = ON` would be added if we ever reference other tables;
+  it's intentionally omitted here because there are no FK relationships yet.
+* Schema is created lazily on first request via the `get_db` dependency so
+  tests can use `:memory:` and ephemeral file DBs without a bootstrap step.
+"""
+from __future__ import annotations
+
+import sqlite3
+from contextlib import contextmanager
+from pathlib import Path
+from typing import Iterator
+
+# Public so tests can import the canonical schema string for assertions.
+SCHEMA_SQL = """
+CREATE TABLE IF NOT EXISTS todos (
+    id            INTEGER PRIMARY KEY AUTOINCREMENT,
+    title         TEXT    NOT NULL,
+    description   TEXT    NOT NULL DEFAULT '',
+    completed     INTEGER NOT NULL DEFAULT 0 CHECK (completed IN (0, 1)),
+    created_at    TEXT    NOT NULL DEFAULT (datetime('now')),
+    updated_at    TEXT    NOT NULL DEFAULT (datetime('now'))
+);
+
+CREATE TRIGGER IF NOT EXISTS todos_updated_at
+AFTER UPDATE ON todos
+FOR EACH ROW
+BEGIN
+    UPDATE todos SET updated_at = datetime('now') WHERE id = OLD.id;
+END;
+"""
+
+
+def connect(db_path: str | Path) -> sqlite3.Connection:
+    """Open a SQLite connection with row factory set to sqlite3.Row."""
+    conn = sqlite3.connect(str(db_path), check_same_thread=False, isolation_level=None)
+    conn.row_factory = sqlite3.Row
+    # WAL gives better concurrency for a small server even though we run one
+    # request per connection here; it's cheap insurance.
+    conn.execute("PRAGMA journal_mode=WAL;")
+    conn.execute("PRAGMA synchronous=NORMAL;")
+    return conn
+
+
+def init_schema(conn: sqlite3.Connection) -> None:
+    """Create tables/triggers if missing. Idempotent."""
+    conn.executescript(SCHEMA_SQL)
+
+
+@contextmanager
+def session_scope(conn: sqlite3.Connection) -> Iterator[sqlite3.Connection]:
+    """Transaction helper. Commits on success, rolls back on exception."""
+    try:

+        yield conn
+        conn.execute("COMMIT;")
+    except Exception:

+"""In-process token-bucket-ish rate limiter.
+
+Why in-process and not slowapi/Redis:
+* The task asks for a simple rate limit on a single-process demo project.
+* A pure-stdlib implementation keeps the dependency surface zero and is easy
+  to unit-test deterministically (no time mocking needed for the window).
+* Drop-in: pass a `Limiter` instance via FastAPI dependency, swap for a Redis
+  implementation later if you scale out.
+
+Algorithm
+---------
+Sliding-window counter per key. We keep a deque of request timestamps per
+key and reject when `len(deque) >= limit` within `window_seconds`.
+
+Tradeoffs:
+* O(requests_in_window) memory per key. Fine for ≤ a few hundred per key.
+* Not cluster-safe — if you run multiple uvicorn workers, swap for Redis.
+* Clock is monotonic (`time.monotonic()`) so it isn't affected by NTP jumps.
+"""
+from __future__ import annotations
+
+import threading
+import time
+from collections import deque
+from dataclasses import dataclass
+
+
+@dataclass(frozen=True)
+class LimitDecision:
+    allowed: bool
+    remaining: int
+    retry_after_seconds: float  # 0 when allowed
+
+
+class Limiter:
+    def __init__(self, *, limit: int, window_seconds: float):
+        if limit <= 0:
+            raise ValueError("limit must be > 0")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be > 0")
+        self.limit = limit
+        self.window = float(window_seconds)
+        self._lock = threading.Lock()
+        # key -> deque[float]
+        self._hits: dict[str, deque[float]] = {}
+
+    # Test seam: force a virtual "now" so we can assert deterministic eviction.
+    def _now(self) -> float:
+        return time.monotonic()
+
+    def check(self, key: str, *, now: float | None = None) -> LimitDecision:
+        """Register a hit and decide whether to allow it.
+
+        Returns LimitDecision; caller is responsible for translating to HTTP.
+        """
+        ts = self._now() if now is None else now
+        cutoff = ts - self.window
+        with self._lock:
+            q = self._hits.setdefault(key, deque())
+            # evict expired timestamps
+            while q and q[0] <= cutoff:
+                q.popleft()
+
+            if len(q) >= self.limit:
+                # retry_after = how long until the oldest hit expires
+                retry = max(0.0, (q[0] + self.window) - ts)
+                return LimitDecision(

+# Implementation Notes — Todo API
+
+This file records the **design reasoning**, **tradeoffs**, and **future-work**
+items behind the FastAPI todo API in this directory. It complements the
+high-level `README.md` with the *why*, not just the *what*.
+
+---
+
+## 1. High-level shape
+
+```
+┌─────────────┐   HTTP    ┌──────────────┐   SQL    ┌──────────┐
+│  TestClient │ ───────▶  │  FastAPI app │ ───────▶ │  SQLite  │
+│  / curl     │           │  app/main.py │          │  file DB │
+└─────────────┘           │  + routers/  │          └──────────┘
+                          └──────────────┘
+                                  │
+                          ┌───────┴────────┐
+                          │ repository.py  │  (no FastAPI imports)
+                          │ rate_limit.py  │  (no FastAPI imports)
+                          │ db.py          │  (sqlite3 stdlib)
+                          └────────────────┘
+```
+
+Three architectural seams:
+
+1. **App factory** (`app/main.py:create_app`) — every component (DB, repo,
+   limiter) is attached to `app.state`. Tests build a fresh app per run.
+2. **Repository layer** — the SQL boundary. Routers never write SQL.
+3. **Limiter as a dependency** — `enforce_rate_limit` is a FastAPI
+   `Depends`, so it can be disabled or replaced per-route in tests.
+
+---
+
+## 2. SQLite schema
+
+```sql
+CREATE TABLE todos (
+    id          INTEGER PRIMARY KEY AUTOINCREMENT,
+    title       TEXT    NOT NULL,
+    description TEXT    NOT NULL DEFAULT '',
+    completed   INTEGER NOT NULL DEFAULT 0 CHECK (completed IN (0, 1)),
+    created_at  TEXT    NOT NULL DEFAULT (datetime('now')),
+    updated_at  TEXT    NOT NULL DEFAULT (datetime('now'))
+);
+
+CREATE TRIGGER todos_updated_at
+AFTER UPDATE ON todos
+FOR EACH ROW
+BEGIN
+    UPDATE todos SET updated_at = datetime('now') WHERE id = OLD.id;
+END;
+```
+
+### Why these choices
+
+- **`INTEGER` for `completed`** — SQLite has no native bool, and 0/1 plays
+  nicer with Pydantic adapters than storing `'t'`/`'f'`. The router
+  converts to `bool` in `_row_to_read`.
+- **`CHECK` constraint** — defends against bad writes at the lowest layer.
+- **`updated_at` via trigger** — keeps the application layer focused on
+  domain logic, and there's exactly one place where the rule lives.
+- **`journal_mode=WAL` + `synchronous=NORMAL`** — WAL gives better
+  read/write concurrency on a busy server, and NORMAL is durable enough
+  for a demo app without sacrificing throughput.
