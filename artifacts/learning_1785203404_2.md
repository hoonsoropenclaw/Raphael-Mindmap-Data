+"""Minimal dependency-free Telegram weather bot using wttr.in."""
+from __future__ import annotations
+
+import json
+import logging

+import signal
+import time
+from dataclasses import dataclass
+from urllib.error import HTTPError, URLError
+from urllib.parse import quote
+from urllib.request import Request, urlopen
+
+LOG = logging.getLogger("telegram-weather-bot")
+
+@dataclass(frozen=True)
+class Weather:
+    location: str
+    condition: str
+    temperature_c: str
+    feels_like_c: str
+    humidity: str
+    wind_kmph: str
+
+
+def _http_json(url: str, timeout: int = 20) -> dict:
+    request = Request(url, headers={"User-Agent": "telegram-weather-bot/1.0"})
+    with urlopen(request, timeout=timeout) as response:
+        return json.load(response)
+
+
+def fetch_weather(location: str) -> Weather:
+    location = location.strip()
+    if not location or len(location) > 100:
+        raise ValueError("請輸入 1–100 個字元的地點。")
+    data = _http_json(f"https://wttr.in/{quote(location)}?format=j1")
+    current = data["current_condition"][0]
+    return Weather(
+        location=location,
+        condition=current.get("weatherDesc", [{"value": "未知"}])[0]["value"],
+        temperature_c=current["temp_C"],
+        feels_like_c=current["FeelsLikeC"],
+        humidity=current["humidity"],
+        wind_kmph=current["windspeedKmph"],
+    )
+
+
+def format_weather(weather: Weather) -> str:
+    return (
+        f"📍 {weather.location}\n"
+        f"🌤️ 天氣：{weather.condition}\n"
+        f"🌡️ 溫度：{weather.temperature_c}°C（體感 {weather.feels_like_c}°C）\n"
+        f"💧 濕度：{weather.humidity}%\n"
+        f"💨 風速：{weather.wind_kmph} km/h"
+    )
+
+
+class TelegramAPI:
+    def __init__(self, token: str):

+# Telegram 天氣機器人實作思路
+
+## 架構
+- 使用 Python 標準函式庫 `urllib` 呼叫 Telegram Bot API 與 wttr.in，零第三方依賴。
+- 使用 long polling（`getUpdates`），適合沒有公開 HTTPS endpoint 的主機。
+- Telegram 與天氣 API 分離在同一腳本的不同責任函式：`TelegramAPI`、`fetch_weather`、`format_weather`、`run`。
+
+## 主要流程
+1. 從 `TELEGRAM_BOT_TOKEN` 讀取 Bot token。
+2. 以 `getUpdates(timeout=30)` 接收文字訊息。
+3. 使用 `offset = update_id + 1` 避免重複處理更新。
+4. 將訊息文字 URL encode 後查詢 wttr.in JSON API。
+5. 提取目前天氣、溫度、體感、濕度與風速，回傳 Telegram。
+6. 捕捉網路、JSON、API 與輸入錯誤，避免 polling loop 因單次錯誤終止。
+7. SIGINT/SIGTERM 設定 graceful shutdown flag。
+
+## 防護
+- `TELEGRAM_ALLOWED_USERS` 可用逗號分隔的 Telegram user ID 建立 allowlist；未設定時允許所有使用者。
+- 每位使用者 3 秒冷卻，避免濫用免費天氣 API。
+- 地點輸入限制 1–100 字元。
+- 直接使用 Unicode emoji，避免 surrogate pair 編碼錯誤。
+
+## 執行
+```bash
+export TELEGRAM_BOT_TOKEN='從 BotFather 取得的 token'
+# 可選：限制使用者，例如 export TELEGRAM_ALLOWED_USERS='123456789'
+python3 telegram_weather_bot.py
+```
+
+## 驗證
+```bash
+python3 -m py_compile telegram_weather_bot.py
+```
+
+注意：沒有真實 Telegram token 時，只能完成語法與靜態驗證，不能聲稱已連線到 Telegram 實測。
