📊 為什麼要優化?

2D 遊戲的物理瓶頸常見於:GC 抖動、子彈/敵人爆量、O(n²) 碰撞、跨硬體不一致。本實作以 Godot 4.x GDScript 為主,但核心概念跨引擎通用(Unity / Unreal 也用同樣的招式)。

策略解決的痛點預期效益
Object Pool子彈/粒子 new+delete 抖動消除 GC spike、配置時間歸零
Spatial HashO(n²) 碰撞偵測1000 物件從 1M → 幾千次配對
Fixed Timestep不同幀率物理行為不一致跨硬體行為確定性
LOD遠距離物件無謂更新CPU 從 80% → 30%
Batch Sprite大量 Sprite draw calldraw_call 砍半以上
Signal Debounce輸入/碰撞訊號高頻觸發輸入平滑、技能冷卻

## 從池中取得物件
func acquire() -> Node2D:
    _total_acquires += 1
    var obj: Node2D = null
    if _available.is_empty():
        if auto_grow and get_pool_total() < max_size:
            obj = pooled_scene.instantiate() as Node2D
            add_child(obj)
            _total_grew += 1
        else: return null  # 池滿
    else:
        obj = _available.pop_back()  # O(1) 取出
    obj.visible = true
    obj.set_process(true)
    return obj

## 歸還物件
func release(obj: Node2D) -> void:
    obj.visible = false
    obj.set_process(false)
    _available.append(obj)  # O(1) 放回

func _cell_key(pos: Vector2) -> Vector2i:
    return Vector2i(
        int(floor(pos.x / cell_size)),
        int(floor(pos.y / cell_size))
    )

func query_nearby(pos: Vector2, radius: float) -> Array:
    var results: Array = []
    var center = _cell_key(pos)
    var cell_radius = int(ceil(radius / cell_size))
    for dx in range(-cell_radius, cell_radius + 1):
        for dy in range(-cell_radius, cell_radius + 1):
            var key = _key_to_str(Vector2i(
                center.x + dx, center.y + dy))
            if _buckets.has(key):
                results.append_array(_buckets[key])
    return results

## Glenn Fiedler 的 "Fix Your Timestep!" 經典實作
func advance(real_delta: float) -> void:
    var clamped = clampf(real_delta, 0.0, 0.1)
    _accumulator += clamped
    while _accumulator >= fixed_dt \
        and _steps_this_frame < max_steps_per_frame:
        step.emit(fixed_dt)         # 固定 dt = 1/60
        _accumulator -= fixed_dt
        _steps_this_frame += 1

const PROCESS_FREQ = {
    Level.L0_NEAR: 1,    # 每幀
    Level.L1_MID:  2,    # 每 2 幀
    Level.L2_FAR:  6,    # 每 6 幀
    Level.L3_SLEEP: 999  # 凍結
}

func should_process_this_frame(obj) -> bool:
    var entry = _entries[obj.get_instance_id()]
    var freq = PROCESS_FREQ[entry.current_level]
    if freq == 999: return false
    entry.frame_counter += 1
    if entry.frame_counter >= freq:
        entry.frame_counter = 0
        return true
    return false

func _draw() -> void:
    var rect = Rect2(-sprite_size * 0.5, sprite_size)
    for inst in _instances:
        var xform = Transform2D(inst.rot, inst.pos)

## Throttle:第一個會過,後續 N 秒內都擋下
func try_throttle(name: StringName, now: float) -> bool:
    if _last_emit_time.has(name) \
        and now - _last_emit_time[name] < _interval[name]:
        _total_dropped += 1
        return false
    _last_emit_time[name] = now
    throttled.emit(name)
    return true

## Cooldown:第一次 emit,等待 N 秒後才發 ready
func try_cooldown(name, now) -> bool:
    if _cooldown_end.has(name) and now < _cooldown_end[name]:
        return false
    _cooldown_end[name] = now + _interval[name]
    var timer = get_tree().create_timer(_interval[name])
    timer.timeout.connect(func(): cooldown_ready.emit(name))
    return true
