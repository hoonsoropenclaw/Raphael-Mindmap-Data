// app/api/transcribe/route.ts
// Next.js 14 App Router · Edge-safe 音訊轉錄端點
import { NextRequest, NextResponse } from 'next/server';
import { transcribeAudio, getProvider } from '@/lib/whisper';

// 關閉預設 body parser（音訊是 multipart/form-data）
export const config = {
  api: { bodyParser: false },
  // Whisper-large 可能跑 60s+，Edge runtime 會 timeout
  runtime: 'nodejs',
  maxDuration: 60,
};

export async function POST(req: NextRequest) {
  try {
    const form = await req.formData();

    // 1. 取得音訊檔
    const file = form.get('file') as File | null;
    if (!file) {
      return NextResponse.json(
        { error: 'No audio file in form-data' },
        { status: 400 }
      );
    }

    // 2. 解析可選參數
    const language = form.get('language')?.toString();
    const model = form.get('model')?.toString() ?? 'whisper-1';
    const withTimestamps = form.get('timestamps') === 'true';
    const prompt = form.get('prompt')?.toString();

    // 3. 取得 provider 實例（單例模式快取）
    const provider = getProvider(process.env.WHISPER_PROVIDER ?? 'openai');

    // 4. 呼叫 Whisper
    const result = await transcribeAudio(provider, {
      file,
      language,
      model,
      prompt,
      responseFormat: withTimestamps ? 'verbose_json' : 'json',
    });

    // 5. 統一回傳格式
    return NextResponse.json({
      ok: true,
      provider,
      text: result.text,
      language: result.language,
      duration: result.duration,
      segments: withTimestamps ? result.segments : undefined,
      words: withTimestamps ? result.words : undefined,
      model: result.model,
      timestamp: new Date().toISOString(),
    });
  } catch (err: any) {
    console.error('[transcribe]', err);
    return NextResponse.json(
      { ok: false, error: err.message ?? 'Transcription failed' },
      { status: 500 }
    );
  }
}

// 健康檢查端點
export async function GET() {
  const provider = process.env.WHISPER_PROVIDER ?? 'openai';
  return NextResponse.json({
    ok: true,
    provider,
    model: process.env.WHISPER_MODEL ?? 'whisper-1',
    version: '1.0.0',
  });
}

    


    


// lib/whisper.ts
// 統一抽象層：OpenAI Whisper / Local whisper.cpp 都用同一個介面
import OpenAI from 'openai';
import { exec } from 'child_process';
import { promisify } from 'util';
import { mkdtemp, readFile, unlink, writeFile } from 'fs/promises';
import { tmpdir } from 'os';
import { join } from 'path';

const execAsync = promisify(exec);

export type WhisperSegment = {
  id: number;
  start: number;
  end: number;
  text: string;
  no_speech_prob?: number;
};

export type TranscribeResult = {
  text: string;
  language?: string;
  duration?: number;
  segments?: WhisperSegment[];
  words?: Array<{ word: string; start: number; end: number }>;
  model: string;
};

export type TranscribeOptions = {
  file: File | Buffer;
  filename?: string;
  language?: string;
  model?: string;
  prompt?: string;
  responseFormat?: 'json' | 'verbose_json' | 'text' | 'srt' | 'vtt';
};

// 單例快取
let _openai: OpenAI | null = null;
function getOpenAI(): OpenAI {
  if (!_openai) {
    _openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
  }
  return _openai;
}

/** OpenAI Whisper API */
async function transcribeWithOpenAI(opts: TranscribeOptions): Promise<TranscribeResult> {
  const client = getOpenAI();
  const file =
    opts.file instanceof File
      ? opts.file
      : new Blob([opts.file]) as any;

  const resp = await client.audio.transcriptions.create({
    file: file as any,
    model: opts.model ?? 'whisper-1',
    language: opts.language || undefined,
    prompt: opts.prompt,
    response_format: opts.responseFormat ?? 'verbose_json',
    timestamp_granularities: opts.responseFormat === 'verbose_json'
      ? ['segment', 'word']
      : undefined,
  });

  // @ts-ignore — verbose_json 才有這些欄位
  return {
    text: (resp as any).text,
    language: (resp as any).language,
    duration: (resp as any).duration,
    segments: (resp as any).segments?.map((s: any) => ({
      id: s.id, start: s.start, end: s.end,
      text: s.text, no_speech_prob: s.no_speech_prob,
    })),
    words: (resp as any).words,
    model: opts.model ?? 'whisper-1',
  };
}

/** Local whisper.cpp（透過 CLI） */
async function transcribeWithLocal(opts: TranscribeOptions): Promise<TranscribeResult> {
  const dir = await mkdtemp(join(tmpdir(), 'whisper-'));
  const inputPath = join(dir, opts.filename ?? 'audio.webm');

  await writeFile(
    inputPath,
    opts.file instanceof File ? Buffer.from(await opts.file.arrayBuffer()) : opts.file
  );

  const modelPath = process.env.LOCAL_WHISPER_MODEL ?? '/models/ggml-large-v3.bin';
  const outputPrefix = join(dir, 'out');

  // whisper.cpp CLI：--output-json --print-progress false
  const cmd = [
    `whisper-cli`,
    `-m "${modelPath}"`,
    `-f "${inputPath}"`,
    `--output-json`,
    `--output-file "${outputPrefix}"`,
    `--language ${opts.language ?? 'auto'}`,
    `--print-progress false`,
  ].join(' ');

  try {
    await execAsync(cmd);
    const raw = await readFile(outputPrefix + '.json', 'utf8');
    const parsed = JSON.parse(raw);

    return {
      text: parsed.transcription?.map((s: any) => s.text).join(' ') ?? '',
      language: parsed.language,
      duration: parsed.transcription?.at(-1)?.end,
      segments: parsed.transcription?.map((s: any, i: number) => ({
        id: i, start: s.start, end: s.end, text: s.text.trim(),
      })),
      model: opts.model ?? 'local',
    };
  } finally {
    await unlink(inputPath).catch(() => {});
    await unlink(outputPrefix + '.json').catch(() => {});
  }
}

// 統一入口
export async function transcribeAudio(
  provider: 'openai' | 'local',
  opts: TranscribeOptions
): Promise<TranscribeResult> {
  if (provider === 'openai') return transcribeWithOpenAI(opts);
  if (provider === 'local') return transcribeWithLocal(opts);
  throw new Error(`Unknown provider: ${provider}`);
}

export function getProvider(name: string): 'openai' | 'local' {
  return name === 'local' ? 'local' : 'openai';
}

    


    


// components/VoiceInput.tsx
// React 19 + TypeScript · MediaRecorder + FormData 上傳
'use client';

import { useState, useRef, useCallback } from 'react';

type Props = {
  onResult?: (text: string) => void;
  language?: string;
  model?: string;
};

export function VoiceInput({ onResult, language, model }: Props) {
  const [recording, setRecording] = useState(false);
  const [busy, setBusy] = useState(false);
  const mediaRef = useRef<MediaRecorder | null>(null);
  const chunksRef = useRef<Blob[]>([]);

  const start = useCallback(async () => {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    const rec = new MediaRecorder(stream, { mimeType: 'audio/webm;codecs=opus' });
    mediaRef.current = rec;
    chunksRef.current = [];
    rec.ondataavailable = (e) => { if (e.data.size) chunksRef.current.push(e.data); };
    rec.start(100); // 每 100ms 切片，方便長時間錄音
    setRecording(true);
  }, []);

  const stop = useCallback(async () => {
    const rec = mediaRef.current;
    if (!rec) return;
    setRecording(false);
    setBusy(true);

    await new Promise<void>((r) => { rec.onstop = () => r(); rec.stop(); });
    rec.stream.getTracks().forEach(t => t.stop());

    const blob = new Blob(chunksRef.current, { type: 'audio/webm' });
    const form = new FormData();
    form.append('file', blob, 'recording.webm');
    if (language) form.append('language', language);
    if (model) form.append('model', model);

    try {
      const r = await fetch('/api/transcribe', { method: 'POST', body: form });
      const data = await r.json();
      if (!r.ok) throw new Error(data.error);
      onResult?.(data.text);
    } catch (err) {
      console.error(err);
      alert('轉錄失敗：' + (err as Error).message);
    } finally {
      setBusy(false);
    }
  }, [language, model, onResult]);

  return (
    <div>
      <button onClick={recording ? stop : start} disabled={busy}>
        {busy ? '處理中...' : recording ? '⏹ 停止' : '🎤 開始'}
      </button>
    </div>
  );
}

      📐 架構設計筆記
    

    

      

        
為什麼走 Next.js Route Handler

        
    
          
  • 同源部署、API key 不洩漏到前端
    • 
          
  • App Router 支援 runtime: 'nodejs'（跑 multipart）
    • 
          
  • Vercel / 自架都可，部署單一 codebase
    • 
          
  • 之後加 LLM 摘要 / 翻譯就在同一個 route 裡
    • 
        

      

      

        
Provider 抽象層設計

        
    
          
  • transcribeAudio(provider, opts) 統一介面
    • 
          
  • OpenAI API：JSON response、開發快、付費
    • 
          
  • Local whisper.cpp：完全離線、隱私、需 GPU
    • 
          
  • 兩者都回傳 TranscribeResult 統一格式
    • 
        

      

      

        
前端 UX 流程

        
    
          
  • 錄音 / 上傳 / URL 三種入口，符合不同使用情境
    • 
          
  • MediaRecorder 切片（100ms）避免長錄音 memory 爆炸
    • 
          
  • 錄音中即時 waveform 視覺化，提升「有在做事」感
    • 
          
  • 轉錄後 6 種下載動作（SRT/VTT/JSON/摘要/TTS/複製）
    • 
        

      

      

        
部署與資安

        
    
          
  • 限制上傳大小（25MB）防止 OOM
    • 
          
  • CORS 白名單只開自家 origin
    • 
          
  • API key 只放 .env.local，不入 git
    • 
          
  • 未來加 rate limit：每 IP 10 req/min
    •
