import { findUserByEmail } from "@/lib/users";
import { verifyTotp } from "@/lib/totp";
import { logAudit } from "@/lib/db";

export async function POST(req: NextRequest) {
  const ip = req.headers.get("x-forwarded-for");
  const ua = req.headers.get("user-agent");
  const body = await req.json().catch(() => null) as { code?: string } | null;
  const code = String(body?.code ?? "").trim();
  if (!/^\\d{6}$/.test(code)) {
    return NextResponse.json({ status: "invalid_format" }, { status: 400 });
  }

  const c = await cookies();
  const pending = c.get("mfa_pending")?.value;
  if (!pending) {
    logAudit({ event: "mfa_verify_failure", ip, userAgent: ua, outcome: "failure", detail: { reason: "no pending token" } });
    return NextResponse.json({ status: "no_pending_session" }, { status: 401 });
  }

  let claims;
  try {
    claims = await verifyMfaPendingToken(pending);
  } catch (e: any) {
    logAudit({ event: "mfa_verify_failure", ip, userAgent: ua, outcome: "failure", detail: { reason: "expired_or_invalid", error: e.message } });
    return NextResponse.json({ status: "pending_expired" }, { status: 401 });
  }

  const user = findUserByEmail(claims.email);
  if (!user || !user.mfa_secret) {
    return NextResponse.json({ status: "user_misconfigured" }, { status: 500 });
  }

  if (!verifyTotp(user.mfa_secret, code)) {
    logAudit({ event: "mfa_verify_failure", userId: user.id, email: user.email, ip, userAgent: ua, outcome: "failure" });
    return NextResponse.json({ status: "wrong_code" }, { status: 401 });
  }

  logAudit({ event: "mfa_verify_success", userId: user.id, email: user.email, ip, userAgent: ua, outcome: "success" });

  const session = await signSessionToken(user.id, user.email, user.name ?? undefined, 3600);
  const res = NextResponse.json({ status: "ok" });

    httpOnly: true,
    sameSite: "lax",
    path: "/",
    maxAge: 3600,
  });
  res.cookies.set("mfa_pending", "", { path: "/", maxAge: 0 });
