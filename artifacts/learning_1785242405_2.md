+ *   ADD        → 新增 todo

+ *   DELETE     → 刪除 todo
+ *   EDIT       → 編輯 todo 標題

+function todosReducer(state, action) {
+  switch (action.type) {
+    case "ADD":
+      if (!action.payload.text.trim()) return state;
+      return {
+        ...state,
+        todos: [
+          ...state.todos,
+          {
+            id: crypto.randomUUID
+              ? crypto.randomUUID()
+              : Date.now().toString(36) + Math.random().toString(36).slice(2),
+            text: action.payload.text.trim(),
+            completed: false,
+            createdAt: new Date().toISOString(),
+          },
+        ],
+      };
+
+    case "TOGGLE":
+      return {
+        ...state,
+        todos: state.todos.map((t) =>

+      };

+    case "DELETE":

+        todos: state.todos.filter((t) => t.id !== action.payload.id),

+  const [draft, setDraft] = useState(todo.text);
+  const inputRef = useRef(null);
+
+  // 進入編輯模式時 focus + 選取
+  useEffect(() => {
+    if (isEditing && inputRef.current) {
+      inputRef.current.focus();
+      inputRef.current.select();
+    }
+  }, [isEditing]);

+  // 同步外部 todo 變更(例如來自 reducer 的 EDIT)

+    if (!isEditing) setDraft(todo.text);
+  }, [todo.text, isEditing]);
+
+  const commit = () => {
+    const trimmed = draft.trim();
+    if (trimmed && trimmed !== todo.text) {
+      onEdit(todo.id, trimmed);
+    } else {
+      setDraft(todo.text);
+      onEdit(todo.id, todo.text); // 退出編輯模式
+    }
+  };
+
+  const onKeyDown = (e) => {
+    if (e.key === "Enter") {
+      e.preventDefault();
+      commit();

+      setDraft(todo.text);

+    }
+  };

+## 狀態管理架構
+
+```
+[useTodosReducer custom hook]
+       ↓
+[useReducer + 6 action types]
+       ↓
+[localStorage sync via useEffect]
+       ↓
+[useMemo derived state: filtered / remaining / completedCount]
+       ↓
+[useCallback handlers: handleAdd / Toggle / Delete / Edit / Filter / Clear]
+       ↓
+[<TodoList> view] → <TodoItem>
+```
+
+### 動作類型
+
+| Type | 用途 | Payload |
+|------|------|---------|
+| `ADD` | 新增 todo | `{ text }` |
+| `TOGGLE` | 切換完成狀態 | `{ id }` |
+| `DELETE` | 刪除 todo | `{ id }` |
+| `EDIT` | 編輯標題 | `{ id, text }` |
+| `SET_FILTER` | 切換 filter | `{ filter }` |
+| `CLEAR_COMPLETED` | 清除已完成 | 無 |
+
+## 功能
+
+- ✅ 新增 / 編輯 / 刪除 / 完成切換

+- ✅ Enter 確認、Esc 取消
