+def _split_csv(raw: str) -> frozenset[int]:
+    if not raw:
+        return frozenset()
+    out: set[int] = set()

+        if not chunk:

+"""Async token-bucket rate limiter, one bucket per chat_id.
+
+Architect feedback applied:
+  #1 — Explicit init per chat_id (no defaultdict lazy burst risk)
+  #4 — asyncio.Lock() guards bucket mutation (true async concurrency)
+  #7 — TTL eviction sweep prevents memory leak from inactive buckets
+"""
+from __future__ import annotations
+
+import asyncio
+import time
+from dataclasses import dataclass
+
+
+@dataclass
+class _Bucket:
+    tokens: float
+    last_refill: float
+    last_seen: float
+
+
+class RateLimiter:
+    """Async-safe token bucket per key (typically chat_id)."""
+
+    def __init__(
+        self,
+        rate: int,
+        capacity: int,
+        *,
+        ttl_seconds: float = 600.0,
+        clock=time.monotonic,
+    ) -> None:
+        if rate <= 0 or capacity <= 0:
+            raise ValueError("rate and capacity must be positive")
+        self.rate = float(rate)
+        self.capacity = float(capacity)
+        self.ttl_seconds = float(ttl_seconds)
+        self._clock = clock
+        self._buckets: dict[str, _Bucket] = {}
+        self._lock = asyncio.Lock()
+
+    async def acquire(self, key: str) -> bool:
+        """Returns True if the request is allowed, False if rate-limited.
+
+        Refills tokens based on elapsed time, then attempts to consume one.
+        """
+        now = self._clock()
+        async with self._lock:
+            bucket = self._buckets.get(key)
+            if bucket is None:
+                # Explicit init (architect feedback #1 — no defaultdict race).
+                bucket = _Bucket(
+                    tokens=self.capacity - 1.0,
+                    last_refill=now,
+                    last_seen=now,
+                )
+                self._buckets[key] = bucket
+                return True
+
+            elapsed = max(0.0, now - bucket.last_refill)
+            bucket.tokens = min(self.capacity, bucket.tokens + elapsed * self.rate)
+            bucket.last_refill = now
+            bucket.last_seen = now
+
+            if bucket.tokens >= 1.0:
+                bucket.tokens -= 1.0
+                return True
+            return False
+

+        """Evict buckets idle longer than ttl_seconds. Returns count evicted."""

+"""Mock transport for httpx so tests never touch the real Open-Meteo API."""
+from __future__ import annotations
+
+import json
+from dataclasses import dataclass
+from typing import Any, Callable
+
+import httpx
+
+
+@dataclass
+class _Route:
+    url_contains: str
+    handler: Callable[[httpx.Request], httpx.Response]
+
+
+class MockTransport(httpx.AsyncBaseTransport):
+    """httpx transport that matches by substring and returns canned responses."""
+
+    def __init__(self) -> None:
+        self._routes: list[_Route] = []
+        self.calls: list[httpx.Request] = []
+
+    def add(self, url_contains: str, response: httpx.Response) -> None:
+        self._routes.append(_Route(url_contains, lambda req: response))
+
+    def add_json(self, url_contains: str, payload: Any, status: int = 200) -> None:
+        body = json.dumps(payload).encode("utf-8")
+        resp = httpx.Response(
+            status_code=status,
+            content=body,
+            headers={"content-type": "application/json"},
+            request=httpx.Request("GET", "https://mock"),
+        )
+        self.add(url_contains, resp)
+
+    def add_exception(self, url_contains: str, exc: Exception) -> None:
+        async def _raise(_req: httpx.Request) -> httpx.Response:
+            raise exc
+
+        self._routes.append(_Route(url_contains, _raise))
+
+    async def handle_async_request(self, request: httpx.Request) -> httpx.Response:
+        self.calls.append(request)
+        for route in self._routes:
+            if route.url_contains in str(request.url):
+                return await route.handler(request)

+            status_code=599,
+            content=b"no mock route matched",

+| 1 | Lazy `defaultdict` bucket init is unsafe (burst risk) | `RateLimiter.acquire` uses **explicit init** — `if bucket is None: self._buckets[key] = _Bucket(...)`. No defaultdict. |

+| 5 | X-Forwarded-For edge cases | Translated to Telegram analog: `_is_authorized` enforces an `allowed_user_ids` allowlist; non-allowlisted users are **silently dropped** (no reply, no leak that the bot exists). Configurable via `ALLOWED_USER_IDS` env. |

+| 7 | Rate-limiter memory leak (no TTL eviction) | `RateLimiter.sweep()` evicts buckets idle > `ttl_seconds` (default 600s). `main()` runs `sweep()` every 60s in a background task spawned from `post_init`, cancelled in `post_shutdown`. |
