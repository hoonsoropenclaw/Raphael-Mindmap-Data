+#!/usr/bin/env bash
+# run_monitor.sh — wrapper for cron / manual run
+#
+# 用法：
+#   ./run_monitor.sh             # 跑一次，append 到 monitor.log，沒告警就安靜
+#   ./run_monitor.sh --verbose   # 不論有沒有告警都印到 stdout
+#   ./run_monitor.sh --self-test # 驗證告警路徑（給 alert channel / monitoring 對接用）
+#
+# 設計：
+#   - 用 --quiet 模式呼叫 monitor.py（只印 WARN/CRIT）
+#   - 結果 append 到 ./monitor.log（永遠寫，方便事後 audit）
+#   - exit code 沿用 monitor.py：0 ok / 1 alert / 2 script error
+#   - 出 alert 時 echo 一行到 stderr（讓 cron 透過 MAILTO 收信）
+
+set -euo pipefail
+
+HERE="$(cd "$(dirname "$0")" && pwd)"
+PY="${HERE}/monitor.py"
+LOG="${HERE}/monitor.log"
+CFG="${HERE}/monitor.yaml"
+
+QUIET="--quiet"
+WRITE_LOG=1
+
+for arg in "$@"; do
+    case "$arg" in
+        --verbose) QUIET="" ;;
+        --self-test) QUIET="" ;;
+        --no-log) WRITE_LOG=0 ;;
+        --help|-h)
+            sed -n '2,15p' "$0"
+            exit 0
+            ;;
+    esac
+done
+
+mkdir -p "$(dirname "$LOG")"
+TS="$(date -Iseconds)"
+
+# 跑（含錯誤處理：不要讓 set -e 把 stderr 吞掉）
+if python3 "$PY" --config "$CFG" $QUIET > /tmp/monitor.out.$$ 2> /tmp/monitor.err.$$; then
+    rc=0
+elif [ $? -eq 1 ]; then
+    rc=1
+else
+    rc=2
+fi
+
+# 取實際 exit code
+python3 "$PY" --config "$CFG" $QUIET > /tmp/monitor.out.$$ 2> /tmp/monitor.err.$$ || rc=$?
+
+# 把 stdout 寫進 log（含 header）
+if [ "$WRITE_LOG" -eq 1 ]; then
+    {
+        echo "----- ${TS} (rc=${rc}) -----"
+        cat /tmp/monitor.out.$$
+        if [ -s /tmp/monitor.err.$$ ]; then
+            echo "[stderr]"
+            cat /tmp/monitor.err.$$
+        fi
+    } >> "$LOG"
+fi
+
+# verbose 模式：把結果也印到 caller 的 stdout
+if [ -z "$QUIET" ]; then
+    cat /tmp/monitor.out.$$
+fi
+
+# alert 時額外提示到 stderr（cron 友善）
+if [ "$rc" -eq 1 ]; then
+    echo "[run_monitor] ⚠ ALERT detected — see $LOG" >&2
+elif [ "$rc" -eq 2 ]; then
+    echo "[run_monitor] ✗ script error — see $LOG / /tmp/monitor.err.$$" >&2
+fi
+
+rm -f /tmp/monitor.out.$$ /tmp/monitor.err.$$
+exit "$rc"

+# architect_notes.md — Linux 系統日誌與資源監控腳本
+
+> **總工程師交付摘要**（極限超頻模式）
+> 任務：撰寫自動化檢查 Linux 系統日誌與資源的維運腳本，並在超過閾值時發出警告。
+> 交付物：`monitor.py` + `monitor.yaml` + `run_monitor.sh`，全部已實機驗證可執行。
+
+---
+
+## 一、設計目標與非目標
+
+### 目標
+1. **單檔可跑** — `python3 monitor.py` 一行就能跑，不依賴 psutil / pip 安裝
+2. **Cron 友善** — 退出碼（0/1/2）+ 自動 append log + `--quiet` 模式
+3. **機器讀** — `--json` 模式讓 Prometheus / Slack bot / 其他 agent 可解析
+4. **失敗不連坐** — 任何一個 checker 拋例外，其他 checker 仍要跑完
+5. **可閾值化** — 全部閾值集中在 `monitor.yaml`，不必改程式碼
+
+### 非目標（明確不做）
+- ❌ 不做趨勢預測 / ML anomaly detection（這是 monitoring，不是 observability platform）
+- ❌ 不做自動修復（`Restart-Service`、`systemctl restart`）—— 違反 SRE 邊界
+- ❌ 不做遠端推送（Slack / PagerDuty）—— 用 stdout + exit code 1 交給 cron MAILTO 即可
+- ❌ 不取代 Prometheus / node_exporter —— 這是「輕量、無外部依賴」路線
+
+---
+
+## 二、架構決策
+
+### 1. 為什麼不用 psutil？
+- 標準庫 `os.statvfs` + `/proc/*` 已能拿所有需要的指標
+- 多一個依賴 = 多一個 apt 套件 / 多一個 venv / 多一個升級風險面
+- **If** 部署到陌生 Linux（無 pip、無 venv）**Then** stdlib-only 是 zero-install 路徑
+
+### 2. 為什麼走 dataclass 而不是 dict？
+- `Finding` 跟 `Report` 用 `@dataclass` 強制型別
+- `Report.to_dict()` 顯式提供序列化入口，JSON 模式不會誤把 private attribute 倒出來
+- 未來加欄位時（如：trace_id、run_id）不會破壞向後相容
+
+### 3. 為什麼每個 checker 獨立 try/except？
+監控腳本最常見的失敗模式是「某個 metric 拿不到，把整個報告吃掉」。例如：
+- `dmesg` 在 container 內讀不到 kernel ring buffer → PermissionError
+- `lastb` 沒裝或 btmp 不存在 → FileNotFoundError
+- `/var/log/btmp` 是 syslog:adm owned → PermissionError
+
+**每個 checker 獨立 try/except + 錯誤收集到 `report.errors`**，確保：
+- 一個失敗不影響其他
+- 退出碼 2（腳本錯誤）vs 1（業務警告）分清楚
+- 使用者能在報告最後看到「哪個 checker 死了」，不必看 stderr
+
+### 4. 為什麼 graceful degradation 用「找不到檔案就 INFO」而非 raise？
+- 開發機 / 客製化 server 不一定有 `/var/log/kern.log`
+- 與其讓整個 log scanner 死掉，不如 INFO「檔案不存在」並繼續下一個
+
+### 5. 為什麼 log scan 用 regex 而不是簡單 substring？
+- `\bOOM\b` 比 `OOM` 精準（不會誤中 `BOOM` `ZOOM`）
+- `\bkilled process\b` 不會誤中 `process killed by watchdog`
+- 啟發式關鍵字（`Call Trace`、`WARNING:`）都是業界 kernel developer 經驗值
+
+### 6. 為什麼時間窗過濾（24h）？
+- syslog 可能 9 MB 大，全檔 grep 慢
+- 舊的 OOM / segfault 早已處理過，現在掃沒意義
+- 只看「近 24h」大幅降低噪音、聚焦在「最近發生了什麼」
+- **窗寬可調**（`monitor.yaml` 的 `log_scan_window_hours`）
+
+### 7. 為什麼 mount point 跳過 tmpfs / sysfs / proc？
+- 這些虛擬 fs 的 `total` 通常是 0 或極小，會誤報 100% / 0%
+- `shutil.disk_usage` 在 bind mount 重複條目會出現多次，已用 `seen` 集合去重
+- 配置檔 `disk_ignore_mounts` 開放使用者自訂跳過清單
