+import { useCallback, useEffect, useRef, useState } from 'react';
+
+/**
+ * useLocalStorage — syncs a value with window.localStorage.
+ *
+ * Behaviour notes:
+ *  - SSR-safe: returns the initial value when window is undefined.
+ *  - Resilient to JSON parse errors (falls back to initial).
+ *  - Cross-tab sync via the native `storage` event.
+ *  - Debounces writes (300ms) to avoid hammering disk during rapid
+ *    edits (typing, dragging).
+ */
+export function useLocalStorage(key, initialValue, options = {}) {
+  const { debounceMs = 300, syncAcrossTabs = true } = options;
+  const isClient = typeof window !== 'undefined';
+
+  const [value, setValue] = useState(() => {
+    if (!isClient) return initialValue;
+    try {
+      const raw = window.localStorage.getItem(key);
+      return raw === null ? initialValue : JSON.parse(raw);
+    } catch (err) {
+      console.warn(`useLocalStorage: failed to parse ${key}`, err);
+      return initialValue;
+    }
+  });
+
+  const timerRef = useRef(null);
+  const latestRef = useRef(value);
+  latestRef.current = value;
+
+  // Debounced write
+  useEffect(() => {
+    if (!isClient) return;
+    if (timerRef.current) clearTimeout(timerRef.current);
+    timerRef.current = setTimeout(() => {
+      try {
+        window.localStorage.setItem(key, JSON.stringify(latestRef.current));
+      } catch (err) {
+        console.warn(`useLocalStorage: failed to persist ${key}`, err);
+      }
+    }, debounceMs);
+    return () => {
+      if (timerRef.current) clearTimeout(timerRef.current);
+    };
+  }, [key, value, debounceMs, isClient]);
+
+  // Cross-tab sync
+  useEffect(() => {
+    if (!isClient || !syncAcrossTabs) return;
+    const onStorage = (e) => {
+      if (e.key !== key || e.newValue === null) return;
+      try {
+        const next = JSON.parse(e.newValue);
+        // Only update if actually different (avoid infinite loops).
+        setValue((prev) => (JSON.stringify(prev) === e.newValue ? prev : next));
+      } catch (_) {
+        /* ignore malformed payloads from other tabs */
+      }
+    };
+    window.addEventListener('storage', onStorage);
+    return () => window.removeEventListener('storage', onStorage);
+  }, [key, isClient, syncAcrossTabs]);
+
+  const update = useCallback((next) => {
+    setValue((prev) => (typeof next === 'function' ? next(prev) : next));
+  }, []);
+
+  return [value, update];

+# React TaskBoard — Complex State Management Showcase
+
+A production-grade React UI component with non-trivial state management,
+designed as a single self-contained demo. Built from scratch to showcase:
+
+- **useReducer** as the single source of truth for ~14 action types
+- **Context API** with a hand-rolled provider, memoised selectors, and
+  stable action creators
+- **Custom hooks** (`useLocalStorage`, `useKeybindings`, `useThemeEffect`,
+  `useSystemTheme`, `usePrevious`)
+- **Undo / redo** via a bounded history stack
+- **Drag-to-reorder**, optimistic updates, debounced persistence
+- **Light / dark theme** with OS-preference detection
+- **Accessibility-first**: ARIA roles, keyboard shortcuts, focus rings,
+  reduced-motion respect, semantic HTML
+- **Pure CSS design system** with custom properties for theming, no
+  framework dependency (no Tailwind, no styled-components)
+
