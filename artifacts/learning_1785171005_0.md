+// Context Provider: 把 [state, dispatch] 透過 useReducer + Context 跨元件傳遞
+// 設計理由：useReducer 集中狀態邏輯，Context 解決 prop drilling
+// 持久化策略：useEffect 監聽 state 變化 → 寫入 localStorage；初次 mount 從 localStorage 還原
+
+import { createContext, useContext, useEffect, useReducer, useRef } from 'react';
+import { todoReducer, initialState } from './todoReducer.js';
+
+const STORAGE_KEY = 'hermes-todo-state-v1';
+const TodoContext = createContext(null);
+
+export function TodoProvider({ children }) {
+  // lazy init: 同步從 localStorage 讀（避免 hydration 閃爍）
+  const [state, dispatch] = useReducer(todoReducer, initialState, (init) => {
+    if (typeof window === 'undefined') return init;
+    try {
+      const raw = window.localStorage.getItem(STORAGE_KEY);
+      if (!raw) return init;
+      const parsed = JSON.parse(raw);
+      // 簡單 schema 守衛
+      if (!parsed || !Array.isArray(parsed.todos)) return init;
+      return { ...init, ...parsed };
+    } catch (e) {
+      // localStorage 壞掉（隱私模式、quota 滿、parse error）→ 退回 init
+      console.warn('[TodoProvider] localStorage read failed:', e);
+      return init;
+    }
+  });
+
+  // 持久化：debounce 500ms 避免每 keystroke 都寫
+  const writeTimerRef = useRef(null);
+  useEffect(() => {
+    if (writeTimerRef.current) clearTimeout(writeTimerRef.current);
+    writeTimerRef.current = setTimeout(() => {
+      try {
+        window.localStorage.setItem(STORAGE_KEY, JSON.stringify({
+          todos: state.todos,
+          filter: state.filter
+        }));
+      } catch (e) {
+        // quota exceeded 等錯誤 — 用 console.warn 而非 throw，避免破壞 UI
+        console.warn('[TodoProvider] localStorage write failed:', e);
+      }
+    }, 500);
+
+    return () => {
+      if (writeTimerRef.current) clearTimeout(writeTimerRef.current);
+    };
+  }, [state.todos, state.filter]);
+
+  // 跨分頁同步：監聽 storage event
+  useEffect(() => {
+    const onStorage = (e) => {
+      if (e.key !== STORAGE_KEY || !e.newValue) return;
+      try {
+        const parsed = JSON.parse(e.newValue);
+        dispatch({ type: 'HYDRATE', payload: parsed });
+      } catch { /* ignore */ }
+    };
+    window.addEventListener('storage', onStorage);
+    return () => window.removeEventListener('storage', onStorage);
+  }, []);
+

+    <TodoContext.Provider value={{ state, dispatch }}>
+      {children}
+    </TodoContext.Provider>
+  );

+  const [value, setValue] = useState(target);
+  const fromRef = useRef(target);
+  const startRef = useRef(0);
+  const rafRef = useRef(0);

+  useEffect(() => {
+    cancelAnimationFrame(rafRef.current);
+    fromRef.current = value;
+    startRef.current = performance.now();
+    const tick = (now) => {
+      const elapsed = now - startRef.current;
+      const t = Math.min(1, elapsed / duration);
+      // ease-out cubic
+      const eased = 1 - Math.pow(1 - t, 3);

+# React Todo Showcase
+
+一個展示 **React 18 複雜狀態管理** 的完整 UI 元件實作。
+
+由 ⚕ Hermes 在「極限超頻模式」訊息脈絡下交付，採用 TodoList 為例展示 4 種狀態管理協作方式。

+## 複雜狀態管理展示
+
+| 技術 | 用在哪 | 為什麼選這個 |
+|------|--------|--------------|
+| **`useReducer`** | `todoReducer.js` 管理 todos 陣列所有 CRUD | 集中狀態邏輯，方便 debug / time-travel / 測試 |
+| **`Context API`** | `TodoContext.jsx` 跨元件傳遞 [state, dispatch] | 解決 prop drilling，讓深層子元件直接拿到 dispatch |
+| **`useState`** | 各 UI 子元件自己的暫存（input、edit draft、theme） | 元件本地 UI 狀態不需要污染全域 |
+| **`useEffect` + `localStorage`** | `TodoContext` 內 debounced 持久化 + 跨分頁 storage 事件 | 體驗一致、reload 後狀態不丟失 |
+
+自訂 hook `useTodos(state, dispatch)` 把 reducer 包成領域 API（`add / toggle / edit / setPriority / ...`）。
+
