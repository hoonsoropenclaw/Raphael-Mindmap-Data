+"""Fixed-window in-process rate limiter for the todo API.
+
+The limiter is intentionally kept tiny and dependency-free so it can be unit
+tested deterministically without external services. It tracks request counts
+per client identifier inside a process-local dictionary guarded by a lock.
+"""
+
+from __future__ import annotations
+
+import threading
+import time
+from collections import defaultdict, deque
+from collections.abc import Iterable
+from dataclasses import dataclass
+
+
+@dataclass(frozen=True, slots=True)
+class RateLimitDecision:
+    """Outcome of evaluating a request against the limiter."""
+
+    allowed: bool
+    remaining: int
+    limit: int
+    retry_after_seconds: int
+
+
+class FixedWindowRateLimiter:
+    """Thread-safe fixed-window limiter keyed by client identifier."""
+
+    def __init__(self, *, limit: int, window_seconds: int) -> None:
+        if limit <= 0:
+            raise ValueError("limit must be a positive integer")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be a positive integer")
+        self._limit = limit
+        self._window_seconds = window_seconds
+        self._events: dict[str, deque[float]] = defaultdict(deque)
+        self._lock = threading.Lock()
+
+    @property
+    def limit(self) -> int:
+        return self._limit
+
+    @property
+    def window_seconds(self) -> int:
+        return self._window_seconds
+
+    def check(self, client_id: str, *, now: float | None = None) -> RateLimitDecision:

+# Todo API
+
+A small, well-tested FastAPI service that exposes a JSON todo resource
+backed by SQLite. It demonstrates three production-shaped concerns in a
+single file: structured configuration, a thread-safe rate limiter, and
+parameterised persistence with WAL journaling.
+
+## Layout
+
+```
+todo_api/
+  app.py          # FastAPI factory + route handlers + rate-limit middleware
+  config.py       # AppSettings value object (frozen dataclass)
+  rate_limit.py   # FixedWindowRateLimiter + client-IP resolver
+  repository.py   # SQLite repository (parameterised SQL, WAL, foreign keys)
+  schemas.py      # Pydantic v2 request/response models
+tests/
+  test_todos.py        # CRUD + persistence + schema PRAGMA checks
+  test_rate_limit.py   # fixed-window behaviour, header exposure, exemptions
+scripts/
+  smoke_asgi.py        # end-to-end ASGI smoke test (httpx ASGITransport)
+```
+
+## Run the server
+
+```bash
+python3 -m venv .venv
+.venv/bin/pip install -e ".[dev]"
+.venv/bin/uvicorn todo_api:create_app --factory --host 0.0.0.0 --port 8000
+```
+
+The factory accepts the same keyword arguments as `create_app` in
+`todo_api/app.py`; the most useful ones for production are
+`rate_limit_requests`, `rate_limit_window_seconds`,
+`trust_proxy_headers`, and `rate_limit_exempt_paths`.
+
+## Test
+
+```bash
+.venv/bin/python -m pytest tests
+.venv/bin/python scripts/smoke_asgi.py
+```
+
+## Endpoints
+
+| Method | Path             | Description                              |
+|--------|------------------|------------------------------------------|
+| GET    | `/health`        | Liveness probe (exempt from rate limit). |
+| POST   | `/todos`         | Create a todo.                           |
+| GET    | `/todos`         | List todos. `?completed=true\|false`.    |
+| GET    | `/todos/{id}`    | Fetch a single todo.                     |
+| PATCH  | `/todos/{id}`    | Partial update; 422 on empty body.       |
+| DELETE | `/todos/{id}`    | Delete a todo; 204 on success.           |
+
+Responses use trailing-`Z` UTC ISO 8601 timestamps. Validation errors use

+## 2. 設計決策與依據
+
+### 2.1 同步 FastAPI + 同步 SQLite repository（不採 async）
+
+* 候選：async SQLAlchemy + `async def` endpoints。
+* 取捨：
+  * 此 CRUD 沒有外部 I/O 等待、沒有 WebSocket 串接、沒有 fan-out 任務。
+  * `trial-and-error/references/by-category/fastapi-async-session-pitfalls.md` 列出 3 個非同步必踩的雷（identity map stale read、async + sync session deadlock、`expire_on_commit` 漏設）。
+  * `trial-and-error/references/by-category/fastapi-sync-vs-async-pattern-20260727.md` 引用 Quote API 在 72 個測試都用 sync 的實戰證據，sync 的複雜度只 1 個 `def` + `db.execute()`、async 至少多 3 個 pitfall 類別。
+* 結論：採 **sync**。所有 route handler 用 `def`、repository 用 `sqlite3` 直接驅動。
+
+### 2.2 Rate limit：固定視窗 + 程序內字典 + 鎖
+
+* 不引第三方套件（題目要求「簡單」、避免 `slowapi`/`limits` 額外相依與設定）。
+* `FixedWindowRateLimiter` 維護 `client_id → deque[monotonic timestamp]`，每命中一次 `popleft()` 過期項；`threading.Lock` 保護 race。
+* 視窗大小以 `time.monotonic()` 為基準（不受系統時鐘跳動影響）。
+* `Retry-After` 用 `int(window_seconds - (now - 最早命中))` 取下限 1 秒。
+* 為了防止 `X-Forwarded-For` 被惡意客戶端偽造而繞過，預設 `trust_proxy_headers=False`，必須**顯式**開啟才信任反向代理 header。
+
+### 2.3 `rate_limit_exempt_paths` 預設值位置：放 `create_app` 而非 `AppSettings`
+
+* 嘗試 1：把 `frozenset({"/health"})` 放 `AppSettings` 預設 → 測試難以「關掉」這個預設。
+* 嘗試 2：保留在 `AppSettings` 但把 `create_app` 改成「呼叫者沒傳就套用」 → 仍有「`/health` 靜默被 exempt」的隱藏行為。
+* 最終決策：`create_app` 內顯式 `if rate_limit_exempt_paths is None: overrides["rate_limit_exempt_paths"] = frozenset({"/health"})`。
+* 理由：production 友善的預設仍保留（LB 探活不會被限流），但測試與自訂部署**必須**用 `rate_limit_exempt_paths=frozenset()` 顯式表達「我不要任何豁免」。這同時也讓綠區測試 4（`test_health_can_be_exempted_from_rate_limit`）必須主動加 `/health`，是**正向**依賴不是隱藏耦合。
+
+### 2.4 SQLite schema：WAL + foreign keys + 索引
+
+* `PRAGMA journal_mode = WAL` → 多讀單寫、降低寫入鎖。
+* `PRAGMA foreign_keys = ON` → 雖然目前無外鍵，未來加 user/tag 表時直接生效。
+* `idx_todos_completed` 索引 `completed` 欄位 → list filter 走索引而非全表掃描。
+* `time` 欄位以 ISO 8601 字串儲存（`detect_types=PARSE_DECLTYPES` 可在後續改 native timestamp）。
+
+### 2.5 Pydantic v2：`TodoUpdate.extra="forbid"` + 422 on empty PATCH
+
+* 422 是 FastAPI 對「空 body」+「`has_changes()` 為 False」兩個情境的共同回應。
+* 這比 Quote API 慣例的 400 更嚴格：Pydantic 自己會對未知欄位拋 422，避免客戶端 typo 不會被默默忽略。
+* `TitleStr = Annotated[str, StringConstraints(strip_whitespace=True, min_length=1, max_length=200)]` → 空白字串一定會 422，連「全部空白」也被擋。
+
+### 2.6 時間序列化：trailing `Z`
+
+* `app.py::_format_utc` 強制 `astimezone(timezone.utc).isoformat().replace("+00:00", "Z")`。
+* 測試斷言 `body["created_at"].endswith("Z")` 確保**未來** Python 升級到 ISO 8601 預設改回 `+00:00` 時仍會壞掉（警示測試守住行為）。
