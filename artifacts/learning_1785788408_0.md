        ctx = await browser.new_context(viewport={"width":1440,"height":900})
        page = await ctx.new_page()

        console_msgs = []
        errors = []
        page.on("console", lambda m: console_msgs.append((m.type, m.text)))
        page.on("pageerror", lambda e: errors.append(str(e)))

        await page.goto(HTML, wait_until="networkidle", timeout=30000)

        edge_count = await page.locator(".react-flow__edge").count()
        minimap = await page.locator(".react-flow__minimap").count()
        controls = await page.locator(".react-flow__controls").count()
        custom_node = await page.locator(".ff-node").count()
        palette = await page.locator(".palette-item").count()

        stat_nodes = await page.locator(".stat-card").count()
