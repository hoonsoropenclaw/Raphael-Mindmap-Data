+// App Router middleware — gates UI routes by role
+// Page-level check happens in layout / page components,
+// this gates URLs only (no token = redirect to /login)
+
+import { NextRequest, NextResponse } from "next/server";
+
+const PROTECTED_PREFIXES = ["/dashboard", "/admin", "/console"];
+const PUBLIC_PATHS = ["/login", "/api/login", "/api/audit", "/api/health", "/_next", "/favicon"];
+
+export function middleware(req: NextRequest) {
+  const { pathname } = req.nextUrl;
+  if (PUBLIC_PATHS.some(p => pathname.startsWith(p))) return NextResponse.next();
+  if (!PROTECTED_PREFIXES.some(p => pathname.startsWith(p))) return NextResponse.next();
+
+  const auth = req.headers.get("authorization") ?? "";
+  const cookieToken = req.cookies.get("portal_token")?.value;
+  const token = auth.replace(/^Bearer\s+/i, "") || cookieToken;
+  if (!token) {
+    const url = req.nextUrl.clone();
+    url.pathname = "/login";
+    return NextResponse.redirect(url);
+  }
+  return NextResponse.next();
+}

+// Server-side check: user → platform → action → 200/403
+
+import { NextRequest, NextResponse } from "next/server";
+import { verify } from "@/lib/auth";
+import { can, methodToAction, requiredRoleFor } from "@/lib/rbac";
+import { withAudit } from "@/lib/audit";
+import type { Action, Platform } from "@/lib/rbac";
+
+const VALID_PLATFORMS = ["github", "vercel", "google", "stripe"] as const;
+
+export async function handle(req: NextRequest, platform: string, rest: string[]) {
+  if (!VALID_PLATFORMS.includes(platform as Platform)) {
+    return NextResponse.json({ error: "unknown_platform" }, { status: 404 });
+  }
+
+  const token = req.headers.get("authorization")?.replace(/^Bearer\s+/i, "")
+    ?? req.cookies.get("portal_token")?.value;
+  const user = await verify(token);
+  if (!user) {
+    return withAudit(
+      { username: "anonymous", name: "", email: "", role: "viewer" },
+      req.method, platform, "/" + (rest?.join("/") ?? ""),
+      methodToAction(req.method), false,
+      NextResponse.json({ error: "unauthorized" }, { status: 401 }),
+    );
+  }
+
+  const action = methodToAction(req.method);
+  const allowed = can(user.role, platform as Platform, action);
+
+  if (!allowed) {
+    return withAudit(
+      user, req.method, platform, "/" + (rest?.join("/") ?? ""), action, false,
+      NextResponse.json({
+        error: "forbidden",
+        message: `Role '${user.role}' cannot ${action.toUpperCase()} on platform '${platform}'`,
+        required_role: requiredRoleFor(action as Action),
+        current_role: user.role,
+      }, { status: 403 }),
+    );
+  }
+
+  // In production: forward to actual upstream API with proper creds
+  // For demo: return a mock payload
+  const data = mockResponse(platform, rest ?? [], req.method);
+  return withAudit(
+    user, req.method, platform, "/" + (rest?.join("/") ?? ""), action, true,
+    NextResponse.json(data, { status: 200 }),
+  );

+# Next.js RBAC Multi-Platform API Portal
+
+Role-Based Access Control integration hub that proxies authenticated requests to multiple third-party platforms (GitHub, Vercel, Google, Stripe) with a single source-of-truth permission matrix.
+
+## Run
+
+```bash
+npm install
+npm run dev      # http://localhost:3000
+```
+
+## Demo accounts
+
+| Username | Password | Role      | What they can do                          |
+|----------|----------|-----------|-------------------------------------------|
+| `admin`  | `admin`  | admin     | read / write / delete / admin (everything)|
+| `dev`    | `dev`    | developer | read / write on all platforms             |
+| `viewer` | `viewer` | viewer    | read-only on all platforms                |
+
+## Architecture
+
+| Path | Purpose |
+|------|---------|
+| `lib/rbac.ts` | Single source of truth — `ROLE_PERMS[role][platform][action]` matrix |
+| `lib/auth.ts` | JWT (HS256) login/verify via `jose` |
+| `lib/audit.ts` | In-memory ring-buffer audit log |
+| `middleware.ts` | Cookie-based route guard (`/dashboard`, `/admin`, `/console` require token) |
+| `app/api/login/route.ts` | POST `{username, password}` → sets `portal_token` HttpOnly cookie |
+| `app/api/logout/route.ts` | Clears the cookie |
+| `app/api/audit/route.ts` | Lists audit entries (auth required) |
+| `app/api/[platform]/[...path]/route.ts` | RBAC-gated proxy — returns `200` mock data if allowed, `403` with reason if denied |
+| `app/login/` | Public login page |
+| `app/dashboard/` | Authenticated dashboard with per-platform action buttons (UI buttons disabled when RBAC denies) |
+
+## How RBAC is enforced (defense in depth)
+
+1. **Middleware** — anonymous user can't reach `/dashboard` (redirected to `/login`).
+2. **Page-level** — `app/dashboard/page.tsx` calls `verify()` server-side and redirects if token invalid.
+3. **API-level** — `app/api/[platform]/[...path]/route.ts` checks `can(role, platform, action)` BEFORE returning data; writes to audit regardless.
+4. **UI-level** — buttons rendered `disabled` for actions the role can't perform.
+

+| | `admin` | `developer` | `viewer` |

+
+### RBAC 4 層防禦（任務要求 = 真實驗證，不是擺設）
+
+| 層 | 元件 | 角色 |
+|---|---|---|
+| 1 | `middleware.ts` | 路由級 — 無 token 重導向 `/login` |
+| 2 | `app/dashboard/page.tsx` 的 `verify()` | 頁面級 — server-side check |
+| 3 | `app/api/[platform]/[...path]/route.ts` 的 `can()` | API 級 — 檢查後才 mock data；不通過回 `403 {error:"forbidden", required_role, current_role}` |
+| 4 | HTML 內按鈕的 `disabled` 屬性 | UI 級 — viewer 看不到 delete 按鈕能按 |
+
+### 流程
+
+```
+login (/api/login, POST {username, password})
+  → jose HS256 sign → 設 HttpOnly cookie "portal_token" (1h)
+dashboard 讀 cookie → verify() → 渲染已 disable 的按鈕
+user 按 "List" → formAction /api/{platform}/{path} GET
+  → mock API route 用 verify() 確認 token, 用 can(role, platform, action) 確認權限
+  → 不通過: 403 + audit log + 不返回敏感資料
+  → 通過: 200 + mock payload + audit log
+```
+
+### 審計日誌
+每次 API 呼叫寫一筆:`{ ts, user, role, method, platform, path, action, allowed, status }`。
+Web 版在記憶體,Next.js 版在 server process 記憶體 ring buffer(MAX 500 條)。

+## 4. 漏洞清單（誠實列出沒做的）
+
+| 項目 | 原因 | 修法 |
+|---|---|---|
+| **真實金鑰未連接** | 4 個平台(GitHub / Vercel / Google / Stripe)都沒真實 OAuth flow,只用 mock data | 見下方「生產 checklist」 |
+| **Audit 用 in-memory** | server restart 就掉 | 接 Postgres / Vercel KV / CloudWatch |
+| **無 refresh token** | 1 小時硬過期 | 加 refresh flow + revoke list |
+| **無 CSRF 防護** | cookie-based auth 預設受 SameSite=Lax | 加 double-submit cookie 或 origin check |
+| **無 rate limit** | 一個 token 可以每秒狂打 | 加 token bucket / Redis counter |
+| **無 RBAC 時間邊界** | 帳號「明天升 viewer」= 立刻生效? | 加 effective_from / effective_until |
+| **per-platform credentials scoping 未實作** | 全部走 mock | 每平台 API key 應分開存 (1Password / KMS) |
+| **無 audit log retention** | 暫存 100-500 條就 ring-buffer 覆蓋 | 寫到 Loki / CloudWatch / S3 |
+| **無 audit log 不可竄改性** | 記憶體可改 | 加 hash chain (Merkle root per day) |
+| **viewer 仍能呼叫 API 看到 mock data** | mock 不含敏感資料所以 OK,真實資料下會外洩 | mock 應明示 `_mock:true` 標籤,real impl 過 proxy |
