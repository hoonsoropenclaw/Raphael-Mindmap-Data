+"""Structured logging setup for docwatch.
+
+Provides a single ``get_logger(name)`` factory that emits JSON-friendly
+log lines with timestamp, level, module, and message. Console output uses
+colourised human format; file output is plain for ingestion.
+"""
+from __future__ import annotations
+
+import logging
+import os
+import sys
+from pathlib import Path
+from typing import Optional
+
+__all__ = ["get_logger", "setup_logging"]
+
+
+_INITIALISED = False
+
+
+def setup_logging(
+    level: str = "INFO",
+    log_file: Optional[str] = None,
+    log_dir: Optional[str] = None,
+) -> None:
+    """Initialise root logging once. Idempotent."""
+    global _INITIALISED
+    if _INITIALISED:
+        return
+
+    root = logging.getLogger()
+    root.setLevel(getattr(logging, level.upper(), logging.INFO))
+
+    # Remove any prior handlers (e.g. from previous imports)
+    for h in list(root.handlers):
+        root.removeHandler(h)
+
+    fmt = "[%(asctime)s] %(levelname)-7s %(name)s — %(message)s"
+    datefmt = "%Y-%m-%d %H:%M:%S"
+
+    console = logging.StreamHandler(sys.stderr)
+    console.setFormatter(logging.Formatter(fmt, datefmt=datefmt))
+    root.addHandler(console)
+
+    if log_file or log_dir:
+        log_dir = log_dir or str(Path(log_file).parent)  # type: ignore[arg-type]
+        Path(log_dir).mkdir(parents=True, exist_ok=True)
+        file_path = log_file or str(Path(log_dir) / "docwatch.log")
+        fh = logging.FileHandler(file_path, encoding="utf-8")
+        fh.setFormatter(logging.Formatter(fmt, datefmt=datefmt))
+        root.addHandler(fh)
+
+    # Tame noisy third-party loggers
+    logging.getLogger("watchdog.observers").setLevel(logging.WARNING)
+    logging.getLogger("apscheduler.scheduler").setLevel(logging.WARNING)
+    logging.getLogger("engineio").setLevel(logging.WARNING)
+    logging.getLogger("socketio").setLevel(logging.WARNING)

+
+

+docwatch — System-level document monitor with xdotool automation
+================================================================
+
+A lightweight, layered tool that watches files on disk and triggers
+desktop automation in response. Built around two battle-tested
+foundations:
+
+* **watchdog** — cross-platform file-system observer (inotify on Linux)
+* **subprocess wrappers** — xdotool / ydotool / wtype / AT-SPI / mock
+
+Built to run on a desktop session, a headless server, or a CI runner —
+the same code path handles all three modes through an explicit backend
+abstraction.
+
+Architecture
+============
+
+```
+  ┌──────────────────────┐  inotify events  ┌──────────────────────┐
+  │  FileMonitor         │──────────────────▶  Debouncer           │
+  │  watchdog Observer   │                  │  collapse 2-5 evt    │
+  └──────────────────────┘                  └──────────┬───────────┘
+                                                       │ FileBatchEvent
+                                                       ▼
+  ┌──────────────────────┐  match when  ┌──────────────────────┐
+  │  RuleEngine          │◀──────────────│  Rule (YAML)         │
+  │  (rules.yaml)        │               └──────────────────────┘
+  └──────────┬───────────┘
+             │ dispatch action
+             ▼
+  ┌──────────────────────┐            ┌──────────────────────┐
+  │  AutomationBackend   │ sandboxed  │  EventStore (SQLite) │
+  │  xdotool / ydotool   │  logging    │  events + actions    │
+  └──────────────────────┘            └──────────────────────┘
+             │
+             ▼
+  ┌──────────────────────┐
+  │  Flask-SocketIO      │  REST + WebSocket →  web_output.html
+  └──────────────────────┘
+```
+
+Modules
+-------
+
+| Module | Purpose |
+|--------|---------|
+| `core/automation_backends.py` | 5 backends (xdotool, ydotool, wtype, atspi, mock) with a unified protocol |
+| `core/file_monitor.py`       | watchdog-based observer + debouncer + path filter |

+docwatch — deployment notes
+============================
+
+Option A: systemd (recommended for desktop sessions)
+----------------------------------------------------
+
+1. Place the project at `/home/<user>/.docwatch/app/`.
+2. Copy `docwatch.service` to `~/.config/systemd/user/docwatch.service`.

+4. `systemctl --user enable --now docwatch`
+5. `journalctl --user -u docwatch -f`
+

+can talk to the user's X11 session. If you're on Wayland, you'll need
+to set `WAYLAND_DISPLAY` and the service will auto-select `ydotool`.

+Option B: docker-compose (mainly for the dashboard / CI)
+-------------------------------------------------------
+
+1. `cp config.yaml.example deploy/config.yaml` and edit.
+2. `cp examples/rules.yaml deploy/rules.yaml` and edit.
+3. `docker compose up -d`
+4. Open `http://localhost:8765/`.
+
+The container starts in `--backend mock` because Docker has no
+display server. To get real xdotool action, run `pip install -e .`
+on the host directly and use Option A.

+  logs/events-*.jsonl  — periodic exports
+  logs/docwatch.log  — running log
