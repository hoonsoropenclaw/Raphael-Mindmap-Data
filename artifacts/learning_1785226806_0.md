+"""In-memory token-bucket rate limiter.
+
+設計：
+- 演算法：token bucket（比 fixed window 平滑，比 leaky bucket 允許 burst）
+- 維度：per client IP（從 request.client.host 取）
+- 狀態：dict[ip, (tokens, last_refill_ts)]，全部在記憶體
+- 限制：單進程 only；多 worker（uvicorn --workers N）會各算一份（demo 級夠用）
+- 可注入：透過 app.state.rate_limiter，測試可以傳 mock 或 reset
+
+公開介面：
+- RateLimiter(capacity, refill_rate)  → .check(ip) -> (allowed: bool, retry_after: float)
+"""
+from __future__ import annotations
+
+import time
+from dataclasses import dataclass
+from threading import Lock
+
+
+@dataclass
+class _Bucket:
+    tokens: float
+    last_refill: float  # monotonic time
+
+
+class RateLimiter:
+    """Thread-safe token bucket rate limiter."""
+
+    def __init__(self, capacity: int = 10, refill_rate: float = 1.0):
+        """
+        capacity   : 桶子最大 token 數（burst 上限）
+        refill_rate: 每秒回填多少 token（steady-state QPS）
+        """
+        if capacity <= 0:
+            raise ValueError("capacity must be > 0")
+        if refill_rate <= 0:
+            raise ValueError("refill_rate must be > 0")
+        self.capacity = float(capacity)
+        self.refill_rate = float(refill_rate)
+        self._buckets: dict[str, _Bucket] = {}
+        self._lock = Lock()
+
+    def _refill(self, bucket: _Bucket, now: float) -> None:
+        elapsed = now - bucket.last_refill
+        if elapsed > 0:
+            bucket.tokens = min(
+                self.capacity, bucket.tokens + elapsed * self.refill_rate
+            )
+            bucket.last_refill = now
+
+    def check(self, key: str, cost: float = 1.0) -> tuple[bool, float]:
+        """嘗試扣 cost 個 token。
+
+        Returns:
+            (allowed, retry_after)

+            - allowed=False 表示拒絕，retry_after=還要等幾秒才有下一個 token
+        """
+        now = time.monotonic()
+        with self._lock:
+            bucket = self._buckets.get(key)
+            if bucket is None:

+                self._buckets[key] = bucket

+
+設計：
+- create_app(db_path, rate_capacity, rate_refill) 工廠函式 → 測試可以傳 in-memory DB + 小桶子
+- RateLimitMiddleware 攔截所有 request，/health 不限流
+- lifespan：startup 開 DB / shutdown 關 DB
+"""
+from __future__ import annotations
+
+import os
+from contextlib import asynccontextmanager
+from typing import AsyncIterator
+
+from fastapi import FastAPI, Request
+from fastapi.responses import JSONResponse
+
+from .db import Database
+from .ratelimit import RateLimiter
+from .routers import router as todos_router
+
+
+# ---------- middleware ----------
+
+class RateLimitMiddleware:
+    """純 ASGI middleware；從 app.state.rate_limiter 取得實例。"""
+
+    def __init__(self, app, exempt_paths: tuple[str, ...] = ("/health",)):
+        self.app = app
+        self.exempt_paths = set(exempt_paths)
+
+    async def __call__(self, scope, receive, send):
+        if scope["type"] != "http":
+            await self.app(scope, receive, send)
+            return
+
+        request = Request(scope, receive=receive)
+        path = request.url.path
+
+        # 不限流路徑
+        if path in self.exempt_paths:
+            await self.app(scope, receive, send)
+            return
+
+        limiter: RateLimiter = request.app.state.rate_limiter
+        # client.host 在測試環境可能為 None → fallback "testclient"
+        client = request.client
+        ip = client.host if client else "testclient"
+
+        allowed, retry_after = limiter.check(ip)
+        if not allowed:
+            response = JSONResponse(
+                status_code=429,
+                content={
+                    "detail": "Rate limit exceeded",
+                    "retry_after_seconds": round(retry_after, 3),
+                },
+                headers={"Retry-After": str(max(1, int(retry_after) + 1))},
+            )
+            await response(scope, receive, send)
+            return
+
+        await self.app(scope, receive, send)
+
+
+# ---------- factory ----------
+
+def create_app(
+    db_path: str | None = None,
+    rate_capacity: int | None = None,

+## 🏗️ 架構決策（為什麼這樣寫）
+
+### 1. 為什麼拆成 5 個模組而不是單檔？
+- **db.py** 純資料層，沒有 FastAPI 依賴 → 可獨立 unit test、可換 DB 後端（換成 Postgres 只改這檔）
+- **ratelimit.py** 同樣零外部依賴 → 純演算法、可重用
+- **routers.py** 只負責 HTTP <-> DB 翻譯
+- **main.py** 只負責 wire-up（middleware、lifespan、router include）
+- **schemas.py** 集中所有 Pydantic model → 之後加新欄位只動這檔
+
+### 2. 為什麼用 token bucket 而不是 FastAPI-Limiter 套件？
+- **零外部依賴** — 42 行寫完，不引入 `slowapi`/`fastapi-limiter`（那些要裝 Redis 或額外設定）
+- **可控可測** — 純 Python class、可以 inject mock clock、可以在測試裡 reset
+- **demo 場景足夠** — 單機單 process，in-memory dict 就夠
+
+### 3. 為什麼 Rate Limit 寫成 middleware 而不是 dependency？
+- **Middleware 早於 route 處理** — 即使 Pydantic 422 也會被限流（避免有人用畸形 payload 攻擊 token 解析）
+- **可以一次攔截所有路徑** — `/health` 是用 exempt list 排除，比每個 route 加 dependency 乾淨
+- **缺點**：拿不到 `Depends` 注入的東西（但限流本來就不需要）
+
+### 4. 為什麼 SQLite 不用 SQLAlchemy？
+- **零 ORM 設定成本** — 一個 `aiosqlite.connect()` 就開工
+- **SQL 夠直觀** — 5 個 CRUD 用原生 SQL 不到 30 行
+- **aiosqlite 已經 async** — 不需要 `run_in_executor` 包同步 sqlite3
+- **升級路徑**：之後真的要換 Postgres / 加關聯 → 換 SQLAlchemy Core（不用換 ORM，保留 SQL-first 風格）
+
+### 5. 為什麼測試用 in-memory SQLite + capacity=3 小桶子？
+- **隔離** — 每個測試都是 fresh DB，不互相污染
+- **速度** — 不用建檔案、沒有磁碟 I/O
+- **rate limit 邊界測試** — 桶子只塞 3 個 token，4 個 request 就觸發 429，測試不用 sleep 等 refill
+
+### 6. 為什麼用 `create_app(db_path, capacity, refill)` factory？
+- **測試可注入** — 每個測試可以拿到不同配置的 app
+- **環境變數可覆寫** — production 用 `os.environ.get(...)` 讀 `TODO_DB_PATH` 等
+- **簡單** — 不需要 `Depends(get_settings)` 那一層複雜度
