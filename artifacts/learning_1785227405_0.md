+#!/usr/bin/env bash
+# sysmon-cron.sh — 把 sysmon 包成 cron / systemd-timer 友善的呼叫
+#
+# 用法（兩種模式任選）：
+#   (a) cron — 5 分鐘跑一次、有 alert 才寄信
+#       */5 * * * * /home/hoonsoropenclaw/.hermes/projects/learning_1785227405_0/sysmon-cron.sh /path/to/alert.log
+#
+#   (b) 背景 daemon（自帶 --watch）
+#       /home/hoonsoropenclaw/.hermes/projects/learning_1785227405_0/sysmon-cron.sh --watch /var/log/sysmon.log
+#
+# 第一個參數是 --watch → 進 daemon 模式；否則把 sysmon.py --once 的 stdout / stderr
+# 導到指定檔（mode 0644、但用 append）。只有 exit code != 0 時，把該次輸出複製到
+# /var/log/sysmon-alerts.log，方便郵件 / 監控系統撈。
+
+set -euo pipefail
+
+HERE="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
+PY="${PYTHON:-python3}"
+SCRIPT="${HERE}/sysmon.py"
+
+if [[ "${1:-}" == "--watch" ]]; then
+    log="${2:-/var/log/sysmon.log}"
+    mkdir -p "$(dirname "$log")"
+    exec "$PY" "$SCRIPT" --watch --interval "${INTERVAL:-60}" --log-file "$log"
+fi
+
+log="${1:-/var/log/sysmon.log}"
+mkdir -p "$(dirname "$log")"
+
+set +e
+output="$("$PY" "$SCRIPT" --once --log-file "$log" 2>&1)"
+rc=$?
+set -e
+
+if [[ $rc -ne 0 ]]; then
+    alert_log="/var/log/sysmon-alerts.log"
+    mkdir -p "$(dirname "$alert_log")"
+    {
+        echo "===== $(date -u +%FT%TZ) ====="
+        echo "$output"
+        echo
+    } >> "$alert_log"
+fi
+
+exit $rc
