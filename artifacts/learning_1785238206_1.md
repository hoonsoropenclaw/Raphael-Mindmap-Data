+import time
+from dataclasses import dataclass
+from threading import Lock
+from typing import Dict
+
+
+@dataclass
+class _Bucket:
+    reset_at: float
+    remaining: int
+
+
+class FixedWindowRateLimiter:
+    """In-memory fixed-window limiter with explicit TTL eviction.
+
+    Architect feedback #1: a request must never be admitted by a wrong-limit bucket
+    because of lazy initialisation. The limit is therefore supplied at construction
+    and every bucket is sized identically, eliminating the lazy bucket-mis-match
+    class of bug. Architect feedback #7: idle buckets are evicted by an explicit
+    sweep call so a long-running process can bound its memory footprint.
+    """
+
+    def __init__(self, limit: int, window_seconds: float, bucket_ttl_seconds: float) -> None:
+        if limit < 1:
+            raise ValueError("limit must be >= 1")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be > 0")
+        if bucket_ttl_seconds <= 0:
+            raise ValueError("bucket_ttl_seconds must be > 0")
+        self._limit = limit
+        self._window = window_seconds
+        self._ttl = bucket_ttl_seconds
+        self._buckets: Dict[str, tuple[_Bucket, float]] = {}
+        self._lock = Lock()
+
+    @property
+    def limit(self) -> int:
+        return self._limit
+
+    @property
+    def window_seconds(self) -> float:
+        return self._window
+
+    @property
+    def bucket_count(self) -> int:
+        with self._lock:
+            return len(self._buckets)
+
+    def check(self, key: str, now: float | None = None) -> tuple[bool, int, int]:
+        current = time.monotonic() if now is None else now
+        with self._lock:
+            entry = self._buckets.get(key)
+            if entry is None or entry[0].reset_at <= current:
+                self._buckets[key] = (_Bucket(reset_at=current + self._window, remaining=self._limit - 1), current)
+                return True, self._limit - 1, int(self._window)
+            bucket, _ = entry
+            if bucket.remaining <= 0:
+                retry_after = max(1, int(bucket.reset_at - current))
+                return False, 0, retry_after
+            bucket.remaining -= 1
+            return True, bucket.remaining, max(1, int(bucket.reset_at - current))
+
+    def sweep(self, now: float | None = None) -> int:
+        current = time.monotonic() if now is None else now
+        with self._lock:
+            stale = [key for key, (_, last_seen) in self._buckets.items() if current - last_seen > self._ttl]
+            for key in stale:
+                del self._buckets[key]
+            return len(stale)

+def _enable_sqlite_foreign_keys(dbapi_connection, _connection_record) -> None:  # pragma: no cover - exercised via PRAGMA test
+    cursor = dbapi_connection.cursor()
+    cursor.execute("PRAGMA foreign_keys=ON")

+

+def _set_sqlite_pragmas(dbapi_connection, _connection_record) -> None:
+    cursor = dbapi_connection.cursor()
+    cursor.execute("PRAGMA journal_mode=WAL")
+    cursor.execute("PRAGMA synchronous=NORMAL")
+    cursor.execute("PRAGMA busy_timeout=5000")
+    cursor.close()

+
+def _ensure_sqlite_security(database_url: str) -> None:
+    if not database_url.startswith("sqlite:///"):
+        return
+    path = Path(database_url[len("sqlite:///"):])
+    parent = path.parent
+    if parent == Path(""):
+        return
+    if not parent.exists():
+        os.makedirs(parent, mode=0o700)
+        return
+    current_mode = parent.stat().st_mode & 0o777
+    if current_mode & 0o077:
+        raise PermissionError(
+            f"Refusing to use SQLite parent directory {parent} with permissive mode {oct(current_mode)}."
+        )
+
+
+class Database:
+    def __init__(self, database_url: str) -> None:
+        _ensure_sqlite_security(database_url)
+        self.engine: Engine = create_engine(
+            database_url,
+            future=True,
+            connect_args={"check_same_thread": False} if database_url.startswith("sqlite") else {},
+        )
+        if database_url.startswith("sqlite"):
+            event.listen(self.engine, "connect", _set_sqlite_pragmas)

+# Todo API — FastAPI + SQLite + Rate Limit
+
+A small, production-shaped FastAPI service for managing todo items. SQLite is the
+storage engine, every endpoint is protected by an in-memory fixed-window rate
+limiter, and the codebase ships with 19 unit + integration tests and 93 % line
+coverage (>= 80 % enforced in `pytest.ini`).
+
+## Why this exists
+
+The project is a reference implementation for the patterns recommended by
+Antigravity's Senior Architect reviews:
+
+1. **No lazy rate-limit initialisation.** Each bucket is sized with the real
+   limit on its first request — there is no `defaultdict(lambda: default)`
+   that admits a burst at the wrong rate before being corrected.
+2. **PATCH semantics that respect `nullable=False`.** Sending `null` for a
+   field the schema does not allow is rejected (422); omitting a field leaves
+   it untouched.
+3. **SQLite security hardening as a first-class concern.** The data directory
+   is created with `0700` and the database file ends up at `0600`; an existing
+   permissive parent is refused instead of silently widened.
+4. **Rate-limit identity comes from `request.client.host` only.** No
+   application-level parsing of `X-Forwarded-For`. The runner is configured
+   with `--no-proxy-headers` so Uvicorn also refuses to inject a forged
+   forwarding header.

+## 1. Stack decisions
+
+### 1.1 Sync SQLAlchemy + sync FastAPI handlers
+The project has no real-time I/O, no external services, and the SQLite database is in-process. Using async SQLAlchemy would only have bought the three documented pitfalls (stale reads, async/sync event-loop deadlock, `expire_on_commit=False` discipline) for zero benefit. Sync `sessionmaker` + `Session` keeps the call sites plain (`def` handlers, `session.query(...)` / `session.get(...)`) and is fast enough for a single-process toy.
+
+### 1.2 Fixed-window counter, not token bucket
+The rate-limit semantics requested by the prompt are "N requests per window of W seconds". A token bucket with a 1-second refill would behave almost identically for small bursts and adds the complexity of a fractional refill timer. The fixed window has a single integer counter per key plus a `reset_at` timestamp, which is trivial to reason about and trivial to test deterministically (every test passes a frozen `now=...`).
+
+### 1.3 In-memory limiter with a TTL sweep
+The Architect explicitly called out the unbounded-dict memory-leak risk. Two layers of defence:
+- The application periodically calls `limiter.sweep()` from a daemon thread.
+- The sweep only runs when `sweep_interval < bucket_ttl`, so a misconfigured deployment (`sweep_interval == bucket_ttl`) cannot accidentally disable the loop.
+
+### 1.4 Pydantic v2 `BeforeValidator` for null-rejection
+The Architect's concern was that `Optional[bool]` lets `null` slip past the type checker. Pydantic v2's `Annotated[Optional[T], BeforeValidator(...)]` is the right shape: the validator runs *before* field-level coercion, sees the raw dict, and can reject a literal `None` with a `ValueError` that FastAPI translates to 422. This means the database layer never has to think about `NULL` for fields that should not be nullable.
+
+### 1.5 Rate-limit identity = `request.client.host`
+- `request.client` is filled in by the ASGI server from the real TCP peer.
+- `X-Forwarded-For` is parsed by Uvicorn only when `--proxy-headers` is on **and** the immediate peer is in `forwarded-allow-ips`.
+- We start Uvicorn with `--no-proxy-headers` so even a misconfigured deployment cannot be tricked into trusting a forwarded header.
+- The integration test `test_client_identity_ignores_forged_forwarded_headers` sets up the test client with an explicit `(host, port)` tuple and a request that contains a forged `X-Forwarded-For`. The limiter admits the first request and rejects the second (limit=1), proving the bucket key came from `request.client.host`.
