+
+A pipeline is just a list of stages.  Each stage is a callable
+`Stage(item: ParsedItem | None) -> ParsedItem | None`:
+
+    - return item to pass it down
+    - return None to drop it
+    - raise to bubble an error (caught by RetryStage if upstream)
+

+
+    Pipeline(stages=[DedupStage(), ...], sinks=[JsonlSink("out.jsonl")])

+
+import csv
+import json
+import logging
+import os
+import time

+from dataclasses import asdict
+from pathlib import Path
+from typing import Protocol
+
+from .result import ParsedItem
+
+log = logging.getLogger(__name__)
+
+
+class Stage(Protocol):
+    """A pure transformation on ParsedItem."""
+
+    def __call__(self, item: ParsedItem | None) -> ParsedItem | None: ...
+
+
+class Sink(Protocol):
+    """Terminal stage — persists items, returns None to stop the chain."""
+
+    def __call__(self, item: ParsedItem | None) -> None: ...
+
+
+class Pipeline:
+    """Chains stages → sinks.  `feed` is the main entry point."""
+
+    def __init__(
+        self,
+        stages: Iterable[Stage] | None = None,
+        sinks: Iterable[Sink] | None = None,
+    ) -> None:
+        self.stages: list[Stage] = list(stages or [])
+        self.sinks: list[Sink] = list(sinks or [])
+
+    def add_stage(self, stage: Stage) -> "Pipeline":
+        self.stages.append(stage)
+        return self
+
+    def add_sink(self, sink: Sink) -> "Pipeline":
+        self.sinks.append(sink)
+        return self
+
+    def feed(self, item: ParsedItem | None) -> ParsedItem | None:
+        """Run one item through every stage, then every sink."""

+# smart_crawler — a modular web scraping framework with API reverse-engineering
+
+**BeautifulSoup + Selenium + reverse-engineering for the post-SPA web.**
+
+This framework treats scraping as a pipeline problem:
+
+```
+fetch ─▶ parse ─▶ transform ─▶ sink
+```
+
+…and bolts an **API reverse-engineering** layer on the side that finds the
+XHR/JSON endpoints a JS-rendered page actually uses, infers their schema,
+and lets you replay them directly with `requests` — no browser required.
+
+## Layout
+
+```
+crawler/
+├── core/             # FetchResult, ParsedItem, Pipeline, CrawlerEngine
+├── fetchers/         # HTTPFetcher, BrowserFetcher (Selenium), APIDirectFetcher
+├── parsers/          # BS4Parser, JSONParser, RegexParser
+├── interceptors/     # SeleniumHAR — convert browser performance log → HAR
+├── pipeline/         # (in core/) JsonlSink, CsvSink, MemorySink, DedupStage
+├── api_reverse/      # APIReverseEngineer — HAR/perf-log/HTML → Endpoint catalog
+├── config/           # YAML site profiles (SiteConfig)
+├── utils/            # URL normalization, throttling, content-type sniffing
+├── demo/             # Flask backend for the web UI
+└── cli.py            # python -m crawler.cli {crawl,reverse,serve}
+```
+
+## Install
+
+```bash
+pip install beautifulsoup4 selenium requests pyyaml flask
+# Optional, only for BrowserFetcher:
+pip install webdriver-manager  # or install chromedriver manually
+```
+
+## Quickstart
+

+
