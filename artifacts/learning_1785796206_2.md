+// Standalone core modules for headless testing (mirrors web_output.html)
+const ErrorKind = Object.freeze({
+  NETWORK: 'NETWORK', TIMEOUT: 'TIMEOUT', ABORTED: 'ABORTED',
+  CLIENT_4XX: 'CLIENT_4XX', RATE_LIMIT: 'RATE_LIMIT',
+  SERVER_5XX: 'SERVER_5XX', PARSE: 'PARSE', UNKNOWN: 'UNKNOWN'
+});
+function classifyError(err, status, body) {
+  if (err?.name === 'AbortError') return { kind: ErrorKind.ABORTED, retryable: false };
+  if (err?.name === 'TimeoutError' || /timeout/i.test(err?.message || '')) return { kind: ErrorKind.TIMEOUT, retryable: true };
+  if (err instanceof TypeError) return { kind: ErrorKind.NETWORK, retryable: true };
+  if (status === 0 || status === undefined) return { kind: ErrorKind.NETWORK, retryable: true };
+  if (status === 408 || status === 429) return { kind: ErrorKind.RATE_LIMIT, retryable: true, status };
+  if (status >= 400 && status < 500) return { kind: ErrorKind.CLIENT_4XX, retryable: false, status };
+  if (status >= 500 && status < 600) return { kind: ErrorKind.SERVER_5XX, retryable: true, status };
+  if (body !== undefined && !validBody(body)) return { kind: ErrorKind.PARSE, retryable: true };
+  return { kind: ErrorKind.UNKNOWN, retryable: true };
+}
+function validBody(body) {
+  if (body === undefined || body === null) return false;
+  if (typeof body === 'object') return true;
+  const s = String(body).trim();
+  return s.length > 0;
+}
+function computeBackoff({ attempt, baseDelay, maxDelay, jitterRatio, consecutiveFailures }) {
+  let delay = baseDelay * Math.pow(2, Math.max(0, attempt - 1));
+  const escalation = 1 + Math.min(consecutiveFailures, 6) * 0.25;
+  delay *= escalation;
+  const jitter = jitterRatio / 100;
+  delay = delay * (1 - jitter) + Math.random() * delay * jitter;
+  return Math.min(Math.round(delay), maxDelay);
+}
+class CircuitBreaker {
+  constructor(threshold, cooldownMs = 5000) {
+    this.threshold = threshold; this.cooldownMs = cooldownMs;
+    this.failures = 0; this.state = 'CLOSED'; this.openedAt = 0;
+  }
+  canPass() {
+    if (this.state === 'CLOSED') return true;
+    if (this.state === 'OPEN' && Date.now() - this.openedAt >= this.cooldownMs) {
+      this.state = 'HALF_OPEN'; return true;
+    }
+    return this.state === 'HALF_OPEN';
+  }
+  recordSuccess() { this.failures = 0; this.state = 'CLOSED'; }
+  recordFailure() {
+    this.failures++;
+    if (this.failures >= this.threshold && this.state !== 'OPEN') {
+      this.state = 'OPEN'; this.openedAt = Date.now();
+    }
+  }
+}
+function dispatch({ attempt, maxAttempts, classification, breaker, consecutiveFailures, baseDelay, maxDelay, jitterRatio }) {
+  if (classification.kind === ErrorKind.ABORTED) return { type: 'abort', reason: 'AbortSignal received' };
+  if (!breaker.canPass() && breaker.state === 'OPEN') return { type: 'abort', reason: `Circuit breaker OPEN (${breaker.failures} consecutive failures)` };
+  if (classification.kind === ErrorKind.CLIENT_4XX) return { type: 'abort', reason: `Non-retryable HTTP ${classification.status}` };
