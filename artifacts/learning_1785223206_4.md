+// useLocalStorage - persistent state with SSR safety and error handling
+import { useEffect, useState, useRef, useCallback } from 'react';
+
+export function useLocalStorage<T>(
+  key: string,
+  initialValue: T,
+): [T, (value: T | ((prev: T) => T)) => void, () => void] {
+  // Use a ref to track if we've already loaded from storage
+  const loadedRef = useRef(false);
+  const [storedValue, setStoredValue] = useState<T>(initialValue);
+
+  // Load from storage on mount (client-side only)
+  useEffect(() => {
+    if (typeof window === 'undefined') return;
+    try {
+      const item = window.localStorage.getItem(key);
+      if (item !== null) {
+        setStoredValue(JSON.parse(item) as T);
+      }
+    } catch (err) {
+      console.warn(`useLocalStorage: failed to read key "${key}"`, err);
+    } finally {
+      loadedRef.current = true;
+    }
+  }, [key]);
+
+  // Persist to storage when value changes (but only after initial load)
+  useEffect(() => {
+    if (!loadedRef.current) return;
+    if (typeof window === 'undefined') return;
+    try {
+      window.localStorage.setItem(key, JSON.stringify(storedValue));
+    } catch (err) {
+      console.warn(`useLocalStorage: failed to write key "${key}"`, err);
+    }
+  }, [key, storedValue]);
+
+  const setValue = useCallback((value: T | ((prev: T) => T)) => {
+    setStoredValue((prev) => {
+      const next = value instanceof Function ? value(prev) : value;
+      return next;
+    });
+  }, []);
+
+  const clearValue = useCallback(() => {
+    setStoredValue(initialValue);
+    if (typeof window !== 'undefined') {
+      window.localStorage.removeItem(key);
+    }
+  }, [initialValue, key]);
+
+  return [storedValue, setValue, clearValue];

+// useDebounce - debounce a value (for search input)
+import { useEffect, useState } from 'react';
+
+export function useDebounce<T>(value: T, delayMs: number = 250): T {
+  const [debounced, setDebounced] = useState(value);
+
+  useEffect(() => {
+    const timer = setTimeout(() => setDebounced(value), delayMs);
+    return () => clearTimeout(timer);
+  }, [value, delayMs]);
+
+  return debounced;

+export type ShortcutHandler = (event: KeyboardEvent) => void;
+
+export interface Shortcut {
+  key: string;
+  meta?: boolean; // cmd on mac, ctrl on win/linux
+  shift?: boolean;
+  handler: ShortcutHandler;
+  description?: string;
+}
+
+export function useKeyboardShortcuts(shortcuts: Shortcut[]): void {
+  useEffect(() => {
+    function onKeyDown(event: KeyboardEvent) {
+      for (const shortcut of shortcuts) {
+        const metaMatch = shortcut.meta
+          ? event.metaKey || event.ctrlKey
+          : !event.metaKey && !event.ctrlKey;
+        if (!metaMatch) continue;
+        if (shortcut.shift !== undefined && event.shiftKey !== shortcut.shift) continue;
+        if (event.key.toLowerCase() === shortcut.key.toLowerCase()) {
+          event.preventDefault();
+          shortcut.handler(event);
+          return;
+        }
+      }
+    }
+
+    window.addEventListener('keydown', onKeyDown);
+    return () => window.removeEventListener('keydown', onKeyDown);
+  }, [shortcuts]);

+# Task Management Dashboard — Implementation Rationale
+
+## 1. 整體架構 (Architecture Overview)
+
+採用**分層 reducer + custom hooks + 純函式 utility** 的三層架構，避免把所有邏輯塞進單一巨型元件。
+
+```
+TaskDashboard (orchestrator)
+├── useTaskReducer()        ← 集中狀態管理（任務、篩選、表單、選取）
+│   ├── useReducer + Action types
+│   └── useLocalStorage()   ← 持久化 hook
+├── useDebounce()           ← 搜尋輸入去抖
+├── useKeyboardShortcuts()  ← 快捷鍵 (N 新增、Esc 關閉、/ 搜尋)
+├── computeStats()          ← 純函式，從 tasks 計算統計
+├── filterTasks()           ← 純函式，篩選 + 排序 pipeline
+└── <TaskList>              ← 顯示層
+    ├── <TaskCard>          ← 單筆任務（展開/收合/切狀態/編輯/刪除）
+    │   └── <Badge>         ← 狀態/優先度小標籤
+    ├── <TaskForm>          ← 新增/編輯 modal
+    ├── <FilterBar>         ← 搜尋 + 篩選 + 排序 + tag filter
+    └── <StatsBar>          ← 4 個關鍵指標
+```
+
+**為什麼這樣切**:
+- 邏輯跟 UI 分離 → reducer 純函式可獨立測試
+- 工具函式 (`filterTasks`, `computeStats`) 可重複使用、可 memoize
+- Custom hooks 封裝副作用（localStorage、debounce、keyboard）
+- 元件只負責 render，props in / events out
+
+## 2. 狀態管理設計 (State Management Design)
+
+### 2.1 為什麼選 `useReducer` 而不是 `useState` 多個
+
+- 16 種 actions 全部走單一 reducer，每個 action 都有對應的 case
+- 中央化狀態轉換邏輯，比散在多個 `useState` setter 更易追蹤
+- 容易擴展：新增 action type 只要在 reducer 裡加一個 case
+
+### 2.2 狀態結構
+
+```typescript
+interface AppState {
+  tasks: Task[];            // 主資料
+  filter: FilterState;      // 篩選條件（search/status/priority/tags/sort）
+  selectedTaskId: string | null;  // 選取中的任務
+  isFormOpen: boolean;      // 表單開關
+  editingTaskId: string | null;   // 編輯中的任務 id
+}
+```
+
+**單一選擇**：把 UI 控制狀態（modal 開關、選取、編輯）跟資料（tasks）放同一個 state tree。**優點**：所有 state transition 集中；**代價**：modal 開關的細粒度變化會 re-render 整個 dashboard → 用 `useMemo` 在 `TaskDashboard` 裡把 `visibleTasks` 跟 `editingTask` 算出來，避免不必要的子元件 re-render。
+
+### 2.3 Action 類型設計
+
+15 個 actions 涵蓋所有 mutation：
+- `ADD_TASK` / `UPDATE_TASK` / `DELETE_TASK` — CRUD
+- `SET_STATUS` / `TOGGLE_SUBTASK` — 部分更新
+- `BULK_DELETE` / `BULK_SET_STATUS` / `CLEAR_COMPLETED` — 批量操作
+- `SELECT_TASK` / `OPEN_FORM` / `CLOSE_FORM` — UI 控制
+- `SET_FILTER` / `RESET_FILTER` / `TOGGLE_TAG_FILTER` — 篩選
+- `IMPORT_TASKS` — 重置/匯入
+
+每個 action 都是 discriminated union，TypeScript 強制保證 payload 型別正確。
+
+### 2.4 複雜的衍生狀態 (Derived State)
+
+`visibleTasks`、`stats`、`editingTask` 都是從 `state.tasks` + `state.filter` 衍生，**不存進 state**。每次 render 重算（用 `useMemo` 防止浪費）。
+
+```typescript
+const visibleTasks = useMemo(() => filterTasks(tasks, filter), [tasks, filter]);
+const stats = useMemo(() => computeStats(tasks), [tasks]);
+const editingTask = useMemo(() => tasks.find(t => t.id === editingTaskId), [tasks, editingTaskId]);
+```
