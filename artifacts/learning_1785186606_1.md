+from __future__ import annotations
+
+import math
+import threading
+import time
+from dataclasses import dataclass
+from typing import Callable
+
+
+@dataclass(slots=True)
+class Bucket:
+    tokens: float
+    updated_at: float
+
+
+class TokenBucketLimiter:
+    def __init__(
+        self,
+        capacity: int,
+        refill_per_second: float,
+        clock: Callable[[], float] = time.monotonic,
+    ) -> None:
+        if capacity <= 0 or refill_per_second <= 0:
+            raise ValueError("rate limit values must be positive")
+        self.capacity = capacity
+        self.refill_per_second = refill_per_second
+        self.clock = clock
+        self._buckets: dict[str, Bucket] = {}
+        self._lock = threading.Lock()
+
+    def consume(self, key: str) -> tuple[bool, int]:
+        with self._lock:
+            now = self.clock()
+            bucket = self._buckets.setdefault(key, Bucket(float(self.capacity), now))
+            elapsed = max(0.0, now - bucket.updated_at)
+            bucket.tokens = min(
+                float(self.capacity), bucket.tokens + elapsed * self.refill_per_second
+            )
+            bucket.updated_at = now
+            if bucket.tokens >= 1.0:
+                bucket.tokens -= 1.0
+                return True, 0
+            retry_after = max(1, math.ceil((1.0 - bucket.tokens) / self.refill_per_second))
+            return False, retry_after

+        return

+# 實作思路與架構審查回應
+
+## 1. 規格解讀
+
+任務中的「讀取」與「寫入」路徑是空白模板，因此沒有猜測不存在的目的地。專案直接建立於本輪工作目錄；實作說明放在本檔。已實際讀取 `/home/hoonsoropenclaw/.hermes/architect_feedback.md`。
+
+## 2. 核心架構
+
+- `app/main.py`：application factory、middleware、CRUD endpoints。
+- `app/database.py`：engine/session factory、WAL、busy timeout、檔案權限。
+- `app/models.py`：SQLAlchemy Todo model。
+- `app/schemas.py`：Pydantic v2 input/output validation。
+- `app/rate_limit.py`：per-client token bucket，使用 `threading.Lock` 保證 burst 原子性。
+- `tests/`：CRUD、錯誤路徑、rate limit、HTTP 並發 burst、SQLite 權限。
+
+## 3. 為何採同步 SQLAlchemy
+
+此服務只有本機 SQLite CRUD，沒有單次請求內的大量外部 I/O。使用同步 `sessionmaker` 搭配同步 `def` endpoint，FastAPI 會放入 thread pool；避免 `async def` 混用同步 DB 所造成的 event-loop blocking/deadlock，也避免 async session identity-map 與 lifecycle 複雜度。
+
+## 4. Senior Architect 回饋吸收
+
+### 並發壓力
+
+不只測順序請求：
+- `test_threaded_burst_never_overgrants`：100 次 concurrent consume，容量 5 時精確只放行 5 次。
+- `test_concurrent_http_burst_is_capped`：50 個 concurrent ASGI HTTP requests，驗證 5 個 200、45 個 429。
+
+### SQLite WAL/SHM 權限
+
+三層防護：
+1. `app/__init__.py` 與入口 `main.py` 設 `os.umask(0o077)`。
+2. SQLite 開啟前，以 `os.open(..., 0o600)` 預建主 DB。
+3. connect、session close、shutdown 時執行 `secure_sqlite_files()`，將 DB、`-wal`、`-shm` 固定為 `0600`。
+
+只對本程式新建的 data 目錄設 `0700`，不修改既有共享父目錄；這是 TDD 時捕捉到的 `/tmp chmod` 邊界案例。
+
+## 5. Rate limit 設計
+
+Token bucket 允許短暫 burst，並以固定速率恢復。key 優先順序：`X-API-Key` → `X-Forwarded-For` 第一個 IP → socket client IP。超限時回標準 `429` 與向上取整的 `Retry-After`。本實作適合單 process；多 worker/多主機部署需改 Redis/Lua 原子 bucket。
+
+## 6. TDD 與驗證鏈
+
+1. 先寫測試，首次執行得到 `ModuleNotFoundError`（RED）。
+2. 實作後得到 4 個 422，定位為 postponed annotations 使 factory-local dependency alias 無法被 FastAPI 正確解析。
+3. 移除該 annotation 延遲並修正共享父目錄權限後，全綠。
+4. 最終執行 coverage 與獨立 smoke test；結果以實際命令輸出為準。
+
+## 7. 已知邊界
+
+- rate-limit state 位於 process memory；重啟會清空，多 worker 不共享。
+- SQLite 適合單機中小型服務；高寫入併發應換 PostgreSQL。
+- 無 authentication；`X-API-Key` 此處只作 rate-limit key，不是身份驗證。
+- schema 目前用 `create_all`；正式演進應加入 Alembic migration。
