+import os
+from collections import defaultdict, deque
+from threading import Lock
+from time import monotonic
+
+from fastapi import Request
+from fastapi.responses import JSONResponse
+
+
+class SlidingWindowRateLimiter:
+    def __init__(self, max_requests: int, window_seconds: int) -> None:
+        self.max_requests = max_requests
+        self.window_seconds = window_seconds
+        self._requests: dict[str, deque[float]] = defaultdict(deque)
+        self._lock = Lock()
+
+    def reset(self) -> None:
+        with self._lock:
+            self._requests.clear()
+
+    def check(self, key: str) -> tuple[bool, int]:
+        now = monotonic()
+        cutoff = now - self.window_seconds
+        with self._lock:
+            timestamps = self._requests[key]
+            while timestamps and timestamps[0] <= cutoff:
+                timestamps.popleft()
+            if len(timestamps) >= self.max_requests:
+                retry_after = max(1, int(timestamps[0] + self.window_seconds - now) + 1)
+                return False, retry_after
+            timestamps.append(now)
+            return True, 0
+
+
+limiter = SlidingWindowRateLimiter(
+    max_requests=int(os.getenv("RATE_LIMIT_MAX_REQUESTS", "5")),
+    window_seconds=int(os.getenv("RATE_LIMIT_WINDOW_SECONDS", "60")),
+)
+
+
+async def rate_limit_middleware(request: Request, call_next):
+    client_ip = request.client.host if request.client else "unknown"
+    allowed, retry_after = limiter.check(client_ip)
+    if not allowed:
+        return JSONResponse(
+            status_code=429,
+            content={"detail": "Rate limit exceeded"},
+            headers={"Retry-After": str(retry_after)},
+        )
+    return await call_next(request)

+from fastapi import Depends, FastAPI, HTTPException, Response, status
+from sqlalchemy import select
+from sqlalchemy.orm import Session
+
+from .database import Base, engine, get_db
+from .models import Todo
+from .rate_limit import rate_limit_middleware
+from .schemas import TodoCreate, TodoRead, TodoUpdate
+
+
+Base.metadata.create_all(bind=engine)
+
+app = FastAPI(title="Todo API", version="1.0.0")
+app.middleware("http")(rate_limit_middleware)
+
+
+@app.get("/health")
+def health() -> dict[str, str]:
+    return {"status": "ok"}
+
+
+@app.post("/todos", response_model=TodoRead, status_code=status.HTTP_201_CREATED)
+def create_todo(payload: TodoCreate, db: Session = Depends(get_db)) -> Todo:
+    todo = Todo(title=payload.title)
+    db.add(todo)
+    db.commit()
+    db.refresh(todo)
+    return todo
+
+
+@app.get("/todos", response_model=list[TodoRead])
+def list_todos(db: Session = Depends(get_db)) -> list[Todo]:
+    return list(db.scalars(select(Todo).order_by(Todo.id)))
+
+
+def require_todo(todo_id: int, db: Session) -> Todo:
+    todo = db.get(Todo, todo_id)
+    if todo is None:
+        raise HTTPException(status_code=404, detail="Todo not found")
+    return todo
+
+
+@app.get("/todos/{todo_id}", response_model=TodoRead)
+def get_todo(todo_id: int, db: Session = Depends(get_db)) -> Todo:
+    return require_todo(todo_id, db)
+
+
+@app.patch("/todos/{todo_id}", response_model=TodoRead)
+def update_todo(todo_id: int, payload: TodoUpdate, db: Session = Depends(get_db)) -> Todo:
+    todo = require_todo(todo_id, db)
+    changes = payload.model_dump(exclude_unset=True)
+    for field, value in changes.items():
+        setattr(todo, field, value)
+    db.commit()
+    db.refresh(todo)
+    return todo
+
+
+@app.delete("/todos/{todo_id}", status_code=status.HTTP_204_NO_CONTENT)
+def delete_todo(todo_id: int, db: Session = Depends(get_db)) -> Response:
+    todo = require_todo(todo_id, db)
+    db.delete(todo)
+    db.commit()
+    return Response(status_code=status.HTTP_204_NO_CONTENT)

+# Implementation Notes
+
+## Design
+
+- FastAPI provides HTTP routing, validation, OpenAPI documentation, and dependency injection.
+- SQLAlchemy 2.0 uses a synchronous session because SQLite operations are small and endpoints are synchronous `def` handlers; FastAPI executes them in its thread pool, avoiding event-loop blocking and async/sync session deadlocks.
+- SQLite persists data in `todos.db`; `DATABASE_URL` can override the location.
+- Pydantic schemas separate request/response contracts from ORM models and reject blank or oversized titles.
+- The rate limiter is an in-process, per-client-IP sliding window guarded by a lock. It returns HTTP 429 and `Retry-After` when exhausted. Defaults are configurable through `RATE_LIMIT_MAX_REQUESTS` and `RATE_LIMIT_WINDOW_SECONDS`.
+
+## API behavior
+
+- `POST /todos`: create a todo, HTTP 201.
+- `GET /todos`: list todos ordered by ID.
+- `GET /todos/{id}`: retrieve one todo or HTTP 404.
+- `PATCH /todos/{id}`: partially update title/completed state.
+- `DELETE /todos/{id}`: delete and return HTTP 204.
+- `GET /health`: lightweight health endpoint.
+
+## Testing strategy
+
+Tests were written before production implementation and first failed because `app.main` did not exist (RED). They cover creation/listing, update/deletion, missing-record 404 behavior, and rate-limit 429 behavior. An autouse fixture resets limiter state between tests while preserving state within each test. SQLite integration is exercised through the real API and ORM rather than mocked repositories.
+
+## Trade-offs and production path
+
+The in-memory limiter is intentionally simple and correct only per process. Multi-worker or multi-host deployment should replace it with Redis plus an atomic Lua script or a managed gateway limiter. SQLite is appropriate for a small single-node API; higher write concurrency should move to PostgreSQL with migrations. Schema creation currently uses `create_all`; production deployments should use Alembic migrations.
