+def parse_time(value: str) -> datetime:
+    if value.endswith("Z"):

+    dt = datetime.fromisoformat(value)
+    if dt.tzinfo is None:
+        raise ValueError(f"datetime must include timezone offset: {value}")
+    return dt.astimezone(timezone.utc)
