+function adjacencyFor(nodes, edges) {
+  const ids = new Set(nodes.map((node) => node.id));

+  for (const edge of edges) {
+    if (!ids.has(edge.source) || !ids.has(edge.target)) continue;
+    adjacency.get(edge.source).push(edge);
+    incoming.set(edge.target, incoming.get(edge.target) + 1);
+  }
+
+  return { adjacency, incoming };

+export function topologicalSort(nodes, edges) {
+  const { adjacency, incoming } = adjacencyFor(nodes, edges);
+  const queue = nodes
+    .filter((node) => incoming.get(node.id) === 0)
+    .map((node) => node.id);
+  const order = [];
+
+  while (queue.length > 0) {
+    const id = queue.shift();
+    order.push(id);
+
+    for (const edge of adjacency.get(id) ?? []) {
+      const nextDegree = incoming.get(edge.target) - 1;
+      incoming.set(edge.target, nextDegree);
+      if (nextDegree === 0) queue.push(edge.target);
+    }
+  }
+
+  const ordered = new Set(order);
+  const cyclicNodeIds = nodes
+    .map((node) => node.id)
+    .filter((id) => !ordered.has(id));
+
+  return {
+    order,
+    hasCycle: cyclicNodeIds.length > 0,
+    cyclicNodeIds,
+  };
