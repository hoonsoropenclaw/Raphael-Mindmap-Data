+# 跨瀏覽器視覺回歸測試自動化流程
+
+使用 **Playwright** 跨三大瀏覽器 (Chromium / Firefox / WebKit) 抓取頁面截圖，並以 **pixelmatch** 進行像素級 diff,自動產出互動式 HTML 報告。
+
+---
+
+## 📐 架構總覽
+
+```
+┌─────────────────────────────────────────────────────────────┐
+│                  tests/visual/run.js                        │
+│                  (CLI entrypoint)                           │
+└──────────────────────┬──────────────────────────────────────┘
+                       │
+       ┌───────────────┼───────────────┐
+       ▼               ▼               ▼
+  ┌─────────┐    ┌──────────┐    ┌──────────┐
+  │chromium │    │ firefox  │    │  webkit  │   ← 3 引擎平行
+  └────┬────┘    └─────┬────┘    └─────┬────┘
+       │               │               │
+       └───────────────┼───────────────┘
+                       ▼
+              ┌─────────────────┐
+              │   screenshot    │   ← page.screenshot()
+              │   capture       │      fullPage + 固定 viewport
+              └────────┬────────┘
+                       ▼
+              ┌─────────────────┐
+              │  baseline 對比  │   ← 第一次跑會建立 baseline
+              │  + pixelmatch   │      之後每次跟 baseline diff
+              └────────┬────────┘
+                       ▼
+              ┌─────────────────┐
+              │ HTML 報告產生器 │   ← 互動式 diff viewer
+              │ reports/index   │      baseline / actual / diff
+              └─────────────────┘      三圖並排,可放大
+```
+

+/**
+ * 啟動單一瀏覽器實例 (headless,固定 viewport)

+async function launchEngine(engineName, viewport) {
+  const launchers = { chromium, firefox, webkit };
+  const browser = await launchers[engineName].launch({ headless: true });
+  const context = await browser.newContext({
+    viewport,
+    // 關閉動畫,避免 race
+    reducedMotion: 'reduce',
+    // 統一 device scale factor
+    deviceScaleFactor: 1,
+  });
+  return { browser, context };
+}

+ * 對單一瀏覽器/viewport 抓取截圖
+ * @returns {Promise<{name, engine, viewport, png:Buffer, width, height, timestamp}>}
+ */
+async function captureOne(targetName, browserKey, url) {
+  const cfg = BROWSER_MATRIX[browserKey];
+  if (!cfg) throw new Error(`Unknown browser key: ${browserKey}`);
+
+  const { browser, context } = await launchEngine(cfg.engine, cfg.viewport);
+  const page = await context.newPage();
+
+  try {
+    await page.goto(url, { waitUntil: 'networkidle', timeout: 30_000 });
+    // 額外等待字體 / 圖片就緒

+    await page.evaluate(() => document.fonts.ready);

+    const png = await page.screenshot({ fullPage: true, type: 'png' });

+      browserKey,

+      viewport: cfg.viewport,

+      height: cfg.viewport.height,
