+/**
+ * Next.js Middleware（Edge Runtime）
+ *
+ * 這是「nextjs_rbac_middleware」對應實作：
+ * - 從 cookie 讀 session token
+ * - 解析 user 角色
+ * - 對照 ROUTE_POLICIES 判定是否能進
+ * - 不能進 → 導向 /login?from=<原路徑>&missing=<缺少的權限>
+ */
+
+import { NextRequest, NextResponse } from 'next/server';
+import {
+  parseUserFromToken,
+  hasPermissions,
+  ROLE_PERMISSIONS,
+  ROUTE_POLICIES,
+  type Permission,
+} from './lib/rbac-matrix';
+
+export const config = {
+  matcher: [
+    /*
+     * 匹配所有 dashboard 與 api 路由
+     * 但排除靜態資源、_next 內部檔案、根路徑（首頁 /login 公開）
+     */
+    '/dashboard/:path*',
+    '/api/:path*',
+  ],
+};
+
+export function middleware(req: NextRequest) {
+  const { pathname } = req.nextUrl;
+
+  // 1) 找出此路由對應的 policy
+  const policy = ROUTE_POLICIES.find((p) => pathname.startsWith(p.pattern));
+  if (!policy) {
+    // 沒被 policy 涵蓋的路由 = 公開（首頁、/login 等）
+    return NextResponse.next();
+  }
+
+  // 2) 解析當前使用者
+  const token = req.cookies.get('session')?.value;
+  const user = parseUserFromToken(token);
+
+  // 3) 未登入 → 導向 /login
+  if (!user) {
+    const url = req.nextUrl.clone();
+    url.pathname = '/login';
+    url.searchParams.set('from', pathname);
+    return NextResponse.redirect(url);
+  }
+
+  // 4) 權限檢查
+  if (!hasPermissions(user, policy.requires)) {
+    const url = req.nextUrl.clone();
+    url.pathname = '/403';
+    const granted = ROLE_PERMISSIONS[user.role] ?? [];
+    const missing = policy.requires.filter((p) => !granted.includes(p));
+    url.searchParams.set('missing', missing.join(','));
+    url.searchParams.set('role', user.role);
+    return NextResponse.redirect(url);
+  }
+
+  // 5) 放行 + 在 header 註明 user 身份（server component 可讀）
+  const res = NextResponse.next();
+  res.headers.set('x-user-id', user.id);
+  res.headers.set('x-user-role', user.role);
+  return res;
