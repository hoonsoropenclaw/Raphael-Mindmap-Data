+"""OpenWeatherMap REST API client。
+
+故意不依賴任何第三方 SDK — `requests` 已經是底層足夠好用的東西，
+自己包一層可控性更高、錯誤訊息更貼近使用者。
+"""
+
+from __future__ import annotations
+
+import logging
+from typing import Any
+
+import requests
+
+logger = logging.getLogger(__name__)
+
+OPENWEATHER_BASE = "https://api.openweathermap.org/data/2.5/weather"
+
+
+class WeatherError(Exception):
+    """所有天氣 API 失敗的根類別。"""
+
+
+class CityNotFoundError(WeatherError):
+    """城市查不到 — 通常是拼字錯或該城市不在 OWM 資料庫。"""
+
+
+class APIKeyError(WeatherError):
+    """API key 無效、quota 用完、或 5xx 等伺服器端問題。"""
+
+
+class NetworkError(WeatherError):
+    """網路層失敗 — DNS / TLS / timeout。"""
+
+
+def fetch_weather(city: str, api_key: str, units: str = "metric", lang: str = "zh_tw", timeout: float = 8.0) -> dict[str, Any]:
+    """查詢指定城市的天氣，回傳結構化 dict。
+
+    拋出對應的 WeatherError 子類別讓上層決定回什麼給使用者。
+    """
+    if not city or not city.strip():
+        raise WeatherError("城市名稱不可為空")
+
+    params = {
+        "q": city.strip(),
+        "appid": api_key,
+        "units": units,
+        "lang": lang,
+    }
+
+    try:
+        resp = requests.get(OPENWEATHER_BASE, params=params, timeout=timeout)
+    except requests.exceptions.Timeout as exc:
+        raise NetworkError(f"天氣 API 逾時（>{timeout}s）") from exc
+    except requests.exceptions.ConnectionError as exc:
+        raise NetworkError(f"無法連線到天氣 API：{exc}") from exc
+    except requests.exceptions.RequestException as exc:
+        raise NetworkError(f"天氣 API 請求失敗：{exc}") from exc
+
+    # OWM 用 HTTP status code 表達業務錯誤
+    if resp.status_code == 404:
+        raise CityNotFoundError(f"找不到城市：{city!r}")
+    if resp.status_code == 401:
+        raise APIKeyError("OpenWeatherMap API key 無效（401）")
+    if resp.status_code == 429:
+        raise APIKeyError("OpenWeatherMap API 已達免費額度上限（429）")
+    if resp.status_code >= 500:
+        raise APIKeyError(f"OpenWeatherMap 伺服器錯誤（{resp.status_code}）")
+
+    try:
+        resp.raise_for_status()
+    except requests.exceptions.HTTPError as exc:
+        # 沒歸類到上面的 4xx — 一律當 API key / 配額 / 未知錯誤處理
+        raise APIKeyError(f"OpenWeatherMap 回應錯誤（{resp.status_code}）") from exc
+
+    try:
+        return resp.json()
+    except ValueError as exc:
+        raise APIKeyError(f"OpenWeatherMap 回應不是合法 JSON：{exc}") from exc
