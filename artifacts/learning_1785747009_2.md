+"""EventBus 異步事件分發器。
+
+設計目標：
+- 輕量 pub/sub；支援多訂閱者、wildcard、非同步 handler
+- 錯誤隔離：某 handler 拋例外不影響其他訂閱者（事件總線必須存活）
+- 訂閱/退訂為 O(1) 查找、並發發布使用 asyncio.gather 平行派送
+"""
+from __future__ import annotations
+
+import asyncio
+import inspect
+import logging
+from dataclasses import dataclass, field
+from typing import Any, Awaitable, Callable, Union
+
+logger = logging.getLogger(__name__)
+
+Handler = Callable[["Event"], Union[Awaitable[None], None]]
+
+
+@dataclass(frozen=True)
+class Event:
+    """不可變事件物件：type 必填、payload 為任意 JSON-serializable 物件。"""
+
+    type: str
+    payload: Any = None
+
+
+class EventBus:
+    """異步事件總線。
+
+    內部結構：type -> list[handler] 的字典，'*' 為 wildcard。
+    publish 流程：
+      1. 複製一份該 type 的 handler 清單（避免 unsubscribe 競態）
+      2. asyncio.gather 平行執行，return_exceptions=True 達成錯誤隔離
+    """
+
+    def __init__(self) -> None:
+        self._subscribers: dict[str, list[Handler]] = {}
+
+    def subscribe(self, event_type: str, handler: Handler) -> None:
+        """訂閱指定 type。支援 '*' wildcard。"""
+        if event_type not in self._subscribers:
+            self._subscribers[event_type] = []
+        if handler not in self._subscribers[event_type]:
+            self._subscribers[event_type].append(handler)
+
+    def unsubscribe(self, event_type: str, handler: Handler) -> None:
+        """取消訂閱。若 handler 不存在則 no-op。"""
+        handlers = self._subscribers.get(event_type, [])
+        if handler in handlers:
+            handlers.remove(handler)
+
+    def listener_count(self, event_type: str) -> int:
+        """回傳該 type 的訂閱者數（含 wildcard 訂閱者）。"""
+        direct = len(self._subscribers.get(event_type, []))
+        wildcard = len(self._subscribers.get("*", []))
+        return direct + wildcard
