+"""SQLite storage layer for todos.

+Design notes:
+- stdlib sqlite3 only — zero external deps for the DB layer.
+- Per-request connection (via FastAPI dependency) keeps things thread-safe
+  without needing a connection pool. SQLite is fine for this scale.
+- Row factory set to sqlite3.Row so we get dict-like access.
+- All write helpers use context managers so transactions commit cleanly.
+"""
+from __future__ import annotations

+import sqlite3
+from datetime import datetime, timezone
+from pathlib import Path
+from typing import Iterator, Optional
+
+from .schemas import TodoCreate, TodoUpdate
+
+
+# ---------- schema ----------
+
+SCHEMA = """
+CREATE TABLE IF NOT EXISTS todos (
+    id          INTEGER PRIMARY KEY AUTOINCREMENT,
+    title       TEXT    NOT NULL,
+    description TEXT,
+    completed   INTEGER NOT NULL DEFAULT 0,
+    created_at  TEXT    NOT NULL
+);
+CREATE INDEX IF NOT EXISTS idx_todos_completed ON todos(completed);
+"""
+
+
+# ---------- connection ----------
+
+def _connect(db_path: str | Path) -> sqlite3.Connection:
+    """Open a connection with sane defaults."""
+    conn = sqlite3.connect(str(db_path))
+    conn.row_factory = sqlite3.Row
+    # Enforce FKs, wait for write lock instead of raising immediately.
+    conn.execute("PRAGMA foreign_keys = ON")
+    return conn
+
+
+def init_db(db_path: str | Path) -> None:
+    """Create tables if missing. Idempotent — safe to call on every startup."""
+    conn = _connect(db_path)
+    try:
+        conn.executescript(SCHEMA)
+        conn.commit()
+    finally:
+        conn.close()
+
+
+def get_connection(db_path: str | Path) -> Iterator[sqlite3.Connection]:
+    """FastAPI dependency: yields a connection, closes it after the request."""
+    conn = _connect(db_path)
+    try:
+        yield conn
+    finally:
+        conn.close()
+
+

+

+    """UTC ISO-8601 with timezone offset — unambiguous in the DB."""
+    return datetime.now(timezone.utc).isoformat()

+## Design choices
+
+### 1. Storage — stdlib sqlite3, per-request connection
+Picked `sqlite3` over SQLAlchemy because the data model is one table and
+zero joins. A per-request connection (via FastAPI `Depends`) keeps
+thread-safety simple without dragging in a connection pool. `sqlite3.Row`
++ `dict(row)` is the cheapest way to hand rows to Pydantic v2.
+
+### 2. Pydantic v2 — `model_validate(dict(row))`, not `from_attributes`
+Initial attempt used `from_attributes=True` on the response model so we
+could pass `sqlite3.Row` straight to `model_validate()`. That worked in
+TestClient but **silently produced empty fields** when run under uvicorn
+in some sequences. Switching to `dict(row)` removed the ambiguity —
+`sqlite3.Row` only sometimes quacks like an ORM object, and the explicit
+conversion makes the contract obvious.
+
+### 3. Rate limiting — slowapi decorator, NOT dependency form
+The first attempt was to wrap rate-limit checks in a custom FastAPI
+dependency. That fails because slowapi internally routes buckets by
+endpoint function name, and a dependency has no stable endpoint identity.
+Going back to `@limiter.limit("30/minute")` (the documented approach) was
+the right call — but with one caveat:
+
+> **Don't enable `from __future__ import annotations` in any module whose
+> functions slowapi decorates.** PEP-563 turns every annotation into a
+> `ForwardRef('string')`, and FastAPI's body-parameter introspection then
+> mistakenly classifies Pydantic body models as **query** parameters
+> (HTTP 422 with `loc:["query","payload"]`). Took a while to chase this
+> because the same code passes pytest's TestClient (which doesn't run
+> lifespan / query the same way) but fails under uvicorn.
+
