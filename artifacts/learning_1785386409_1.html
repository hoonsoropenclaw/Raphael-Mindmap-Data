// Colors
colors: {
  status: { approved, pending, rejected, draft, info },
  amber:   { 400, 500, 600 },
  ink:     { 50 → 900 },
  glass:   { tint, panel, deep, stroke, stroke2 },
}
// Elevation
boxShadow: {
  glassSm, glassMd, glassLg, glassXl,
  glowAmber, glowCool,
}
backdropBlur: {
  xs: '2px', md: '14px',
  '2xl': '32px', '3xl': '48px',
}
// Motion
keyframes: {
  drift, pulseSoft,
  shimmer, rise, ticker, float,
}
animation: {
  drift, 'drift-slow', pulseSoft,
  shimmer, rise, ticker, float,
}
