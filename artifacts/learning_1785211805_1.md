+/**
+ * Sync state slice into localStorage.  Failure-tolerant — quota errors,

+  // 避免 StrictMode 雙渲染 + hydration race 寫入空白覆蓋既有資料
+  const isFirstRender = useRef(true);
+
+  useEffect(() => {
+    if (isFirstRender.current) {
+      isFirstRender.current = false;
+      return;
+    }
+    try {
+      const payload: PersistShape = {
+        todos: state.todos,
+        filter: state.filter,
+        search: state.search,
+        theme: state.theme,
+      };
+      localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
+    } catch (err) {
+      // eslint-disable-next-line no-console
+      console.warn("[todo] localStorage write failed:", err);
+    }
+  }, [state.todos, state.filter, state.search, state.theme]);
