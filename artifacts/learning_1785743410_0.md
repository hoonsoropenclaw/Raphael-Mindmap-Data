+#!/usr/bin/env python3
+"""Deterministic PNG visual comparator using Pillow; emits JSON for the runner."""
+import json, sys
+from pathlib import Path
+from PIL import Image, ImageChops
+
+def main():
+    if len(sys.argv) != 4:
+        print(json.dumps({'error': 'usage: visual_diff.py baseline current diff'})); return 2
+    baseline, current, diff = map(Path, sys.argv[1:])
+    try:
+        a, b = Image.open(baseline).convert('RGBA'), Image.open(current).convert('RGBA')
+    except Exception as exc:
+        print(json.dumps({'error': f'PNG_READ_ERROR: {exc}'})); return 2
+    if a.size != b.size:
+        print(json.dumps({'error': 'SIZE_MISMATCH', 'baselineSize': a.size, 'currentSize': b.size})); return 0
+    delta = ImageChops.difference(a, b)
+    bbox = delta.getbbox()
+    if not bbox:
+        pixels = 0
+    else:
+        pixels = sum(1 for px in delta.getdata() if px != (0, 0, 0, 0))
+        visual = Image.new('RGBA', a.size, (255, 32, 80, 0))
+        visual.putalpha(delta.getchannel('A').point(lambda x: min(255, x * 4)))
+        diff.parent.mkdir(parents=True, exist_ok=True)
+        Image.alpha_composite(b, visual).save(diff)
+    total = a.width * a.height
+    print(json.dumps({'diffPixels': pixels, 'totalPixels': total, 'ratio': pixels / total if total else 0, 'diffImage': str(diff) if bbox else None}))
+    return 0
+
+if __name__ == '__main__': raise SystemExit(main())

+```bash
+npm test
+```

+## 設計決策
+
+1. 使用 plain `playwright` 而非強綁 `@playwright/test`，讓 harness 能在目前已存在的 Hermes Playwright toolchain 中獨立執行。
+2. 所有互動元素都有 `data-testid` 或穩定 aria 名稱，避免文字空白與 UI 文案改動造成脆弱選擇器。
+3. 每次截圖前停用 animation/transition/caret，降低跨執行的非決定性差異。
+4. 視覺 baseline 不會在一般測試中自動覆蓋；必須明確傳入 `--update-baseline`。
