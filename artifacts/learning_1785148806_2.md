+"""Small stdlib-only Telegram weather bot using long polling."""
+from __future__ import annotations
+
+import json
+import logging

+import time
+from dataclasses import dataclass

+from urllib.parse import quote, urlencode
+from urllib.request import Request, urlopen
+
+LOG = logging.getLogger("telegram_weather_bot")
+
+
+class BotError(RuntimeError):
+    """Expected error from Telegram or the weather provider."""
+
+
+@dataclass(frozen=True)
+class Config:
+    telegram_token: str
+    weather_url: str = "https://wttr.in"
+    request_timeout: float = 35.0
+
+    @classmethod
+    def from_env(cls) -> "Config":
+        token = os.environ.get("TELEGRAM_BOT_TOKEN", "").strip()
+        if not token:
+            raise ValueError("TELEGRAM_BOT_TOKEN is required")
+        return cls(token, os.environ.get("WEATHER_URL", "https://wttr.in").rstrip("/"))
+
+
+def http_json(url: str, *, timeout: float, method: str = "GET", payload: dict[str, Any] | None = None) -> Any:
+    body = None
+    headers = {"User-Agent": "stdlib-telegram-weather-bot/1.0", "Accept": "application/json"}
+    if payload is not None:
+        body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
+        headers["Content-Type"] = "application/json"
+    request = Request(url, data=body, headers=headers, method=method)
+    try:
+        with urlopen(request, timeout=timeout) as response:
+            return json.loads(response.read().decode("utf-8"))
+    except (HTTPError, URLError, TimeoutError, json.JSONDecodeError) as exc:
+        raise BotError(f"HTTP request failed: {exc}") from exc
+
+
+def weather(location: str, *, base_url: str = "https://wttr.in", timeout: float = 20.0) -> str:
+    location = " ".join(location.split()).strip()
+    if not location or len(location) > 100:
+        raise ValueError("請輸入有效的城市名稱，例如：台北")
+    url = f"{base_url}/{quote(location, safe='')}",
+    query = urlencode({"format": "j1"})
+    data = http_json(url[0] + "?" + query, timeout=timeout)
+    current = data.get("current_condition", [{}])[0]
+    desc = current.get("weatherDesc", [{"value": "未知"}])[0].get("value", "未知")
+    temp = current.get("temp_C", "?")
+    feels = current.get("FeelsLikeC", "?")
+    humidity = current.get("humidity", "?")
+    wind = current.get("windspeedKmph", "?")
+    return (
+        f"📍 {location}\n"
+        f"🌤️ {desc}\n"
+        f"🌡️ 溫度：{temp}°C（體感 {feels}°C）\n"
+        f"💧 濕度：{humidity}%\n"
+        f"💨 風速：{wind} km/h"
+    )

+# Telegram Weather Bot
+
+stdlib-only Telegram bot: long polling receives text, queries wttr.in, and replies with weather data.
+
+## Run
+
+```bash
+export TELEGRAM_BOT_TOKEN='從 @BotFather 取得的 token'
+chmod +x run.sh
+./run.sh
+```
+
+輸入城市名稱即可查詢，也支援 `/weather 台北`、`/start`、`/help`。
+
+## Optional settings
+
+- `WEATHER_URL`: change the weather provider base URL (default `https://wttr.in`)
+- `LOG_LEVEL`: default `INFO`
+
+## Design notes
+
+- Uses only Python standard library (`urllib`, `json`, `signal`, `dataclasses`).
+- Long polling avoids requiring a public HTTPS endpoint.
+- Uses absolute `update_id + 1` offsets to prevent duplicate updates.
+- Network failures are logged and retried with a bounded 3-second delay; invalid user input is returned as a user-facing error.
+- SIGINT/SIGTERM stop the polling loop cleanly.
+- The bot token is read only from the environment and is never logged.

+# Senior Architect feedback absorbed
+
+## Architecture
+
+- Keep the implementation stdlib-only for this simple message → API → reply flow.
+- Separate weather lookup from Telegram polling so the lookup can be tested independently.
+- Use long polling on a headless host; use `offset = update_id + 1`, not incremental guessing.
+- Add bounded network timeouts, retry handling for transient failures, and signal-based graceful shutdown.
+- Keep secrets in environment variables; never hard-code or log the Telegram token.
+- Handle Unicode with direct literals rather than surrogate-pair escapes, and normalize user input before constructing the provider URL.
+
+## Verification checklist
+
+- Python syntax/lint check passed when `telegram_weather_bot.py` was written.
+- The script has no third-party dependencies.
+- `run.sh` fails fast when `TELEGRAM_BOT_TOKEN` is missing.
