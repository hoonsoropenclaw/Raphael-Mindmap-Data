+#!/usr/bin/env python3
+"""
+OCR Engine for PDF 公務文件 (Official Document) Processing
+============================================================
+Hierarchical extraction strategy:
+  Tier 1: Direct text extraction via pymupdf (text-based PDFs)
+  Tier 2: OCR via tesseract on rasterized pages (scanned PDFs)
+  Tier 3: Layout-aware markdown via pymupdf4llm (tables, structured docs)
+
+Plus a key data extractor using regex tuned for Taiwan official documents:
+  - 發文字號 (Document numbers): e.g., 府授文號字第1130123456號
+  - 中華民國日期 (ROC dates): e.g., 中華民國113年5月20日
+  - ISO/西元 dates: 2026-05-20, 2026/5/20
+  - 統一編號 / 身分證字號 (8-digit / 2-letter+8-digit)
+  - 金額 (Amounts): NT$ / 新台幣 with commas
+  - 電話 / Email / 地址
+
+Usage:
+    python3 ocr_engine.py <pdf_path>           # print extracted text + key data JSON
+    python3 ocr_engine.py <pdf_path> --json    # JSON only
+    python3 ocr_engine.py <pdf_path> --md      # Markdown
+    python3 ocr_engine.py <pdf_path> --ocr     # Force tesseract OCR on all pages
+"""

+import sys
+import os
+import re
+import json
+import argparse

+from datetime import datetime

+# ──────────────────────────────────────────────────────────────────
+# Tier 1: pymupdf direct extraction
+# ──────────────────────────────────────────────────────────────────
+def extract_text_pymupdf(pdf_path):
+    """Extract text directly from text-based PDFs."""
+    import pymupdf
+    doc = pymupdf.open(pdf_path)
+    pages_text = []
+    total_chars = 0
+    for i, page in enumerate(doc):
+        text = page.get_text("text")
+        pages_text.append({"page": i + 1, "text": text, "char_count": len(text)})
+        total_chars += len(text)
+    metadata = {
+        "pages": len(doc),
+        "title": doc.metadata.get("title", ""),
+        "author": doc.metadata.get("author", ""),
+        "subject": doc.metadata.get("subject", ""),
+        "creator": doc.metadata.get("creator", ""),
+        "producer": doc.metadata.get("producer", ""),
+        "total_chars": total_chars,
+    }
+    doc.close()
+    return pages_text, metadata
+
+
+def extract_markdown_pymupdf4llm(pdf_path):
+    """Extract markdown via pymupdf4llm (better table/layout preservation)."""
+    try:
+        import pymupdf4llm
+        md = pymupdf4llm.to_markdown(pdf_path)
+        return md
+    except Exception as e:
+        return f"[pymupdf4llm failed: {e}]"
+
+
+# ──────────────────────────────────────────────────────────────────
+# Tier 2: tesseract OCR for scanned/image PDFs
+# ──────────────────────────────────────────────────────────────────
+def is_scanned_pdf(pages_text, min_chars_per_page=20):

+    if not pages_text:
+        return True
+    avg = sum(p["char_count"] for p in pages_text) / len(pages_text)
+    return avg < min_chars_per_page

+## 📋 系統組成
+
+| 檔案 | 角色 | 說明 |
+|------|------|------|

+## 🏗️ 三層管線架構 (Three-Tier Pipeline)
+
+```
+┌──────────────────────────────────────────────────────────┐
+│ 輸入 PDF                                                 │
+└──────────────────────────────────────────────────────────┘
+   │
+   ▼
+┌──────────────────────────────────────────────────────────┐
+│ Tier 1: pymupdf 直接萃取 (~25MB,毫秒級)                  │
+│   - 文字型 PDF: 100% 準確、保留結構                       │
+│   - 偵測 chars/page < 20 → 視為掃描件 → 跳到 Tier 2     │
+└──────────────────────────────────────────────────────────┘
+   │ (掃描件)
+   ▼
+┌──────────────────────────────────────────────────────────┐
+│ Tier 2: tesseract OCR (pytesseract, 支援 chi_tra+eng)   │
+│   - 柵格化 PDF 頁面 → 圖片 → OCR                          │
+│   - 處理時間: ~2-5 秒/頁 (CPU)                            │
+└──────────────────────────────────────────────────────────┘
+   │
+   ▼
+┌──────────────────────────────────────────────────────────┐
+│ Tier 3: pymupdf4llm markdown 萃取                        │
+│   - 並行處理: 表格保留、閱讀順序偵測、Markdown 輸出       │
+└──────────────────────────────────────────────────────────┘
+   │
+   ▼
+┌──────────────────────────────────────────────────────────┐
+│ 關鍵欄位萃取 (Regex) — 針對台灣公文調校                   │
+│   - 發文字號 / 中華民國日期 (轉 ISO) / 西元日期 / 金額    │
+│   - 統一編號 / 身分證字號 / 電話 / Email / 地址            │
+│   - 受文者 / 主旨                                         │
+└──────────────────────────────────────────────────────────┘
+   │
+   ▼
+┌──────────────────────────────────────────────────────────┐
+│ 結構化 JSON 輸出 (含 metadata + full_text + key_data)    │
+└──────────────────────────────────────────────────────────┘
+```
