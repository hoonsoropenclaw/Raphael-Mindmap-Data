+const fs = require('fs');

+  '.html': 'text/html; charset=utf-8',
+  '.js':   'application/javascript; charset=utf-8',
+  '.css':  'text/css; charset=utf-8',
+  '.json': 'application/json; charset=utf-8',
+  '.png':  'image/png',
+  '.svg':  'image/svg+xml',
+};
+
+http.createServer((req, res) => {
+  let urlPath = decodeURIComponent((req.url || '/').split('?')[0]);
+  if (urlPath === '/' || urlPath === '') urlPath = '/web_output.html';
+  const filePath = path.join(ROOT, urlPath);
+  if (!filePath.startsWith(ROOT)) {
+    res.writeHead(403); res.end('forbidden'); return;
+  }
+  fs.readFile(filePath, (err, data) => {
+    if (err) { res.writeHead(404); res.end('not found'); return; }
+    res.writeHead(200, { 'Content-Type': MIME[path.extname(filePath)] || 'application/octet-stream' });
+    res.end(data);
+  });
+}).listen(PORT, '127.0.0.1', () => {
+  console.log(`[vrt-server] http://127.0.0.1:${PORT}/web_output.html`);

+# Glassmorphism Visual Regression Framework
+
+A self-contained **Playwright cross-browser** + **Pillow pixel-diff** testing
+framework, built around a Glassmorphism UI target. Catches unintended visual
+drift across Chromium, Firefox, WebKit, and two mobile emulators.
+
+```
+web_output.html   ← Glassmorphism test target (hero, dashboard, themes, tasks)
+server.js         ← zero-dep static file server (port 4179)
+playwright.config.js
+tests/visual.spec.js   ← 3 tests × 5 browser projects = 15 runs
+compare.py        ← Pillow diff engine + self-contained HTML report
+report.js         ← `npm run report` wrapper
+```
+
+## Quick start
+
+```bash
+npm install
+npx playwright install chromium firefox webkit
+npm test                    # 10 baseline tests pass (5 engines × 2 tests)
+RUN_VISUAL_PROBE=1 npm test # add 5 probe tests that detect intentional drift
+npm run report              # → ./regression-report.html
+```
+
+## How it works
+
+1. **`web_output.html`** — a Glassmorphism dashboard. Every interactive
+   element carries a stable `data-testid` so tests can address it without
+   selector churn. `backdrop-filter` is applied **only to top-level panels**
+   (per the architect's performance note) — leaf nodes use plain backgrounds.
+2. **`server.js`** — minimal zero-dep static server on `127.0.0.1:4179`
+   (chosen to avoid the busy 4173/4174 ports on this host).
+3. **`playwright.config.js`** — five projects:
+   - `chromium` (Desktop Chrome)
+   - `firefox` (Desktop Firefox)
+   - `webkit` (Desktop Safari)
+   - `mobile-chromium` (Pixel 5)
+   - `mobile-webkit` (iPhone 13)
+4. **`tests/visual.spec.js`** — three tests, run on every project:
+   - `hero + dashboard renders` — captures the baseline screenshot, asserts
+     DOM structure (5 tasks visible).
+   - `theme switch — aurora` — clicks the aurora theme swatch, asserts
+     `data-theme="aurora"`, captures a state-changed screenshot.
+   - `probe — intentional visual drift must be caught` *(opt-in)* — injects
+     a `hue-rotate(120deg)` CSS filter on the hero title, captures, and
+     asserts the byte-stream differs from baseline. When you turn it on
+     with `RUN_VISUAL_PROBE=1`, this validates the diff engine end-to-end.
+5. **Animation freezing** — every screenshot runs after injecting a CSS
+   `* { animation: none !important; transition: none !important; }` rule
+   and waiting 400 ms, so gradients/hover states are deterministic.
+6. **`compare.py`** — Pillow-based pixel diff (per `visual_regression_with_pillow`
+   skill #34). For each current screenshot, it:
+   - Loads the matching golden baseline from `baselines_golden/`.
+   - Computes per-channel max difference.
+   - Counts non-zero diff pixels.
+   - If over `MAX_DIFF_PIXELS=5000` or `MAX_DIFF_RATIO=0.5%`, marks FAIL and
+     writes a red-overlay diff image to `diff/`.
+   - Embeds baseline / current / diff inline (base64) into a single
+     `regression-report.html` for human review.
+   - On the first run it **promotes** all current screenshots into the
+     golden directory so subsequent runs have a baseline to compare against.
