+"""Per-host token-bucket rate limiter with 429 / 5xx penalty slots.
+
+Skeleton adapted from SKILL_CATALOG §Advanced HTTP Request Management §2,
+with three fixes:
+
+1.  The original `acquire()` did `await asyncio.sleep(self._refill_time)`
+    inside a spinloop — it could busy-loop if the lock was contended.
+    We use an `asyncio.Event` + monotonic-time re-check to avoid that.
+2.  The original only supported global penalize. We keep per-host penalty
+    slots so one slow host doesn't block other hosts.
+3.  We expose a `HostPenaltyBook` separately so tests can drive penalty
+    durations deterministically without juggling clocks.
+"""
+from __future__ import annotations
+
+import asyncio
+import time
+from dataclasses import dataclass, field
+from typing import Dict
+
+
+@dataclass
+class HostPenaltyBook:
+    """Per-host cooldown tracker.
+
+    `penalize(host, until)` records `host` as unavailable until `until`.
+    `is_blocked(host)` returns True iff the cooldown is still in effect.
+    """
+
+    _until: Dict[str, float] = field(default_factory=dict)
+
+    def penalize(self, host: str, seconds: float) -> None:
+        """Block `host` for `seconds` from now."""
+        if seconds <= 0:
+            return
+        deadline = time.monotonic() + seconds
+        # Take the max — don't shorten an existing longer penalty.
+        prev = self._until.get(host, 0.0)
+        if deadline > prev:
+            self._until[host] = deadline
+
+    def is_blocked(self, host: str) -> bool:
+        deadline = self._until.get(host, 0.0)
+        if deadline == 0.0:
+            return False
+        if time.monotonic() >= deadline:
+            # Expired — clean up so the dict doesn't grow forever.
+            self._until.pop(host, None)
+            return False
+        return True
+
+    def remaining(self, host: str) -> float:
+        return max(0.0, self._until.get(host, 0.0) - time.monotonic())
+
+    def blocked_hosts(self) -> list[str]:
+        return [h for h in self._until if self.is_blocked(h)]

+"""Dedup store — abstract interface + memory + SQLite backends.
+
+The dedup contract has exactly two operations:
+    * `add(url) -> bool`  — atomic test-and-set; returns False if already seen
+    * `seen(url) -> bool`  — non-mutating lookup
+
+The key is normalised via `url_normalize_key()` so that trivial URL variations
+(query order, trailing slash, default port) don't bypass dedup.
+
+Backend selection is driven by config (memory vs sqlite). The memory backend
+uses an LRU cap so long-running crawls don't OOM; the SQLite backend is the
+durable choice for multi-process crawls.
+"""
+from __future__ import annotations
+
+import asyncio
+import sqlite3
+import threading
+from collections import OrderedDict
+from dataclasses import dataclass, field
+from enum import Enum
+from typing import Optional, Protocol
+from urllib.parse import urlsplit, urlunsplit
+
+
+def url_normalize_key(url: str) -> str:
+    """Return a canonical form suitable as a dedup key.
+
+    Steps:
+      - lower-case scheme + host
+      - drop default ports (80/443)
+      - sort query params alphabetically
+      - drop empty fragment
+    """
+    parts = urlsplit(url)
+    scheme = parts.scheme.lower()
+    netloc = parts.netloc.lower()
+    if (scheme == "http" and netloc.endswith(":80")) or (
+        scheme == "https" and netloc.endswith(":443")
+    ):
+        netloc = netloc.rsplit(":", 1)[0]
+    query = "&".join(sorted(p for p in parts.query.split("&") if p))
+    fragment = "" if parts.fragment in ("", "/") else parts.fragment
+    return urlunsplit((scheme, netloc, parts.path, query, fragment))
+
+
+class DedupBackend(str, Enum):
+    MEMORY = "memory"
+    SQLITE = "sqlite"
+
+
+class _Backend(Protocol):
+    def add(self, key: str) -> bool: ...
+    def seen(self, key: str) -> bool: ...
+    def close(self) -> None: ...
+
+
+@dataclass
+class _MemoryBackend:
+    max_items: int = 1_000_000
+    _data: OrderedDict[str, None] = field(default_factory=OrderedDict)
+
+    def add(self, key: str) -> bool:
+        if key in self._data:
+            self._data.move_to_end(key)
+            return False
+        self._data[key] = None
+        if len(self._data) > self.max_items:
+            # Pop oldest — LRU eviction. Not "first-time-seen" friendly, but
+            # bounded memory is the explicit tradeoff.
+            self._data.popitem(last=False)
+        return True
+
+    def seen(self, key: str) -> bool:
+        return key in self._data
+
+    def close(self) -> None:
+        self._data.clear()
