+import { createServer } from 'node:http';
+import { readFile } from 'node:fs/promises';
+import { resolve } from 'node:path';
+
+const root = resolve(process.cwd());
+const port = Number(process.env.PORT || 4173);
+const server = createServer(async (request, response) => {
+  const pathname = request.url === '/' ? '/web_output.html' : request.url;
+  const file = resolve(root, `.${pathname}`);
+  if (!file.startsWith(root)) {
+    response.writeHead(403, { 'content-type': 'text/plain; charset=utf-8' });
+    response.end('Forbidden');
+    return;
+  }
+  try {
+    const body = await readFile(file);
+    const type = file.endsWith('.html') ? 'text/html; charset=utf-8' : 'application/octet-stream';
+    response.writeHead(200, { 'content-type': type, 'cache-control': 'no-store' });
+    response.end(body);
+  } catch {
+    response.writeHead(404, { 'content-type': 'text/plain; charset=utf-8' });
+    response.end('Not found');
+  }
+});

+# Northstar — Cross-Browser Playwright with Visual Regression
+
+A self-contained Playwright workspace that runs functional and visual regression
+tests against `web_output.html` across Chromium, Firefox, and WebKit on a
+headless Linux box.
+
+## Layout
+
+```
+web_output.html         Test fixture (responsive, no external assets, no network)
+server.mjs              Tiny static server (127.0.0.1:4173)
+playwright.config.js    3 projects, dark/reduced-motion, webServer auto-start
+tests/northstar.spec.js 18 tests = 12 functional + 6 visual
+scripts/inspect-mobile.mjs  One-off DOM inspector used to debug CSS asserts
+package.json            @playwright/test, scripts: test, test:headed, test:update
+```
+
+## Install
+
+```bash
+npm install
+# browsers are read from ~/.cache/ms-playwright; nothing to download on this box
+ls ~/.cache/ms-playwright | grep -E 'chromium-|firefox-|webkit-'
+```
+
+## Run
+
+```bash
+# First time — write the visual baseline (run once per browser, per viewport)
+BASE_URL=http://127.0.0.1:4173 npx playwright test --update-snapshots
+
+# Every subsequent run — diff against the baseline
+BASE_URL=http://127.0.0.1:4173 npx playwright test
+```
+
+The server auto-starts via `webServer.command`; pass `CI=1` to require a fresh
+process and to enable 2 retries per test.
+
+## What's covered
+

+| ------- | ---------- | -------------- | ------------- |
+| chromium | 4 | 1 | 1 |
+| firefox | 4 | 1 | 1 |

+

+- mobile (390x844) layout: no horizontal scroll, topbar wrap, full-width button

+Visual suite uses `toHaveScreenshot` with `animations: 'disabled'`,
