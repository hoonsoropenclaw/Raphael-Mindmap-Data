+#!/usr/bin/env python3
+"""Small stdlib-only Telegram weather bot using wttr.in.
+
+Required environment:
+  TELEGRAM_BOT_TOKEN       BotFather token
+Optional:
+  TELEGRAM_ALLOWED_USERS   Comma-separated numeric Telegram user IDs
+  WEATHER_API_BASE         Override wttr.in endpoint for testing
+"""
+from __future__ import annotations
+
+import json
+import logging
+import os
+import signal
+import time
+from dataclasses import dataclass
+from urllib.error import HTTPError, URLError
+from urllib.parse import quote
+from urllib.request import Request, urlopen
+
+LOG = logging.getLogger("weather-bot")
+
+
+@dataclass(frozen=True)
+class Weather:
+    location: str
+    description: str
+    temperature_c: str
+    feels_like_c: str
+    humidity: str
+    wind_kmh: str
+
+
+class WeatherError(RuntimeError):
+    pass
+
+
+def _clean(value: object) -> str:
+    return str(value).encode("utf-16", "surrogatepass").decode("utf-16", "replace").strip()
+
+
+def fetch_weather(location: str, timeout: int = 15) -> Weather:
+    location = _clean(location).strip()
+    if not location or len(location) > 100:
+        raise WeatherError("請輸入有效的地點（最多 100 字元）。")
+    base = os.getenv("WEATHER_API_BASE", "https://wttr.in")
+    url = f"{base.rstrip('/')}/{quote(location)}?format=j1&lang=zh"
+    request = Request(url, headers={"User-Agent": "telegram-weather-bot/1.0"})
+    try:
+        with urlopen(request, timeout=timeout) as response:
+            payload = json.load(response)
+    except (HTTPError, URLError, TimeoutError, json.JSONDecodeError, OSError) as exc:
+        raise WeatherError("天氣服務目前無法使用，請稍後再試。") from exc
+    try:
+        current = payload["current_condition"][0]
+        area = payload.get("nearest_area", [{}])[0]
+        area_name = area.get("areaName", [{"value": location}])[0].get("value", location)
+        return Weather(
+            location=_clean(area_name),
+            description=_clean(current.get("weatherDesc", [{"value": "未知"}])[0]["value"]),
+            temperature_c=_clean(current["temp_C"]),
+            feels_like_c=_clean(current["FeelsLikeC"]),
+            humidity=_clean(current["humidity"]),
+            wind_kmh=_clean(current["windspeedKmph"]),
+        )
+    except (KeyError, IndexError, TypeError) as exc:
+        raise WeatherError("天氣服務回傳了無法辨識的資料。") from exc
+
+
+def format_weather(weather: Weather) -> str:
+    return (
+        f"📍 {weather.location}\n"
+        f"🌤️ {weather.description}\n"
+        f"🌡️ 溫度：{weather.temperature_c}°C（體感 {weather.feels_like_c}°C）\n"
+        f"💧 濕度：{weather.humidity}%\n"
+        f"💨 風速：{weather.wind_kmh} km/h"
+    )

+## Setup
+
+```bash
+export TELEGRAM_BOT_TOKEN='token-from-BotFather'
+# Optional defense-in-depth allowlist:
+export TELEGRAM_ALLOWED_USERS='123456789,987654321'
+chmod +x run.sh

+```
+
+Send `/start` or `/help`, then send a city name. The bot uses Telegram long polling and wttr.in; no third-party Python packages are required.
+
+## Design notes
+
+- `weather_bot.py` separates weather retrieval, Telegram transport, formatting, and polling.
+- `offset = update_id + 1` prevents duplicate update processing.
+- SIGINT/SIGTERM stop polling cleanly.
+- Location input is bounded and sanitized; API failures become user-safe messages.
+- `TELEGRAM_ALLOWED_USERS` provides an optional user-ID allowlist. Keep the bot token in the environment, never in source control.
+- `WEATHER_API_BASE` can point to a local mock endpoint for deterministic tests.
+

+# Implementation review
+
+1. Chose Python standard library only (`urllib`, `json`, `signal`, `dataclasses`) because this is a simple message → weather API → message pipeline; this avoids dependency installation and supply-chain overhead.
+2. Split responsibilities inside `weather_bot.py`: `fetch_weather()` handles wttr.in, `TelegramAPI` handles Telegram transport, `format_weather()` handles presentation, and `run()` owns long polling and shutdown.
+3. Used Telegram long polling because it does not require a public HTTPS endpoint. The polling cursor is absolute (`update_id + 1`) to prevent duplicate delivery.
+4. Added graceful SIGINT/SIGTERM handling so container/service shutdowns stop the loop rather than leaving a process running.
+5. Added input length validation, Unicode sanitization, safe user-facing API errors, and an optional `TELEGRAM_ALLOWED_USERS` allowlist.
+6. Kept credentials out of source code. `TELEGRAM_BOT_TOKEN` must be supplied through the environment.
