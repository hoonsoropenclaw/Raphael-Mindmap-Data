+// src/state/todoReducer.js
+// Single source of truth for todo state transitions.
+// Pure reducer — no side effects, no DOM, no storage. Easy to unit-test.
+
+// Action types are exported as constants so the reducer switch and the
+// dispatchers in the hook can never typo a string and silently fall
+// through the default branch.
+export const ACTIONS = Object.freeze({
+  ADD: 'ADD',
+  TOGGLE: 'TOGGLE',
+  EDIT: 'EDIT',
+  DELETE: 'DELETE',
+  CLEAR_COMPLETED: 'CLEAR_COMPLETED',
+  TOGGLE_ALL: 'TOGGLE_ALL',
+  REORDER: 'REORDER',
+  RESTORE: 'RESTORE', // for undo / hydration from localStorage
+});
+
+let _idSeed = 0;
+// Stable id generator: combine timestamp with an in-memory counter so two
+// todos added in the same millisecond still get distinct ids.
+export const nextId = () =>
+  `${Date.now().toString(36)}-${(++_idSeed).toString(36)}`;
+
+export const todoReducer = (state, action) => {
+  switch (action.type) {
+    case ACTIONS.ADD: {
+      const text = (action.text ?? '').trim();
+      if (!text) return state;
+      return [
+        {
+          id: nextId(),
+          text,
+          done: false,
+          createdAt: Date.now(),
+        },
+        ...state,
+      ];
+    }
+    case ACTIONS.TOGGLE:
+      return state.map((t) =>
+        t.id === action.id ? { ...t, done: !t.done } : t,
+      );
+    case ACTIONS.EDIT:
+      return state.map((t) =>
+        t.id === action.id
+          ? { ...t, text: (action.text ?? '').trim() || t.text }
+          : t,
+      );
+    case ACTIONS.DELETE:
+      return state.filter((t) => t.id !== action.id);
+    case ACTIONS.CLEAR_COMPLETED:
+      return state.filter((t) => !t.done);
+    case ACTIONS.TOGGLE_ALL: {
+      const targetDone = Boolean(action.done);
+      // Only flip todos whose state differs — avoids creating new array
+      // entries for items already in the target state.
+      return state.map((t) => (t.done === targetDone ? t : { ...t, done: targetDone }));
+    }
+    case ACTIONS.REORDER: {
+      const { fromId, toId } = action;
+      if (fromId === toId) return state;
+      const fromIdx = state.findIndex((t) => t.id === fromId);
+      const toIdx = state.findIndex((t) => t.id === toId);
+      if (fromIdx === -1 || toIdx === -1) return state;
+      const next = state.slice();

+      next.splice(toIdx, 0, moved);

+  const state = useTodosState();
