+    async def acquire(self, n: float = 1.0, timeout: Optional[float] = None) -> bool:
+        """取得 n 個 token；timeout 秒內等不到就放棄。"""
+        deadline = time.monotonic() + timeout if timeout else None
+        while True:
+            async with self._lock:

+                    self.tokens -= n
+                    return True

+                wait = (n - self.tokens) / self.refill_rate
+            if deadline is not None and time.monotonic() + wait > deadline:
+                return False
+            await asyncio.sleep(min(wait, 0.5))

+# ============================================================
+# TTL 緩存層
+# ============================================================
+@dataclass
+class CacheEntry:
+    value: Any
+    expires_at: float
+    created_at: float = field(default_factory=time.monotonic)
+    hits: int = 0

+class TTLCache:
+    """簡單的異步安全 TTL 緩存。
+
+    - get(key) -> Optional[value]：過期回 None
+    - set(key, value, ttl)
+    - invalidate(key) / clear()
+    """
+
+    def __init__(self, max_size: int = 1024) -> None:
+        self.max_size = max_size
+        self._store: Dict[str, CacheEntry] = {}
+        self._lock = asyncio.Lock()
+        self.stats = {"hits": 0, "misses": 0, "evictions": 0, "expirations": 0}
+
+    @staticmethod
+    def make_key(*parts: Any) -> str:
+        """產生確定性 key。"""
+        raw = json.dumps(parts, sort_keys=True, default=str)
+        return hashlib.sha256(raw.encode()).hexdigest()[:16]
+
+    async def get(self, key: str) -> Optional[Any]:
+        async with self._lock:
+            entry = self._store.get(key)
+            if entry is None:
+                self.stats["misses"] += 1
+                return None
+            if time.monotonic() >= entry.expires_at:
+                del self._store[key]
+                self.stats["expirations"] += 1
+                self.stats["misses"] += 1
+                return None
+            entry.hits += 1
+            self.stats["hits"] += 1
+            return entry.value
+

+                oldest_key = min(self._store, key=lambda k: self._store[k].created_at)
+                del self._store[oldest_key]
+                self.stats["evictions"] += 1
+            self._store[key] = CacheEntry(
+                value=value, expires_at=time.monotonic() + ttl
+            )

+        async with self._lock:
+            self._store.clear()
