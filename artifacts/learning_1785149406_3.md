+"""SQLite 資料庫層 — todo items 的 CRUD。
+
+使用 sqlite3 標準庫 + context manager。
+不依賴 ORM，保持簡單與可移植。
+"""
+import sqlite3
+from contextlib import contextmanager
+from typing import Iterator
+
+# 預設 schema：每個 todo 有 id / title / done flag / created_at
+SCHEMA = """
+CREATE TABLE IF NOT EXISTS todos (
+    id         INTEGER PRIMARY KEY AUTOINCREMENT,
+    title      TEXT    NOT NULL,
+    done       INTEGER NOT NULL DEFAULT 0,
+    created_at TEXT    NOT NULL DEFAULT CURRENT_TIMESTAMP
+);
+"""
+
+
+@contextmanager
+def get_conn(db_path: str) -> Iterator[sqlite3.Connection]:
+    """連線 context manager，結束時自動 commit（無錯時）+ close。
+
+    Row factory 設為 sqlite3.Row，方便測試用欄位名取值。
+    """
+    conn = sqlite3.connect(db_path)
+    conn.row_factory = sqlite3.Row
+    try:
+        yield conn
+        conn.commit()
+    finally:
+        conn.close()
+
+
+def init_db(db_path: str) -> None:
+    """建立 schema。idempotent。"""
+    with get_conn(db_path) as conn:
+        conn.executescript(SCHEMA)
+
+
+def create_todo(db_path: str, title: str) -> dict:
+    """新增 todo，回傳完整 record（含 id 與 created_at）。"""
+    with get_conn(db_path) as conn:
+        cur = conn.execute(
+            "INSERT INTO todos (title) VALUES (?)", (title,)
+        )
+        todo_id = cur.lastrowid
+        row = conn.execute(
+            "SELECT id, title, done, created_at FROM todos WHERE id = ?",
+            (todo_id,),
+        ).fetchone()
+    return dict(row)
+
+
+def list_todos(db_path: str) -> list[dict]:
+    """列出全部 todo，依 id 排序。"""
+    with get_conn(db_path) as conn:
+        rows = conn.execute(
+            "SELECT id, title, done, created_at FROM todos ORDER BY id"
+        ).fetchall()
+    return [dict(r) for r in rows]
+
+
+def get_todo(db_path: str, todo_id: int) -> dict | None:
