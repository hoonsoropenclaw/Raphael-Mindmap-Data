+from __future__ import annotations
+
+from collections import defaultdict, deque
+from collections.abc import Callable
+from dataclasses import dataclass
+from math import ceil
+from threading import Lock
+from time import monotonic
+
+
+@dataclass(frozen=True, slots=True)
+class RateLimitDecision:
+    """Result of one rate-limit check."""
+
+    allowed: bool
+    limit: int
+    remaining: int
+    retry_after: float
+    reset_after: float
+
+
+class SlidingWindowRateLimiter:
+    """Thread-safe, process-local sliding-window rate limiter."""
+
+    def __init__(
+        self,
+        *,
+        limit: int,
+        window_seconds: float,
+        clock: Callable[[], float] | None = None,
+    ) -> None:
+        if limit <= 0:
+            raise ValueError("limit must be greater than zero")
+        if window_seconds <= 0:
+            raise ValueError("window_seconds must be greater than zero")
+
+        self.limit = limit
+        self.window_seconds = float(window_seconds)
+        self._clock = clock or monotonic
+        self._requests: dict[str, deque[float]] = defaultdict(deque)
+        self._lock = Lock()
+
+    def check(self, key: str) -> RateLimitDecision:
+        """Record one request when allowed and return the current decision."""
+        now = self._clock()
+        cutoff = now - self.window_seconds
+
+        with self._lock:
+            timestamps = self._requests[key]
+            while timestamps and timestamps[0] <= cutoff:

+# FastAPI Todo API
+
+A small, executable Todo REST API built with FastAPI, SQLAlchemy, SQLite, and a thread-safe sliding-window rate limiter.
+
+## Features
+
+- Todo CRUD: create, list, filter, retrieve, patch, and delete
+- SQLite persistence through synchronous SQLAlchemy 2.x sessions
+- Pydantic request validation and separate ORM/response models
+- Per-client sliding-window rate limiting on all `/todos` routes
+- Standard rate-limit response headers and `429 Too Many Requests`
+- Health endpoint and automatically generated OpenAPI documentation
+- Isolated unit/integration tests with 90% minimum coverage enforcement
+
+## Requirements
+
+- Python 3.11 or newer
+
+This project was verified with Python 3.12.3.
+
+## Setup
+
+```bash
+python3 -m venv .venv
+.venv/bin/python -m pip install --upgrade pip
+.venv/bin/python -m pip install -e '.[dev]'
+```
+
+## Run
+
+```bash
+.venv/bin/uvicorn app.main:app --host 127.0.0.1 --port 8000
+```
+
+Then open:
+
+- API documentation: http://127.0.0.1:8000/docs
+- OpenAPI JSON: http://127.0.0.1:8000/openapi.json
+- Health check: http://127.0.0.1:8000/health
+
+## Configuration
+
+All settings are optional environment variables.
+
+| Variable | Default | Purpose |
+|---|---:|---|
+| `DATABASE_URL` | `sqlite:///./todos.db` | SQLAlchemy database URL |
+| `RATE_LIMIT_REQUESTS` | `60` | Requests allowed in each window |
+| `RATE_LIMIT_WINDOW_SECONDS` | `60` | Sliding-window duration in seconds |
+
+Example:
+
+```bash
+DATABASE_URL='sqlite:///./data/todos.db' \
+RATE_LIMIT_REQUESTS=30 \
+RATE_LIMIT_WINDOW_SECONDS=60 \
+.venv/bin/uvicorn app.main:app --host 127.0.0.1 --port 8000

+### 3.1 Application factory

+### 3.2 Persistence
+
+`Database` owns engine/session creation. SQLite connections use:
