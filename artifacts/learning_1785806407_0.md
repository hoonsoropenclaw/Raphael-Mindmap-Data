+def load_image(path: Path) -> Image.Image:
+    """Load PNG and normalize to RGBA."""
+    img = Image.open(path)
+    if img.mode != "RGBA":
+        img = img.convert("RGBA")
+    return img
+
+
+def normalize_size(img_a: Image.Image, img_b: Image.Image) -> Tuple[Image.Image, Image.Image]:
+    """Pad/crop images to the same size (use the max dimensions)."""
+    w = max(img_a.width, img_b.width)
+    h = max(img_a.height, img_b.height)
+    canvas_a = Image.new("RGBA", (w, h), (255, 255, 255, 255))
+    canvas_b = Image.new("RGBA", (w, h), (255, 255, 255, 255))
+    canvas_a.paste(img_a, (0, 0), img_a)
+    canvas_b.paste(img_b, (0, 0), img_b)
+    return canvas_a, canvas_b
+
+
+def pixel_diff_ratio(img_a: Image.Image, img_b: Image.Image) -> float:
+    """Return fraction of pixels that differ (0.0 - 1.0)."""
+    a, b = normalize_size(img_a, img_b)
+    diff = ImageChops.difference(a, b)
+    bbox = diff.getbbox()
+    if not bbox:
+        return 0.0
+    # Count pixels with any channel difference > threshold
+    total = a.width * a.height
+    pixels = diff.getdata()
+    threshold = 8  # anti-aliasing tolerance per channel
+    diff_count = sum(1 for p in pixels if any(c > threshold for c in p[:3]))
+    return diff_count / total
