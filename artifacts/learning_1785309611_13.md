+"""
+retry.py — 重試策略
+=====================
+
+兩種主流策略：
+
+1. ExponentialBackoffRetry
+   delay = min(base * 2^attempt, cap)
+   典型: 1s → 2s → 4s → 8s → 16s (cap=30s)
+   適用: 多數 transient 場景
+
+2. JitteredRetry
+   delay = random.uniform(0, min(base * 2^attempt, cap))
+   目的: 避免 thundering herd（多個 client 同時重試壓垮下游）
+   適用: 大量 worker 同時重試同一個下游服務
+
+決策原則（來自 trial-and-error/multi-agent-failure-recovery-20260720.md）：
+- Transient error → 重試
+- Permanent error → 不重試
+- Critical error → 整鏈停止，不重試
+
+delay 計算採用純函式，便於測試與重用。
+"""
+
+from __future__ import annotations
+
+import random
+import time
+from abc import ABC, abstractmethod
+from dataclasses import dataclass
+from typing import Callable, Optional
+
+from .core import ErrorClass, ErrorClassifier
+
+
+@dataclass
+class RetryConfig:
+    max_attempts: int = 5
+    base_delay_s: float = 1.0
+    max_delay_s: float = 30.0
+    jitter: bool = True  # 加 jitter 避免 thundering herd
+
+
+class RetryStrategy(ABC):
+    @abstractmethod
+    def compute_delay(self, attempt: int) -> float:
+        """計算第 N 次失敗後要等多久。attempt 是 1-indexed。"""
+
+    def should_retry(self, attempt: int, exc: BaseException) -> bool:
+        if attempt >= self.config.max_attempts:
+            return False
+        return ErrorClassifier.classify(exc) == ErrorClass.TRANSIENT
+
+
+@dataclass
+class ExponentialBackoffRetry(RetryStrategy):
+    config: RetryConfig
+
+    def compute_delay(self, attempt: int) -> float:
+        # 1-indexed: attempt=1 → 1*base, attempt=2 → 2*base, attempt=3 → 4*base
+        delay = self.config.base_delay_s * (2 ** (attempt - 1))
+        delay = min(delay, self.config.max_delay_s)
+        if self.config.jitter:
+            # 對稱抖動: ±25%
+            delay = delay * random.uniform(0.75, 1.25)
+        return delay

+    能更有效打散重試波峰。

+recovery.py — 降級與替代方案
+==============================
+
+1. FallbackChain

+   適用場景：
+     - Primary DB 掛了 → 試 Replica DB → 試 Cache → 試靜態預設值
+     - 主 API 限流 → 試備用 API endpoint → 回 cached result
