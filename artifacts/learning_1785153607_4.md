+ * ─────────────────────────────────────────────────────────────
+ * - 寫入時 throttle，避免每個 keystroke 都打 localStorage

+ * - 解析失敗時不 throw，回傳 fallback + console.warn

+import { useCallback, useEffect, useRef, useState } from "react";

+export function useLocalStorage<T>(
+  key: string,
+  initial: T,
+  options: { throttleMs?: number } = {}
+): [T, (value: T | ((prev: T) => T)) => void] {
+  const { throttleMs = 200 } = options;
+  const [value, setValue] = useState<T>(() => {
+    if (typeof window === "undefined") return initial;
+    try {
+      const raw = window.localStorage.getItem(key);
+      if (raw === null) return initial;
+      return JSON.parse(raw) as T;
+    } catch (err) {
+      console.warn(`[useLocalStorage] failed to parse "${key}":`, err);
+      return initial;
+    }
+  });
+
+  const lastWrite = useRef(0);
+  const pendingTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
+  const latestRef = useRef(value);

+  useEffect(() => {
+    if (typeof window === "undefined") return;

+      try {

+      } catch (err) {

+      }

+    if (elapsed >= throttleMs) {

+export type ShortcutMap = Record<string, (e: KeyboardEvent) => void>;

+function shouldIgnoreTarget(target: EventTarget | null): boolean {
+  if (!(target instanceof HTMLElement)) return false;
+  const tag = target.tagName;
+  if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return true;
+  if (target.isContentEditable) return true;
+  return false;
+}

+export function useKeyboardShortcuts(
+  shortcuts: ShortcutMap,
+  enabled: boolean = true

+  useEffect(() => {
+    if (!enabled) return;

+      // 組出組合字串，例如 "Ctrl+Z" / "Cmd+/"
+      const parts: string[] = [];

+      if (e.shiftKey) parts.push("Shift");
+      if (e.altKey) parts.push("Alt");
+      // key 標準化：Esc → Escape, 空白 → Space
+      let key = e.key;
+      if (key === "Esc") key = "Escape";
+      if (key === " ") key = "Space";

+# TaskFlow — 實作思路與架構設計
+
+> 一個展示「複雜狀態管理」的 React UI 元件，採看板 (Kanban) 形式呈現任務流。
+> 重點不是 UI 漂亮就好，而是展示：reducer、undo/redo、drag-and-drop、localStorage

+## 1. 為什麼選這個題目
+
+一個合格的「複雜狀態管理」展示，必須同時具備：
+
+| 能力 | 在 TaskFlow 怎麼呈現 |
+|---|---|
+| 結構化狀態 | `TasksState = { byId: Record<string, Task>, allIds: string[] }` — 標準正規化結構 |
+| 不可變更新 | 所有 reducer 動作都用 `{...state}` 展開，不做 mutation |
+| Action / Reducer 模式 | 5 種 action：ADD / UPDATE / DELETE / MOVE / BULK_REPLACE / CLEAR_DONE |
+| History (Undo/Redo) | `useUndoable` hook 包 useState + 雙向 stack |
+| 衍生狀態 (derived state) | `selectByStatus`、`sortTasks`、`applyFilters` 純函式 selectors |
+| 持久化 | `useLocalStorage` hook：throttled write + 跨分頁 storage 事件 |
+| 副作用管理 | `useEffect` 監聽 keyboard、storage、focus |
+| Drag & Drop 跨欄 | HTML5 DnD + 計算插入位置（drop indicator） |
+| 鍵盤快捷鍵 | `useKeyboardShortcuts` hook：自動避開 input |
+| 鍵盤 modal 操作 | Esc 關閉、Enter 提交、auto-focus |
+| 樂觀更新 | 全部 reducer 同步，無 async race condition |
+| 預設資料 | 4 筆 sample tasks，首次開啟就有東西看 |
