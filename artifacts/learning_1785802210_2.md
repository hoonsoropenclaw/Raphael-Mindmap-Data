+"""Filesystem monitor — watchdog Observer.
+
+Anti-pattern (blocking-io): the watchdog event handler MUST NOT do OCR inline.
+It only validates the file extension and pushes (path, event_type) onto a thread-safe
+queue. OCR happens inside ocr_worker.Worker.
+
+Cross-platform: PollingObserver is used when inotify/ReadDirectoryChangesW isn't
+available (Windows native FS, some network mounts, Docker bind mounts in some setups).
+We try Observer first, then fall back to PollingObserver.
+"""
+from __future__ import annotations
+
+import threading
+import time
+from pathlib import Path
+from queue import Queue
+from typing import Optional
+
+from watchdog.events import FileSystemEventHandler, FileSystemEvent
+from watchdog.observers import Observer
+from watchdog.observers.polling import PollingObserver
+

+class _Handler(FileSystemEventHandler):
+    """Pure event handler — only enqueues. NEVER calls tesseract."""
+
+    def __init__(self, queue: "Queue", image_exts, debounce_seconds: float):
+        super().__init__()
+        self._queue = queue
+        self._exts = tuple(ext.lower() for ext in image_exts)
+        self._debounce = debounce_seconds
+        self._last_seen: dict[str, float] = {}
+        self._lock = threading.Lock()
+
+    # watchdog calls these from its own thread pool; keep them cheap.
+    def on_created(self, event: FileSystemEvent) -> None:
+        self._enqueue(event.src_path, "created")
+
+    def on_modified(self, event: FileSystemEvent) -> None:
+        self._enqueue(event.src_path, "modified")
+
+    def on_moved(self, event: FileSystemEvent) -> None:
+        # dest_path is set on move; treat like a new file
+        dest = getattr(event, "dest_path", None) or event.src_path
+        self._enqueue(dest, "moved")
+
+    def _enqueue(self, raw_path: str, kind: str) -> None:
+        path = Path(raw_path)
+        if path.suffix.lower() not in self._exts:
+            return
+        if path.is_dir():
+            return
+
+        now = time.monotonic()
+        with self._lock:
+            last = self._last_seen.get(str(path), 0.0)
+            # debounce: editors write the same file multiple times during save
+            if now - last < self._debounce:
+                # still re-arm the debounce so the final write gets through
+                self._last_seen[str(path)] = now
+                return
+            self._last_seen[str(path)] = now
+
+        # queue is bounded (maxsize); if full, drop oldest by polling semantics
+        self._queue.put((str(path), kind, now))

+# OCR Sentinel
+
+Cross-platform desktop automation tool that watches a directory, OCRs any image
+that lands in it, classifies the content (INVOICE / RECEIPT / ID_CARD / CONTRACT
+/ RESUME / NOTE), and moves the file into a category folder. A live web dashboard
+streams every OCR result via SSE.
+
+```
+ ┌──────────────┐    put(path,kind)    ┌─────────────────┐    OCR     ┌──────────┐
+ │ watchdog     │ ───────────────────► │  bounded Queue  │ ────────►  │ workers  │
+ │ Observer     │   (anti-pattern:     │  (max 1024)     │            │ N=2      │
+ │ / Polling    │    no blocking I/O   │                 │ ◄───────   │ threads  │
+ └──────────────┘    in handler)       └─────────────────┘  result    └────┬─────┘
+                                                                             │
+                                                              publish()      ▼
+                                                            ┌──────────────────────┐
+                                                            │ _BroadcastHub        │
+                                                            │  subs: Queue[...]    │
+                                                            └────────┬─────────────┘
+                                                                     │ SSE
+                                                                     ▼
+                                                            ┌──────────────────────┐
+                                                            │ ThreadingHTTPServer  │
+                                                            │  /                  │
+                                                            │  /status            │
+                                                            │  /events  (SSE)     │
+                                                            │  /ocr/<stem>.txt    │
+                                                            └──────────────────────┘
+```
+
+## Why this design
+
+It reuses five micro-skills from `/home/hoonsoropenclaw/.hermes/data_repo/skills/`:
+
+| Skill                  | How it's applied                                     |
+|------------------------|------------------------------------------------------|
+| `ocr_daemon_init`      | `config.ensure_dirs()` creates watch/.ocr/logs/exports before any thread starts |
+| `ocr_status_polling`   | `state_snapshot()` reads `.ocr/*.txt` + `events.ndjson` for live counts |
+| `file_rename_rule`     | `Worker._maybe_rename()` matches `INVOICE` / `RECEIPT` / etc. and moves the file |
+| `web_output_dashboard` | `dashboard.py` = `BaseHTTPRequestHandler` + SSE, zero external deps |
+| `anti-pattern (io/loop)` | Event handler only enqueues; worker loop has stop flag + `queue.get(timeout=…)` |
+
+It also applies lessons from `trial-and-error`:
+
+- **stdlib-first** (lesson 36): no `requests`, no `flask`, only stdlib + watchdog + pytesseract + Pillow.
+- **anti-blocking-io** (lesson): watchdog handler never runs tesseract.
+- **anti-infinite-loop**: worker loop polls with `timeout=0.5`, and exits when `_stop` event is set.
+- **bounded concurrency**: queue `maxsize=1024`, `workers=2` (configurable via `OCR_WORKERS`).
+- **cross-platform**: tries `Observer` first (inotify / FSEvents), falls back to
+  `PollingObserver` for Windows-native FS or sandbox bind mounts.
