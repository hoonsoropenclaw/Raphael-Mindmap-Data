#!/usr/bin/env python3
"""
Visual Regression — Pillow pixel-diff engine.

Micro-skill: #39 Visual Regression with Pillow
Extended from catalog snippet with:
  - Per-pixel tolerance (avoid false positives from AA)
  - Bounding-box of diff for fast triage
  - RGBA normalization (Pillow quirk)
  - Heatmap visualization (red overlay on different pixels)
  - JSON summary for downstream HTML report

Usage:
  python3 pixel_diff.py diff <baseline.png> <current.png> <diff.png> [tolerance]
  python3 pixel_diff.py baseline <source.png> <baseline.png>
  python3 pixel_diff.py info <image.png>
Exit code:
  0 = identical (within tolerance)
  1 = different (within caller-set policy)
  2 = error (file missing / size mismatch)
"""

import sys
import json
import argparse
from pathlib import Path
from PIL import Image, ImageChops, ImageDraw, ImageStat


# ---------- core ops ----------

def normalize(img: Image.Image) -> Image.Image:
    """Catalog snippet uses .convert('RGBA') — keep that anchor contract."""
    return img.convert("RGBA")


def compare(baseline_path: str, current_path: str, diff_path: str,
            tolerance: int = 5) -> dict:
    """
    Compare two PNGs. Returns a structured summary dict.

    Algorithm:
      1. Normalize both to RGBA
      2. Enforce size match (catalog warns about this)
      3. Pixel-by-pixel diff via ImageChops.difference
      4. Tolerance: any channel diff within `tolerance` is treated as equal
      5. Compute stats: changed pixels, max channel diff, bbox
      6. Render heatmap with red overlay on truly-different pixels
    """
    base = Path(baseline_path)
    curr = Path(current_path)
    out  = Path(diff_path)
    if not base.exists():
        return {"ok": False, "error": f"baseline missing: {base}"}
    if not curr.exists():
        return {"ok": False, "error": f"current missing: {curr}"}

    a = normalize(Image.open(base))
    b = normalize(Image.open(curr))

    if a.size != b.size:
        # Common case: fullPage height varies slightly when content shifts
        # (e.g. text rewrites). If widths match, crop the taller to match
        # the shorter so we can still produce a meaningful pixel diff. If
        # widths differ → that's a real change in browser rendering at the
        # layout level, reported as a hard error.
        if a.size[0] == b.size[0]:
            target_h = min(a.size[1], b.size[1])
            if a.size[1] != target_h:
                a = a.crop((0, 0, a.size[0], target_h))
            if b.size[1] != target_h:
                b = b.crop((0, 0, b.size[0], target_h))
        else:
            return {
                "ok": False,
                "error": f"width mismatch: baseline={a.size} current={b.size}",
                "baseline": str(base),
                "current": str(curr),
            }

    diff = ImageChops.difference(a, b)
    raw_bbox = diff.getbbox()  # raw diff bbox (no tolerance)

    # Compute per-pixel "is_changed" using tolerance
    # We do this channel-by-channel via split (faster than iterating in Python)
    a_ch = a.split()
    b_ch = b.split()
    width, height = a.size
    total_pixels = width * height
    changed_pixels = 0
    max_channel_diff = 0
    diff_mask = Image.new("L", a.size, 0)

    # Vectorize via PIL: get raw band data
    import numpy as np  # Pillow pulls numpy as optional; fall back if absent
    try:
        a_arr = np.array(a, dtype=np.int16)
        b_arr = np.array(b, dtype=np.int16)
        # Any channel outside tolerance = changed
        per_pixel_max = np.abs(a_arr - b_arr).max(axis=2)
        changed_mask = per_pixel_max > tolerance
        changed_pixels = int(changed_mask.sum())
        max_channel_diff = int(per_pixel_max.max())
        diff_mask = Image.fromarray((changed_mask * 255).astype(np.uint8), mode="L")
    except ImportError:
        # Fallback: pure-Pillow band math (slower but works without numpy)
        mask = Image.new("L", a.size, 0)
        for ch_a, ch_b in zip(a_ch, b_ch):
            ch_diff = ImageChops.difference(ch_a, ch_b)
            mask = ImageChops.add(mask, ch_diff)
        # mask now holds sum of channel diffs; threshold conservatively
        # threshold = tolerance * 4 (4 channels)
        threshold = tolerance * 4
        diff_mask = mask.point(lambda v: 255 if v > threshold else 0)
        # Count via getbbox + ImageStat
        bbox2 = diff_mask.getbbox()
        # Approximate count via ImageStat (mean * total)
        stat = ImageStat.Stat(diff_mask)
        approx_changed = (stat.mean[0] / 255.0) * total_pixels
        changed_pixels = int(approx_changed)
        max_channel_diff = 0  # not computed in fallback
        raw_bbox = bbox2

    # Generate heatmap: red overlay on changed pixels
    out.parent.mkdir(parents=True, exist_ok=True)
    overlay = b.copy()
    red = Image.new("RGBA", a.size, (255, 0, 0, 0))
    draw = ImageDraw.Draw(red)
    # Use the mask: red where mask > 0
    solid_red = Image.new("RGBA", a.size, (239, 68, 68, 140))  # red-500 @ ~55% alpha
    overlay.paste(solid_red, (0, 0), diff_mask)
    overlay.save(out)

    pct_changed = (changed_pixels / total_pixels) * 100.0 if total_pixels else 0.0
    # identical iff no raw diff AND no tolerance-respecting pixels flagged
    identical = (raw_bbox is None) and (changed_pixels == 0)

    return {
        "ok": True,
        "identical": identical,
        "baseline": str(base),
        "current": str(curr),
        "diff": str(out),
        "width": width,
        "height": height,
        "total_pixels": total_pixels,
        "changed_pixels": changed_pixels,
        "pct_changed": round(pct_changed, 4),
        "max_channel_diff": max_channel_diff,
        "tolerance": tolerance,
        "bbox": raw_bbox,
    }


def create_baseline(source_path: str, baseline_path: str) -> dict:
    """Snapshot the source PNG as the new baseline (first-run capture)."""
    src = Path(source_path)
    dst = Path(baseline_path)
    if not src.exists():
        return {"ok": False, "error": f"source missing: {src}"}
    dst.parent.mkdir(parents=True, exist_ok=True)
    img = normalize(Image.open(src))
    img.save(dst)
    return {
        "ok": True,
        "baseline": str(dst),
        "source": str(src),
        "width": img.size[0],
        "height": img.size[1],
    }


def image_info(path: str) -> dict:
    p = Path(path)
    if not p.exists():
        return {"ok": False, "error": f"missing: {p}"}
    img = Image.open(p)
    return {
        "ok": True,
        "path": str(p),
        "width": img.size[0],
        "height": img.size[1],
        "mode": img.mode,
        "format": img.format,
        "size_bytes": p.stat().st_size,
    }


# ---------- CLI ----------

def main():
    parser = argparse.ArgumentParser(description="Visual regression pixel-diff engine")
    sub = parser.add_subparsers(dest="cmd", required=True)

    p_diff = sub.add_parser("diff", help="compare baseline vs current")
    p_diff.add_argument("baseline")
    p_diff.add_argument("current")
    p_diff.add_argument("diff_out")
    p_diff.add_argument("--tolerance", "-t", type=int, default=5,
                       help="per-channel tolerance (default 5)")

    p_base = sub.add_parser("baseline", help="snapshot current as new baseline")
    p_base.add_argument("source")
    p_base.add_argument("baseline")

    p_info = sub.add_parser("info", help="show image metadata")
    p_info.add_argument("path")

    args = parser.parse_args()

    if args.cmd == "diff":
        result = compare(args.baseline, args.current, args.diff_out, args.tolerance)
    elif args.cmd == "baseline":
        result = create_baseline(args.source, args.baseline)
    else:
        result = image_info(args.path)

    print(json.dumps(result, indent=2, ensure_ascii=False))
    if not result.get("ok"):
        sys.exit(2)
    if args.cmd == "diff" and not result.get("identical"):
        sys.exit(1)


if __name__ == "__main__":
    main()

#!/usr/bin/env node
/**
 * Visual Regression Framework — orchestrator
 *
 * Pipeline:
 *   1. Start a tiny HTTP server that serves tests/pages/*.html
 *   2. Spin up each browser project (chromium, firefox, webkit, mobile-*).
 *   3. For each page × browser combo, capture a full-page screenshot.
 *   4. For each capture, call pixel_diff.py to compare against baseline.
 *   5. Aggregate results into a single HTML report.
 *   6. Exit 0 if all pass; exit 1 if any drift detected.
 *
 * Modes:
 *   --baseline          : overwrite baselines with current captures (first run)
 *   --report-only       : just regenerate the report from existing JSON
 *   --serve             : start the HTTP server and stop (useful for manual QA)
 *   --browsers=LIST     : comma-separated subset (chromium,firefox,...)
 *   --tolerance=N       : per-channel diff tolerance (default 5)
 *   --bail              : exit on first diff (default: collect all)
 */

import { chromium, firefox, webkit, devices } from 'playwright';
import { spawn, spawnSync } from 'node:child_process';
import { existsSync, mkdirSync, writeFileSync, readFileSync, statSync } from 'node:fs';
import { resolve, join, basename, extname } from 'node:path';
import { createServer } from 'node:http';
import { readdirSync } from 'node:fs';

// ---------- config ----------
const ROOT = resolve(import.meta.dirname);
const PAGES_DIR = join(ROOT, 'tests', 'pages');
const SCREENSHOTS_DIR = join(ROOT, 'screenshots');
const BASELINES_DIR = join(ROOT, 'baselines');
const DIFFS_DIR = join(ROOT, 'diffs');
const REPORTS_DIR = join(ROOT, 'reports');
const REPORTS_DATA = join(REPORTS_DIR, 'results.json');
const PYTHON_DIFF = join(ROOT, 'lib', 'pixel_diff.py');

for (const d of [SCREENSHOTS_DIR, BASELINES_DIR, DIFFS_DIR, REPORTS_DIR]) {
  mkdirSync(d, { recursive: true });
}

const args = new Set(process.argv.slice(2));
const arg = (k, def) => {
  const found = process.argv.slice(2).find(a => a.startsWith(`--${k}=`));
  return found ? found.split('=')[1] : def;
};
const MODE_BASELINE = args.has('--baseline');
const MODE_REPORT = args.has('--report-only');
const MODE_SERVE = args.has('--serve');
const MODE_BAIL = args.has('--bail');
const BROWSERS = (arg('browsers', 'chromium,firefox,webkit,mobile-chromium,mobile-webkit')).split(',');
const TOLERANCE = parseInt(arg('tolerance', '5'), 10);
const PORT = parseInt(arg('port', '4173'), 10);

// ---------- static server ----------
function startServer() {
  const server = createServer((req, res) => {
    // Strip query, normalize path
    let urlPath = decodeURIComponent(req.url.split('?')[0]);
    if (urlPath === '/') urlPath = '/index.html';
    let filePath = join(PAGES_DIR, urlPath);
    if (!existsSync(filePath)) {
      // allow index inside subdirs
      if (existsSync(filePath + '.html')) filePath += '.html';
      else if (existsSync(join(filePath, 'index.html'))) filePath = join(filePath, 'index.html');
      else { res.statusCode = 404; res.end('404'); return; }
    }
    const ext = extname(filePath).toLowerCase();
    const mime = {
      '.html': 'text/html; charset=utf-8',
      '.js':   'application/javascript',
      '.css':  'text/css',
      '.svg':  'image/svg+xml',
      '.png':  'image/png',
      '.jpg':  'image/jpeg',
      '.json': 'application/json',
    }[ext] || 'application/octet-stream';
    res.setHeader('Content-Type', mime);
    res.setHeader('Cache-Control', 'no-store');
    res.end(readFileSync(filePath));
  });
  return new Promise((resolveServer) => {
    server.listen(PORT, '127.0.0.1', () => {
      console.log(`[server] http://127.0.0.1:${PORT}  serving ${PAGES_DIR}`);
      resolveServer(server);
    });
  });
}

// ---------- browser launcher ----------
function getBrowserAndProfile(projectName) {
  // Map project name → (browserType, profile)
  switch (projectName) {
    case 'chromium':         return { browserType: chromium, profile: devices['Desktop Chrome'] };
    case 'firefox':          return { browserType: firefox,  profile: devices['Desktop Firefox'] };
    case 'webkit':           return { browserType: webkit,   profile: devices['Desktop Safari'] };
    case 'mobile-chromium':  return { browserType: chromium, profile: devices['Pixel 5'] };
    case 'mobile-webkit':    return { browserType: webkit,   profile: devices['iPhone 13'] };
    default: throw new Error(`Unknown project: ${projectName}`);
  }
}

// ---------- single capture ----------
async function captureOne(browserType, profile, projectName, pageFile) {
  const baseName = basename(pageFile, '.html');
  const capturePath = join(SCREENSHOTS_DIR, `${projectName}_${baseName}.png`);
  const baselinePath = join(BASELINES_DIR, `${projectName}_${baseName}.png`);
  const diffPath = join(DIFFS_DIR, `${projectName}_${baseName}.png`);

  const browser = await browserType.launch({ headless: true });
  const context = await browser.newContext({ ...profile, reducedMotion: 'reduce' });
  const page = await context.newPage();
  const url = `http://127.0.0.1:${PORT}/${baseName}.html`;
  const errors = [];
  page.on('pageerror', e => errors.push(`pageerror: ${e.message}`));
  page.on('console', m => { if (m.type() === 'error') errors.push(`console: ${m.text()}`); });

  const t0 = Date.now();
  await page.goto(url, { waitUntil: 'networkidle' });
  // Wait for Tailwind CDN to inject styles (avoids unstyled flash)
  await page.waitForTimeout ? await page.waitForTimeout(300) : await new Promise(r => setTimeout(r, 300));
  await page.screenshot({ path: capturePath, fullPage: true });

  await browser.close();
  const t1 = Date.now();

  return { projectName, baseName, capturePath, baselinePath, diffPath, url, errors, ms: t1 - t0 };
}

// ---------- pixel diff helpers ----------
function diffImages(baselinePath, currentPath, diffPath) {
  const r = spawnSync('python3', [PYTHON_DIFF, 'diff', baselinePath, currentPath, diffPath, '--tolerance', String(TOLERANCE)], {
    encoding: 'utf-8',
  });
  // Exit 0 = identical, 1 = different, 2 = error
  const stdout = (r.stdout || '').trim();
  let parsed = {};
  try { parsed = JSON.parse(stdout); } catch { parsed = { ok: false, error: 'parse-fail', raw: stdout }; }
  parsed.exit_code = r.status;
  return parsed;
}

function snapshotBaseline(sourcePath, baselinePath) {
  const r = spawnSync('python3', [PYTHON_DIFF, 'baseline', sourcePath, baselinePath], { encoding: 'utf-8' });
  const stdout = (r.stdout || '').trim();
  let parsed = {};
  try { parsed = JSON.parse(stdout); } catch { parsed = { ok: false, error: 'parse-fail', raw: stdout }; }
  parsed.exit_code = r.status;
  return parsed;
}

function escapeHtml(s) {
  return String(s ?? '').replace(/[&<>"']/g, c => ({ '&': '&', '<': '<', '>': '>', '"': '"', "'": ''' }[c]));
}

// ---------- main pipeline ----------
async function runPipeline() {
  const pages = readdirSync(PAGES_DIR).filter(f => f.endsWith('.html')).map(f => join(PAGES_DIR, f));
  if (pages.length === 0) {
    console.error(`[fatal] no .html files under ${PAGES_DIR}`);
    process.exit(2);
  }

  console.log(`[plan] pages = ${pages.length}, browsers = ${BROWSERS.length}, mode = ${MODE_BASELINE ? 'BASELINE' : 'COMPARE'}`);
  const server = await startServer();
  const results = [];

  try {
    for (const projectName of BROWSERS) {
      const { browserType, profile } = getBrowserAndProfile(projectName);
      for (const pageFile of pages) {
        const label = `${projectName} :: ${basename(pageFile)}`;
        process.stdout.write(`  [capture] ${label} ... `);
        let capResult;
        try {
          capResult = await captureOne(browserType, profile, projectName, pageFile);
        } catch (e) {
          console.log(`BROWSER ERROR`);
          results.push({
            project: projectName,
            baseName: basename(pageFile, '.html'),
            capture: null, baseline: null, diff: null,
            status: 'error',
            error: e.message,
          });
          if (MODE_BAIL) break;
          continue;
        }

        if (capResult.errors.length > 0) {
          console.log(`PAGE ERRORS (${capResult.errors.length})`);
          results.push({
            project: projectName,
            baseName: capResult.baseName,
            capture: capResult.capturePath,
            baseline: null, diff: null,
            status: 'error',
            error: capResult.errors.join(' | '),
          });
          if (MODE_BAIL) break;
          continue;
        }

        if (MODE_BASELINE) {
          const r = snapshotBaseline(capResult.capturePath, capResult.baselinePath);
          if (!r.ok) {
            console.log(`BASELINE WRITE ERROR`);
            results.push({ project: projectName, baseName: capResult.baseName,
              capture: capResult.capturePath, baseline: null, diff: null,
              status: 'error', error: r.error });
          } else {
            console.log(`baseline written (${capResult.ms}ms)`);
            results.push({ project: projectName, baseName: capResult.baseName,
              capture: capResult.capturePath, baseline: r.baseline, diff: null,
              status: 'pass', diff_summary: { pct_changed: '0.0', changed_pixels: 0 } });
          }
        } else {
          // compare mode — needs baseline
          if (!existsSync(capResult.baselinePath)) {
            console.log(`NO BASELINE (run --baseline first)`);
            results.push({ project: projectName, baseName: capResult.baseName,
              capture: capResult.capturePath, baseline: null, diff: null,
              status: 'error', error: 'baseline missing — run with --baseline first' });
          } else {
            const r = diffImages(capResult.baselinePath, capResult.capturePath, capResult.diffPath);
            if (!r.ok) {
              console.log(`DIFF ERROR`);
              results.push({ project: projectName, baseName: capResult.baseName,
                capture: capResult.capturePath, baseline: capResult.baselinePath, diff: null,
                status: 'error', error: r.error || 'unknown' });
            } else if (r.identical) {
              console.log(`PASS (${capResult.ms}ms)`);
              results.push({ project: projectName, baseName: capResult.baseName,
                capture: capResult.capturePath, baseline: capResult.baselinePath, diff: capResult.diffPath,
                status: 'pass', diff_summary: r });
            } else {
              console.log(`FAIL — ${r.changed_pixels} px (${r.pct_changed}%)`);
              results.push({ project: projectName, baseName: capResult.baseName,
                capture: capResult.capturePath, baseline: capResult.baselinePath, diff: capResult.diffPath,
                status: 'fail', diff_summary: r });
              if (MODE_BAIL) break;
            }
          }
        }
      }
    }
  } finally {
    server.close();
  }

  // Persist JSON + report
  writeFileSync(REPORTS_DATA, JSON.stringify(results, null, 2));
  const { reportPath, summary } = renderReport(results);
  console.log(`\n[summary] total=${summary.total} pass=${summary.passed} fail=${summary.failed} error=${summary.errored} (${summary.passRate}%)`);
  console.log(`[report]  ${reportPath}`);

  return { results, summary, reportPath };
}

// ---------- entry ----------
async function main() {
  if (MODE_SERVE) {
    const server = await startServer();
    console.log('[serve] press Ctrl+C to stop');
    process.on('SIGINT', () => { server.close(); process.exit(0); });
    return;
  }

  if (MODE_REPORT) {
    if (!existsSync(REPORTS_DATA)) {
      console.error('[fatal] no results.json — run a real test first');
      process.exit(2);
    }
    const results = JSON.parse(readFileSync(REPORTS_DATA, 'utf-8'));
    const { reportPath, summary } = renderReport(results);
    console.log(`[report] regenerated ${reportPath} (${summary.passRate}%)`);
    return;
  }

  const { summary } = await runPipeline();
  process.exit(summary.failed > 0 || summary.errored > 0 ? 1 : 0);
}

main().catch(e => { console.error('[fatal]', e); process.exit(2); });
