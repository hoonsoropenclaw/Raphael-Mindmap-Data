# Re-validate treating void elements as self-closing regardless of slash
from html.parser import HTMLParser
VOID = {'br','hr','img','input','meta','link','source','area','base','col','embed','param','track','wbr'}

class V(HTMLParser):
    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.stack = []
        self.errors = []
    def handle_starttag(self, tag, attrs):
        if tag not in VOID:
            self.stack.append((tag, self.getpos()))
    def handle_startendtag(self, tag, attrs):
        pass  # self-closing, do nothing
    def handle_endtag(self, tag):
        if tag in VOID:
            return  # ignore stray closing tags for void elements (Parser sees <input /> as <input>)
        if not self.stack:
            self.errors.append(f'Unexpected </{tag}> at {self.getpos()}')
            return
        top, pos = self.stack[-1]
        if top != tag:
            self.errors.append(f'Mismatched </{tag}> at {self.getpos()} (expected </{top}> opened at {pos})')
        self.stack.pop()

v = V()
v.feed(open('web_output.html').read())
print('Unclosed tags at EOF:', v.stack if v.stack else 'NONE  ✓')
print('Errors:', v.errors if v.errors else 'NONE  ✓')
print('OK' if not v.errors and not v.stack else 'FAIL')
