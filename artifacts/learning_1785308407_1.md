+"""In-process rate limiter for the Todo API.
+
+Design decisions
+----------------
+* Implements a sliding-window counter in pure Python (deque + dict).
+  No external dependencies, no Redis — appropriate for a single-process
+  FastAPI demo. For multi-worker / multi-host deployments, swap this
+  for a Redis-backed limiter (e.g. ``slowapi`` with a Redis storage).
+* The limiter is intentionally injectable: ``Limiter(time_source=...)``
+  lets tests freeze time so they don't have to ``sleep()`` to verify
+  the rolling window.
+* Exposed to FastAPI as a dependency so individual routes can opt in
+  via ``dependencies=[Depends(rate_limit)]`` or a route-level decorator
+  pattern.
+"""
+
+from __future__ import annotations
+
+from collections import deque
+from dataclasses import dataclass
+from time import monotonic
+from typing import Callable, Deque, Dict, Optional
+
+
+class RateLimitExceeded(Exception):
+    """Raised when a client exceeds the allowed request budget."""
+
+    def __init__(self, retry_after: float) -> None:
+        super().__init__(f"rate limit exceeded, retry after {retry_after:.2f}s")
+        self.retry_after = retry_after
+
+
+@dataclass
+class RateLimitConfig:
+    """Configuration for the limiter.
+
+    Attributes
+    ----------
+    requests:
+        Maximum number of requests allowed within ``window_seconds``.
+    window_seconds:
+        Width of the rolling window in seconds.
+    """
+
+    requests: int = 60
+    window_seconds: float = 60.0
+
+
+class Limiter:
+    """Sliding-window rate limiter keyed by an arbitrary string (typically an IP).
+
+    The implementation keeps, per key, a ``deque`` of monotonic timestamps
+    for each accepted request. A request is allowed iff the deque contains
+    fewer than ``config.requests`` entries after we drop those that fell
+    out of the window. This is exact (not approximate) and cheap for the
+    sizes we care about (≤ a few hundred entries per key).
+    """
+
+    def __init__(
+        self,
+        config: Optional[RateLimitConfig] = None,
+        *,
+        time_source: Callable[[], float] = monotonic,
+    ) -> None:

+        self._time = time_source
+        self._hits: Dict[str, Deque[float]] = {}

+# Todo API
+
+A small but production-shaped FastAPI service that manages a list of
+todo items, persists them in SQLite, and applies a per-client sliding
+window rate limit on every endpoint.
+
+```
+┌────────────┐    HTTP     ┌──────────────┐    SQL    ┌─────────┐
+│  Client    │ ─────────▶ │  FastAPI     │ ────────▶ │ SQLite  │
+│  (curl /   │ ◀───────── │  + limiter   │ ◀──────── │ file    │
+│  browser)  │   JSON     └──────────────┘           └─────────┘
+└────────────┘
+```
+
+## Features
+
+* CRUD endpoints (`POST/GET/PUT/DELETE`) for `/todos`
+* SQLite-backed persistence with auto-created schema
+* Sliding-window rate limiter (default `60 req / 60 s` per client)
+* Pydantic v2 request validation (`title` cannot be empty/whitespace)
+* 100 % line coverage on `app/` (58 tests)
+
+## Layout
+
+```
+todo_api/
+├── app/
+│   ├── __init__.py
+│   ├── database.py     # schema, CRUD, pydantic models
+│   ├── main.py         # FastAPI app factory + routes
+│   └── rate_limit.py   # in-process sliding-window limiter
+├── tests/
+│   ├── conftest.py     # shared fixtures (FakeClock, tmp DB)
+│   ├── test_api.py     # end-to-end HTTP tests
+│   ├── test_database.py
+│   └── test_rate_limit.py
+├── pytest.ini
+├── requirements.txt
+└── README.md

+## 2. 設計取捨
+
+### 2.1 ORM vs 裸 `sqlite3`
+選 **裸 `sqlite3`**。這是一個小型 reference project,SQLAlchemy 會模糊掉 persistence 層,讀者反而看不到 schema、index、prepared statement 是怎麼寫的。對 production 我會選 SQLAlchemy 2.x async + Alembic;對教學/小服務,裸 `sqlite3` + Pydantic 是最佳投資。
+
+### 2.2 Connection 管理
+- 每個 request 開一個 connection(`get_db` 是 generator dependency)
+- 透過 `try/except/finally` 處理 commit / rollback / close
+- 測試用 pytest 的 `tmp_path` 自動提供乾淨 DB,**從不碰 host filesystem**
+
+### 2.3 Rate limiter 演算法
+選 **sliding window counter with `deque`**:
+- 精確(非近似)
+- 對小流量(< 1000 entries/key)cheap
+- 不需要 Redis 或外部依賴
+- 接受 `time_source=...` injection,測試可注入 fake clock → **測試不必 `sleep()`**
+
+Token bucket / leaky bucket 也可以,但 sliding window 對「API rate limit」這個場景語意最自然(「過去 60 秒內不超過 60 次」)。
+
+### 2.4 Rate-limit key
+優先順序:`X-Forwarded-For` 第一跳 → `request.client.host` → `"anonymous"`。
+理由:production 通常在 reverse proxy(Nginx / Cloudflare)後面,直接讀 `request.client.host` 只會看到 proxy 的 IP,所有使用者會共享單一 bucket。讀 `XFF` 才是「per-client」的本意。
+
+### 2.5 Validation
+- `title` 用 `field_validator` 拒絕空白字串(`min_length=1` 擋不住 `"   "`)
+- `description` 預設空字串、max 2000
+- 錯誤一律回 `422 Unprocessable Entity`(FastAPI/Pydantic 預設)
+
+### 2.6 `/healthz` 排除 rate limit
+k8s liveness probe 預設每秒打一次,跑滿 60 req/60s 預算會直接誤報 pod unhealthy。所以 `/healthz` 故意 `dependencies=[]`。
+
+### 2.7 204 No Content
+FastAPI 對 `status_code=204` 不允許帶 response body。修法是 `response_class=Response` + 顯式 `return Response(status_code=204)`。
