+Functions take an open sqlite3.Connection and return plain dicts.
+This keeps the layer pure-Python and trivially testable.

+def _row_to_dict(row: sqlite3.Row) -> dict:
+    """Convert a sqlite3.Row to a JSON-friendly dict.
+
+    `done` is stored as INTEGER 0/1 in SQLite but exposed as bool in the
+    API surface — clients don't care about the storage encoding.
+    """
+    return {
+        "id": row["id"],
+        "title": row["title"],
+        "done": bool(row["done"]),
+        "created_at": row["created_at"],
+        "updated_at": row["updated_at"],
+        "deleted_at": row["deleted_at"],
+    }
+
+
+def list_todos(conn: sqlite3.Connection, include_deleted: bool = False) -> list[dict]:
+    """Return todos ordered by id ascending.
+
+    By default soft-deleted rows are filtered out — that's what most
+    API consumers want. `include_deleted=True` is mainly for tests
+    and admin paths.
+    """
+    if include_deleted:
+        sql = "SELECT * FROM todos ORDER BY id ASC"
+        rows = conn.execute(sql).fetchall()
+    else:
+        sql = "SELECT * FROM todos WHERE deleted_at IS NULL ORDER BY id ASC"
+        rows = conn.execute(sql).fetchall()
+    return [_row_to_dict(r) for r in rows]
+
+
+def get_todo(conn: sqlite3.Connection, todo_id: int, include_deleted: bool = False) -> Optional[dict]:
+    """Fetch a single todo by id, or None if not found / soft-deleted."""
+    if include_deleted:
+        row = conn.execute("SELECT * FROM todos WHERE id = ?", (todo_id,)).fetchone()
+    else:
+        row = conn.execute(
+            "SELECT * FROM todos WHERE id = ? AND deleted_at IS NULL", (todo_id,)
+        ).fetchone()
+    return _row_to_dict(row) if row else None
