+def wait_for_stable(path: Path, poll_interval: float = 0.2) -> bool:
+    """等到檔案大小穩定為止 (防止讀到寫到一半的檔)"""
+    deadline = time.time() + WRITE_MAX_WAIT
+    last_size = -1
+    while time.time() < deadline:
+        try:
+            size = path.stat().st_size
+        except FileNotFoundError:
+            return False
+        if size == last_size and size > 0:
+            time.sleep(WRITE_QUIET_WINDOW)
+            try:
+                if path.stat().st_size == size:
+                    return True
+            except FileNotFoundError:
+                return False
+        last_size = size
+        time.sleep(poll_interval)
+    return False
