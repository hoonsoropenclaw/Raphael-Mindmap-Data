+"""In-process sliding-window rate limiter.
+
+Design notes
+------------
+We use a simple sliding-window counter keyed by client IP. This keeps the
+implementation dependency-free and easy to test. For multi-worker / multi-host
+deployments, swap this for Redis-backed limits; the FastAPI dependency surface
+is intentionally tiny so the swap is local.
+
+The limiter lives on ``app.state`` so tests can inject a fresh instance per
+session.
+"""
+from __future__ import annotations
+
+import time
+from collections import deque
+from dataclasses import dataclass, field
+from typing import Deque, Dict
+
+
+@dataclass
+class RateLimitState:
+    """Per-key sliding window of recent request timestamps."""
+
+    window_seconds: float
+    max_requests: int
+    hits: Deque[float] = field(default_factory=deque)
+
+    def allow(self, now: float | None = None) -> bool:
+        ts = now if now is not None else time.monotonic()
+        # Drop entries that fell out of the window.
+        cutoff = ts - self.window_seconds
+        while self.hits and self.hits[0] <= cutoff:
+            self.hits.popleft()
+        if len(self.hits) >= self.max_requests:
+            return False
+        self.hits.append(ts)
+        return True
+
+
+@dataclass
+class RateLimiter:
+    """Stores a separate ``RateLimitState`` per client key."""
+
