+        self,
+        *,
+        capacity: int,
+        refill_per_sec: float,
+        clock: Callable[[], float] = time.monotonic,
+    ) -> None:
+        if capacity <= 0:
+            raise ValueError("capacity must be greater than zero")
+        if refill_per_sec <= 0:
+            raise ValueError("refill_per_sec must be greater than zero")
+        self.capacity = capacity
+        self.refill_per_sec = refill_per_sec
+        self._clock = clock
+        self._tokens = float(capacity)
+        self._last_refill = clock()
+        self._lock = threading.Lock()
+
+    def _refill_locked(self, now: float) -> None:
+        elapsed = max(0.0, now - self._last_refill)
+        self._tokens = min(
+            float(self.capacity),
+            self._tokens + elapsed * self.refill_per_sec,
+        )
+        self._last_refill = now
+
+    def try_consume(self, *, tokens: float = 1.0) -> RateLimitDecision:
+        if tokens <= 0:
+            raise ValueError("tokens must be greater than zero")
+        now = self._clock()
+        with self._lock:
+            self._refill_locked(now)
+            if self._tokens >= tokens:
+                self._tokens -= tokens
+                return RateLimitDecision(
+                    allowed=True,
+                    remaining=max(0, int(self._tokens)),
+                    retry_after=0.0,
+                    limit=self.capacity,
+                )
+            deficit = tokens - self._tokens
+            return RateLimitDecision(
+                allowed=False,
+                remaining=max(0, int(self._tokens)),
+                retry_after=deficit / self.refill_per_sec,
+                limit=self.capacity,
+            )

+
+## 3. Architecture decisions
+
+### 3.1 Synchronous SQLAlchemy-style repository, not async ORM
+
+The trial-and-error catalog (`trial-and-error/references/by-category/fastapi-async-session-pitfalls.md`, `fastapi-sync-vs-async-pattern-20260727.md`) calls out three async pitfalls — stale read, async/sync deadlock, `expire_on_commit=False` — and shows that for a small CRUD service the sync pattern is the safer choice. I went further: I dropped SQLAlchemy entirely and used the stdlib `sqlite3` module. Three reasons:
+
+- The data model is one table with five columns. An ORM would add a model class, a session lifecycle, and a dialect abstraction for zero practical benefit.
+- The handlers are declared `def` (synchronous), so FastAPI dispatches them through its threadpool; the `threading.RLock` around the shared connection keeps writes atomic. No async means none of the `expire_on_commit` / `await db.refresh()` traps.
+- Parameterized SQL (`cursor.execute(sql, params)`) is exactly as injection-safe as the ORM, and the verification chain in `tests/test_db.py` is shorter.
+
+### 3.2 Token bucket as a `Depends`, not middleware
+
+The Senior Architect's existing implementation also chose `Depends`. I kept that choice because it lets the limiter be tuned per endpoint (a future read-only endpoint could share a slower bucket) and because the `HTTPException` it raises carries the structured `Retry-After` + `X-RateLimit-*` headers. A small `add_rate_limit_headers` middleware then copies those headers onto every response, including the `429` itself.
+
+The key is `X-API-Key` → first `X-Forwarded-For` hop → peer IP. The API key is hashed with `sha256` before becoming a bucket key, so the raw key is not stored in the limiter's registry.
+
+### 3.3 Three-layer file-permission defense
+
+| Layer | Code | Guarantee |
+|---|---|---|
+| `os.umask(0o077)` | `create_app` and `init_db` | Every file subsequently created by this process is `0xxx` not `0644`, regardless of how the program is launched |
+| Explicit `0o600` for the main file | `os.open(path, O_CREAT | O_RDWR, 0o600)` before SQLite opens the connection | The main db is locked even if the umask is reset between this call and SQLite's first write |
+| `scrub_db_permissions` | walks the data dir after every write and `chmod 0o600` on the main file and the `-wal`/`-shm` sidecars | Any leaked sidecar file is corrected at the next CRUD call |
+
+The `_positive_int` / `_positive_float` config helpers and the `Settings.from_env` validator are pure-function unit tests, so a misconfigured deployment fails fast at startup.
+
+### 3.4 Validation strategy
+
+Pydantic v2 with `extra="forbid"` on every model. `TodoCreate` strips the title and bounds it to 200 characters; `TodoUpdate` re-validates blank-after-strip titles and rejects empty bodies through a `model_validator`. Whitespace stripping and length caps are documented in `README.md` so the API contract is discoverable from the openapi schema.
+
+### 3.5 Test layers
+
+| Layer | File | Coverage target |
+|---|---|---|
+| Token bucket math + lock | `tests/test_rate_limit.py` | refilling, isolation, threaded burst, asyncio burst |
+| Repository + permissions | `tests/test_db.py` | CRUD, transactions, WAL/SHM scrub |
+| Pydantic models | exercised through the API tests, plus a standalone `tests/test_config.py` for env loading |
+| End-to-end API | `tests/test_api.py` | validation, 404, 204, rate-limit headers, asyncio.gather against the live ASGI app |
+| Live HTTP | `scripts/smoke.py` | seven happy-path checks plus a manual 5-token burst run via curl |
+
+`pytest.ini` enforces `--cov-fail-under=80`; the current run achieves 91.85 %.
+
+### 3.6 Why I rejected the alternatives
+
+- **SQLAlchemy ORM** — extra model layer for one table; introduces a session object whose lifecycle must be coordinated with the FastAPI handler threadpool.
+- **`slowapi` rate limiter** — extra dependency, opaque key resolution, and the prior senior-architect review already showed that an in-process `threading.Lock` is the correct mental model.
+- **Async handlers with `asyncpg`** — none of the endpoints do any I/O that benefits from async; the only external call is to SQLite, which serializes writes anyway. The async pattern would add the three pitfalls for no real benefit.
+- **Sharing one Python process across multiple workers** — out of scope and documented in `README.md`; the rate limiter and SQLite connection are process-local by design.
