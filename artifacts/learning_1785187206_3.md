+#!/usr/bin/env python3
+"""github_trending.py - Fetch GitHub Trending repositories and emit a markdown table.
+
+GitHub does not expose an official trending API; we scrape https://github.com/trending
+and parse the public HTML using only the Python standard library (urllib + re).
+That keeps the tool dependency-free so it runs in any Python 3.8+ environment
+(including hermes-agent sandboxes that may not have requests installed).
+
+Usage:
+    github_trending.py                          # daily, all languages
+    github_trending.py --since weekly            # weekly
+    github_trending.py --language python         # python only
+    github_trending.py --limit 10 --output out.md
+    github_trending.py --json                    # JSON instead of markdown
+
+Exit codes:
+    0 = success (rendered markdown / JSON)
+    1 = network error (curl-style: tried, failed, told the user)
+    2 = parse error (page fetched but contained 0 repos - layout drifted)
+    3 = usage error (bad CLI flags)
+"""

+    "(KHTML, like Gecko) Chrome/120.0 Safari/537.36"

+@dataclass
+class TrendingRepo:
+    owner: str
+    name: str
+    full_name: str          # owner/name
+    url: str                # https://github.com/owner/name
+    description: str        # may be empty
+    language: str           # may be empty
+    total_stars: int        # cumulative stars
+    stars_in_period: int    # stars today / this week / this month
+    period_label: str       # "today" | "this week" | "this month"
+    built_by: List[str]     # contributor avatars' alt text (usernames)
+
+
+def build_url(since: str, language: Optional[str]) -> str:
+    url = GITHUB_TRENDING_URL + "/" + (since or "daily")
+    if language:
+        url += "/" + language
+    return url
+
+
+def fetch_html(url: str, timeout: int = 30, max_retries: int = 3) -> str:
+    """Fetch URL with retry/backoff. Raises urllib.error.URLError on hard failure."""
+    last_err: Optional[Exception] = None
+    for attempt in range(1, max_retries + 1):
+        try:
+            req = urllib.request.Request(url, headers={"User-Agent": DEFAULT_USER_AGENT,
+                                                       "Accept-Language": "en-US,en;q=0.8"})
+            with urllib.request.urlopen(req, timeout=timeout) as resp:
+                if resp.status != 200:
+                    raise urllib.error.HTTPError(
+                        url, resp.status, "HTTP " + str(resp.status), resp.headers, None)
+                return resp.read().decode("utf-8", errors="replace")
+        except (urllib.error.URLError, urllib.error.HTTPError, TimeoutError) as e:
+            last_err = e
+            if attempt < max_retries:
+                time.sleep(2 ** attempt)
