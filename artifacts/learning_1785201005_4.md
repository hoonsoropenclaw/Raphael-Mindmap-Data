+"""SQLAlchemy engine + session factory.
+
+設計重點：
+- 測試用 in-memory SQLite + StaticPool 才能跨 session 共用同一個連線
+- 正式跑時從環境變數 `TODO_DB_URL` 讀，預設 `sqlite:///./todos.db`
+- `get_db` 走 generator pattern 確保 session 一定 close
+"""
+from __future__ import annotations
+
+import os
+from typing import Generator
+
+from sqlalchemy import create_engine
+from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker
+
+
+class Base(DeclarativeBase):
+    pass
+
+
+def make_engine(url: str | None = None):
+    """Build an engine. In-memory 需要 StaticPool 才能讓多個 session 看見同一份資料."""
+    db_url = url or os.environ.get("TODO_DB_URL", "sqlite:///./todos.db")
+    if db_url.endswith(":memory:") or db_url == "sqlite:///:memory:":
+        from sqlalchemy.pool import StaticPool
+
+        return create_engine(
+            db_url,
+            connect_args={"check_same_thread": False},
+            poolclass=StaticPool,
+            future=True,
+        )
+    return create_engine(db_url, future=True)
+
+
+def make_session_factory(engine) -> sessionmaker:
+    return sessionmaker(bind=engine, autoflush=False, autocommit=False, expire_on_commit=False)
+
+
+def get_db(session_factory: sessionmaker) -> Generator[Session, None, None]:
+    db = session_factory()
+    try:
+        yield db
+    finally:
+        db.close()

+## 關鍵技術決策
+
+### 1. SQLAlchemy 2.0 sync（非 async）
+- SQLite + sync 模式下測試不需要 `pytest-asyncio` loop policies
+- `StaticPool` + in-memory SQLite 解決「每個 session 拿到全新空 DB」問題
+- `session_factory` 由 `create_app(session_factory=...)` 注入 → 測試可控，正式環境自動建檔
+
+### 2. `Annotated[Session, Depends(db_dep)]` 失敗的修正
+第一版程式用 `db: DbSession` 結果 FastAPI 0.140 把 `db` 當 query param（422 "missing field"）。FastAPI 0.140+ 對 nested type 提示判定比舊版嚴。要用：
+```python
+def f(payload: ..., db: Session = Depends(db_dep)):
+```
+而不是 type alias。修完一次綠。
+
+### 3. slowapi 中間件掛法
+光建 `Limiter(default_limits=[...])` 不會自動套用，要：
+```python
+app.add_middleware(SlowAPIMiddleware)
+app.add_exception_handler(RateLimitExceeded, _rate_limit_exceeded_handler)
+```
+兩個缺一就 429 不會觸發。
+
+### 4. 測試端 rate-limit 配額
+- CRUD 測試把 limit 調成 `1000/minute` 不干擾
+- rate-limit 測試用獨立 fixture 灌 `tight_app` limit = `3/minute` 來觸發 429
+
+### 5. PATCH 走 `model_dump(exclude_unset=True)`
+保證「沒給的欄位不覆寫既有值」，partial update 才正確。
+
+### 6. `Base.metadata.create_all(bind=...)` 而非 Alembic
+單進程小工具夠用；正式部署應該改 Alembic migration，這是已知 trade-off，IMPL_NOTES 透明紀錄。
