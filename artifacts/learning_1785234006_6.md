+"""Database engine + session factory.
+
+We use SQLAlchemy 2.0 typed-style. Engine is created once per process and
+reused via `get_db` dependency. For tests, the engine is rebuilt against an
+in-memory SQLite URL via `init_engine`.
+"""
+from __future__ import annotations
+
+from collections.abc import Generator
+from contextlib import contextmanager
+
+from sqlalchemy import Engine, create_engine
+from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker
+
+
+class Base(DeclarativeBase):
+    """Declarative base for all ORM models."""
+
+
+_engine: Engine | None = None
+_SessionLocal: sessionmaker[Session] | None = None
+
+
+def init_engine(database_url: str = "sqlite:///./todos.db") -> Engine:
+    """Initialize (or replace) the global engine + session factory.
+
+    Called once at app startup and again in tests against an in-memory DB.
+    """
+    global _engine, _SessionLocal
+    # `check_same_thread=False` is required for SQLite when used with FastAPI's
+    # threadpool. SQLAlchemy handles thread-safety via the connection pool.
+    connect_args = {"check_same_thread": False} if database_url.startswith("sqlite") else {}
+    _engine = create_engine(
+        database_url,
+        connect_args=connect_args,
+        future=True,
+        echo=False,
+    )
+    _SessionLocal = sessionmaker(
+        bind=_engine,
+        autoflush=False,
+        autocommit=False,
+        expire_on_commit=False,
+    )
+    return _engine
+
+
+def get_engine() -> Engine:
+    """Return the active engine, raising if init_engine was never called."""
+    if _engine is None:
+        raise RuntimeError("Engine not initialized. Call init_engine() first.")
+    return _engine
+
+
+def get_db() -> Generator[Session, None, None]:
+    """FastAPI dependency that yields a Session and ensures cleanup.
+
+    Pattern: generator-style dependency so the session is always closed even
+    if the route raises. We do NOT commit here — the CRUD layer commits
+    explicitly so it can roll back on error.
+    """

+        raise RuntimeError("Session factory not initialized. Call init_engine() first.")
+    db = _SessionLocal()
+    try:
+        yield db
+    finally:
+        db.close()

+"""Sliding-window in-memory rate limiter as a FastAPI dependency.
+
+Design choices (rationale so future maintainers don't have to reverse-engineer):
+
+  * **Sliding window over fixed window** — fixed-window allows a 2x burst at the
+    boundary (10 reqs at 0:59 + 10 reqs at 1:00 = 20 in 1s). Sliding window
+    enforces the limit more fairly.
+
+  * **In-memory over Redis** — this is a single-process app. Adding Redis for
+    one instance is over-engineering. Document the limitation so a future
+    multi-worker deployment knows to upgrade.
+
+  * **Per-client key (IP-based) with thread-safe `deque`** — we use a `dict[str,
+    deque[float]]` and protect mutation with a `threading.Lock`. This is good
+    enough for moderate traffic; for very high RPS you'd want token buckets.
+
+  * **Exposes remaining + retry_after headers** — clients can self-throttle.
+"""
+from __future__ import annotations
+
+import threading
+import time
+from collections import defaultdict, deque
+from collections.abc import Callable
+from dataclasses import dataclass
+
+from fastapi import Depends, HTTPException, Request, status

+class RateLimitConfig:
+    """Configuration for a rate limiter instance.
+
+    `max_requests` is the request count permitted in the rolling window.
+    `window_seconds` is the size of that window.
+    """
+
+    max_requests: int
+    window_seconds: float
+
+
+class SlidingWindowRateLimiter:
+    """Thread-safe sliding-window rate limiter keyed by client identifier."""
+
+    def __init__(self, config: RateLimitConfig, *, key_fn: Callable[[Request], str] | None = None) -> None:
+        self._config = config
+        self._buckets: dict[str, deque[float]] = defaultdict(deque)
+        self._lock = threading.Lock()
+        # Default key: client IP. `Request.client.host` is set by uvicorn for
+        # both direct connections and X-Forwarded-For if you set `proxy_headers`.
+        self._key_fn = key_fn or (lambda req: req.client.host if req.client else "anonymous")
+
+    @property
+    def config(self) -> RateLimitConfig:
+        return self._config
+
+    def _check(self, key: str) -> tuple[bool, int, float]:
+        """Atomically check + record. Returns (allowed, remaining, retry_after).
+
+        The caller is expected to call this from within a request handler. The
+        lock here is short-lived (microseconds) so contention is negligible.
+        """
+        now = time.monotonic()
+        window_start = now - self._config.window_seconds
+        with self._lock:
+            bucket = self._buckets[key]
+            # Drop timestamps outside the window (oldest first since we append).
+            while bucket and bucket[0] <= window_start:
+                bucket.popleft()
+            if len(bucket) >= self._config.max_requests:
+                # The earliest timestamp will exit the window soonest.
+                retry_after = max(0.0, bucket[0] - window_start)
+                return False, 0, retry_after
+            bucket.append(now)
+            remaining = self._config.max_requests - len(bucket)
+            return True, remaining, 0.0
