+// src/context/todoReducer.ts
+// 用 useReducer 而不是 useState — 因為狀態變化是「動作型」而非「值型」
+// 對應前次 API：POST/PUT/DELETE/GET 各自對應一個 action type
+
+import type { Todo, TodoDraft } from "../types/todo";
+import { newId } from "../utils/id";
+import { nowIso } from "../utils/date";
+
+export type TodoAction =
+  | { type: "ADD"; payload: TodoDraft }
+  | { type: "UPDATE"; payload: { id: string; patch: Partial<TodoDraft> } }
+  | { type: "TOGGLE"; payload: { id: string } }
+  | { type: "DELETE"; payload: { id: string } }
+  | { type: "CLEAR_COMPLETED" }
+  | { type: "REPLACE_ALL"; payload: Todo[] }
+  | { type: "RESET" };
+
+export const initialTodos: Todo[] = [];
+
+export function todoReducer(state: Todo[], action: TodoAction): Todo[] {
+  switch (action.type) {
+    case "ADD": {
+      const now = nowIso();
+      const todo: Todo = {
+        ...action.payload,
+        id: newId(),
+        createdAt: now,
+        updatedAt: now,
+      };
+      return [todo, ...state]; // newest first（對應 API GET /todos newest first）
+    }
+
+    case "UPDATE": {
+      return state.map((t) =>
+        t.id === action.payload.id
+          ? { ...t, ...action.payload.patch, updatedAt: nowIso() }
+          : t
+      );
+    }
+
+    case "TOGGLE": {
+      return state.map((t) =>
+        t.id === action.payload.id
+          ? { ...t, completed: !t.completed, updatedAt: nowIso() }
+          : t
+      );
+    }
+
+    case "DELETE": {
+      return state.filter((t) => t.id !== action.payload.id);
+    }
+
+    case "CLEAR_COMPLETED": {
+      return state.filter((t) => !t.completed);
+    }
+
+    case "REPLACE_ALL": {
+      return action.payload;
+    }
+
+    case "RESET": {
+      return initialTodos;
+    }
+
+    default: {
+      // exhaustive check
+      const _exhaustive: never = action;
+      return state;
+    }
+  }
