+ * Domain types for the Kanban board.

+ * IDs are stable strings — generated on the client (no backend coupling in this demo).

+// Minimal static file server for the production build.
+// Why a script and not vite preview: vite preview has surprising SPA-fallback
+// behaviour for nested routes and isn't trivial to deploy independently. This
+// module needs zero deps and survives behind a read-only filesystem.
+import { createServer } from 'node:http';
+import { readFile } from 'node:fs/promises';
+import { extname, join, normalize, resolve } from 'node:path';
+import { fileURLToPath } from 'node:url';
+
+const __dirname = fileURLToPath(new URL('.', import.meta.url));
+const distDir = resolve(__dirname, '..', 'dist');
+const port = Number.parseInt(process.env.PORT || '4180', 10);
+
+const MIME = {
+  '.html': 'text/html; charset=utf-8',
+  '.js': 'application/javascript; charset=utf-8',
+  '.mjs': 'application/javascript; charset=utf-8',
+  '.css': 'text/css; charset=utf-8',
+  '.svg': 'image/svg+xml',
+  '.png': 'image/png',
+  '.jpg': 'image/jpeg',
+  '.json': 'application/json; charset=utf-8',
+};
+
+const server = createServer(async (req, res) => {
+  try {
+    // Strip query and decode.
+    const url = new URL(req.url || '/', `http://${req.headers.host}`);
+    let pathname = decodeURIComponent(url.pathname);
+    if (pathname.endsWith('/')) pathname += 'index.html';
+
+    const safe = normalize(pathname).replace(/^([\\/]\.\.)+/, '');
+    const filePath = join(distDir, safe);
+    const data = await readFile(filePath);
+    res.writeHead(200, {
+      'Content-Type': MIME[extname(filePath).toLowerCase()] || 'application/octet-stream',
+      'Cache-Control': 'no-cache',
+    });
+    res.end(data);
+  } catch (err) {
+    // SPA fallback: serve index.html for unknown paths so client routing works.
+    try {
+      const data = await readFile(join(distDir, 'index.html'));
+      res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
+      res.end(data);
+    } catch {
+      res.writeHead(404, { 'Content-Type': 'text/plain' });
+      res.end(`404 — ${err?.message || 'not found'}`);
+    }
+  }
+});
+
+server.listen(port, '127.0.0.1', () => {
+  // eslint-disable-next-line no-console
+  console.log(`Serving dist on http://127.0.0.1:${port}/`);
+});

+# Kanban Board
+
+Drag-and-drop Kanban board built with React 18 + TypeScript + Vite, with undo/redo, bulk operations, keyboard shortcuts, filters, and a dual light/dark theme.
+
+## Run
+
+```bash
+npm install
+npm run dev     # http://127.0.0.1:5180
+npm test        # 38 vitest tests
+npm run build   # dist/ (~170 kB JS, 17 kB CSS)
+```
+
+## Features
+
+- **Drag and drop** — Native HTML5 DnD (no library). Cross-column move + in-column reorder, with rAF-throttled drop-target preview.
+- **Undo / Redo** — Cmd/Ctrl+Z, Cmd/Ctrl+Shift+Z. Bounded at 50 snapshots.
+- **Keyboard shortcuts** — `N` new task, `/` focus search, `Esc` cancel layer by layer.
+- **Multi-select** — Cmd-click toggle, Shift-click range; bulk move, bulk delete, bulk priority.
+- **Filters** — Free-text search across title/description/tag, priority chips, tag chips, assignee, hide-done toggle. Filter changes are independent of undo history.
+- **Persistence** — Tasks + filters + theme persist in `localStorage`. History is memory-only.
+- **Theming** — Full light + dark via CSS variables on `[data-theme]`. No flash on reload.
+- **Inline edit** — Click a card title to edit, `Enter` commits, `Esc` reverts.
+- **Inline tags + assignee** — Editable per-card without a detail modal.
+
+## Architecture
+
+- `src/state/boardReducer.ts` — single source of truth for state mutation. 27 action types.
+- `src/state/selectors.ts` — denormalised read views (column view with visibility/total/hidden counts, all-tags, all-assignees).
+- `src/state/BoardProvider.tsx` — React context + lazy init from `localStorage` + side-effect: persist slice on change.
+- `src/hooks/useDrag.ts` — drop index + rAF throttle.
+- `src/hooks/useKeyboardShortcuts.ts` — global hotkeys with input-aware skip.
+- `src/components/{Board,Column,Card,Modal,Toolbar}/` — one folder per component, each with `*.tsx` + `*.css`.
+- `src/styles/global.css` — design tokens + reset + reduced-motion defaults.
+
